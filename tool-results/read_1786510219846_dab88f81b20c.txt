    16→ *   - Genre tags (for auto-selection by learning system)

    19→// ─── Types ─────────────────────────────────────────────────────────────────
    20→export type EngineType = 'DRUM' | 'SYNTH' | 'FM' | 'NOISE' | 'WAVETABLE';
    21→export type DrumType = 'kick' | 'snare' | 'clap' | 'hatC' | 'hatO' | 'tom' | 'rim' | 'glitch' | 'shaker' | 'riser' | 'impact' | 'downlifter';
    22→export type FilterType = 'lowpass' | 'highpass' | 'bandpass' | 'notch' | 'allpass';
    23→export type LFODest = 'off' | 'cutoff' | 'pitch' | 'amp' | 'pan';
    24→export type Genre = 'PSYTRANCE' | 'TECHNO' | 'TRANCE' | 'PROGRESSIVE' | 'DARK-PSY' | 'GOA' | 'ANY';
    25→export type Category = 'drum' | 'bass' | 'lead' | 'pad' | 'pluck' | 'arp' | 'fx' | 'texture';
    26→
    27→export interface SoundPreset {
    28→  id: string;
    29→  name: string;
    30→  genre: Genre;
    31→  cat: Category;
    32→  engine: EngineType;
    33→
    34→  // Drum-specific
    35→  drumType?: DrumType;
    36→  tune?: number;
    37→  decay?: number;
    38→  tone?: number;
    39→  punch?: number;
    40→
    41→  // Synth oscillators
    42→  wave1?: OscillatorType;
    43→  wave2?: OscillatorType;
    44→  oct2?: number;       // octave offset for osc2
    45→  detune?: number;     // cents
    46→  fmAmount?: number;   // for FM engine
    47→  fmRatio?: number;    // carrier:modulator ratio
    48→
    49→  // Filter
    50→  fType?: FilterType;
    51→  cutoff?: number;
    52→  res?: number;
    53→  fEnvAmt?: number;    // filter envelope amount
    54→  fDecay?: number;     // filter envelope decay
    55→
    56→  // Amp envelope (ADSR)
    57→  atk?: number;
    58→  dec?: number;
    59→  sus?: number;
    60→  rel?: number;
    61→  gate?: number;       // note gate (multiplier of step duration)
    62→
    63→  // LFO
    64→  lfoRate?: number;
    65→  lfoDepth?: number;
    66→  lfoDest?: LFODest;
    67→  lfoShape?: OscillatorType;
    68→
    69→  // Effects sends
    70→  sendDelay?: number;  // 0-1
    71→  sendReverb?: number; // 0-1
    72→
    73→  // Polyphony
    74→  poly?: number;
    75→
    76→  // Velocity sensitivity
    77→  velSens?: number;    // 0-1
    78→
    79→  // Musical context (for learning system auto-selection)
    80→  scaleDegrees?: number[]; // which scale degrees this fits (e.g., [0, 4, 7] = root, 3rd, 5th)
    81→  energyLevel?: number;    // 0-1, how energetic
    82→  moodTags?: string[];     // ['dark', 'hypnotic', 'uplifting', 'aggressive', 'ethereal']
    83→}

   653→/**
   654→ * Auto-select presets by detected scale + genre + energy.
   655→ * Used by the learning system to pick the best sounds for the current context.
   656→ */
   657→export function autoSelect(
   658→  scaleDegrees: number[],
   659→  genre: Genre = 'PSYTRANCE',
   660→  energy: number = 0.7,
   661→  cat?: Category,
   662→): SoundPreset[] {
   663→  let candidates = SOUND_BANK;
   664→  if (cat) candidates = candidates.filter(p => p.cat === cat);
   665→  // Prefer matching genre, fall back to ANY
   666→  const genreMatch = candidates.filter(p => p.genre === genre || p.genre === 'ANY');
   667→  candidates = genreMatch.length > 0 ? genreMatch : candidates;
   668→  // Filter by energy (±0.2 tolerance)
   669→  candidates = candidates.filter(p => Math.abs((p.energyLevel || 0.5) - energy) < 0.25);
   670→  // Sort by how many scale degrees match
   671→  candidates.sort((a, b) => {
   672→    const aMatch = (a.scaleDegrees || []).filter(d => scaleDegrees.includes(d)).length;
   673→    const bMatch = (b.scaleDegrees || []).filter(d => scaleDegrees.includes(d)).length;
   674→    return bMatch - aMatch;
   675→  });
   676→  return candidates.slice(0, 5);
   677→}

   679→export function bankStats() {
   680→  const cats: Record<string, number> = {};
   681→  const genres: Record<string, number> = {};
   682→  SOUND_BANK.forEach(p => {
   683→    cats[p.cat] = (cats[p.cat] || 0) + 1;
   684→    genres[p.genre] = (genres[p.genre] || 0) + 1;
   685→  });
   686→  return { total: SOUND_BANK.length, cats, genres };
   687→}
