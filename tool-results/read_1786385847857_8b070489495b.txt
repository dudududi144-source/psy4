     1→/**
     2→ * WorkletEngine — Task W1 unified AudioWorklet bridge.
     3→ *
     4→ * Replaces the 1054-node Web Audio NODE GRAPH inside Psy4EngineV2 with a
     5→ * SINGLE AudioWorkletNode that contains ALL DSP (Moog ladder, polyBLEP,
     6→ * Schroeder reverb, stereo delay, bus processors, master chain, 18 voice
     7→ * types including the new FM voice).
     8→ *
     9→ * Architecture (PSY5 model — single worklet, RT-safe):
    10→ *   Main thread (Psy4EngineV2 — musical logic only)
    11→ *     ├─ Harmony, melody, style detection, learning, DJ sync, flow, director
    12→ *     ├─ tick() → ask director for window notes → build Float64Array batch
    13→ *     └─ port.postMessage({type:'events', events: Float64Array})  ← BATCHED
    14→ *   AudioWorklet (psy4-engine.js — DSP only)
    15→ *     ├─ 256-slot ring buffer (PSY5 proven size)
    16→ *     ├─ Preallocated voice pools (zero per-block allocation)
    17→ *     ├─ Polynomial ftanh (Pade approx — 10x cheaper than Math.tanh)
    18→ *     ├─ CPU-load monitoring + dynamic voice budget
    19→ *     └─ Bus processors + master chain + Schroeder reverb + ping-pong delay
    20→ *     ↓ stereo output
    21→ *   AnalyserNode → AudioContext.destination
    22→ *
    23→ * The main thread decides WHAT notes to play (musical logic); the worklet
    24→ * decides HOW they sound (DSP). This is the PSY5 separation of concerns.
    25→ *
    26→ * Why this is 5-10x more efficient than the node graph:
    27→ *   - 1 AudioWorkletNode vs 1054 createOscillator/Gain/Filter nodes
    28→ *   - No per-hit node creation (voices are preallocated pools inside worklet)
    29→ *   - No GC pressure (zero allocation in process())
    30→ *   - No AudioParam connection overhead (params are direct function args)
    31→ *   - No main-thread scheduling jitter (worklet runs on audio thread)
    32→ *
    33→ * Voice ID constants mirror public/worklets/psy4-engine.js. Keep in sync.
    34→ */
    35→

    45→// ─── Voice IDs (mirror worklet) ────────────────────────────────────────────
    46→export const VOICE = {
    47→  KICK: 0, BASS: 1, LEAD: 2, ACID: 3, PAD: 4,
    48→  HAT: 5, HAT_OPEN: 6, CLAP: 7, PERC: 8, SHAKER: 9,
    49→  TEXTURE: 10, RISER: 11, IMPACT: 12, SWEEP: 13,
    50→  ZAP: 14, BLIP: 15, DOWNLIFTER: 16, FM: 17,
    51→} as const;
    52→export type VoiceId = typeof VOICE[keyof typeof VOICE];
    53→
    54→// ─── Track → Voice mapping (used by Psy4EngineV2 facade) ───────────────────
    55→// Track indices match Psy4EngineV2's 8-track model:
    56→//   0=KICK  1=SNARE/CLAP  2=HATS  3=PERC  4=BASS  5=LEAD  6=PAD  7=ARP
    57→// The worklet voice IDs are different (see VOICE above), so we map.
    58→export function trackToVoiceId(track: number, opts?: { fmLead?: boolean; fmArp?: boolean }): VoiceId {
    59→  switch (track) {
    60→    case 0: return VOICE.KICK;
    61→    case 1: return VOICE.CLAP;
    62→    case 2: return VOICE.HAT;
    63→    case 3: return VOICE.PERC;
    64→    case 4: return VOICE.BASS;
    65→    case 5: return opts?.fmLead ? VOICE.FM : VOICE.LEAD;
    66→    case 6: return VOICE.PAD;
    67→    case 7: return opts?.fmArp ? VOICE.FM : VOICE.ACID;
    68→    default: return VOICE.KICK;
    69→  }
    70→}

   107→// ─── WorkletEngine ─────────────────────────────────────────────────────────
   108→
   109→/**
   110→ * Unified AudioWorklet bridge. Owns the AudioContext, the AudioWorkletNode,
   111→ * and an AnalyserNode tap. Exposes a minimal, typed API that Psy4EngineV2
   112→ * (the facade) calls into.
   113→ *
   114→ * Lifecycle:
   115→ *   const engine = new WorkletEngine();
   116→ *   await engine.init();              // loads worklet module, creates node
   117→ *   engine.start('dark-psy');         // sends 'play' to worklet
   118→ *   engine.setBpm(145);
   119→ *   engine.setWorld({ kickFundamental: 50, ... });
   120→ *   engine.sendEventBatch(float64Array); // notes to play
   121→ *   engine.stop();
   122→ *
   123→ * The AudioContext is created lazily inside init() (after a user gesture —
   124→ * required by browsers for audio playback).
   125→ */
   126→export class WorkletEngine implements AudioBackend {
   127→  private ctx: AudioContext | null = null;
   128→  private node: AudioWorkletNode | null = null;
   129→  private analyser: AnalyserNode | null = null;
   130→  private stats: WorkletStats = {
   131→    playing: false, step: 0, activeVoices: 0, eventCount: 0,
   132→    currentFrame: 0, cpuLoad: 0, voiceBudget: 32, processMs: 0,
   133→  };
   134→  private statsListeners: Array<(stats: WorkletStats) => void> = [];
   135→  private moduleLoaded = false;
   136→  private loadPromise: Promise<boolean> | null = null;
   137→  /**
   138→   * Task F1-F3: per-tick event batch builder. The engine calls
   139→   * triggerDrum/triggerSynth during scheduleStep(); events accumulate here.
   140→   * flushEvents() sends them in ONE postMessage (PSY5 batched) at the end
   141→   * of tick(). This is the bridge between main-thread musical logic and
   142→   * audio-thread DSP.
   143→   */
   144→  private eventBatch: EventBatchBuilder = new EventBatchBuilder();
   145→  /**
   146→   * Task F1-F3: effective params cache. Mirrors what we've pushed via
   147→   * setWorld / setFX so getParams() can return them without a round-trip
   148→   * to the worklet (the worklet is on the audio thread — we can't query it
   149→   * synchronously). Updated on every setWorld / setFX call.
   150→   */
   151→  private effectiveWorld: Record<string, number> = {};
   152→  private effectiveFX: AudioBackendFXConfig = {};
   153→  private effectiveMacros: Record<string, number> = {};
   154→
   155→  /** The AudioContext (created in init()). Exposed for the facade. */
   156→  get context(): AudioContext | null { return this.ctx; }
   157→
   158→  /** The AudioWorkletNode (null until init() completes). */
   159→  get workletNode(): AudioWorkletNode | null { return this.node; }
   160→
   161→  /** True if the worklet module + node are ready for events. */
   162→  get ready(): boolean { return this.node !== null && this.moduleLoaded; }
   163→
   164→  /**
   165→   * Initialize the AudioContext, load the worklet module, and create the
   166→   * AudioWorkletNode + AnalyserNode. Safe to call multiple times — subsequent
   167→   * calls are no-ops if already initialized.
   168→   *
   169→   * MUST be called from a user-gesture handler (click/keypress) so the
   170→   * AudioContext starts in 'running' state.
   171→   *
   172→   * @param latencyHintOrCtx Either an AudioContextLatencyCategory for a new
   173→   *   context, OR an existing AudioContext to reuse (e.g., shared with a
   174→   *   legacy engine facade). Reusing avoids creating two AudioContexts (which
   175→   *   would double the audio thread overhead).
   176→   * @returns true if the worklet is ready; false if loading failed.
   177→   */
   178→  async init(
   179→    latencyHintOrCtx: AudioContextLatencyCategory | AudioContext = 'interactive',
   180→  ): Promise<boolean> {
   181→    if (this.ctx && this.node && this.moduleLoaded) return true;
   182→
   183→    // Create or reuse AudioContext
   184→    if (!this.ctx) {
   185→      if (latencyHintOrCtx instanceof AudioContext) {
   186→        // Reuse an existing context (facade pattern — Psy4EngineV2 shares
   187→        // its context so we don't double the audio-thread overhead).
   188→        this.ctx = latencyHintOrCtx;
   189→      } else {
   190→        // Create a new context (lazy — must be from a user gesture)
   191→        const w = typeof window !== 'undefined' ? window : undefined;
   192→        const Ctx: (typeof AudioContext) | undefined =
   193→          (w && (w.AudioContext || (w as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext)) ||
   194→          undefined;
   195→        if (!Ctx) {
   196→          console.warn('[WorkletEngine] No AudioContext available');
   197→          return false;
   198→        }
   199→        this.ctx = new Ctx({ latencyHint: latencyHintOrCtx });
   200→      }
   201→    }
   202→
   203→    // Resume if suspended (autoplay policy)
   204→    if (this.ctx.state === 'suspended') {
   205→      try { await this.ctx.resume(); } catch { /* user gesture required */ }
   206→    }
   207→
   208→    // Load the worklet module (cached per-context)
   209→    if (!this.moduleLoaded && !this.loadPromise) {
   210→      this.loadPromise = this.ctx.audioWorklet.addModule('/worklets/psy4-engine.js')
   211→        .then(() => { this.moduleLoaded = true; return true; })
   212→        .catch((e) => {
   213→          console.warn('[WorkletEngine] Failed to load worklet module:', e);
   214→          this.loadPromise = null;
   215→          return false;
   216→        });
   217→    }
   218→    if (this.loadPromise) {
   219→      const ok = await this.loadPromise;
   220→      if (!ok) return false;
   221→    }
   222→
   223→    // Create the AudioWorkletNode
   224→    if (!this.node) {
   225→      try {
   226→        this.node = new AudioWorkletNode(this.ctx!, 'psy4-engine', {
   227→          numberOfInputs: 0,
   228→          numberOfOutputs: 1,
   229→          outputChannelCount: [2],
   230→          channelCount: 2,
   231→          channelCountMode: 'explicit',
   232→          channelInterpretation: 'speakers',
   233→        });
   234→      } catch (e) {
   235→        console.warn('[WorkletEngine] Failed to create AudioWorkletNode:', e);
   236→        return false;
   237→      }
   238→
   239→      // Wire stats messages from the worklet
   240→      this.node.port.onmessage = (e: MessageEvent) => {
   241→        const msg = e.data as Partial<WorkletStats> & { type: string };
   242→        if (msg.type === 'stats') {
   243→          // Merge incoming stats (worklet sends a subset; preserve local defaults)
   244→          this.stats = {
   245→            playing: msg.playing ?? false,
   246→            step: msg.step ?? 0,
   247→            activeVoices: msg.activeVoices ?? 0,
   248→            eventCount: msg.eventCount ?? 0,
   249→            currentFrame: msg.currentFrame ?? 0,
   250→            cpuLoad: msg.cpuLoad ?? 0,
   251→            voiceBudget: msg.voiceBudget ?? 32,
   252→            processMs: msg.processMs ?? 0,
   253→          };
   254→          // Notify listeners (UI pulls this on its own cadence; the worklet
   255→          // posts at ~10 Hz so this is cheap).
   256→          for (const fn of this.statsListeners) {
   257→            try { fn(this.stats); } catch { /* listener error — ignore */ }
   258→          }
   259→        }
   260→      };
   261→    }
   262→
   263→    // Create analyser + connect: worklet → analyser → destination
   264→    if (!this.analyser) {
   265→      this.analyser = this.ctx!.createAnalyser();
   266→      this.analyser.fftSize = 2048;
   267→      this.analyser.smoothingTimeConstant = 0.78;
   268→      this.node!.connect(this.analyser);
   269→      // Only connect to destination if this is OUR context (not a shared one
   270→      // — the facade's context already has its own destination wiring, and
   271→      // double-connecting would cause the worklet output to play twice).
   272→      // We ALWAYS connect the analyser to destination because the worklet's
   273→      // output must reach the speakers. The facade (Psy4EngineV2) does NOT
   274→      // connect anything else to destination when useWorklet=true — the
   275→      // worklet is the sole audio source.
   276→      this.analyser.connect(this.ctx!.destination);
   277→    }
   278→
   279→    return true;
   280→  }
   281→
   282→  /**
   283→   * Start playback. Sends 'play' to the worklet. The worklet resets its
   284→   * transport (step=0, currentSample=0).
   285→   *
   286→   * @param worldId Optional world id (sent for the worklet's reference; the
   287→   *   worklet doesn't use it directly but main thread uses it for tracking).
   288→   */
   289→  start(_worldId?: string): void {
   290→    if (!this.node) return;
   291→    this.stats.playing = true;
   292→    this.node.port.postMessage({ type: 'play' });
   293→  }
   294→
   295→  /** Stop playback. Sends 'stop' to the worklet — all voices deactivated. */
   296→  stop(): void {
   297→    if (!this.node) return;
   298→    this.stats.playing = false;
   299→    this.node.port.postMessage({ type: 'stop' });
   300→  }
   301→
   302→  /**
   303→   * Send a batch of events to the worklet. PSY5 pattern: ALL step events go
   304→   * in ONE message (not one message per event) to minimize postMessage
   305→   * overhead.
   306→   *
   307→   * Event layout (Float64Array, 6 floats per event):
   308→   *   [time, voice, note, velocity, duration, param,  // event 0
   309→   *    time, voice, note, velocity, duration, param,  // event 1
   310→   *    ...]
   311→   *
   312→   * `time` is an absolute AudioContext.currentTime-relative seconds value
   313→   * (must be in the future, within the worklet's 256-event ring buffer
   314→   * capacity — typically < 100ms ahead).
   315→   *
   316→   * The Float64Array buffer is transferred (zero-copy) — the caller MUST NOT
   317→   * reuse it after calling this.
   318→   */
   319→  sendEventBatch(events: Float64Array): void {
   320→    if (!this.node || events.length === 0) return;
   321→    // Transfer the underlying ArrayBuffer (zero-copy) — PSY5 optimization.
   322→    // The worklet receives ownership; we detach our reference.
   323→    const buffer = events.buffer;
   324→    // Only transfer if the buffer is not already detached (defensive —
   325→    // a detached buffer has byteLength 0).
   326→    const transferList = buffer.byteLength > 0 ? [buffer] : [];
   327→    this.node.port.postMessage({ type: 'events', events }, transferList);
   328→  }
   329→
   330→  /**
   331→   * Trigger a single immediate event (for UI actions like "Drop now").
   332→   * Uses 'trigger' message — the worklet enqueues it directly.
   333→   */
   334→  triggerImmediate(
   335→    voice: VoiceId,
   336→    note: number = 0,
   337→    velocity: number = 1,
   338→    duration: number = 0.2,
   339→    param: number = 0,
   340→  ): void {
   341→    if (!this.node || !this.ctx) return;
   342→    const time = this.ctx.currentTime + 0.02; // 20ms ahead for immediate response
   343→    this.node.port.postMessage({
   344→      type: 'trigger', time, voice, note, velocity, duration, param,
   345→    });
   346→  }

   406→  // ─── Task F1-F3: AudioBackend note triggering ─────────────────────────
   407→  //
   408→  // The engine calls these during scheduleStep(). Events accumulate in
   409→  // eventBatch (PSY5 batched postMessage — minimizes main→audio thread
   410→  // communication). flushEvents() sends the batch at the end of tick().
   411→  //
   412→  // Track → voice ID mapping (mirror of trackToVoiceId):
   413→  //   0=KICK 1=CLAP 2=HAT 3=PERC 4=BASS 5=LEAD/FM 6=PAD 7=ACID/FM
   414→
   415→  /** AudioBackend: trigger a drum voice. Enqueues into eventBatch. */
   416→  triggerDrum(track: number, time: number, vel: number, decay?: number): void {
   417→    if (!this.node) return;
   418→    const voiceId = trackToVoiceId(track);
   419→    const v = Number.isFinite(vel) ? Math.max(0, Math.min(1, vel)) : 0;
   420→    // The worklet's kick voice takes (time, amp, fund, decay, sr) — we encode
   421→    // decay as the `duration` field (in seconds). Other drums use duration as
   422→    // a generic decay envelope length.
   423→    const dur = (decay !== undefined && Number.isFinite(decay) && decay > 0)
   424→      ? decay
   425→      : (track === 0 ? 0.2 : 0.15);
   426→    this.eventBatch.add(time, voiceId, 0, v, dur, 0);
   427→    // Kick triggers sidechain duck — tell the worklet.
   428→    if (track === 0) this.triggerDuck();
   429→  }
   430→
   431→  /** AudioBackend: trigger a synth voice. Enqueues into eventBatch. */
   432→  triggerSynth(
   433→    track: number,
   434→    time: number,
   435→    midi: number,
   436→    vel: number,
   437→    dur: number,
   438→    _timbre?: SynthTimbre,
   439→    opts?: TriggerSynthOpts,
   440→  ): void {
   441→    if (!this.node) return;
   442→    // The worklet's event format carries (time, voice, note, vel, dur, param).
   443→    // Per-note timbre (cutoff/res/drive) isn't in the event — the worklet's
   444→    // voices use their internal worldParams for cutoff. The engine pushes
   445→    // timbre changes via setWorld ({ leadCutoff, bassCutoff, ... }) which
   446→    // the worklet applies to its voice pool.
   447→    const fm = opts?.fm === true;
   448→    const voiceId: VoiceId = trackToVoiceId(track, { fmLead: fm, fmArp: fm });
   449→    const v = Number.isFinite(vel) ? Math.max(0, Math.min(1, vel)) : 0;
   450→    const note = Number.isFinite(midi) ? midi : 60;
   451→    const duration = (Number.isFinite(dur) && dur > 0) ? dur : 0.1;
   452→    // `param` encodes the FM ratio (×10) for V_FM voice; 0 otherwise.
   453→    const param = voiceId === VOICE.FM ? 20 : 0; // ratio 2.0
   454→    this.eventBatch.add(time, voiceId, note, v, duration, param);
   455→  }
   456→

   458→  triggerRiser(time: number, dur: number): void {

   461→  }

   464→  triggerImpact(time: number): void {

   467→  }

   470→  triggerReverseImpact(time: number, intensity: number): void {
   471→    if (!this.node) return;
   472→    const i = Number.isFinite(intensity) ? Math.max(0, Math.min(1, intensity)) : 0.5;
   473→    this.eventBatch.add(time, VOICE.IMPACT, 0, 0.5 * i, 0.6, 0);
   474→  }
   475→
   476→  /**
   477→   * AudioBackend: flush the accumulated event batch to the worklet.
   478→   * PSY5 batched postMessage — ALL step events go in ONE message (not one
   479→   * per event) to minimize main→audio thread communication overhead.
   480→   *
   481→   * Called by the engine at the end of tick(). The Float64Array buffer is
   482→   * TRANSFERRED (zero-copy) — the worklet takes ownership.
   483→   */
   484→  flushEvents(): void {
   485→    if (!this.node || this.eventBatch.empty) return;
   486→    const events = this.eventBatch.build();
   487→    this.eventBatch.reset();
   488→    if (events.length === 0) return;
   489→    // Transfer the underlying ArrayBuffer (zero-copy) — PSY5 optimization.
   490→    const buffer = events.buffer;
   491→    const transferList = buffer.byteLength > 0 ? [buffer] : [];
   492→    this.node.port.postMessage({ type: 'events', events }, transferList);
   493→  }

   495→  // ─── Task F1-F3: legacy-only effect methods (worklet no-ops) ──────────
   496→  // The worklet has its own internal DSP (Moog ladder, Schroeder reverb,
   497→  // bus processors). The rich effects system (E1/T1/A1 — per-track racks,
   498→  // multiband, send effects) is a LEGACY-ONLY construct. These methods are
   499→  // no-ops in worklet mode; the engine calls them unconditionally and the
   500→  // worklet silently ignores them. The worklet's setWorld / setFX / setMacros
   501→  // cover the equivalent functionality.

   557→  /**
   558→   * Task F1-F3: AudioBackend.getParams() — return the effective params
   559→   * currently pushed to the worklet. Used by the learning memory to
   560→   * snapshot "what the engine is actually doing right now".
   561→   *
   562→   * The worklet is on the audio thread — we can't query it synchronously.
   563→   * Instead we return the cached values from setWorld / setFX calls.
   564→   * This is an approximation (the worklet may have applied smoothing), but
   565→   * it's accurate enough for the learning memory's purpose (storing
   566→   * "good" param configurations for later recall).
   567→   */
   568→  getParams(): AudioBackendParams {
   569→    // Average per-bus reverb/delay sends into a single "sendReverb/sendDelay"
   570→    // value (the learning memory stores averaged melodic sends).
   571→    const reverbSends = this.effectiveFX.reverbSends;
   572→    const delaySends = this.effectiveFX.delaySends;
   573→    let sendReverb = 0;
   574→    let sendDelay = 0;
   575→    if (reverbSends && reverbSends.length >= 5) {
   576→      // music bus (index 2) is the main melodic treatment.
   577→      sendReverb = reverbSends[2] ?? 0;
   578→    }
   579→    if (delaySends && delaySends.length >= 5) {
   580→      sendDelay = delaySends[2] ?? 0;
   581→    }
   582→    return {
   583→      kickFundamental: this.effectiveWorld.kickFundamental,
   584→      kickDecay: this.effectiveWorld.kickDecay,
   585→      bassCutoff: this.effectiveWorld.bassCutoff,
   586→      bassResonance: this.effectiveWorld.bassResonance,
   587→      leadCutoff: this.effectiveWorld.leadCutoff,
   588→      leadDetune: this.effectiveWorld.leadDetune,
   589→      padCutoff: this.effectiveWorld.padCutoff,
   590→      padAttack: this.effectiveWorld.padAttack,
   591→      padDetune: this.effectiveWorld.padDetune,
   592→      padEvolveRate: this.effectiveWorld.padEvolveRate,
   593→      duck: this.effectiveWorld.duck,
   594→      sendReverb,
   595→      sendDelay,
   596→      reverbWet: this.effectiveFX.reverbWet,
   597→      delayWet: this.effectiveFX.delayWet,
   598→      delayFeedback: this.effectiveFX.delayFeedback,
   599→    };
   600→  }

   618→  /**
   619→   * Dispose the engine — disconnect the worklet, close the AudioContext.
   620→   * After dispose(), the engine cannot be reused (create a new instance).
   621→   */
   622→  async dispose(): Promise<void> {
   623→    if (this.node) {
   624→      try { this.node.disconnect(); } catch { /* already disconnected */ }
   625→      try { this.node.port.close(); } catch { /* port already closed */ }
   626→      this.node = null;
   627→    }
   628→    if (this.analyser) {
   629→      try { this.analyser.disconnect(); } catch { /* already disconnected */ }
   630→      this.analyser = null;
   631→    }
   632→    if (this.ctx) {
   633→      try { await this.ctx.close(); } catch { /* already closed */ }
   634→      this.ctx = null;
   635→    }
   636→    this.moduleLoaded = false;
   637→    this.loadPromise = null;
   638→    this.statsListeners.length = 0;
   639→    this.stats = {
   640→      playing: false, step: 0, activeVoices: 0, eventCount: 0,
   641→      currentFrame: 0, cpuLoad: 0, voiceBudget: 32, processMs: 0,
   642→    };
   643→  }

   646→// ─── Event batch builder ───────────────────────────────────────────────────
   647→/**
   648→ * Helper for the main thread to build a Float64Array batch incrementally.
   649→ * Psy4EngineV2's tick() loop creates one of these per tick, appends events
   650→ * as the director yields notes, then calls sendEventBatch() to flush.
   651→ *
   652→ * Preallocates a fixed-capacity backing array (no per-event allocation).
   653→ * If the batch overflows, the oldest events are dropped (defensive — should
   654→ * never happen with 256 slots at 145 BPM).
   655→ */
   656→export class EventBatchBuilder {
   657→  static readonly EVENT_SIZE = 6;
   658→  static readonly MAX_EVENTS = 256;
   659→
   660→  private buf: Float64Array;
   661→  private count = 0;
   662→
   663→  constructor() {
   664→    this.buf = new Float64Array(EventBatchBuilder.MAX_EVENTS * EventBatchBuilder.EVENT_SIZE);
   665→  }
   666→
   667→  /** Number of events currently in the batch. */
   668→  get size(): number { return this.count; }
   669→
   670→  /** True if the batch is empty. */
   671→  get empty(): boolean { return this.count === 0; }
   672→
   673→  /**
   674→   * Append an event. If the batch is full, the event is dropped (returns false).
   675→   * PSY7 safety: clamps velocity to [0,1], duration to [0.001, 10].
   676→   */
   677→  add(time: number, voice: VoiceId, note: number, velocity: number, duration: number, param: number = 0): boolean {
   678→    if (this.count >= EventBatchBuilder.MAX_EVENTS) return false;
   679→    // PSY7: clamp + finite-check to prevent NaN crashes in the worklet
   680→    if (!Number.isFinite(time) || !Number.isFinite(voice) || !Number.isFinite(note) ||
   681→        !Number.isFinite(velocity) || !Number.isFinite(duration)) {
   682→      return false;
   683→    }
   684→    const base = this.count * EventBatchBuilder.EVENT_SIZE;
   685→    this.buf[base] = time;
   686→    this.buf[base + 1] = voice;
   687→    this.buf[base + 2] = note;
   688→    this.buf[base + 3] = Math.max(0, Math.min(1, velocity));
   689→    this.buf[base + 4] = Math.max(0.001, Math.min(10, duration));
   690→    this.buf[base + 5] = Number.isFinite(param) ? param : 0;
   691→    this.count++;
   692→    return true;
   693→  }
   694→
   695→  /**
   696→   * Build the Float64Array to send to the worklet. Returns a VIEW of the
   697→   * internal buffer (so the caller must sendEventBatch() immediately and
   698→   * not hold the reference). After sendEventBatch(), the buffer is
   699→   * transferred to the worklet — the caller MUST call reset() to reuse.
   700→   */
   701→  build(): Float64Array {
   702→    if (this.count === 0) return new Float64Array(0);
   703→    // Return a subarray view (no copy). sendEventBatch() will transfer the
   704→    // underlying buffer's slice — but Transferable only works on the FULL
   705→    // ArrayBuffer, so we copy the filled portion into a fresh buffer.
   706→    // This is a per-tick allocation, but tick() runs at ~40 Hz, not per-sample.
   707→    const out = new Float64Array(this.count * EventBatchBuilder.EVENT_SIZE);
   708→    out.set(this.buf.subarray(0, this.count * EventBatchBuilder.EVENT_SIZE));
   709→    return out;
   710→  }
   711→
   712→  /** Reset the batch for reuse (after build() + sendEventBatch()). */
   713→  reset(): void {
   714→    this.count = 0;
   715→  }
   716→}
