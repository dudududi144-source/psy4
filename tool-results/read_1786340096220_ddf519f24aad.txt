   180→/**
   181→ * Map a reference spectral centroid (Hz) to a target synth cutoff (Hz)
   182→ * using log-linear interpolation. Anchor points:
   183→ *   500Hz  ->  800 Hz cutoff (dark / warm)
   184→ *   2000Hz -> 3000 Hz cutoff (balanced)
   185→ *   5000Hz -> 6000 Hz cutoff (bright / cutting)
   186→ * Falls back to clamp(centroid, 200, 12000) if out of typical range.
   187→ */
   188→function centroidToCutoff(centroid: number): number {
   189→  if (!isFinite(centroid) || centroid <= 0) return 1500;
   190→  const lc = Math.log(clamp(centroid, 80, 16000));
   191→  // Fit using endpoints (500,800) and (5000,6000):
   192→  //   ln(cut) = 0.8753 * ln(centroid) + 1.245
   193→  const lcut = 0.8753 * lc + 1.245;
   194→  return clamp(Math.exp(lcut), 200, 12000);
   195→}

   302→// ─── Synth voice pool ───────────────────────────────────────────────────────
   303→//
   304→// The synth pool uses AdvancedSynthVoice (from ./advancedVoice) which supports
   305→// 4 modes: classic (drop-in 2-osc), fm (carrier+modulator), supersaw (5-7 detuned
   306→// saws with stereo spread), and wavetable (2 crossfading periodic waves with LFO
   307→// morph). All nodes are preallocated per voice — zero per-note allocation.
   308→//
   309→// Each voice holds up to 7 OscillatorNodes. With 20 voices that's 140 oscillators
   310→// max, well within modern browser limits.
   311→
   312→// ─── Pooled Drum Voice (from PSY6 — multi-type) ────────────────────────────
   313→
   314→class PooledDrumVoice {
   315→  noise: AudioBufferSourceNode;
   316→  noiseGain: GainNode;
   317→  nFilter: BiquadFilterNode;
   318→  osc: OscillatorNode;
   319→  oscGain: GainNode;
   320→  out: GainNode;
   321→  bus: GainNode | null = null;
   322→
   323→  constructor(ctx: AudioContext, noiseBuffer: AudioBuffer) {
   324→    this.noise = ctx.createBufferSource();
   325→    this.noise.buffer = noiseBuffer;
   326→    this.noise.loop = true;
   327→    this.noiseGain = ctx.createGain();
   328→    this.noiseGain.gain.value = 0;
   329→    this.nFilter = ctx.createBiquadFilter();
   330→    this.nFilter.type = 'bandpass';
   331→    this.noise.connect(this.nFilter);
   332→    this.nFilter.connect(this.noiseGain);
   333→    this.osc = ctx.createOscillator();
   334→    this.osc.type = 'sine';
   335→    this.oscGain = ctx.createGain();
   336→    this.oscGain.gain.value = 0;
   337→    this.osc.connect(this.oscGain);
   338→    this.out = ctx.createGain();
   339→    this.noiseGain.connect(this.out);
   340→    this.oscGain.connect(this.out);
   341→    this.noise.start();
   342→    this.osc.start();
   343→  }
   344→
   345→  connect(bus: GainNode) {
   346→    if (this.bus !== bus) {
   347→      this.out.disconnect();
   348→      this.out.connect(bus);
   349→      this.bus = bus;
   350→    }
   351→  }
   352→
   353→  /**
   354→   * Trigger this voice. The optional `decayOverride` lets callers (e.g. reference
   355→   * pursuit) replace the preset decay with a blended value to chase the radio's
   356→   * kick decay. It is guarded against NaN/zero/out-of-range values.
   357→   */
   358→  hit(p: DrumPreset, when: number, vel: number, bus: GainNode, decayOverride?: number) {
   359→    this.connect(bus);
   360→    const tune = p.tune || 1;
   361→    const candidateDecay = (typeof decayOverride === 'number'
   362→      && isFinite(decayOverride) && decayOverride > 0.001 && decayOverride < 50)
   363→      ? decayOverride
   364→      : null;
   365→    const decay = candidateDecay ?? p.decay ?? 1;
   366→    const tone = p.tone || 1;
   367→    const punch = p.punch || 0;
   368→    const type = p.type;
   369→    const ng = this.noiseGain.gain;
   370→    const og = this.oscGain.gain;
   371→    ng.cancelScheduledValues(when);
   372→    og.cancelScheduledValues(when);
   373→    ng.setValueAtTime(0, when);
   374→    og.setValueAtTime(0, when);
   375→
   376→    if (type === 'kick') {
   377→      const dur = 0.12 + 0.5 * decay;
   378→      this.osc.frequency.setValueAtTime(180 * tune, when);
   379→      this.osc.frequency.exponentialRampToValueAtTime(Math.max(36 * tune, 24), when + 0.09);
   380→      og.setValueAtTime(vel * 1.1, when);
   381→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   382→      if (punch > 0) {
   383→        ng.setValueAtTime(vel * punch * 0.8, when);
   384→        ng.exponentialRampToValueAtTime(0.0001, when + 0.02);
   385→        this.nFilter.frequency.setValueAtTime(2500, when);
   386→      }
   387→    } else if (type === 'clap') {
   388→      const dur = 0.25 + 0.15 * decay;
   389→      this.nFilter.type = 'bandpass';
   390→      this.nFilter.frequency.setValueAtTime(1150 * tone, when);
   391→      this.nFilter.Q.value = 1.3;
   392→      ng.setValueAtTime(0, when);
   393→      [0, 0.014, 0.03].forEach(t2 => {
   394→        ng.setValueAtTime(0, when + t2);
   395→        ng.linearRampToValueAtTime(vel * 0.9, when + t2 + 0.002);
   396→        ng.exponentialRampToValueAtTime(0.02, when + t2 + 0.012);
   397→      });
   398→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   399→    } else if (type === 'hatC' || type === 'hatO') {
   400→      const open = type === 'hatO';
   401→      const dur = open ? 0.26 + 0.5 * decay : 0.03 + 0.05 * decay;
   402→      this.nFilter.type = 'highpass';
   403→      this.nFilter.frequency.setValueAtTime(7200 * Math.sqrt(tone), when);
   404→      ng.setValueAtTime(vel * (open ? 0.4 : 0.5), when);
   405→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   406→    } else if (type === 'tom') {
   407→      const dur = 0.22 + 0.35 * decay;
   408→      this.osc.type = 'sine';
   409→      this.osc.frequency.setValueAtTime(180 * tune, when);
   410→      this.osc.frequency.exponentialRampToValueAtTime(92 * tune, when + dur * 0.7);
   411→      og.setValueAtTime(vel * 0.9, when);
   412→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   413→    } else if (type === 'shaker') {
   414→      const dur = 0.04 + 0.07 * decay;
   415→      this.nFilter.type = 'highpass';
   416→      this.nFilter.frequency.setValueAtTime(6000 * tone, when);
   417→      ng.setValueAtTime(vel * 0.45, when);
   418→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   419→    } else if (type === 'glitch') {
   420→      const dur = 0.08 + 0.14 * decay;
   421→      this.nFilter.type = 'bandpass';
   422→      this.nFilter.frequency.setValueAtTime(1500 * tone + 800, when);
   423→      this.nFilter.Q.value = 4;
   424→      ng.setValueAtTime(vel * 0.7, when);
   425→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   426→    }
   427→  }
   428→
   429→  panic(ctx: AudioContext) {
   430→    try {
   431→      this.noiseGain.gain.cancelScheduledValues(0);
   432→      this.noiseGain.gain.setValueAtTime(0, ctx.currentTime);
   433→      this.oscGain.gain.cancelScheduledValues(0);
   434→      this.oscGain.gain.setValueAtTime(0, ctx.currentTime);
   435→    } catch {}
   436→  }
   437→}

   842→
   843→    // Track buses (8 tracks) — each gets a full TrackEffectsRack insert chain.
   844→    this.duckGain = c.createGain();
   845→    this.duckGain.gain.value = 1.0;
   846→    // duckGain connects to master ONCE (not twice — was causing feedback)
   847→    this.duckGain.connect(this.master);
   848→
   849→    // Build per-track rack configs from the current world. The configs encode
   850→    // per-track EQ/comp/sat/pan/Haas + per-track send levels + per-world
   851→    // modulations (dark-psy → more distortion, goa → more phaser, etc.).
   852→    const rackConfigs = buildTrackRackConfigs(this.currentWorld);
   853→    for (let i = 0; i < 8; i++) {
   854→      const rack = new TrackEffectsRack(c, rackConfigs[i]);
   855→      this.racks.push(rack);
   856→      // Voices connect to rack.input; liveTrack/setWorld adjust rack.output.
   857→      this.chains.push(rack.input);
   858→      this.trackGains.push(rack.output);
   859→
   860→      // Wire rack.output → (duckGain for bass | master for others).
   861→      // Bass goes through duckGain for sidechain ducking on kick hits.
   862→      if (i === 4) {
   863→        rack.output.connect(this.duckGain);
   864→      } else {
   865→        rack.output.connect(this.master);
   866→      }
   867→
   868→      // Wire all 6 send taps to the global send bus inputs.
   869→      // reverb/delay are the existing global buses (their gain is the master
   870→      // FX mix controlled by world.fxMix); chorus/phaser/distortion/bitcrush
   871→      // are the new Task E1 buses (their gain stays at 1.0 — per-track send
   872→      // gains do the scaling).
   873→      rack.connectSend('reverb', this.reverbSend);
   874→      rack.connectSend('delay', this.delaySend);
   875→      rack.connectSend('chorus', this.chorusSend);
   876→      rack.connectSend('phaser', this.phaserSend);
   877→      rack.connectSend('distortion', this.distortionSend);
   878→      rack.connectSend('bitcrush', this.bitcrushSend);
   879→    }
   880→    // duckGain already connected to master above
   881→
   882→    // Allocate voice pools (20 synth + 24 drum — from PSY6)
   883→    // Each AdvancedSynthVoice preallocates 7 OscillatorNodes + panners + LFOs.
   884→    // Total: 20 voices × 7 osc = 140 max oscillators (modern browsers handle this).
   885→    for (let i = 0; i < 20; i++) this.synthPool.push(new AdvancedSynthVoice(c, i));
   886→    for (let i = 0; i < 24; i++) this.drumPool.push(new PooledDrumVoice(c, this.noiseBuffer));
   887→
   888→    // Initialize tracks
   889→    this.initTracks();
   890→  }
   891→
   892→  private initTracks(): void {
   893→    const names = ['KICK', 'SNARE', 'HATS', 'PERC', 'BASS', 'LEAD', 'PAD', 'ARP'];
   894→    const presets = ['PS-KICK-TIGHT', 'TR-CLAP', 'PS-HAT', 'PS-PERC', 'PS-BASS-ROLL', 'PS-LEAD-SQUELCH', 'PS-PAD-PSYCH', 'PS-ARP-ACID'];
   895→    // Boosted track volumes for commercial loudness
   896→    const vols = [1.0, 0.6, 0.5, 0.4, 1.2, 0.7, 0.5, 0.5];
   897→    this.tracks = [];
   898→    for (let i = 0; i < 8; i++) {
   899→      this.tracks.push({
   900→        idx: i,
   901→        kind: i < 4 ? 'drum' : 'synth',
   902→        name: names[i],
   903→        presetId: presets[i],
   904→        mix: { vol: vols[i], pan: 0, mute: false, sendA: 0, sendB: 0 },
   905→      });
   906→    }
   907→    // Initialize musical generators with the default key so they exist before
   908→    // the first key-change event arrives from the reference listener.
   909→    this.refreshMusicalGenerators();
   910→  }

   912→  start(worldId?: string): void {
   913→    this.init();
   914→    if (this.ctx!.state === 'suspended') this.ctx!.resume();
   915→
   916→    // ── World-driven configuration (Track A) ──
   917→    this.currentWorld = WORLDS[worldId as WorldId] || WORLDS['dark-psy'];
   918→    this._bpm = this.currentWorld.defaultBpm;
   919→    this.musicalKey = {
   920→      root: Math.floor((this.currentWorld.rootRange[0] + this.currentWorld.rootRange[1]) / 2),
   921→      scale: this.currentWorld.defaultScale,
   922→    };
   923→    // Re-create musical generators with the world's key (MelodyEngine, AcidPattern)
   924→    this.refreshMusicalGenerators();
   925→    this.arpIdx = 0;
   926→    this.bassPatternIdx = 0;
   927→    // Reset phrase-locked rotation counters for a clean start
   928→    this.phraseCounter = 0;
   929→    this.phrasePresetVariant = 0;
   930→    // Reset auto-switch anti-thrash guard
   931→    this.lastAutoSwitchTime = 0;
   932→    this.lastAutoSwitchWorldId = null;
   933→    this.styleMatches = [];
   934→
   935→    // Apply the world's preferred kick/bass/lead presets immediately so the
   936→    // engine starts with the right timbres (Task 15).
   937→    this.applyWorldPresets();
   938→
   939→    // Apply per-world send-effect settings (Task E1) — pushes per-track send
   940→    // levels and global effect parameters for the current world.
   941→    this.applyWorldEffectSettings(this.currentWorld);
   942→
   943→    // Apply world FX mix to reverb and delay sends
   944→    const fxMix = this.currentWorld.fxMix;
   945→    this.reverbSend.gain.setTargetAtTime(0.04 + fxMix * 0.22, this.ctx!.currentTime, 0.05);
   946→    this.delaySend.gain.setTargetAtTime(0.05 + fxMix * 0.30, this.ctx!.currentTime, 0.05);
   947→
   948→    if (this.playing) return;
   949→    this.playing = true;
   950→    this.step = 0;
   951→    this.bar = 0;
   952→    this.sectionIdx = 0;
   953→    this.currentSection = this.arrangement[0].label;
   954→    // Task V2b: reset section-automation state so the first section's static
   955→    // levels get pushed on the first tick. Also clear any leftover lead
   956→    // cutoff override from a previous session so the lead starts clean.
   957→    this.lastAutomationSection = '';
   958→    this.leadCutoffOverride = -1;
   959→    this.onSectionChange?.(this.currentSection);
   960→    this.nextTime = this.ctx!.currentTime + 0.03;
   961→    this.scheduleNextTick();
   962→  }
   963→
   964→  stop(): void {
   965→    this.playing = false;
   966→    // Task V2a: stop the Worker-based scheduler. The SchedulerWorker keeps
   967→    // the underlying Worker instance alive across stop/start cycles (cheap
   968→    // restart) — it just stops posting ticks. We also clear the legacy
   969→    // `timer` field for the (currently unused) setTimeout fallback path.
   970→    this.scheduler.stop();
   971→    if (this.timer) { clearTimeout(this.timer); this.timer = null; }
   972→    if (this.ctx) {
   973→      for (const v of this.synthPool) v.panic(this.ctx);
   974→      for (const v of this.drumPool) v.panic(this.ctx);
   975→    }
   976→  }

   978→  private get bpm(): number {
   979→    return this._bpm || 145;
   980→  }

   982→
   983→  setBpm(bpm: number): void {
   984→    this._bpm = bpm;

   986→
   987→  applyMusicalUnderstanding(understanding: {
   988→    key: { root: number; scale: string; confidence: number };
   989→    bpm: number;
   990→    bpmConfidence: number;
   991→    style?: string;
   992→    styleConfidence?: number;
   993→  }): void {
   994→    // ── KEY PURSUIT — refresh lead/acid generators when key changes ──
   995→    // The radio's key detector returns a CHROMATIC root (0-11, C=0..B=11).
   996→    // Our internal musicalKey.root is a MIDI note (typically 36-60). So we
   997→    // lift the chromatic root into the bass octave (36 + root) when it's in
   998→    // the chromatic range. If the listener ever returns a MIDI note directly
   999→    // (> 11), we trust it as-is. NaN/undefined confidence → no-op.
  1000→    const k = understanding.key;
  1001→    const conf = typeof k?.confidence === 'number' && isFinite(k.confidence) ? k.confidence : 0;
  1002→    const rawRoot = typeof k?.root === 'number' && isFinite(k.root) ? k.root : -1;
  1003→    if (conf > 0.2 && rawRoot >= 0) {
  1004→      // Snap the root into the world's preferred rootRange so the engine
  1005→      // stays in a useful octave for the current world. We pick the octave
  1006→      // of 36 + (root mod 12) that falls inside rootRange.
  1007→      const chroma = ((rawRoot % 12) + 12) % 12;
  1008→      const worldRange = this.currentWorld.rootRange;
  1009→      let newRoot: number;
  1010→      if (rawRoot >= 12 && rawRoot >= worldRange[0] && rawRoot <= worldRange[1]) {
  1011→        // Listener returned a MIDI note already in range — trust it.
  1012→        newRoot = rawRoot;
  1013→      } else {
  1014→        // Lift chroma into an octave inside the world's rootRange.
  1015→        const lo = worldRange[0];
  1016→        const candidate = lo + (((chroma - (lo % 12)) + 12) % 12);
  1017→        newRoot = candidate;
  1018→      }
  1019→      const newScale = (typeof k.scale === 'string' && k.scale.length > 0) ? k.scale : this.musicalKey.scale;
  1020→      const changed = newRoot !== this.musicalKey.root || newScale !== this.musicalKey.scale;
  1021→      // ALWAYS update musicalKey first, then refresh generators — order matters
  1022→      // so the generators pick up the new key.
  1023→      this.musicalKey = { root: newRoot, scale: newScale };
  1024→      // Also store the scale on the ref feature snapshot so the classifier
  1025→      // can use it next time it runs.
  1026→      this.refKeyScale = newScale;
  1027→      if (changed) {
  1028→        this.refreshMusicalGenerators();
  1029→        if (typeof console !== 'undefined') {
  1030→          console.log('[PSY4] Key updated:', this.musicalKey);
  1031→        }
  1032→      }
  1033→    }
  1034→
  1035→    // ── CONTINUOUS BPM TRACKING (smooth ramp for large diffs) ──
  1036→    // Always honor the new BPM when confidence is high. Small diffs (<=2 BPM)
  1037→    // are applied immediately; larger diffs ramp across 4 bars to avoid tempo
  1038→    // glitches and keep the scheduler's lookahead stable.
  1039→    if (understanding.bpm > 0 && understanding.bpmConfidence > 0.5) {
  1040→      const target = Math.round(understanding.bpm);
  1041→      this.refBpm = clamp(target, 30, 220);
  1042→      const diff = target - this._bpm;
  1043→      if (Math.abs(diff) > 2) {
  1044→        this.targetBpm = clamp(target, 60, 200);
  1045→        this.bpmRampBarsLeft = 4;
  1046→        this.bpmRampPerBar = diff / 4;
  1047→      } else {
  1048→        this._bpm = clamp(target, 60, 200);
  1049→        this.targetBpm = 0;
  1050→        this.bpmRampBarsLeft = 0;
  1051→        this.bpmRampPerBar = 0;
  1052→      }
  1053→    }
  1054→
  1055→    // ── STYLE CLASSIFICATION (Task 14) ──
  1056→    // Two paths to determine the style:
  1057→    //   (a) Reference listener provided an explicit style tag with high
  1058→    //       confidence → use it directly (legacy path, kept for compat).
  1059→    //   (b) No style tag (or low confidence) → LEARN the style from the
  1060→    //       acoustic features we've accumulated in liveTrack(). This is the
  1061→    //       new spectral classifier path that addresses the user's complaint
  1062→    //       that "style must be learned, not defined by fat bass".
  1063→    // In both cases we run the classifier on the stored features so the UI
  1064→    // can render the full ranking. Auto-switch only happens when the top
  1065→    // match's confidence exceeds the threshold AND the anti-thrash cooldown
  1066→    // has elapsed.
  1067→    const features = this.buildRefFeatures();
  1068→    if (features) {
  1069→      const matches = classifyStyle(features);
  1070→      this.styleMatches = matches;
  1071→
  1072→      const explicitStyle = understanding.style;
  1073→      const explicitConf = understanding.styleConfidence ?? 0;
  1074→      const topMatch = matches[0];
  1075→
  1076→      if (explicitStyle && explicitConf > 0.4) {
  1077→        // Path (a): use the explicit tag — but still record matches for UI.
  1078→        // Only auto-switch if explicit tag strongly disagrees with current world.
  1079→        const explicitWorldId = styleToWorld(explicitStyle);
  1080→        if (explicitWorldId !== this.currentWorld.id && explicitConf > 0.6) {
  1081→          this.tryAutoSwitch(explicitWorldId, `explicit style tag '${explicitStyle}' (${(explicitConf * 100).toFixed(0)}%)`);
  1082→        }
  1083→      } else if (topMatch &&
  1084→                 topMatch.confidence >= Psy4EngineV2.AUTO_SWITCH_CONFIDENCE_THRESHOLD) {
  1085→        // Path (b): learn from features and auto-switch if confident.
  1086→        const targetWorldId = styleToWorld(topMatch.style);
  1087→        if (targetWorldId !== this.currentWorld.id) {
  1088→          const reason = topMatch.reasons[0] || `style '${topMatch.style}' (${(topMatch.confidence * 100).toFixed(0)}%)`;
  1089→          this.tryAutoSwitch(targetWorldId, reason);
  1090→        }
  1091→      }
  1092→    }
  1093→  }
  1094→
  1095→  /**
  1096→   * Apply a style classification result directly (Task 14).
  1097→   * If the top match's confidence exceeds the auto-switch threshold AND it
  1098→   * differs from the current world, switch worlds smoothly.
  1099→   * Public so the UI/tests can drive it explicitly if desired.
  1100→   */
  1101→  applyStyleClassification(matches: StyleMatch[]): void {
  1102→    this.styleMatches = matches;
  1103→    const top = matches[0];
  1104→    if (!top) return;
  1105→    if (top.confidence >= Psy4EngineV2.AUTO_SWITCH_CONFIDENCE_THRESHOLD) {
  1106→      const targetWorldId = styleToWorld(top.style);
  1107→      if (targetWorldId !== this.currentWorld.id) {
  1108→        const reason = top.reasons[0] || `style '${top.style}' (${(top.confidence * 100).toFixed(0)}%)`;
  1109→        this.tryAutoSwitch(targetWorldId, reason);
  1110→      }
  1111→    }
  1112→  }
  1113→
  1114→  /**
  1115→   * Snapshot of the latest style classification for UI display.
  1116→   * Empty array if no reference features have arrived yet.
  1117→   */
  1118→  getStyleClassification(): StyleMatch[] {
  1119→    return this.styleMatches;
  1120→  }

  1122→  /**
  1123→   * Return the id of the currently active world. Used by the UI to keep its
  1124→   * dropdown in sync with the engine after an auto-switch (Task 14).
  1125→   */
  1126→  getCurrentWorldId(): string {
  1127→    return this.currentWorld.id;
  1128→  }
  1129→
  1130→  /**
  1131→   * Attempt an automatic world switch. Respects the 30-second anti-thrash
  1132→   * cooldown and skips if the target world is the same as the last switch.
  1133→   * This is the only place auto-switches should happen.
  1134→   */
  1135→  private tryAutoSwitch(worldId: string, reason?: string): void {
  1136→    if (!this.ctx) return;
  1137→    // Defensive: validate that worldId is one of the known WorldIds before
  1138→    // attempting to switch. The classifier's styleToWorld() should always
  1139→    // return a valid id, but we don't want a bad id to slip through and
  1140→    // silently no-op in switchWorld().
  1141→    if (!(worldId in WORLDS)) {
  1142→      if (typeof console !== 'undefined') {
  1143→        console.warn('[PSY4] tryAutoSwitch: unknown worldId', worldId);
  1144→      }
  1145→      return;
  1146→    }
  1147→    const now = this.ctx.currentTime * 1000; // ms since ctx start
  1148→    if (this.lastAutoSwitchTime > 0 &&
  1149→        (now - this.lastAutoSwitchTime) < Psy4EngineV2.AUTO_SWITCH_COOLDOWN_MS) {
  1150→      return; // too soon — thrash guard
  1151→    }
  1152→    if (this.lastAutoSwitchWorldId === worldId) {
  1153→      return; // we already switched to this world recently — no-op
  1154→    }
  1155→    this.switchWorld(worldId as WorldId);
  1156→    this.lastAutoSwitchTime = now;
  1157→    this.lastAutoSwitchWorldId = worldId;
  1158→    // Notify the UI that an auto-switch happened.
  1159→    try { this.onWorldChange?.(worldId, reason); } catch {}
  1160→  }
  1161→
  1162→  /**
  1163→   * Switch to a different world smoothly (Task 14). Does NOT restart playback.
  1164→   *   - Updates currentWorld, musicalKey (root + scale), refreshes generators
  1165→   *   - Ramps BPM over 4 bars if the new world's BPM differs by more than 2
  1166→   *   - Applies the new world's FX mix to reverb/delay sends
  1167→   *   - Resets phrase-locked rotation counters (start the new world's first
  1168→   *     phrase cleanly)
  1169→   *   - All audio parameter changes use setTargetAtTime / 4-bar ramp — no jumps
  1170→   */
  1171→  switchWorld(worldId: WorldId): void {
  1172→    const newWorld = WORLDS[worldId];
  1173→    if (!newWorld) return;
  1174→    this.currentWorld = newWorld;
  1175→
  1176→    // Key — keep the root if it's within the world's range; otherwise snap to
  1177→    // the midpoint of the world's rootRange.
  1178→    const newRoot = (this.musicalKey.root >= newWorld.rootRange[0] &&
  1179→                     this.musicalKey.root <= newWorld.rootRange[1])
  1180→      ? this.musicalKey.root
  1181→      : Math.floor((newWorld.rootRange[0] + newWorld.rootRange[1]) / 2);
  1182→    // PRESERVE the listener-detected scale if the reference listener has set
  1183→    // one (refKeyScale) AND the new world allows that scale. This prevents
  1184→    // switchWorld from clobbering the key pursuit when the world auto-switches
  1185→    // (the original bug: radio detected "F major" but the engine reverted to
  1186→    // the world's defaultScale "phrygianDominant" immediately after switching).
  1187→    const listenerScale = this.refKeyScale;
  1188→    const scaleAllowed = (s?: string): s is string =>
  1189→      !!s && newWorld.scales.includes(s);
  1190→    const newScale = scaleAllowed(listenerScale)
  1191→      ? listenerScale
  1192→      : newWorld.defaultScale;
  1193→    const keyChanged = newRoot !== this.musicalKey.root || newScale !== this.musicalKey.scale;
  1194→    this.musicalKey = { root: newRoot, scale: newScale };
  1195→    if (keyChanged) this.refreshMusicalGenerators();
  1196→
  1197→    // BPM — ramp over 4 bars if diff > 2
  1198→    const newBpm = newWorld.defaultBpm;
  1199→    if (Math.abs(newBpm - this._bpm) > 2) {
  1200→      this.targetBpm = clamp(newBpm, 60, 200);
  1201→      this.bpmRampBarsLeft = 4;
  1202→      this.bpmRampPerBar = (newBpm - this._bpm) / 4;
  1203→    } else {
  1204→      this._bpm = clamp(newBpm, 60, 200);
  1205→      this.targetBpm = 0;
  1206→      this.bpmRampBarsLeft = 0;
  1207→      this.bpmRampPerBar = 0;
  1208→    }
  1209→
  1210→    // FX mix — smooth ramp
  1211→    if (this.ctx) {
  1212→      const now = this.ctx.currentTime;
  1213→      const fxMix = newWorld.fxMix;
  1214→      this.reverbSend.gain.setTargetAtTime(0.04 + fxMix * 0.22, now, 0.5);
  1215→      this.delaySend.gain.setTargetAtTime(0.05 + fxMix * 0.30, now, 0.5);
  1216→    }
  1217→
  1218→    // Reset phrase-locked counters — start the new world's first phrase cleanly
  1219→    this.phraseCounter = 0;
  1220→    this.phrasePresetVariant = 0;
  1221→    this.arpIdx = 0;
  1222→    this.bassPatternIdx = 0;
  1223→
  1224→    // Apply the new world's preferred kick/bass/lead presets immediately so
  1225→    // the next phrase starts with the right timbres. The phrase-locked
  1226→    // rotation will then alternate between the two variants from here on.
  1227→    this.applyWorldPresets();
  1228→
  1229→    // Apply per-world send-effect settings (Task E1) — pushes per-track send
  1230→    // levels and global effect parameters for the new world.
  1231→    this.applyWorldEffectSettings(newWorld);
  1232→  }
  1233→
  1234→  /**
  1235→   * Apply the current world's preferred kick/bass/lead/pad/arp presets.
  1236→   * Called by switchWorld() and at start(). Phrase-locked rotation in tick()
  1237→   * will alternate between the two variants every 8 bars from here on.
  1238→   *
  1239→   * Task S1: leads/pads/arp now use ADVANCED_PRESETS:
  1240→   *   - Goa/Acid leads → FM presets (PS-FM-GOA / PS-FM-SQUELCH) — metallic/squelchy
  1241→   *   - Dark worlds    → PS-FM-SQUELCH (acid character)
  1242→   *   - Bright worlds  → PS-FM-BELL (cleaner, bell-like)
  1243→   *   - Pads           → PS-SUPERSAW-PAD (thick, rich, 7-osc supersaw)
  1244→   *   - Arp            → PS-WT-MORPH (wavetable, evolving texture)
  1245→   *   - Bass           → classic (PS-BASS-ROLL/DEEP) — bass doesn't need FM
  1246→   *   - Kick           → classic drum presets (unchanged)
  1247→   *
  1248→   * synthModeOverrides (set by reference pursuit via setSynthMode) take
  1249→   * precedence over the world defaults — applied at triggerSynth() time.
  1250→   */
  1251→  private applyWorldPresets(): void {
  1252→    const id = this.currentWorld.id;
  1253→    // Dark worlds → DEEP kick + ROLL bass; bright worlds → TIGHT kick + DEEP bass.
  1254→    // Mid worlds → mix. Lead swaps for goa/acid (FM metallic) vs others (FM bell).
  1255→    const dark = id === 'dark-psy' || id === 'forest' || id === 'deep-psy' || id === 'hypnotic';
  1256→    const acid = id === 'goa' || id === 'acid-psy';
  1257→    const bright = id === 'morning-psy' || id === 'cosmic' || id === 'organic-psy';
  1258→
  1259→    this.tracks[0].presetId = dark ? 'PS-KICK-DEEP' : 'PS-KICK-TIGHT';
  1260→    // Bass stays classic — bass doesn't need FM/supersaw/wavetable
  1261→    this.tracks[4].presetId = dark ? 'PS-BASS-ROLL' : (bright ? 'PS-BASS-DEEP' : 'PS-BASS-ROLL');
  1262→    // Lead: FM for goa/acid (metallic goa leads), squelchy FM for dark worlds,
  1263→    // bell FM for bright worlds. The synthModeOverrides can still flip a track
  1264→    // to supersaw/wavetable/classic at runtime via setSynthMode().
  1265→    this.tracks[5].presetId = acid
  1266→      ? 'PS-FM-GOA'
  1267→      : (dark ? 'PS-FM-SQUELCH' : 'PS-FM-BELL');
  1268→    // Pad: supersaw for thick rich pads (7-osc with stereo spread)
  1269→    this.tracks[6].presetId = 'PS-SUPERSAW-PAD';
  1270→    // Arp: wavetable for evolving textures that morph over time
  1271→    this.tracks[7].presetId = 'PS-WT-MORPH';
  1272→  }
  1273→
  1274→  /**
  1275→   * Apply per-world send-effect modulations to the existing racks (Task E1).
  1276→   * Called by start() and switchWorld() after applyWorldPresets(). Rebuilds
  1277→   * the per-track rack configs for the new world and pushes the SEND levels
  1278→   * (reverb/delay/chorus/phaser/distortion/bitcrush) to the racks via smooth
  1279→   * ramps. EQ/comp/sat/pan are NOT changed here — they're set once at init()
  1280→   * and left alone so a world switch doesn't reset the per-track tonal balance
  1281→   * mid-phrase.
  1282→   *
  1283→   * Safe to call before init() (no-op when racks is empty).
  1284→   */
  1285→  private applyWorldEffectSettings(world: World): void {
  1286→    if (this.racks.length === 0) return;
  1287→    const cfgs = buildTrackRackConfigs(world);
  1288→    for (let i = 0; i < this.racks.length && i < cfgs.length; i++) {
  1289→      const rack = this.racks[i];
  1290→      const cfg = cfgs[i];
  1291→      // Only send levels are ramped here — tonal chain (EQ/comp/sat/pan) is
  1292→      // set once at init() and left alone for stability.
  1293→      rack.setParameter('sendReverb', cfg.sendReverb);
  1294→      rack.setParameter('sendDelay', cfg.sendDelay);
  1295→      rack.setParameter('sendChorus', cfg.sendChorus);
  1296→      rack.setParameter('sendPhaser', cfg.sendPhaser);
  1297→      rack.setParameter('sendDistortion', cfg.sendDistortion);
  1298→      rack.setParameter('sendBitcrush', cfg.sendBitcrush);
  1299→    }
  1300→
  1301→    // Also nudge the global send-effect parameters based on world character.
  1302→    if (this.chorusEffect && this.phaserEffect && this.distortionEffect && this.bitcrushEffect) {
  1303→      // Bright worlds → faster chorus rate; dark worlds → slower.
  1304→      const chorusRate = 0.35 + world.brightness * 0.5;
  1305→      this.chorusEffect.setParameter('rate', chorusRate);
  1306→      // Psychedelic worlds → faster phaser sweep + more feedback.
  1307→      const phaserRate = 0.15 + world.psychedelia * 0.6;
  1308→      this.phaserEffect.setParameter('rate', phaserRate);
  1309→      this.phaserEffect.setParameter('feedback', 0.25 + world.psychedelia * 0.4);
  1310→      // Aggressive/dark worlds → harder distortion.
  1311→      const distDrive = 2.5 + world.aggression * 5 + world.darkness * 2;
  1312→      this.distortionEffect.setParameter('drive', distDrive);
  1313→      // Dark worlds → coarser bitcrush (fewer bits, longer hold).
  1314→      const bcBits = Math.round(8 - world.darkness * 4);
  1315→      const bcHold = 2 + world.darkness * 6;
  1316→      this.bitcrushEffect.setParameter('bits', bcBits);
  1317→      this.bitcrushEffect.setParameter('holdMs', bcHold);
  1318→    }
  1319→  }
  1320→
  1321→  /**
  1322→   * Adjust a single per-track effect parameter in real-time (Task E1).
  1323→   * Routes to the named rack's setParameter(). Used by the reference pursuit
  1324→   * (and the future automated mixer) to nudge timbre as the radio's character
  1325→   * changes. Unknown trackIdx / effectName → silent no-op.
  1326→   *
  1327→   * Recognized effectNames (see TrackEffectsRack.setParameter):
  1328→   *   eqLowGain, eqMidFreq, eqMidGain, eqMidQ, eqHighGain,
  1329→   *   compThreshold, compRatio, compAttack, compRelease, compKnee,
  1330→   *   satDrive, satMix,
  1331→   *   pan, haasDelayMs, haasMix,
  1332→   *   outputGain,
  1333→   *   sendReverb, sendDelay, sendChorus, sendPhaser, sendDistortion, sendBitcrush
  1334→   *
  1335→   * Task V2b addition: 'cutoff' is special-cased for the LEAD track (idx 5).
  1336→   * The lead's filter cutoff lives inside the AdvancedSynthVoice (not the
  1337→   * rack — the rack is post-voice). So we store the override here and apply
  1338→   * it in triggerSynth() when the lead's voice is about to fire. Pass -1
  1339→   * to clear the override (revert to world timbre + reference pursuit blend).
  1340→   * Any non-negative value (Hz) is clamped to [200, 16000] and used directly.
  1341→   */
  1342→  setTrackEffect(trackIdx: number, effectName: string, value: number): void {
  1343→    if (!Number.isFinite(value)) return;
  1344→    // Task V2b: lead filter cutoff override (for BUILD-section filter sweeps).
  1345→    // Stored on the engine; applied in triggerSynth() when the lead fires.
  1346→    if (effectName === 'cutoff') {
  1347→      if (trackIdx !== 5) return; // only LEAD has a sweepable filter override
  1348→      this.leadCutoffOverride = value < 0 ? -1 : clamp(value, 200, 16000);
  1349→      return;
  1350→    }
  1351→    if (trackIdx < 0 || trackIdx >= this.racks.length) return;
  1352→    this.racks[trackIdx].setParameter(effectName, value);
  1353→  }
  1354→
  1355→  /**
  1356→   * Adjust a per-track SEND level in real-time (Task E1). Subset of
  1357→   * setTrackEffect() specialized for sends — handy for the arrangement
  1358→   * engine to push more reverb in breaks, less in drops, etc.
  1359→   *
  1360→   *   sendName ∈ { 'reverb', 'delay', 'chorus', 'phaser', 'distortion', 'bitcrush' }
  1361→   *   level ∈ [0, 1]
  1362→   */
  1363→  setSendLevel(trackIdx: number, sendName: 'reverb' | 'delay' | 'chorus' | 'phaser' | 'distortion' | 'bitcrush', level: number): void {
  1364→    if (!Number.isFinite(level)) return;
  1365→    if (trackIdx < 0 || trackIdx >= this.racks.length) return;
  1366→    this.racks[trackIdx].setParameter(`send${sendName.charAt(0).toUpperCase()}${sendName.slice(1)}` as any, level);
  1367→  }

  1369→  /**
  1370→   * Adjust a global send-effect parameter (Task E1). Lets the reference
  1371→   * pursuit or arrangement engine tweak the chorus rate / phaser feedback /
  1372→   * distortion drive / bitcrush bits in real time.
  1373→   *
  1374→   *   effectName ∈ { 'chorus', 'phaser', 'distortion', 'bitcrush' }
  1375→   *   param depends on the effect (see each class's setParameter).
  1376→   */
  1377→  setSendEffectParam(effectName: 'chorus' | 'phaser' | 'distortion' | 'bitcrush', param: string, value: number): void {
  1378→    if (!Number.isFinite(value)) return;
  1379→    switch (effectName) {
  1380→      case 'chorus':     this.chorusEffect?.setParameter(param, value);     break;
  1381→      case 'phaser':     this.phaserEffect?.setParameter(param, value);     break;
  1382→      case 'distortion': this.distortionEffect?.setParameter(param, value); break;
  1383→      case 'bitcrush':   this.bitcrushEffect?.setParameter(param, value);   break;
  1384→    }
  1385→  }

  1388→   * Adjust a master multiband compressor parameter in real-time (Task E1).
  1389→   * Recognized names: crossoverLow, crossoverHigh,
  1390→   *   lowThreshold, lowRatio, lowAttack, lowRelease, lowKnee, lowMakeup,
  1391→   *   midThreshold, midRatio, midAttack, midRelease, midKnee, midMakeup,
  1392→   *   highThreshold, highRatio, highAttack, highRelease, highKnee, highMakeup.

  1394→  setMasterParam(name: string, value: number): void {

  1398→  // ─── Task V2b: section-based effects automation ─────────────────────────────
  1399→  //
  1400→  // Commercial psytrance rides the mix alongside the arrangement: reverb
  1401→  // washes out in breaks, sucks dry in drops, delay throws in transitions,
  1402→  // filter sweeps open in builds, chorus thickens in the variation. The
  1403→  // pre-V2b engine left all the per-track send levels STATIC once the world
  1404→  // was applied — so a DROP and a BREAK had the same reverb tail, which is
  1405→  // exactly what makes a generated track sound "produced-but-not-mixed".
  1406→  //
  1407→  // applySectionAutomation() is called every step from scheduleStep(). It
  1408→  // does two things:
  1409→  //   1. When the section CHANGES, push a fresh set of static send levels
  1410→  //      (reverb/delay/chorus/phaser per section type) via setSendLevel().
  1411→  //      These ride on top of the world's per-track send levels (the rack
  1412→  //      uses setTargetAtTime, so the ramps are smooth and click-free).
  1413→  //   2. On EVERY step inside a BUILD section's last 2 bars, compute the
  1414→  //      filter sweep progress and call setTrackEffect(5, 'cutoff', value)
  1415→  //      to ramp the lead filter 800 Hz → 4000 Hz. The lead's noteOn in
  1416→  //      triggerSynth reads leadCutoffOverride and applies it to the voice's
  1417→  //      filter.frequency ramp — so the sweep is sample-accurate per note.
  1418→  //
  1419→  // Section profiles (per the task spec):
  1420→  //   INTRO       : reverb 0.60 / delay 0.10 / no chorus / no phaser
  1421→  //   GROOVE      : reverb 0.40 / delay 0.20 / slight chorus on lead
  1422→  //   BUILD       : rising lead filter cutoff 800→4000 Hz (last 2 bars),
  1423→  //                 increasing delay feedback, riser FX
  1424→  //   DROP        : reverb 0.25 (punchy) / delay 0.30 / full chorus on lead,
  1425→  //                 phaser on arp
  1426→  //   VARIATION   : reverb 0.35 / delay 0.40 (echo throws) / phaser on lead
  1427→  //   BREAK       : reverb 0.70 / delay 0.50 / filter closing on lead
  1428→  //   FINAL DROP  : reverb 0.20 / delay 0.30 / full effects
  1429→  //   OUTRO       : same as INTRO (wash out)
  1430→  //
  1431→  // Tracks: 0=KICK 1=SNARE 2=HATS 3=PERC 4=BASS 5=LEAD 6=PAD 7=ARP.
  1432→  applySectionAutomation(section: ArrangementSection, bar: number, step: number): void {
  1433→    if (!this.ctx) return;
  1434→
  1435→    // ── (1) Static levels — only re-push when the section label changes ──
  1436→    // The rack's setParameter uses setTargetAtTime(0.05s), so re-pushing
  1437→    // every step would be a no-op once the value has settled. We track the
  1438→    // last-applied section to avoid spamming the audio thread.
  1439→    if (section.label !== this.lastAutomationSection) {
  1440→      this.lastAutomationSection = section.label;
  1441→      this.applyStaticSectionLevels(section);
  1442→    }
  1443→
  1444→    // ── (2) BUILD filter sweep — per-step ramp in the last 2 bars ──
  1445→    // Sweep the lead filter cutoff from 800 Hz → 4000 Hz across the last 2
  1446→    // bars of a BUILD section. 2 bars × 16 steps = 32 steps total. We
  1447→    // compute a linear progress 0..1 and interpolate.
  1448→    //
  1449→    // The lead's AdvancedSynthVoice.noteOn() already does an exponential
  1450→    // filter sweep on every note (cut*3 → cut over atk+dec*0.7), so the
  1451→    // sweep value here is the BASELINE cutoff for each note. As the baseline
  1452→    // climbs across the 32 steps, each successive note opens brighter — the
  1453→    // signature "filter opening" build effect.
  1454→    if (section.label === 'BUILD' && bar >= section.bars - 2) {
  1455→      const sweepStartBar = section.bars - 2;
  1456→      const totalSteps = 2 * 16; // 2 bars × 16 steps
  1457→      const stepInSweep = (bar - sweepStartBar) * 16 + step;
  1458→      const progress = clamp(stepInSweep / Math.max(1, totalSteps - 1), 0, 1);
  1459→      // Exponential sweep feels more natural than linear (ears hear log-Hz).
  1460→      const startHz = 800;
  1461→      const endHz = 4000;
  1462→      const sweepHz = startHz * Math.pow(endHz / startHz, progress);
  1463→      this.setTrackEffect(5, 'cutoff', clamp(sweepHz, 200, 16000));
  1464→    } else if (this.leadCutoffOverride > 0) {
  1465→      // Outside a BUILD sweep, clear the override so the lead reverts to its
  1466→      // world timbre + reference pursuit cutoff. This also covers the BREAK
  1467→      // case: we leave the lead cutoff alone (world timbre is already dark
  1468→      // for BREAK energy 0.3) and rely on the high reverb to wash it out.
  1469→      this.leadCutoffOverride = -1;
  1470→    }
  1471→
  1472→    // ── (2b) BREAK filter closing — gradual close across the section ──
  1473→    // Over a 4-bar BREAK, drop the lead cutoff from ~1800 Hz to ~600 Hz for
  1474→    // a "filter closing" release effect. This complements the high reverb
  1475→    // (0.70) and delay (0.50) — the lead recedes into the wash.
  1476→    if (section.label === 'BREAK' && section.bars > 0) {
  1477→      const breakProgress = clamp(bar / Math.max(1, section.bars), 0, 1);
  1478→      const startHz = 1800;
  1479→      const endHz = 600;
  1480→      // Exponential close.
  1481→      const sweepHz = startHz * Math.pow(endHz / startHz, breakProgress);
  1482→      this.setTrackEffect(5, 'cutoff', clamp(sweepHz, 200, 16000));
  1483→    }
  1484→  }
  1485→
  1486→  /**
  1487→   * Apply the static send levels (reverb/delay/chorus/phaser) for a given
  1488→   * section. Called once per section change from applySectionAutomation().
  1489→   *
  1490→   * All ramps are smooth (the rack uses setTargetAtTime(0.05s) internally)
  1491→   * so a section change doesn't produce clicks or zipper noise. The values
  1492→   * are applied IN ADDITION to the world's per-track send levels — the
  1493→   * section automation is layered on top, so a world with extra phaser on
  1494→   * goa leads will still get its goa character plus the section's phaser
  1495→   * boost during drops.
  1496→   *
  1497→   * Tracks: 0=KICK 1=SNARE 2=HATS 3=PERC 4=BASS 5=LEAD 6=PAD 7=ARP.
  1498→   */
  1499→  private applyStaticSectionLevels(section: ArrangementSection): void {
  1500→    // Profile lookup — returns the target send levels per section type.
  1501→    // Each profile is a tuple of [reverb, delay, chorus, phaser] for the
  1502→    // melodic tracks (5/6/7) and a separate (reverb, delay) for the atmos
  1503→    // tracks (1/2/3). Kick (0) and bass (4) always stay dry-ish — they
  1504→    // need to stay punchy and centered regardless of section.
  1505→    interface SectionProfile {
  1506→      // melodic (LEAD/PAD/ARP) sends
  1507→      melReverb: number; melDelay: number; melChorus: number; melPhaser: number;
  1508→      // atmos (SNARE/HATS/PERC) sends
  1509→      atmoReverb: number; atmoDelay: number;
  1510→      // ARP-specific phaser (for DROP, VARIATION)
  1511→      arpPhaser?: number;
  1512→      // LEAD-specific chorus (for GROOVE slight, DROP full)
  1513→      leadChorus?: number;
  1514→    }
  1515→
  1516→    const profiles: Record<string, SectionProfile> = {
  1517→      INTRO:       { melReverb: 0.60, melDelay: 0.10, melChorus: 0.00, melPhaser: 0.00, atmoReverb: 0.30, atmoDelay: 0.05 },
  1518→      GROOVE:      { melReverb: 0.40, melDelay: 0.20, melChorus: 0.15, melPhaser: 0.10, atmoReverb: 0.22, atmoDelay: 0.10, leadChorus: 0.20 },
  1519→      BUILD:       { melReverb: 0.35, melDelay: 0.30, melChorus: 0.20, melPhaser: 0.15, atmoReverb: 0.25, atmoDelay: 0.18 },
  1520→      DROP:        { melReverb: 0.25, melDelay: 0.30, melChorus: 0.30, melPhaser: 0.20, atmoReverb: 0.18, atmoDelay: 0.12, leadChorus: 0.35, arpPhaser: 0.30 },
  1521→      VARIATION:   { melReverb: 0.35, melDelay: 0.40, melChorus: 0.25, melPhaser: 0.25, atmoReverb: 0.22, atmoDelay: 0.18, leadChorus: 0.25, arpPhaser: 0.20 },
  1522→      BREAK:       { melReverb: 0.70, melDelay: 0.50, melChorus: 0.10, melPhaser: 0.10, atmoReverb: 0.50, atmoDelay: 0.30 },
  1523→      'FINAL DROP':{ melReverb: 0.20, melDelay: 0.30, melChorus: 0.32, melPhaser: 0.22, atmoReverb: 0.16, atmoDelay: 0.12, leadChorus: 0.38, arpPhaser: 0.30 },
  1524→      OUTRO:       { melReverb: 0.60, melDelay: 0.10, melChorus: 0.00, melPhaser: 0.00, atmoReverb: 0.30, atmoDelay: 0.05 },
  1525→    };
  1526→
  1527→    const p = profiles[section.label] || profiles.GROOVE;
  1528→
  1529→    // Melodic tracks (5 LEAD, 6 PAD, 7 ARP) — push reverb/delay/chorus/phaser.
  1530→    for (const ti of [5, 6, 7]) {
  1531→      this.setSendLevel(ti, 'reverb', p.melReverb);
  1532→      this.setSendLevel(ti, 'delay', p.melDelay);
  1533→      this.setSendLevel(ti, 'chorus', p.melChorus);
  1534→      this.setSendLevel(ti, 'phaser', p.melPhaser);
  1535→    }
  1536→
  1537→    // LEAD-specific chorus bump (GROOVE = slight, DROP/FINAL DROP = full).
  1538→    if (p.leadChorus !== undefined) {
  1539→      this.setSendLevel(5, 'chorus', p.leadChorus);
  1540→    }
  1541→
  1542→    // ARP-specific phaser bump (DROP = full, VARIATION = medium).
  1543→    if (p.arpPhaser !== undefined) {
  1544→      this.setSendLevel(7, 'phaser', p.arpPhaser);
  1545→    }
  1546→
  1547→    // Atmos tracks (1 SNARE, 2 HATS, 3 PERC) — push reverb/delay only.
  1548→    // Kick (0) and bass (4) are left at their world defaults — they need
  1549→    // to stay punchy and centered regardless of section.
  1550→    for (const ti of [1, 2, 3]) {
  1551→      this.setSendLevel(ti, 'reverb', p.atmoReverb);
  1552→      this.setSendLevel(ti, 'delay', p.atmoDelay);
  1553→    }
  1554→  }

  1608→  private targetEnergy = 0.5;

  1610→  /**
  1611→   * Receive fresh metrics from the reference radio and store them as pursuit
  1612→   * targets. Also applies the smooth sub/high energy gain ramps immediately
  1613→   * (these are the only adjustments that can ramp across long time constants
  1614→   * without per-note cooperation — kick decay / centroid / transient density
  1615→   * are applied per-note in triggerDrum / triggerSynth / scheduleStep).
  1616→   */
  1617→  liveTrack(refMetrics: {
  1618→    lufs: number;
  1619→    kickDecayMs: number;
  1620→    spectralCentroid: number;
  1621→    energy?: number;
  1622→    subEnergy?: number;
  1623→    lowEnergy?: number;
  1624→    midEnergy?: number;
  1625→    highEnergy?: number;
  1626→    airEnergy?: number;
  1627→    transientDensity?: number;
  1628→    bassDecayMs?: number;
  1629→    stereoWidth?: number;
  1630→    bpm?: number;
  1631→    detectedKey?: { root: number; scale: string; confidence: number };
  1632→    // ── Task T1: extended harmonic / transient-shape / stereo fields ──
  1633→    // All optional — older callers (and the V1 listener) won't send them,
  1634→    // in which case the pursuit gracefully no-ops.
  1635→    spectralFlatness?: number;
  1636→    spectralCrest?: number;
  1637→    hnr?: number;
  1638→    inharmonicity?: number;
  1639→    spectralSlopeDb?: number;
  1640→    transientSharpness?: number;
  1641→    transientDecayMs?: number;
  1642→    stereoBalance?: number;
  1643→    stereoCorrelation?: number;
  1644→    msRatio?: number;
  1645→  }): void {
  1646→    if (isFinite(refMetrics.lufs)) this.targetLufs = refMetrics.lufs;
  1647→    if (refMetrics.energy !== undefined && isFinite(refMetrics.energy)) {
  1648→      this.targetEnergy = clamp(refMetrics.energy, 0, 1);
  1649→      this.refEnergy = this.targetEnergy;
  1650→    }
  1651→
