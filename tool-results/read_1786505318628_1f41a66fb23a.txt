   123→
   124→## Beat PLL
   125→
   126→```ts
   127→export interface BeatObservation {
   128→  time: number;
   129→  confidence: number;
   130→}
   131→
   132→export class BeatPLL {
   133→  private bpm = 150;
   134→  private beatTime = 0;
   135→  private initialized = false;
   136→
   137→  // Tune these; don't make them 1.
   138→  private readonly phaseGain = 0.18;
   139→  private readonly tempoGain = 0.025;
   140→
   141→  private confidence = 0;
   142→
   143→  update(obs: BeatObservation) {
   144→    if (obs.confidence < 0.45) return;
   145→
   146→    if (!this.initialized) {
   147→      this.beatTime = obs.time;
   148→      this.initialized = true;
   149→      this.confidence = obs.confidence;
   150→      return;
   151→    }
   152→
   153→    const period = 60 / this.bpm;
   154→    const predicted = this.beatTime + period;
   155→
   156→    let error = obs.time - predicted;
   157→
   158→    // Resolve occasional octave errors.
   159→    const candidates = [
   160→      error,
   161→      error + period,
   162→      error - period,
   163→    ];
   164→
   165→    error = candidates.reduce((best, x) =>
   166→      Math.abs(x) < Math.abs(best) ? x : best
   167→    );
   168→
   169→    // Phase correction.
   170→    this.beatTime += error * this.phaseGain;
   171→
   172→    // Tempo correction.
   173→    const observedPeriod = obs.time - this.beatTime;
   174→
   175→    if (
   176→      observedPeriod > 60 / 190 &&
   177→      observedPeriod < 60 / 80
   178→    ) {
   179→      const observedBpm = 60 / observedPeriod;
   180→
   181→      this.bpm +=
   182→        (observedBpm - this.bpm) * this.tempoGain;
   183→    }
   184→
   185→    this.confidence =
   186→      this.confidence * 0.85 +
   187→      obs.confidence * 0.15;
   188→  }
   189→
   190→  predictNextBeat(): number {
   191→    return this.beatTime + 60 / this.bpm;
   192→  }
   193→
   194→  getBpm() {
   195→    return this.bpm;
   196→  }
   197→
   198→  getConfidence() {
   199→    return this.confidence;
   200→  }
   201→
   202→  getPhase(now: number) {
   203→    const period = 60 / this.bpm;
   204→    const x = (now - this.beatTime) / period;
   205→
   206→    return ((x % 1) + 1) % 1;
   207→  }
   208→}
   209→```
   210→
   211→הנקודה החשובה היא שה-clock משתמש ב-`AudioContext.currentTime`, לא ב-`Date.now()` ולא ב-timestamp של `setInterval`.

   235→function predictBeats(
   236→  clock: BeatPLL,
   237→  now: number,
   238→  horizon = 0.2
   239→) {
   240→  const result: number[] = [];
   241→
   242→  let t = clock.predictNextBeat();
   243→  const period = 60 / clock.getBpm();
   244→
   245→  while (t < now + horizon) {

   248→  }

   280→# Kick detector ב-AudioWorklet
   281→
   282→`public/kick-detector.worklet.js`:
   283→
   284→```js
   285→class KickDetectorProcessor extends AudioWorkletProcessor {
   286→  constructor() {
   287→    super();
   288→
   289→    this.prevEnergy = 0;
   290→    this.cooldown = 0;
   291→
   292→    this.energy = 0;
   293→    this.alpha = 0.92;
   294→  }
   295→
   296→  process(inputs) {
   297→    const input = inputs[0];
   298→
   299→    if (!input || !input[0]) {
   300→      return true;
   301→    }
   302→
   303→    const channel = input[0];
   304→
   305→    let energy = 0;
   306→
   307→    for (let i = 0; i < channel.length; i++) {
   308→      const x = channel[i];
   309→      energy += x * x;
   310→    }
   311→
   312→    energy /= channel.length;
   313→
   314→    // Simple envelope.
   315→    this.energy =
   316→      this.energy * this.alpha +
   317→      energy * (1 - this.alpha);
   318→
   319→    const rise = this.energy - this.prevEnergy;
   320→
   321→    this.cooldown -= channel.length / sampleRate;
   322→
   323→    if (
   324→      this.cooldown <= 0 &&
   325→      this.energy > 0.01 &&
   326→      rise > 0.002
   327→    ) {
   328→      const time = currentTime;
   329→
   330→      this.port.postMessage({
   331→        type: "beat",
   332→        time,
   333→        confidence: Math.min(1, rise * 100),
   334→      });
   335→
   336→      this.cooldown = 0.12;
   337→    }
   338→
   339→    this.prevEnergy = this.energy;
   340→
   341→    return true;
   342→  }
   343→}
   344→
   345→registerProcessor(
   346→  "kick-detector",
   347→  KickDetectorProcessor
   348→);
   349→```
   350→
   351→ב-production הייתי מחליף את ה-energy detector ב-bandpass + spectral flux + adaptive threshold. אבל המבנה הזה הוא הנכון.

  1467→# Scheduler
  1468→
  1469→עכשיו אפשר לבנות scheduler טהור.
  1470→
  1471→```ts
  1472→export interface ScheduledEvent {
  1473→  time: number;
  1474→  step: number;
  1475→  bar: number;
  1476→}
  1477→
  1478→export interface Clock {
  1479→  bpm: number;
  1480→  nextBeatTime: number;
  1481→}
  1482→
  1483→export class MusicScheduler {
  1484→  private lookAhead = 0.20;
  1485→  private interval = 0.025;
  1486→
  1487→  private timer: number | null = null;
  1488→
  1489→  constructor(
  1490→    private ctx: AudioContext,
  1491→    private clock: BeatPLL,
  1492→    private scheduleStep:
  1493→      (step: number, time: number) => void
  1494→  ) {}
  1495→
  1496→  start() {
  1497→    if (this.timer !== null) return;
  1498→
  1499→    this.timer = window.setInterval(
  1500→      () => this.tick(),
  1501→      this.interval * 1000
  1502→    );
  1503→  }
  1504→
  1505→  stop() {
  1506→    if (this.timer !== null) {
  1507→      clearInterval(this.timer);
  1508→      this.timer = null;
  1509→    }
  1510→  }
  1511→
  1512→  private tick() {
  1513→    const now = this.ctx.currentTime;
  1514→
  1515→    const period =
  1516→      60 / this.clock.getBpm();
  1517→
  1518→    let t =
  1519→      this.clock.predictNextBeat();
  1520→
  1521→    while (t < now + this.lookAhead) {
  1522→      for (let step = 0; step < 4; step++) {
  1523→        const stepTime =
  1524→          t + step * period / 4;
  1525→
  1526→        this.scheduleStep(
  1527→          step,
  1528→          stepTime
  1529→        );
  1530→      }
  1531→
  1532→      t += period;
  1533→    }
  1534→  }
  1535→}
  1536→```
  1537→
  1538→ה-`setInterval` כאן הוא **רק wake-up mechanism**.
  1539→
  1540→הוא לא מגדיר את ה-musical timing.
  1541→
  1542→ה-timing האמיתי הוא:
  1543→
  1544→```ts
  1545→osc.start(scheduledAudioTime)
  1546→```
  1547→
  1548→זה בדיוק השימוש הנכון ב-Web Audio scheduling. Web Audio בנוי כגרף מודולרי עם `AudioParam` scheduling, ו-`AudioWorklet` מיועד לעיבוד audio בזמן אמת. ([MDN Web Docs][3])
