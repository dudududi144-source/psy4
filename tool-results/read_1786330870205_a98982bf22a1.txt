    27→    27→function scaleNote(root: number, scale: string, deg: number): number {
    28→    28→  const sc = SCALES[scale] || SCALES.minor;
    29→    29→  const n = sc.length;
    30→    30→  const oct = Math.floor(deg / n);
    31→    31→  const idx = ((deg % n) + n) % n;
    32→    32→  return root + 12 * oct + sc[idx];
    33→    33→}

   116→   116→// ─── Pooled Synth Voice (from PSY6 — persistent oscillators) ────────────────
   117→   117→
   118→   118→class PooledSynthVoice {
   119→   119→  osc1: OscillatorNode;
   120→   120→  osc2: OscillatorNode;
   121→   121→  g1: GainNode;
   122→   122→  g2: GainNode;
   123→   123→  filter: BiquadFilterNode;
   124→   124→  vca: GainNode;
   125→   125→  lfo: OscillatorNode;
   126→   126→  lfoGain: GainNode;
   127→   127→  bus: GainNode | null = null;
   128→   128→
   129→   129→  constructor(ctx: AudioContext) {
   130→   130→    this.osc1 = ctx.createOscillator();
   131→   131→    this.osc2 = ctx.createOscillator();
   132→   132→    this.g1 = ctx.createGain();
   133→   133→    this.g2 = ctx.createGain();
   134→   134→    this.filter = ctx.createBiquadFilter();
   135→   135→    this.filter.type = 'lowpass';
   136→   136→    this.vca = ctx.createGain();
   137→   137→    this.vca.gain.value = 0;
   138→   138→    this.lfo = ctx.createOscillator();
   139→   139→    this.lfoGain = ctx.createGain();
   140→   140→    this.lfoGain.gain.value = 0;
   141→   141→
   142→   142→    this.osc1.connect(this.g1);
   143→   143→    this.osc2.connect(this.g2);
   144→   144→    this.g1.connect(this.filter);
   145→   145→    this.g2.connect(this.filter);
   146→   146→    this.filter.connect(this.vca);
   147→   147→    this.lfo.connect(this.lfoGain);
   148→   148→
   149→   149→    // Start persistent oscillators (never stop them)
   150→   150→    this.osc1.start();
   151→   151→    this.osc2.start();
   152→   152→    this.lfo.start();
   153→   153→  }
   154→   154→
   155→   155→  connect(bus: GainNode) {
   156→   156→    if (this.bus !== bus) {
   157→   157→      this.vca.disconnect();
   158→   158→      this.vca.connect(bus);
   159→   159→      this.lfoGain.disconnect();
   160→   160→      this.lfoGain.connect(this.filter.frequency);
   161→   161→      this.bus = bus;
   162→   162→    }
   163→   163→  }
   164→   164→
   165→   165→  noteOn(p: SynthPreset, when: number, midi: number, vel: number, stepDur: number, bus: GainNode) {
   166→   166→    this.connect(bus);
   167→   167→    const f = mtof(clamp(midi, 12, 108));
   168→   168→    const gate = p.gate || 0.6;
   169→   169→    const dur = stepDur * gate * 2;
   170→   170→    const rel = Math.max(p.rel, 0.02);
   171→   171→    const end = when + dur;
   172→   172→
   173→   173→    this.osc1.type = p.wave1;
   174→   174→    this.osc2.type = p.wave2;
   175→   175→    this.osc1.frequency.setValueAtTime(f, when);
   176→   176→    this.osc2.frequency.setValueAtTime(f * Math.pow(2, p.oct2 || 0), when);
   177→   177→    this.osc2.detune.setValueAtTime(p.detune || 0, when);
   178→   178→    this.g1.gain.setValueAtTime(0.6, when);
   179→   179→    this.g2.gain.setValueAtTime(0.45, when);
   180→   180→
   181→   181→    const cut = clamp(p.cutoff, 60, 16000);
   182→   182→    const res = clamp(p.res, 0.2, 24);
   183→   183→    this.filter.type = p.fType;
   184→   184→    this.filter.Q.setValueAtTime(res, when);
   185→   185→    this.filter.frequency.cancelScheduledValues(when);
   186→   186→    this.filter.frequency.setValueAtTime(Math.min(cut * 3, 16000), when);
   187→   187→    this.filter.frequency.exponentialRampToValueAtTime(cut, when + Math.max((p.atk + p.dec * 0.7), 0.01));
   188→   188→
   189→   189→    if (p.lfoRate > 0 && p.lfoDest === 'cutoff') {
   190→   190→      this.lfo.frequency.setValueAtTime(p.lfoRate, when);
   191→   191→      this.lfoGain.gain.setValueAtTime(p.lfoDepth * 3000, when);
   192→   192→    } else {
   193→   193→      this.lfoGain.gain.setValueAtTime(0, when);
   194→   194→    }
   195→   195→
   196→   196→    const vca = this.vca.gain;
   197→   197→    const atk = Math.max(p.atk, 0.003);
   198→   198→    vca.cancelScheduledValues(when);
   199→   199→    vca.setValueAtTime(0, when);
   200→   200→    vca.linearRampToValueAtTime(vel * 0.5, when + atk);
   201→   201→    vca.setTargetAtTime(vel * 0.5 * p.sus, when + atk, Math.max(p.dec / 3, 0.01));
   202→   202→    vca.setTargetAtTime(0.0001, end, Math.max(rel / 3, 0.008));
   203→   203→  }
   204→   204→
   205→   205→  panic(ctx: AudioContext) {
   206→   206→    try {
   207→   207→      this.vca.gain.cancelScheduledValues(0);
   208→   208→      this.vca.gain.setValueAtTime(0, ctx.currentTime);
   209→   209→    } catch {}
   210→   210→  }
   211→   211→}

   213→   213→// ─── Pooled Drum Voice (from PSY6 — multi-type) ────────────────────────────
   214→   214→
   215→   215→class PooledDrumVoice {
   216→   216→  noise: AudioBufferSourceNode;
   217→   217→  noiseGain: GainNode;
   218→   218→  nFilter: BiquadFilterNode;
   219→   219→  osc: OscillatorNode;
   220→   220→  oscGain: GainNode;
   221→   221→  out: GainNode;
   222→   222→  bus: GainNode | null = null;
   223→   223→
   224→   224→  constructor(ctx: AudioContext, noiseBuffer: AudioBuffer) {
   225→   225→    this.noise = ctx.createBufferSource();
   226→   226→    this.noise.buffer = noiseBuffer;
   227→   227→    this.noise.loop = true;
   228→   228→    this.noiseGain = ctx.createGain();
   229→   229→    this.noiseGain.gain.value = 0;
   230→   230→    this.nFilter = ctx.createBiquadFilter();
   231→   231→    this.nFilter.type = 'bandpass';
   232→   232→    this.noise.connect(this.nFilter);
   233→   233→    this.nFilter.connect(this.noiseGain);
   234→   234→    this.osc = ctx.createOscillator();
   235→   235→    this.osc.type = 'sine';
   236→   236→    this.oscGain = ctx.createGain();
   237→   237→    this.oscGain.gain.value = 0;
   238→   238→    this.osc.connect(this.oscGain);
   239→   239→    this.out = ctx.createGain();
   240→   240→    this.noiseGain.connect(this.out);
   241→   241→    this.oscGain.connect(this.out);
   242→   242→    this.noise.start();
   243→   243→    this.osc.start();
   244→   244→  }
   245→   245→
   246→   246→  connect(bus: GainNode) {
   247→   247→    if (this.bus !== bus) {
   248→   248→      this.out.disconnect();
   249→   249→      this.out.connect(bus);
   250→   250→      this.bus = bus;
   251→   251→    }
   252→   252→  }
   253→   253→
   254→   254→  hit(p: DrumPreset, when: number, vel: number, bus: GainNode) {
   255→   255→    this.connect(bus);
   256→   256→    const tune = p.tune || 1;
   257→   257→    const decay = p.decay || 1;
   258→   258→    const tone = p.tone || 1;
   259→   259→    const punch = p.punch || 0;
   260→   260→    const type = p.type;
   261→   261→    const ng = this.noiseGain.gain;
   262→   262→    const og = this.oscGain.gain;
   263→   263→    ng.cancelScheduledValues(when);
   264→   264→    og.cancelScheduledValues(when);
   265→   265→    ng.setValueAtTime(0, when);
   266→   266→    og.setValueAtTime(0, when);
   267→   267→
   268→   268→    if (type === 'kick') {
   269→   269→      const dur = 0.12 + 0.5 * decay;
   270→   270→      this.osc.frequency.setValueAtTime(180 * tune, when);
   271→   271→      this.osc.frequency.exponentialRampToValueAtTime(Math.max(36 * tune, 24), when + 0.09);
   272→   272→      og.setValueAtTime(vel * 1.1, when);
   273→   273→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   274→   274→      if (punch > 0) {
   275→   275→        ng.setValueAtTime(vel * punch * 0.8, when);
   276→   276→        ng.exponentialRampToValueAtTime(0.0001, when + 0.02);
   277→   277→        this.nFilter.frequency.setValueAtTime(2500, when);
   278→   278→      }
   279→   279→    } else if (type === 'clap') {
   280→   280→      const dur = 0.25 + 0.15 * decay;
   281→   281→      this.nFilter.type = 'bandpass';
   282→   282→      this.nFilter.frequency.setValueAtTime(1150 * tone, when);
   283→   283→      this.nFilter.Q.value = 1.3;
   284→   284→      ng.setValueAtTime(0, when);
   285→   285→      [0, 0.014, 0.03].forEach(t2 => {
   286→   286→        ng.setValueAtTime(0, when + t2);
   287→   287→        ng.linearRampToValueAtTime(vel * 0.9, when + t2 + 0.002);
   288→   288→        ng.exponentialRampToValueAtTime(0.02, when + t2 + 0.012);
   289→   289→      });
   290→   290→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   291→   291→    } else if (type === 'hatC' || type === 'hatO') {
   292→   292→      const open = type === 'hatO';
   293→   293→      const dur = open ? 0.26 + 0.5 * decay : 0.03 + 0.05 * decay;
   294→   294→      this.nFilter.type = 'highpass';
   295→   295→      this.nFilter.frequency.setValueAtTime(7200 * Math.sqrt(tone), when);
   296→   296→      ng.setValueAtTime(vel * (open ? 0.4 : 0.5), when);
   297→   297→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   298→   298→    } else if (type === 'tom') {
   299→   299→      const dur = 0.22 + 0.35 * decay;
   300→   300→      this.osc.type = 'sine';
   301→   301→      this.osc.frequency.setValueAtTime(180 * tune, when);
   302→   302→      this.osc.frequency.exponentialRampToValueAtTime(92 * tune, when + dur * 0.7);
   303→   303→      og.setValueAtTime(vel * 0.9, when);
   304→   304→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   305→   305→    } else if (type === 'shaker') {
   306→   306→      const dur = 0.04 + 0.07 * decay;
   307→   307→      this.nFilter.type = 'highpass';
   308→   308→      this.nFilter.frequency.setValueAtTime(6000 * tone, when);
   309→   309→      ng.setValueAtTime(vel * 0.45, when);
   310→   310→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   311→   311→    } else if (type === 'glitch') {
   312→   312→      const dur = 0.08 + 0.14 * decay;
   313→   313→      this.nFilter.type = 'bandpass';
   314→   314→      this.nFilter.frequency.setValueAtTime(1500 * tone + 800, when);
   315→   315→      this.nFilter.Q.value = 4;
   316→   316→      ng.setValueAtTime(vel * 0.7, when);
   317→   317→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   318→   318→    }
   319→   319→  }
   320→   320→
   321→   321→  panic(ctx: AudioContext) {
   322→   322→    try {
   323→   323→      this.noiseGain.gain.cancelScheduledValues(0);
   324→   324→      this.noiseGain.gain.setValueAtTime(0, ctx.currentTime);
   325→   325→      this.oscGain.gain.cancelScheduledValues(0);
   326→   326→      this.oscGain.gain.setValueAtTime(0, ctx.currentTime);
   327→   327→    } catch {}
   328→   328→  }
   329→   329→}

   423→   423→    this.toneHigh = c.createBiquadFilter();

   519→   519→      rSend.gain.value = i >= 5 ? 0.2 : 0.03;
   520→   520→      bus.connect(rSend);
   521→   521→      rSend.connect(this.reverbSend);

   524→   524→      this.trackGains.push(gain);
   525→   525→    }
   526→   526→    // duckGain already connected to master above (line 458)

   531→   531→

   533→   533→    this.initTracks();

   697→   697→  private tick(): void {
   698→   698→    if (!this.playing || !this.ctx) return;
   699→   699→    const lookahead = 0.06;
   700→   700→    const s16 = 60 / this.bpm / 4;
   701→   701→
   702→   702→    while (this.nextTime < this.ctx.currentTime + lookahead) {
   703→   703→      this.scheduleStep(this.step, this.bar, this.nextTime);
   704→   704→      this.step++;
   705→   705→      this.nextTime += s16;
   706→   706→      if (this.step >= 16) {
   707→   707→        this.step = 0;
   708→   708→        this.bar++;
   709→   709→        const section = this.arrangement[this.sectionIdx % this.arrangement.length];
   710→   710→        if (this.bar >= section.bars) {
   711→   711→          this.sectionIdx++;
   712→   712→          this.bar = 0;
   713→   713→          const next = this.arrangement[this.sectionIdx % this.arrangement.length];
   714→   714→          this.currentSection = next.label;
   715→   715→          this.onSectionChange?.(this.currentSection);
   716→   716→        }
   717→   717→        // Change preset every 4 bars
   718→   718→        if (this.bar % 4 === 0 && this.bar > 0) {
   719→   719→          this.presetIdx = (this.presetIdx + 1) % 3;
   720→   720→          this.rotatePresets();
   721→   721→        }
   722→   722→      }
   723→   723→    }
   724→   724→  }

   736→   736→  private scheduleStep(step: number, bar: number, time: number): void {
   737→   737→    const section = this.arrangement[this.sectionIdx % this.arrangement.length];
   738→   738→    const key = this.musicalKey;
   739→   739→    const root = key.root;
   740→   740→    const sc = key.scale;
   741→   741→    const sd = 60 / this.bpm / 4;
   742→   742→    const isPreDrop = (section.label === 'BUILD' || section.label === 'BUILD 2') && bar >= section.bars - 2;
   743→   743→    const isDropStart = section.label.includes('DROP') && bar === 0 && step === 0;
   744→   744→
   745→   745→    // ── RISER FX (last 2 bars of build) ──
   746→   746→    if (isPreDrop && step === 0 && bar === section.bars - 2) {
   747→   747→      this.triggerRiser(time, sd * 32);
   748→   748→    }
   749→   749→
   750→   750→    // ── IMPACT FX (drop start) ──
   751→   751→    if (isDropStart) {
   752→   752→      this.triggerImpact(time);
   753→   753→    }
   754→   754→
   755→   755→    // ── KICK (track 0) — 4 on the floor ──
   756→   756→    if (step % 4 === 0) {
   757→   757→      this.triggerDrum(0, time, 0.5 + section.density * 0.4);
   758→   758→    }
   759→   759→
   760→   760→    // ── CLAP (track 1) — on 2 and 4 ──
   761→   761→    if ((step === 4 || step === 12) && section.density > 0.4) {
   762→   762→      this.triggerDrum(1, time, 0.3);
   763→   763→    }
   764→   764→
   765→   765→    // ── HATS (track 2) — offbeat with velocity variation ──
   766→   766→    if (step % 2 === 1) {
   767→   767→      const vel = 0.15 + (step % 4 === 3 ? 0.1 : 0) + section.density * 0.1;
   768→   768→      this.triggerDrum(2, time, vel);
   769→   769→    }
   770→   770→
   771→   771→    // ── PERC (track 3) — sparse, with variation ──
   772→   772→    if (section.density > 0.5 && (step === 6 || step === 14) && Math.random() < 0.6) {
   773→   773→      this.triggerDrum(3, time, 0.2);
   774→   774→    }
   775→   775→
   776→   776→    // ── BASS (track 4) — offbeat, evolving pattern with passing tones ──
   777→   777→    if (section.bass && step % 2 === 1) {
   778→   778→      // Rich bass pattern: changes every 4 bars with passing tones and octaves
   779→   779→      let bassPattern: number[];
   780→   780→      const bp = bar % 8;
   781→   781→      if (bp < 2) bassPattern = [0, 0, 0, 0, 0, 0, 4, 0]; // root with fifth
   782→   782→      else if (bp < 4) bassPattern = [0, 0, 2, 0, 4, 0, 3, 0]; // walking
   783→   783→      else if (bp < 6) bassPattern = [0, 0, 0, 7, 0, 0, 4, 3]; // octave + passing
   784→   784→      else bassPattern = [0, 4, 0, 3, 0, 2, 0, 4]; // rolling
   785→   785→
   786→   786→      const bassDeg = bassPattern[Math.floor(step / 2) % bassPattern.length];
   787→   787→      if (bassDeg >= 0) {
   788→   788→        const note = scaleNote(root, sc, bassDeg);
   789→   789→        this.triggerSynth(4, time, note, 0.5, sd);
   790→   790→      }
   791→   791→    }
   792→   792→
   793→   793→    // ── LEAD (track 5) — evolving trance motif with SPACES ──
   794→   794→    if (section.lead) {
   795→   795→      const phraseBar = bar % 8;
   796→   796→      const leadSteps = [0, 6, 10];
   797→   797→      if (leadSteps.includes(step)) {
   798→   798→        let deg: number; let dur: number; let vel: number;
   799→   799→        if (phraseBar < 2) {
   800→   800→          const motif = [7, 5, 3];
   801→   801→          deg = motif[leadSteps.indexOf(step)];
   802→   802→          dur = 0.3; vel = 0.25;
   803→   803→        } else if (phraseBar < 4) {
   804→   804→          const motif = [5, 3, 0];
   805→   805→          deg = motif[leadSteps.indexOf(step)];
   806→   806→          dur = 0.2; vel = 0.3;
   807→   807→        } else if (phraseBar < 6) {
   808→   808→          const motif = [10, 12, 10];
   809→   809→          deg = motif[leadSteps.indexOf(step)];
   810→   810→          dur = 0.15; vel = 0.35;
   811→   811→        } else if (phraseBar === 6) {
   812→   812→          deg = step === 0 ? 0 : step === 6 ? 2 : 0;
   813→   813→          dur = 0.4; vel = 0.4;
   814→   814→        } else {
   815→   815→          return; // rest bar
   816→   816→        }
   817→   817→        const note = scaleNote(root + 12, sc, deg);
   818→   818→        this.triggerSynth(5, time, note, vel, sd, dur);
   819→   819→      }
   820→   820→    }
   821→   821→
   822→   822→    // ── PAD (track 6) — sustained chord every bar in drops ──
   823→   823→    if (section.lead && step === 0) {
   824→   824→      // Chord progression: root, IV, V, III (psytrance progression)
   825→   825→      const progDegs = [0, 3, 4, 2];
   826→   826→      const chordDeg = progDegs[bar % 4];
   827→   827→      const chordRoot = scaleNote(root + 12, sc, chordDeg);
   828→   828→      this.triggerSynth(6, time, chordRoot, 0.25, sd * 4);
   829→   829→      // Also play fifth for full chord
   830→   830→      this.triggerSynth(6, time + 0.01, scaleNote(root + 12, sc, chordDeg + 4), 0.15, sd * 4);
   831→   831→    }
   832→   832→
   833→   833→    // ── ARP (track 7) — rolling arp, 70% density in drops ──
   834→   834→    if (section.lead && step % 2 === 0 && Math.random() < 0.7) {
   835→   835→      const arpDegs = [0, 2, 4, 7, 4, 2, 0, 7];
   836→   836→      const deg = arpDegs[(step / 2) % arpDegs.length];
   837→   837→      const note = scaleNote(root + 24, sc, deg);
   838→   838→      this.triggerSynth(7, time, note, 0.25, sd);
   839→   839→    }
   840→   840→
   841→   841→    // ── SHAKER (track 3 alt) — continuous offbeat in drops ──
   842→   842→    if (section.bass && section.lead && step % 2 === 1 && Math.random() < 0.4) {
   843→   843→      this.triggerDrum(3, time, 0.15);
   844→   844→    }
   845→   845→  }

   847→   847→  private triggerDrum(trackIdx: number, time: number, vel: number): void {
   848→   848→    const track = this.tracks[trackIdx];
   849→   849→    if (track.mix.mute) return;
   850→   850→    const preset = DRUM_PRESETS[track.presetId];
   851→   851→    if (!preset) return;
   852→   852→    const voice = this.drumPool[this.drumIdx];
   853→   853→    this.drumIdx = (this.drumIdx + 1) % this.drumPool.length;
   854→   854→    voice.hit(preset, time, vel * track.mix.vol, this.chains[trackIdx]);
   855→   855→
   856→   856→    // Sidechain: when kick fires, duck the bass
   857→   857→    if (trackIdx === 0 && this.duckGain && this.ctx) {
   858→   858→      this.duckGain.gain.cancelScheduledValues(time);
   859→   859→      this.duckGain.gain.setValueAtTime(1 - 0.4, time); // 40% duck
   860→   860→      this.duckGain.gain.linearRampToValueAtTime(1.0, time + 0.25); // 250ms recovery
   861→   861→    }
   862→   862→  }

   864→   864→  private triggerRiser(time: number, dur: number): void {
   865→   865→    if (!this.ctx) return;
   866→   866→    const c = this.ctx;
   867→   867→    // Noise through filter that opens up
   868→   868→    const noise = c.createBufferSource();
   869→   869→    noise.buffer = this.noiseBuffer;
   870→   870→    noise.loop = true;
   871→   871→    const filter = c.createBiquadFilter();
   872→   872→    filter.type = 'bandpass';
   873→   873→    filter.frequency.setValueAtTime(200, time);
   874→   874→    filter.frequency.exponentialRampToValueAtTime(8000, time + dur);
   875→   875→    filter.Q.value = 2;
   876→   876→    const gain = c.createGain();
   877→   877→    gain.gain.setValueAtTime(0.001, time);
   878→   878→    gain.gain.exponentialRampToValueAtTime(0.3, time + dur);
   879→   879→    gain.gain.exponentialRampToValueAtTime(0.001, time + dur + 0.1);
   880→   880→    noise.connect(filter);
   881→   881→    filter.connect(gain);
   882→   882→    gain.connect(this.master);
   883→   883→    noise.start(time);
   884→   884→    noise.stop(time + dur + 0.2);
   885→   885→  }

   887→   887→  private triggerImpact(time: number): void {
   888→   888→    if (!this.ctx) return;
   889→   889→    const c = this.ctx;
   890→   890→    // Sub boom + noise crack
   891→   891→    const osc = c.createOscillator();
   892→   892→    osc.type = 'sine';
   893→   893→    osc.frequency.setValueAtTime(120, time);
   894→   894→    osc.frequency.exponentialRampToValueAtTime(35, time + 0.5);
   895→   895→    const og = c.createGain();
   896→   896→    og.gain.setValueAtTime(0.8, time);
   897→   897→    og.gain.exponentialRampToValueAtTime(0.001, time + 0.6);
   898→   898→    osc.connect(og);
   899→   899→    og.connect(this.master);
   900→   900→    osc.start(time);
   901→   901→    osc.stop(time + 0.7);
   902→   902→
   903→   903→    const noise = c.createBufferSource();
   904→   904→    noise.buffer = this.noiseBuffer;
   905→   905→    const hp = c.createBiquadFilter();
   906→   906→    hp.type = 'highpass';
   907→   907→    hp.frequency.value = 3000;
   908→   908→    const ng = c.createGain();
   909→   909→    ng.gain.setValueAtTime(0.4, time);
   910→   910→    ng.gain.exponentialRampToValueAtTime(0.001, time + 0.05);
   911→   911→    noise.connect(hp);
   912→   912→    hp.connect(ng);
   913→   913→    ng.connect(this.master);
   914→   914→    noise.start(time);
   915→   915→    noise.stop(time + 0.1);
   916→   916→  }

   918→   918→  private triggerSynth(trackIdx: number, time: number, midi: number, vel: number, stepDur: number, dur?: number): void {
   919→   919→    const track = this.tracks[trackIdx];
   920→   920→    if (track.mix.mute) return;
   921→   921→    const preset = SYNTH_PRESETS[track.presetId];
   922→   922→    if (!preset) return;
   923→   923→    const voice = this.synthPool[this.synthIdx];
   924→   924→    this.synthIdx = (this.synthIdx + 1) % this.synthPool.length;
   925→   925→    const p = dur ? { ...preset, gate: dur / (stepDur * 2) } : preset;
   926→   926→    voice.noteOn(p, time, midi, vel * track.mix.vol, stepDur, this.chains[trackIdx]);
   927→   927→
   928→   928→    // Add sub oscillator for bass track (sine one octave below)
   929→   929→    if (trackIdx === 4 && this.ctx) {
   930→   930→      const subFreq = mtof(midi - 12); // one octave below
   931→   931→      const subOsc = this.ctx.createOscillator();
   932→   932→      const subGain = this.ctx.createGain();
   933→   933→      subOsc.type = 'sine';
   934→   934→      subOsc.frequency.value = subFreq;
   935→   935→      const subDecay = (dur || stepDur * 0.3) + 0.05;
   936→   936→      subGain.gain.setValueAtTime(0.5 * track.mix.vol, time);
   937→   937→      subGain.gain.exponentialRampToValueAtTime(0.001, time + subDecay);
   938→   938→      subOsc.connect(subGain);
   939→   939→      subGain.connect(this.chains[4]);
   940→   940→      subOsc.start(time);
   941→   941→      subOsc.stop(time + subDecay + 0.02);
   942→   942→    }
   943→   943→  }
