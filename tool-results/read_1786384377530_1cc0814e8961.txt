   204→  const connectRef = useCallback(async () => {

   295→  const startEngine = useCallback(async () => {

   321→      // This fixes ROAST-7 bug #2: "If user clicks START before worklet loads,

   337→        a.setEngineBpm((engine as any)._bpm || 145);
   338→        a.onMetrics(m => {
   339→          setSelfMetrics(m);
   340→          a.setEngineBpm((engineRef.current as any)?._bpm || 145);
   341→          if (engineRef.current?.selfTrack) engineRef.current.selfTrack({
   342→            lufs: m.lufs,
   343→            energy: m.energy,
   344→            spectralCentroid: m.spectralCentroid,
   345→            transientDensity: m.transientDensity,
   346→            subEnergy: m.subEnergy,
   347→            highEnergy: m.highEnergy,
   348→          });
   349→          // Refresh pursuit status snapshot for UI display
   350→          if (engineRef.current?.getPursuitStatus) {
   351→            try { setPursuit(engineRef.current.getPursuitStatus()); } catch {}
   352→          }
   353→          // ── Integration UI pulls (Task I1-UI) ──
   354→          // All optional-chained: T1 may or may not have wired the getters yet.
   355→          try {
   356→            if (engineRef.current?.getSynthesisCharacter) {
   357→              setSynthChar(engineRef.current.getSynthesisCharacter());
   358→            }
   359→            if (engineRef.current?.getSynthModeOverrides) {
   360→              setSynthOverrides(engineRef.current.getSynthModeOverrides() || {});
   361→            }
   362→            if (engineRef.current?.getEffectsState) {
   363→              setEffectsState(engineRef.current.getEffectsState());
   364→            }
   365→            if (engineRef.current?.getCurrentChord) {
   366→              setCurrentChord(engineRef.current.getCurrentChord());
   367→            }
   368→            if (engineRef.current?.getCurrentProgression) {
   369→              const chords = engineRef.current.getCurrentProgression();
   370→              const idx = engineRef.current.getChordIdx?.() ?? 0;
   371→              if (Array.isArray(chords)) setProgressionInfo({ chords, idx });
   372→            }
   373→            if (engineRef.current?.getPursuitDashboard) {
   374→              setPursuitDashboard(engineRef.current.getPursuitDashboard());
   375→            }
   376→            if (engineRef.current?.getMelodyState) {
   377→              setMelodyState(engineRef.current.getMelodyState());
   378→            }
   379→            // ── Task A1: pull deep A/B analysis ──
   380→            if (engineRef.current?.getDeepAnalysis) {
   381→              try { setDeepAnalysis(engineRef.current.getDeepAnalysis()); } catch {}
   382→            }
   383→            // ── Task P2 (musical intelligence): pull MusicAnalyzer snapshot ──
   384→            // Optional chaining so we degrade gracefully if P2 isn't merged.
   385→            if (engineRef.current?.getMusicalAnalysis) {
   386→              try { setMusicalAnalysis(engineRef.current.getMusicalAnalysis()); } catch {}
   387→            }
   388→            // ── Task D1: pull DJ sync status + mirror toggle state ──
   389→            // Optional chaining so we degrade gracefully if D1 isn't merged.
   390→            if (engineRef.current?.getSyncStatus) {
   391→              try { setSyncStatus(engineRef.current.getSyncStatus()); } catch {}
   392→            }
   393→            if (engineRef.current?.isSyncEnabled) {
   394→              try { setSyncEnabled(engineRef.current.isSyncEnabled()); } catch {}
   395→            }
   396→            // ── Task P4: pull phrase-sync state ──
   397→            // Optional chaining so we degrade gracefully if P4 isn't merged.
   398→            if (engineRef.current?.getPhraseSyncState) {
   399→              try { setPhraseSyncState(engineRef.current.getPhraseSyncState()); } catch {}
   400→            }
   401→            // ── Task T1 (active learning): pull learning status ──
   402→            // Optional chaining so we degrade gracefully if T1 isn't merged.
   403→            if (engineRef.current?.getLearningStatus) {
   404→              try { setLearningStatus(engineRef.current.getLearningStatus()); } catch {}
   405→            }
   406→            // ── Task P5 (adaptive learning): pull vocabulary stats ──
   407→            // Optional chaining so we degrade gracefully if P5 isn't merged.
   408→            if (engineRef.current?.getVocabularyStats) {
   409→              try { setVocabularyStats(engineRef.current.getVocabularyStats()); } catch {}
   410→            }
   411→          } catch {}
   412→          // Track D: pull style classification + active world on every tick.
   413→          // Optional chaining so we degrade gracefully if Track C isn't merged yet.
   414→          if (engineRef.current?.getStyleClassification) {
   415→            try {
   416→              const matches = engineRef.current.getStyleClassification() as StyleMatch[];
   417→              if (Array.isArray(matches)) setStyleMatches(matches);
   418→            } catch {}
   419→          }
   420→          if (engineRef.current?.getCurrentWorldId) {
   421→            try {
   422→              const cw = engineRef.current.getCurrentWorldId() as string;
   423→              if (cw && cw !== activeWorldRef.current) {
   424→                activeWorldRef.current = cw;
