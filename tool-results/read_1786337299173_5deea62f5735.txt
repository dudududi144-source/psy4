   354→  const pursuitRows: PursuitRow[] = useMemo(() => {

   361→    ];

   363→      const delta = actual - target;
   364→      const absDelta = Math.abs(delta);
   365→      const isActive = target > 0;
   366→      const prev = prevDeltaRef.current[label];

   373→        else if (absDelta > prev + 1e-6) arrow = 'down';
   374→        else arrow = 'idle';
   375→      }

   377→        : absDelta <= tol ? 'text-emerald-400'
   378→        : absDelta <= tol * 3 ? 'text-amber-400'
   379→        : 'text-rose-400';
   380→      return { label, target, actual, unit, tol, delta, absDelta, arrow, color };
   381→    });

   402→  }, [refMetrics, selfMetrics]);
   403→
   404→  const spectralDeltaColor = (d: number | null) =>

   412→

   415→      <Toaster />

   418→          <div className="flex items-center gap-3">

   826→        {/* Learning */}

   862→            </CardContent>
   863→          </Card>
   864→        )}
   865→
   866→        {/* Live Reference Metrics */}
   867→        {mode === 'listen' && refMetrics && (

   889→              {engineOn && (
   890→                <div className="mt-2 p-2 bg-cyan-950/30 border border-cyan-800 rounded text-[10px] font-mono text-cyan-400">
   891→                  ENGINE SYNCED: BPM {engineState.bpm} · Key {engineState.key} · Section {engineState.section} · World {WORLD_NAME[activeWorld] ?? activeWorld}

   896→        )}
   897→      </main>

   899→      <footer className="border-t border-slate-800 bg-slate-900/60 mt-auto">
   900→        <div className="max-w-7xl mx-auto px-4 py-3 text-[10px] text-slate-500 font-mono flex items-center justify-between flex-wrap gap-2">
