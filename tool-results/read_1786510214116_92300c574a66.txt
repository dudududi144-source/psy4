   298→  // ── Audio init (EXACTLY like psy) ──
   299→  private ensureAudio(): void {
   300→    if (this.ctx) { if (this.ctx.state === 'suspended') this.ctx.resume(); return; }
   301→    this.ctx = new (window.AudioContext || (window as any).webkitAudioContext)();
   302→
   303→    // Simple chain: voices → master → analyser → destination
   304→    this.master = this.ctx.createGain();
   305→    this.master.gain.value = 0.9;
   306→    this.analyser = this.ctx.createAnalyser();
   307→    this.analyser.fftSize = 512;
   308→    this.analyser.smoothingTimeConstant = 0.7;
   309→    this.master.connect(this.analyser);
   310→    this.analyser.connect(this.ctx.destination);
   311→
   312→    // Delay (like psy)
   313→    this.delaySend = this.ctx.createGain();
   314→    this.delaySend.gain.value = 1.0;
   315→    this.delay = this.ctx.createDelay(2.0);
   316→    this.delay.delayTime.value = 0.3;
   317→    const wet = this.ctx.createGain(); wet.gain.value = 0.22;
   318→    const fb = this.ctx.createGain(); fb.gain.value = 0.34;
   319→    this.delaySend.connect(this.delay);
   320→    this.delay.connect(wet); wet.connect(this.master);
   321→    this.delay.connect(fb); fb.connect(this.delay);
   322→
   323→    // Noise buffer for hats
   324→    const len = Math.floor(this.ctx.sampleRate * 0.25);
   325→    this.noiseBuf = this.ctx.createBuffer(1, len, this.ctx.sampleRate);
   326→    const nd = this.noiseBuf.getChannelData(0);
   327→    for (let i = 0; i < len; i++) nd[i] = Math.random() * 2 - 1;
   328→
   329→    // Load learning
   330→    this.learningData = loadLearning();
   331→
   332→    // ── PER-ROLE BUSES (from architecture review) ──
   333→    // Each voice connects to its role bus → engineBus → gentle comp → master
   334→    this.kickBus = this.ctx.createGain(); this.kickBus.gain.value = 0.9;
   335→    this.bassBus = this.ctx.createGain(); this.bassBus.gain.value = 0.85;
   336→    this.leadBus = this.ctx.createGain(); this.leadBus.gain.value = 0.7;
   337→    this.hatBus = this.ctx.createGain(); this.hatBus.gain.value = 0.6;
   338→    
   339→    this.engineBus = this.ctx.createGain();
   340→    this.engineBus.gain.value = 0.8;
   341→
   342→    // Gentle compressor on engine bus only (not on radio)
   343→    this.comp = this.ctx.createDynamicsCompressor();
   344→    this.comp.threshold.value = -18;
   345→    this.comp.knee.value = 18;
   346→    this.comp.ratio.value = 2;
   347→    this.comp.attack.value = 0.015;
   348→    this.comp.release.value = 0.12;
   349→
   350→    // Connect: role buses → engineBus → comp → master
   351→    this.kickBus.connect(this.engineBus);
   352→    this.bassBus.connect(this.engineBus);
   353→    this.leadBus.connect(this.engineBus);
   354→    this.hatBus.connect(this.engineBus);
   355→    this.engineBus.connect(this.comp);
   356→    this.comp.connect(this.master);
   357→  }

   359→  // ── Voices — connect to role buses (not master directly) ──
   360→  private kick(t: number): void {
   361→    if (!this.ctx || !this.kickBus) return;
   362→    const osc = this.ctx.createOscillator(), gain = this.ctx.createGain();
   363→    osc.type = 'sine';
   364→    osc.frequency.setValueAtTime(150, t);
   365→    osc.frequency.exponentialRampToValueAtTime(44, t + 0.09);
   366→    gain.gain.setValueAtTime(1.0, t);
   367→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.3);
   368→    osc.connect(gain); gain.connect(this.kickBus);
   369→    osc.start(t); osc.stop(t + 0.32);
   370→  }
   371→
   372→  private hat(t: number, lvl: number): void {
   373→    if (!this.ctx || !this.hatBus || !this.noiseBuf) return;
   374→    const src = this.ctx.createBufferSource(); src.buffer = this.noiseBuf;
   375→    const hp = this.ctx.createBiquadFilter(); hp.type = 'highpass'; hp.frequency.value = 7000;
   376→    const gain = this.ctx.createGain();
   377→    gain.gain.setValueAtTime(Math.max(0.001, lvl), t);
   378→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.05);
   379→    src.connect(hp); hp.connect(gain); gain.connect(this.hatBus);
   380→    src.start(t); src.stop(t + 0.06);
   381→  }
   382→
   383→  private bass(t: number, freq: number, v: Variant): void {
   384→    if (!this.ctx || !this.bassBus) return;
   385→    const osc = this.ctx.createOscillator(); osc.type = v.bassWave; osc.frequency.value = freq;
   386→    const filter = this.ctx.createBiquadFilter(); filter.type = 'lowpass'; filter.Q.value = v.bassQ;
   387→    const fStart = Math.max(60, v.bassCut), fEnd = Math.max(80, v.bassCut * 0.35);
   388→    filter.frequency.setValueAtTime(fStart, t);
   389→    filter.frequency.exponentialRampToValueAtTime(fEnd, t + 0.16);
   390→    const gain = this.ctx.createGain();
   391→    gain.gain.setValueAtTime(0.0001, t);
   392→    gain.gain.exponentialRampToValueAtTime(0.85, t + 0.006);
   393→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.2);
   394→    osc.connect(filter); filter.connect(gain); gain.connect(this.bassBus);
   395→    if (this.delaySend) { const send = this.ctx.createGain(); send.gain.value = 0.12; gain.connect(send); send.connect(this.delaySend); }
   396→    osc.start(t); osc.stop(t + 0.22);
   397→  }
   398→
   399→  private lead(t: number, freq: number, v: Variant, accent: boolean): void {
   400→    if (!this.ctx || !this.leadBus) return;
   401→    const peakCut = Math.max(200, v.leadCut * (accent ? 1.25 : 1));
   402→    const o1 = this.ctx.createOscillator(), o2 = this.ctx.createOscillator();
   403→    o1.type = v.leadWave; o2.type = v.leadWave;
   404→    o1.frequency.value = freq; o2.frequency.value = freq * Math.pow(2, 9 / 1200);
   405→    const filter = this.ctx.createBiquadFilter(); filter.type = 'lowpass'; filter.Q.value = v.leadQ;
   406→    filter.frequency.setValueAtTime(180, t);
   407→    filter.frequency.exponentialRampToValueAtTime(peakCut, t + 0.02);
   408→    filter.frequency.exponentialRampToValueAtTime(240, t + 0.22);
   409→    const gain = this.ctx.createGain();
   410→    const peak = Math.max(0.05, v.leadLvl * (accent ? 1 : 0.7));
   411→    gain.gain.setValueAtTime(0.0001, t);
   412→    gain.gain.exponentialRampToValueAtTime(peak, t + 0.008);
   413→    gain.gain.exponentialRampToValueAtTime(0.001, t + 0.24);
   414→    o1.connect(filter); o2.connect(filter); filter.connect(gain); gain.connect(this.leadBus);
   415→    if (this.delaySend) { const send = this.ctx.createGain(); send.gain.value = 0.3; gain.connect(send); send.connect(this.delaySend); }
   416→    o1.start(t); o2.start(t); o1.stop(t + 0.26); o2.stop(t + 0.26);
   417→  }

   473→  // ── Scheduler — beat-synced via PLL ──
   474→  // When PLL is locked: schedule steps on predicted radio beats
   475→  // When PLL not locked (solo/no radio): use own clock like psy
   476→  private scheduler(): void {
   477→    if (!this.ctx) return;
   478→    try {
   479→      const now = this.ctx.currentTime;
   480→
   481→      // If PLL is locked, sync to radio beats
   482→      if (this.pll.isLocked() && this.radioOn) {
   483→        // Get predicted beats for the next 200ms
   484→        const beats = this.pll.predictBeats(now, 0.2);
   485→        const period = 60 / this.pll.getBpm();
   486→        const stepDur = period / 4; // 16th notes
   487→
   488→        for (const beatTime of beats) {
   489→          // Schedule 4 sixteenth notes per beat
   490→          for (let s = 0; s < 4; s++) {
   491→            const stepTime = beatTime + s * stepDur;
   492→            if (stepTime > now && stepTime < now + this.scheduleAheadTime) {
   493→              // Only schedule if we haven't already scheduled this step
   494→              const stepKey = Math.round(stepTime * 1000);
   495→              if (stepKey > this.lastScheduledStepKey) {
   496→                this.scheduleStep(this.step % 16, stepTime);
   497→                this.lastScheduledStepKey = stepKey;
   498→                this.step = (this.step + 1) % this.totalSteps;
   499→              }
   500→            }
   501→          }
   502→        }
   503→      } else {
   504→        // No PLL lock — use own clock (like psy)
   505→        while (this.nextNoteTime < now + this.scheduleAheadTime) {
   506→          this.scheduleStep(this.step, this.nextNoteTime);
   507→          this.nextNoteTime += this.stepDur();
   508→          this.step = (this.step + 1) % this.totalSteps;
   509→        }
   510→      }
   511→    } catch (e) {}
   512→  }
   513→
   514→  private lastScheduledStepKey = 0;
   515→
   516→  private scheduleStep(step: number, time: number): void {
   517→    const p = this.getPreset();
   518→    const v = this.getVariant();
   519→    const s16 = step % 16;
   520→
   521→    // ── PATTERN MUTATION: evolve every 8 bars ──
   522→    if (s16 === 0) {
   523→      this.barCount++;
   524→      if (this.barCount % 8 === 0) {
   525→        // Time to mutate!
   526→        const basePattern = this.livePattern || p.patterns;
   527→        const mutated = mutatePattern(basePattern, this.occupancy, this.musicState.density);
   528→        if (mutated) {
   529→          this.livePattern = mutated;
   530→          console.log('[MUTATE] pattern evolved, bar', this.barCount);
   531→        }
   532→      }
   533→    }
   534→
   535→    // Use live pattern if available, else preset pattern
   536→    const pat = this.livePattern || p.patterns;
   537→
   538→    // Use detected root if locked, else preset root
   539→    const root = this.harmonicLocked && this.harmonicRoot ? this.harmonicRoot : p.root;
   540→
   541→    // ── ARRANGER: decide what to play based on occupancy ──
   542→    const kickAvailable = this.occupancy.kick < 0.7;
   543→    const bassAvailable = this.occupancy.bass < 0.75;
   544→    const leadAvailable = this.occupancy.lead < 0.85;
   545→    const density = this.musicState.density;
   546→
   547→    // Kick: only if role is available AND density allows
   548→    if (kickAvailable && pat.kick && pat.kick[s16] && Math.random() < density) {
   549→      this.kick(time);
   550→    }
   551→    // Hats: always available (fills high end)
   552→    if (pat.hat && pat.hat[s16]) this.hat(time, v.hatLvl || 0.1);
   553→
   554→    // Bass: only if role is available
   555→    if (bassAvailable) {
   556→      const bn = pat.bass ? pat.bass[s16] : null;
   557→      if (bn !== null && bn !== undefined) this.bass(time, mtof(root + bn), v);
   558→    }
   559→
   560→    // Lead: only if role is available AND density is high enough
   561→    if (leadAvailable && density > 0.4) {
   562→      const ln = pat.lead ? pat.lead[s16] : null;
   563→      if (ln !== null && ln !== undefined) this.lead(time, mtof(root + 24 + ln), v, s16 % 4 === 0);
   564→    }
   565→  }

   647→  private detect(): void {
   648→    if (!this.radioAnalyser || !this.ctx) return;
   649→    if (!this.radioFreqBuf || this.radioFreqBuf.length !== this.radioAnalyser.frequencyBinCount) {
   650→      this.radioFreqBuf = new Uint8Array(this.radioAnalyser.frequencyBinCount);
   651→    }
   652→    const fd = this.radioFreqBuf;
   653→    this.radioAnalyser.getByteFrequencyData(fd);
   654→
   655→    // ── MELODY OBSERVATION (every 4th detect tick — less CPU) ──
   656→    if (this.detectTickCount % 4 === 0) {
   657→      const tdBuf = this.melodyObserver.ensureTimeDomainBuf(this.radioAnalyser);
   658→      this.radioAnalyser.getFloatTimeDomainData(tdBuf);
   659→      const clock = this.pll.getClock(this.ctx.currentTime);
   660→      this.melodyObserver.observe(
   661→        fd, tdBuf, this.ctx.sampleRate, this.radioAnalyser.fftSize,
   662→        this.ctx.currentTime, clock.beatIndex, clock.barIndex,
   663→        this.occupancy
   664→      );
   665→    }
   666→    this.detectTickCount++;
   667→
   668→    let sub = 0;
   669→    for (let i = 0; i < 10; i++) sub += fd[i];
   670→    sub /= (10 * 255);
   671→
   672→    let total = 0, cnt = 0;
   673→    for (let i = 0; i < fd.length; i += 4) { total += fd[i]; cnt++; }
   674→    total /= (cnt * 255);
   675→    this.radioLevel = total;
   676→    this.radioRms = this.radioRms * 0.85 + total * 0.15;
   677→
   678→    const binHz = this.ctx.sampleRate / this.radioAnalyser.fftSize;
   679→    const lowEnd = Math.floor(250 / binHz);
   680→    const midEnd = Math.floor(2500 / binHz);
   681→    let lo = 0, mi = 0, hi = 0, loN = 0, miN = 0, hiN = 0;
   682→    for (let i = 0; i < lowEnd; i += 2) { lo += fd[i]; loN++; }
   683→    for (let i = lowEnd; i < midEnd; i += 4) { mi += fd[i]; miN++; }
   684→    for (let i = midEnd; i < fd.length; i += 8) { hi += fd[i]; hiN++; }
   685→    this.radioBands = {
   686→      low: lo / (Math.max(1, loN) * 255),
   687→      mid: mi / (Math.max(1, miN) * 255),
   688→      high: hi / (Math.max(1, hiN) * 255),
   689→    };
   690→
   691→    // ── OCCUPANCY ANALYSIS (the key insight) ──
   692→    // Instead of "radio loud/quiet", track WHICH ROLES the radio fills
   693→    // Kick: sub-bass energy + transient strength
   694→    // Bass: bass-band energy + stability
   695→    // Lead: mid-band energy + spectral prominence
   696→    // Hats: high-band energy
   697→    const subEnergy = sub;
   698→    const bassEnergy = lo / (Math.max(1, loN) * 255);
   699→    const midEnergy = mi / (Math.max(1, miN) * 255);
   700→    const highEnergy = hi / (Math.max(1, hiN) * 255);
   701→
   702→    // Smooth occupancy (attack fast, release slow)
   703→    const kickTarget = subEnergy > 0.5 ? subEnergy : subEnergy * 0.3;
   704→    const bassTarget = bassEnergy > 0.5 ? bassEnergy : bassEnergy * 0.3;
   705→    const leadTarget = midEnergy > 0.4 ? midEnergy * 0.8 : midEnergy * 0.2;
   706→    const hatsTarget = highEnergy > 0.3 ? highEnergy * 0.7 : highEnergy * 0.2;
   707→
   708→    // Fast attack, slow release
   709→    this.occupancy.kick += (kickTarget - this.occupancy.kick) * (kickTarget > this.occupancy.kick ? 0.5 : 0.05);
   710→    this.occupancy.bass += (bassTarget - this.occupancy.bass) * (bassTarget > this.occupancy.bass ? 0.3 : 0.05);
   711→    this.occupancy.lead += (leadTarget - this.occupancy.lead) * (leadTarget > this.occupancy.lead ? 0.3 : 0.05);
   712→    this.occupancy.hats += (hatsTarget - this.occupancy.hats) * (hatsTarget > this.occupancy.hats ? 0.3 : 0.05);
   713→
   714→    // ── ROLE DUCKING (not sidechain — per-role gain) ──
   715→    // If radio kick is strong → mute engine kick, reduce bass
   716→    // If radio bass is strong → reduce engine bass
   717→    // If radio lead is strong → reduce engine lead
   718→    if (this.kickBus && this.bassBus && this.leadBus && this.hatBus && this.ctx) {
   719→      const now = this.ctx.currentTime;
   720→      // Kick: if radio kick > 0.7, mute engine kick
   721→      const kickGain = this.occupancy.kick > 0.7 ? 0.05 : 0.9;
   722→      this.kickBus.gain.setTargetAtTime(kickGain, now, 0.03);
   723→      // Bass: if radio bass > 0.75, reduce engine bass
   724→      const bassGain = this.occupancy.bass > 0.75 ? 0.35 : 0.85;
   725→      this.bassBus.gain.setTargetAtTime(bassGain, now, 0.05);
   726→      // Lead: if radio mid > 0.85, reduce engine lead
   727→      const leadGain = this.occupancy.lead > 0.85 ? 0.55 : 0.7;
   728→      this.leadBus.gain.setTargetAtTime(leadGain, now, 0.08);
   729→      // Hats: if radio high > 0.9, reduce engine hats
   730→      const hatGain = this.occupancy.hats > 0.9 ? 0.6 : 0.6;
   731→      this.hatBus.gain.setTargetAtTime(hatGain, now, 0.08);
   732→    }
   733→
   734→    // ── ENERGY HISTORY (for relative energy, not absolute) ──
   735→    this.energyHistory.push(total);
   736→    if (this.energyHistory.length > 32) this.energyHistory.shift();
   737→
   738→    // ── MUSIC STATE UPDATE ──
   739→    // Energy slope (rising/falling/stable)
   740→    if (this.energyHistory.length >= 8) {
   741→      const recent = this.energyHistory.slice(-4).reduce((a, b) => a + b, 0) / 4;
   742→      const older = this.energyHistory.slice(-8, -4).reduce((a, b) => a + b, 0) / 4;
   743→      this.musicState.energy = recent;
   744→      this.musicState.energySlope = recent - older;
   745→    }
   746→
   747→    // Update radio roles in music state
   748→    this.musicState.radioRoles = { ...this.occupancy };
   749→    this.musicState.bpm = this.radioBpm || this.engineBpm;
   750→
   751→    // ── STYLE DETECTION (with hysteresis) ──
   752→    const detectedStyle = this.classifyStyle();
   753→    if (detectedStyle) {
   754→      if (detectedStyle !== this.styleCandidate) {
   755→        this.styleCandidate = detectedStyle;
   756→        this.styleCandidateSince = Date.now();
   757→      }
   758→      // Only switch style after 8 seconds of consistent detection
   759→      if (this.styleCandidate && Date.now() - this.styleCandidateSince > 8000) {
   760→        if (this.styleCandidate !== this.currentStyle) {
   761→          this.currentStyle = this.styleCandidate;
   762→          console.log('[STYLE] switched to:', this.currentStyle);
   763→        }
   764→      }
   765→    }
   766→    this.musicState.style = this.currentStyle;
   767→
   768→    // ── COMPETITIVE DENSITY CONTROL ──
   769→    // If radio energy is rising, reduce engine density
   770→    const delta = this.musicState.energySlope;
   771→    if (delta > 0.18) {
   772→      this.musicState.density = Math.max(0.3, this.musicState.density * 0.75);
   773→    } else if (delta < -0.18) {
   774→      // Radio energy falling — engine can fill more
   775→      this.musicState.density = Math.min(0.9, this.musicState.density * 1.15);
   776→    } else {
   777→      // Stable — drift toward 0.7
   778→      this.musicState.density += (0.7 - this.musicState.density) * 0.05;
   779→    }
   780→
   781→    // Kick detection
   782→    this.subBassHistory.push(sub);
   783→    if (this.subBassHistory.length > 50) this.subBassHistory.shift();
   784→    if (this.subBassHistory.length >= 10) {
   785→      const startIdx = Math.max(0, this.subBassHistory.length - 20);
   786→      let sum = 0, max = 0, count = 0;
   787→      for (let i = startIdx; i < this.subBassHistory.length; i++) {
   788→        const v = this.subBassHistory[i];
   789→        sum += v; if (v > max) max = v; count++;
   790→      }
   791→      const avg = sum / count;
   792→      const threshold = avg + (max - avg) * 0.55;
   793→      const prev = this.subBassHistory[this.subBassHistory.length - 2] || 0;
   794→      if (sub > threshold && prev <= threshold) this.onKick();
   795→    }
   796→
   797→    // Bass freq detection
   798→    if (this.kickCount > 0 && this.kickCount % 8 === 0) {
   799→      const minBin = Math.floor(40 / binHz);
   800→      const maxBin = Math.floor(2000 / binHz);
   801→      let pk = 0, pv = 0;
   802→      for (let i = minBin; i <= maxBin && i < fd.length; i++) {
   803→        if (fd[i] > pv) { pv = fd[i]; pk = i; }
   804→      }
   805→      if (pv > 50) {
   806→        this.bassFreq = pk * binHz;
   807→        if (this.learningData) {
   808→          this.learningData = recordBassNote(this.learningData, this.bassFreq);
   809→          this.learningData = deriveInsights(this.learningData);
   810→          saveLearning(this.learningData);
   811→        }
   812→        const midi = freqToMidi(this.bassFreq);
   813→        this.harmonicRoot = midi - 12;
   814→        this.harmonicLocked = true;
   815→      }
   816→    }
   817→
   818→    // Engine level
   819→    if (this.analyser) {
   820→      const d = new Uint8Array(this.analyser.frequencyBinCount);
   821→      this.analyser.getByteFrequencyData(d);
   822→      let s = 0; for (let i = 0; i < d.length; i++) s += d[i];
   823→      this.engineLevel = s / (d.length * 255);
   824→    }
   825→
   826→    this.emit();
   827→  }

   829→  private onKick(): void {
   830→    const now = this.ctx!.currentTime;
   831→    if (this.lastKickTime > 0 && now - this.lastKickTime < 0.25) return;
   832→    this.kickCount++;
   833→
   834→    // ── FEED BEAT OBSERVATION TO PLL ──
   835→    // PLL handles phase + tempo tracking (replaces old median-based BPM)
   836→    const confidence = Math.min(1, this.radioBands.low * 2);
   837→    this.pll.update({ time: now, confidence });
   838→
   839→    // Sync engine BPM from PLL (smooth, not jumping)
   840→    if (this.pll.isLocked()) {
   841→      const pllBpm = this.pll.getBpm();
   842→      this.radioBpm = Math.round(pllBpm);
   843→      // Smooth engine BPM toward PLL (not jump)
   844→      this.engineBpm = this.engineBpm + (pllBpm - this.engineBpm) * 0.3;
   845→      this.updateDelayTime();
   846→      this.syncStatus = 'following';
   847→      this.updateMixMode();
   848→
   849→      if (this.learningData) {
   850→        this.learningData = recordKick(this.learningData, Math.round(pllBpm));
   851→        this.learningData = deriveInsights(this.learningData);
   852→        saveLearning(this.learningData);
   853→      }
   854→    }
   855→
   856→    this.lastKickTime = now;
   857→    this.emit();
   858→  }

   869→  // ── Style classifier (from architecture review) ──
   870→  private classifyStyle(): Style | null {
   871→    const o = this.occupancy;
   872→    const bpm = this.radioBpm || this.engineBpm;
   873→
   874→    // Full On: high kick + high bass + high highs + fast
   875→    if (o.kick > 0.7 && o.bass > 0.6 && o.hats > 0.4 && bpm > 143) {
   876→      return 'fullOn';
   877→    }
   878→    // Dark: high bass + low highs + slow
   879→    if (o.bass > 0.6 && o.hats < 0.3 && bpm < 142) {
   880→      return 'dark';
   881→    }
   882→    // Progressive: moderate everything + stable energy
   883→    if (o.kick < 0.6 && o.bass > 0.4 && this.musicState.energySlope < 0.1) {
   884→      return 'progressive';
   885→    }
   886→    // Acid: mid-heavy + high energy
   887→    if (o.lead > 0.6 && this.musicState.energy > 0.5) {
   888→      return 'acid';
   889→    }
   890→    return null;
   891→  }
