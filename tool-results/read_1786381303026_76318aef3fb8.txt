     1→     1→/**
     2→     2→ * MelodyEngine — professional melodic development engine.
     3→     3→ *
     4→     4→ * Replaces the simple EvolvingSequence-based LeadMotif with a real
     5→     5→ * developmental melodic engine. Implements the classical techniques used
     6→     6→ * by Beethoven, Bach, and modern film/game composers:
     7→     7→ *
     8→     8→ *   - Motif generation with SINGABLE contour (steps over leaps, chord-tone
     9→     9→ *     start/end, resolution-after-leap, octave+3rd range).
    10→    10→ *   - Development techniques: transpose, invert, retrograde, fragment,
    11→    11→ *     elongate (augmentation), shorten (diminution), sequence.
    12→    12→ *   - Phrase structure: A A' B A'' (8-bar developmental phrases, not AABA).
    13→    13→ *   - Tension curves (low/medium/high/peak) drive register, density,
    14→    14→ *     duration, leap probability.
    15→    15→ *   - Call-response: lead plays a "call", arp plays a "response" (descending,
    16→    16→ *     ending on a stable tone) during variation sections.
    17→    17→ *
    18→    18→ * Design notes:
    19→    19→ *   - Motifs store scale degrees (integers, can be negative/>7) so they
    20→    20→ *     transpose cleanly across any scale.
    21→    21→ *   - Strong beats (downbeats, beat 3) snap to chord tones from
    22→    22→ *     PROGRESSIONS[scale] — this keeps the lead compatible with Track H1's
    23→    23→ *     harmony engine (no chord clashes on strong beats).
    24→    24→ *   - Weak beats allow passing tones / neighbor tones.
    25→    25→ *   - All randomness is driven by the supplied SeededRng (deterministic).
    26→    26→ *
    27→    27→ * Task ID: M1 (Melody track).
    28→    28→ */

   160→   160→  // ─── Motif generation ──────────────────────────────────────────────────
   161→   161→
   162→   162→  /**
   163→   163→   * Generate a fresh motif with SINGABLE contour (4-8 notes).
   164→   164→   *
   165→   165→   * Rules:
   166→   166→   *   - Start on a chord tone (1st/3rd/5th).
   167→   167→   *   - Prefer steps (2nds) over leaps (3rds+).
   168→   168→   *   - After a leap, resolve by step in the opposite direction.
   169→   169→   *   - End on a stable tone (1st/3rd/5th).
   170→   170→   *   - Range: within an octave + a 3rd.
   171→   171→   *   - Duration: mix of 8ths/16ths with occasional longer notes.
   172→   172→   *
   173→   173→   * The contour shape is chosen based on `tension`:
   174→   174→   *   - High tension → ascending / wave (build energy)
   175→   175→   *   - Low tension → descending / wave (release)
   176→   176→   *   - Mid → arch (rise then fall)
   177→   177→   */
   178→   178→  generateMotif(energy: number, tension: number): Motif {
   179→   179→    const numNotes = this.rng.int(4, 8);
   180→   180→    const contour = this.pickContour(tension);
   181→   181→
   182→   182→    // Octave shift: high tension lifts the whole motif up; low tension drops it.
   183→   183→    const octShift = tension > 0.65 ? 7 : tension < 0.3 ? -7 : 0;
   184→   184→    // Singable range (scale degrees). -3 (low 3rd) to 9 (octave + 3rd).
   185→   185→    const minDeg = -3 + (octShift < 0 ? 0 : 0);
   186→   186→    const maxDeg = 9 + (octShift > 0 ? 7 : 0);
   187→   187→
   188→   188→    const notes: number[] = [];
   189→   189→    const durations: number[] = [];
   190→   190→    const velocities: number[] = [];
   191→   191→    const rests: number[] = [];
   192→   192→
   193→   193→    // First note: chord tone.
   194→   194→    const startDeg = this.rng.pick(STABLE_DEGREES) + octShift;
   195→   195→    notes.push(startDeg);
   196→   196→    durations.push(this.pickDuration(tension, true));
   197→   197→    velocities.push(this.pickVelocity(energy, tension, true));
   198→   198→    rests.push(0);
   199→   199→
   200→   200→    let prevDelta = 0;
   201→   201→
   202→   202→    for (let i = 1; i < numNotes; i++) {
   203→   203→      const progress = i / numNotes; // 0..1 through motif
   204→   204→      let delta: number;
   205→   205→
   206→   206→      // After a leap (3rd or larger), 75% chance to resolve by step in opposite direction.
   207→   207→      if (Math.abs(prevDelta) >= 2 && this.rng.chance(0.75)) {
   208→   208→        delta = prevDelta > 0 ? -1 : 1;
   209→   209→      } else {
   210→   210→        // Otherwise: prefer steps. Leap probability rises with tension.
   211→   211→        const leapProb = 0.15 + tension * 0.35;
   212→   212→        if (this.rng.chance(leapProb)) {
   213→   213→          // Leap: 3rd, 4th, 5th (up or down).
   214→   214→          delta = this.rng.pick([2, 2, 3, 4, -2, -2, -3, -4]);
   215→   215→        } else {
   216→   216→          // Step: mostly 2nds, occasionally a repeat or skip.
   217→   217→          delta = this.rng.pick([1, 1, 1, -1, -1, -1, 0, 2, -2]);
   218→   218→        }
   219→   219→        // Bias direction by contour shape.
   220→   220→        delta = this.applyContourBias(delta, contour, progress);
   221→   221→      }
   222→   222→
   223→   223→      let nextDeg = notes[i - 1] + delta;
   224→   224→      // Clamp to singable range — bounce back if exceeded.
   225→   225→      if (nextDeg < minDeg) nextDeg = notes[i - 1] + 1;
   226→   226→      if (nextDeg > maxDeg) nextDeg = notes[i - 1] - 1;
   227→   227→
   228→   228→      notes.push(nextDeg);
   229→   229→      durations.push(this.pickDuration(tension, false));
   230→   230→      velocities.push(this.pickVelocity(energy, tension, false));
   231→   231→      rests.push(0);
   232→   232→      prevDelta = delta;
   233→   233→    }
   234→   234→
   235→   235→    // End on a stable tone (1st, 3rd, 5th) — pick nearest stable degree.
   236→   236→    const lastIdx = notes.length - 1;
   237→   237→    notes[lastIdx] = this.nearestStableDegree(notes[lastIdx]);
   238→   238→
   239→   239→    // Insert occasional rests (low tension → more rests).
   240→   240→    const restProb = 0.05 + (1 - tension) * 0.2;
   241→   241→    for (let i = 1; i < notes.length; i++) {
   242→   242→      if (this.rng.chance(restProb)) {
   243→   243→        rests[i] = 1; // mark as rest
   244→   244→      }
   245→   245→    }
   246→   246→
   247→   247→    return {
   248→   248→      notes,
   249→   249→      durations,
   250→   250→      velocities,
   251→   251→      rests: rests.map((r) => r === 1),
   252→   252→    };
   253→   253→  }
   254→   254→
   255→   255→  /** Choose a contour shape based on tension. */
   256→   256→  private pickContour(tension: number): ContourShape {
   257→   257→    if (tension > 0.8) return this.rng.pick<ContourShape>(['ascending', 'wave']);
   258→   258→    if (tension > 0.6) return this.rng.pick<ContourShape>(['ascending', 'arch']);
   259→   259→    if (tension < 0.3) return this.rng.pick<ContourShape>(['descending', 'wave']);
   260→   260→    return this.rng.pick<ContourShape>(['arch', 'wave', 'descending']);
   261→   261→  }
   262→   262→
   263→   263→  /**
   264→   264→   * Bias a step direction by the chosen contour.
   265→   265→   * - ascending: prefer up-steps
   266→   266→   * - descending: prefer down-steps
   267→   267→   * - arch: up in first half, down in second half
   268→   268→   * - wave: alternate direction every 2 notes
   269→   269→   */
   270→   270→  private applyContourBias(delta: number, contour: ContourShape, progress: number): number {
   271→   271→    switch (contour) {
   272→   272→      case 'ascending':
   273→   273→        if (delta < 0 && this.rng.chance(0.6)) return Math.abs(delta);
   274→   274→        return delta;
   275→   275→      case 'descending':
   276→   276→        if (delta > 0 && this.rng.chance(0.6)) return -Math.abs(delta);
   277→   277→        return delta;
   278→   278→      case 'arch':
   279→   279→        if (progress < 0.5 && delta < 0 && this.rng.chance(0.6)) return Math.abs(delta);
   280→   280→        if (progress >= 0.5 && delta > 0 && this.rng.chance(0.6)) return -Math.abs(delta);
   281→   281→        return delta;
   282→   282→      case 'wave': {
   283→   283→        const phase = Math.floor(progress * 4) % 2;
   284→   284→        if (phase === 0 && delta < 0 && this.rng.chance(0.5)) return Math.abs(delta);
   285→   285→        if (phase === 1 && delta > 0 && this.rng.chance(0.5)) return -Math.abs(delta);
   286→   286→        return delta;
   287→   287→      }
   288→   288→    }
   289→   289→    return delta;
   290→   290→  }

   292→   292→  /** Pick a duration (in 16th steps) based on tension. */
   293→   293→  private pickDuration(tension: number, isFirst: boolean): number {
   294→   294→    // Low tension → slow notes (4-8). Peak → 16ths only.
   295→   295→    if (tension < 0.3) return this.rng.pick([4, 6, 8]);
   296→   296→    if (tension < 0.6) return this.rng.pick([2, 4, 4]);
   297→   297→    if (tension < 0.8) return this.rng.pick([1, 2, 2]);
   298→   298→    // Peak: mostly 16ths, occasional 8th.
   299→   299→    return this.rng.pick([1, 1, 2]);
   300→   300→  }
   301→   301→
   302→   302→  /** Pick a velocity based on energy + whether this is a strong beat. */
   303→   303→  private pickVelocity(energy: number, _tension: number, isStrong: boolean): number {
   304→   304→    const base = isStrong ? 0.7 : 0.5;
   305→   305→    return clamp(base * (0.6 + energy * 0.4), 0.1, 1.0);
   306→   306→  }

   308→   308→  /** Find the nearest stable degree (1st/3rd/5th) to a given scale degree. */
   309→   309→  private nearestStableDegree(deg: number): number {
   310→   310→    let nearest = STABLE_DEGREES[0];
   311→   311→    let nearestDist = Infinity;
   312→   312→    for (const s of STABLE_DEGREES) {
   313→   313→      // Search across octaves (steps of 7 scale degrees).
   314→   314→      for (let oct = -7; oct <= 7; oct += 7) {
   315→   315→        const candidate = s + oct;
   316→   316→        const dist = Math.abs(candidate - deg);
   317→   317→        if (dist < nearestDist) {
   318→   318→          nearestDist = dist;
   319→   319→          nearest = candidate;
   320→   320→        }
   321→   321→      }
   322→   322→    }
   323→   323→    return nearest;
   324→   324→  }
   325→   325→
   326→   326→  /** Find the nearest chord tone (root/3rd/5th of `chordDeg`) to `deg`. */
   327→   327→  private nearestChordTone(deg: number, chordDeg: number): number {
   328→   328→    let nearest = chordDeg;
   329→   329→    let nearestDist = Infinity;
   330→   330→    for (const off of CHORD_TONE_OFFSETS) {
   331→   331→      for (let oct = -7; oct <= 7; oct += 7) {
   332→   332→        const candidate = chordDeg + off + oct;
   333→   333→        const dist = Math.abs(candidate - deg);
   334→   334→        if (dist < nearestDist) {
   335→   335→          nearestDist = dist;
   336→   336→          nearest = candidate;
   337→   337→        }
   338→   338→      }
   339→   339→    }
   340→   340→    return nearest;
   341→   341→  }

   345→   345→  /** Transpose a motif by N scale degrees (clean — stays in scale). */
   346→   346→  transpose(motif: Motif, scaleSteps: number): Motif {
   347→   347→    return {
   348→   348→      notes: motif.notes.map((n) => n + scaleSteps),
   349→   349→      durations: [...motif.durations],
   350→   350→      velocities: [...motif.velocities],
   351→   351→      rests: [...motif.rests],
   352→   352→    };
   353→   353→  }
   354→   354→
   355→   355→  /**
   356→   356→   * Melodic inversion — flip the contour upside-down.
   357→   357→   * First note stays; subsequent deltas are negated.
   358→   358→   */
   359→   359→  invert(motif: Motif): Motif {
   360→   360→    if (motif.notes.length === 0) {
   361→   361→      return { notes: [], durations: [], velocities: [], rests: [] };
   362→   362→    }
   363→   363→    const first = motif.notes[0];
   364→   364→    const invertedNotes = [first];
   365→   365→    for (let i = 1; i < motif.notes.length; i++) {
   366→   366→      const delta = motif.notes[i] - motif.notes[i - 1];
   367→   367→      invertedNotes.push(invertedNotes[i - 1] - delta);
   368→   368→    }
   369→   369→    return {
   370→   370→      notes: invertedNotes,
   371→   371→      durations: [...motif.durations],
   372→   372→      velocities: [...motif.velocities],
   373→   373→      rests: [...motif.rests],
   374→   374→    };
   375→   375→  }
   376→   376→
   377→   377→  /** Retrograde — play the motif backwards. */
   378→   378→  retrograde(motif: Motif): Motif {
   379→   379→    return {
   380→   380→      notes: [...motif.notes].reverse(),
   381→   381→      durations: [...motif.durations].reverse(),
   382→   382→      velocities: [...motif.velocities].reverse(),
   383→   383→      rests: [...motif.rests].reverse(),
   384→   384→    };
   385→   385→  }

   387→   387→  /** Fragment — take a sub-range of the motif (a 2-3 note cell). */
   388→   388→  fragment(motif: Motif, startIdx: number, length: number): Motif {
   389→   389→    const end = Math.min(startIdx + length, motif.notes.length);
   390→   390→    const actualStart = Math.max(0, Math.min(startIdx, motif.notes.length - 1));
   391→   391→    return {
   392→   392→      notes: motif.notes.slice(actualStart, end),
   393→   393→      durations: motif.durations.slice(actualStart, end),
   394→   394→      velocities: motif.velocities.slice(actualStart, end),
   395→   395→      rests: motif.rests.slice(actualStart, end),
   396→   396→    };
   397→   397→  }
   398→   398→
   399→   399→  /** Elongate (rhythmic augmentation) — multiply durations by `factor`. */
   400→   400→  elongate(motif: Motif, factor: number): Motif {
   401→   401→    return {
   402→   402→      notes: [...motif.notes],
   403→   403→      durations: motif.durations.map((d) => Math.max(1, Math.round(d * factor))),
   404→   404→      velocities: [...motif.velocities],
   405→   405→      rests: [...motif.rests],
   406→   406→    };
   407→   407→  }
   408→   408→
   409→   409→  /** Shorten (rhythmic diminution) — divide durations by `factor`. */
   410→   410→  shorten(motif: Motif, factor: number): Motif {
   411→   411→    return this.elongate(motif, 1 / factor);
   412→   412→  }
   413→   413→
   414→   414→  /**
   415→   415→   * Sequence — repeat the motif at successively higher/lower scale degrees.
   416→   416→   * The classic melodic development technique (Beethoven, Bach, film composers).
   417→   417→   * Default shift = 2 scale degrees per repeat (up/down a 3rd).
   418→   418→   */
   419→   419→  sequence(motif: Motif, steps: number, direction: 'up' | 'down'): Motif[] {
   420→   420→    const out: Motif[] = [motif];
   421→   421→    const dir = direction === 'up' ? 1 : -1;
   422→   422→    const shiftPerStep = 2;
   423→   423→    for (let i = 1; i < Math.max(1, steps); i++) {
   424→   424→      out.push(this.transpose(motif, dir * shiftPerStep * i));
   425→   425→    }
   426→   426→    return out;
   427→   427→  }

   429→   429→  // ─── Call / Response ────────────────────────────────────────────────────
   430→   430→
   431→   431→  /**
   432→   432→   * Generate a "response" motif that answers a "call" motif.
   433→   433→   * The response:
   434→   434→   *   - Inverts the call's contour (call ascending → response descending).
   435→   435→   *   - Ends on the root (most stable tone).
   436→   436→   *   - Shortens durations for a lighter, "answering" feel.
   437→   437→   */
   438→   438→  generateResponse(prevPhrase: Motif): Motif {
   439→   439→    const inverted = this.invert(prevPhrase);
   440→   440→    // Force the response to end on the root (degree 0) — definitive "answer".
   441→   441→    if (inverted.notes.length > 0) {
   442→   442→      inverted.notes[inverted.notes.length - 1] = 0;
   443→   443→    }
   444→   444→    // Lighter rhythm (twice as fast).
   445→   445→    const response = this.shorten(inverted, 2);
   446→   446→    // Slightly lower velocity for the response (counter-melody feel).
   447→   447→    response.velocities = response.velocities.map((v) => clamp(v * 0.8, 0.1, 1.0));
   448→   448→    return response;
   449→   449→  }
   450→   450→
   451→   451→  // ─── Phrase structure: A A' B A'' ───────────────────────────────────────
   452→   452→
   453→   453→  /**
   454→   454→   * Force a new phrase at section boundaries.
   455→   455→   * Computes internal tension from energy + phraseCount, generates a fresh
   456→   456→   * call motif, and rebuilds the 8-bar developmental phrase.
   457→   457→   */
   458→   458→  newPhrase(energy: number): void {
   459→   459→    this.phraseCount++;
   460→   460→    this.lastEnergy = energy;
   461→   461→    const tension = this.computeTension(energy, this.phraseCount);
   462→   462→    this.lastTension = tension;
   463→   463→    this.previousMotif = this.currentMotif;
   464→   464→    this.currentMotif = this.generateMotif(energy, tension);
   465→   465→    this.buildPhrase(energy, tension);
   466→   466→  }
   467→   467→
   468→   468→  /**
   469→   469→   * Tension curves — maps energy (0..1) to melodic behavior.
   470→   470→   *   - Low (0..0.3): slow, low register, consonant, rests.
   471→   471→   *   - Medium (0.3..0.6): mid register, mostly steps.
   472→   472→   *   - High (0.6..0.8): faster, ascending sequences, leaps.
   473→   473→   *   - Peak (0.8..1.0): fastest, highest register, climbing sequences.
   474→   474→   *
   475→   475→   * We add a small periodic variation so consecutive phrases don't all hit
   476→   476→   * the same tension peak even at the same energy level.
   477→   477→   */
   478→   478→  private computeTension(energy: number, phraseCount: number): number {
   479→   479→    const phase = (phraseCount % 4) / 4;
   480→   480→    const variation = 0.12 * Math.sin(2 * Math.PI * phase);
   481→   481→    return clamp(energy + variation, 0, 1);
   482→   482→  }
   483→   483→
   484→   484→  /**
   485→   485→   * Build the 8-bar phrase events from the current motif.
   486→   486→   *   - A (bars 0-1): state the motif.
   487→   487→   *   - A' (bars 2-3): variation — transpose up a 3rd OR fragment & repeat.
   488→   488→   *   - B (bars 4-5): contrasting motif (fresh, different contour).
   489→   489→   *   - A'' (bars 6-7): return + development — augment + sequence up.
   490→   490→   */
   491→   491→  private buildPhrase(energy: number, tension: number): void {
   492→   492→    // Clear event tables.
   493→   493→    this.phraseEvents = new Array(this.phraseSteps).fill(null);
   494→   494→    this.responseEvents = new Array(this.phraseSteps).fill(null);
   495→   495→
   496→   496→    // A — original motif.
   497→   497→    const sectionA = this.currentMotif;
   498→   498→
   499→   499→    // A' — variation: transpose up a 3rd (2 scale degrees) OR fragment+repeat.
   500→   500→    const aPrime = this.rng.chance(0.5)
   501→   501→      ? this.transpose(sectionA, 2)
   502→   502→      : this.repeatFragment(sectionA);
   503→   503→
   504→   504→    // B — contrasting motif (fresh, with higher tension for contrast).
   505→   505→    const sectionB = this.generateMotif(energy, clamp(tension + 0.2, 0, 1));
   506→   506→
   507→   507→    // A'' — augmented + sequenced up (slower, then sequenced).
   508→   508→    const augmented = this.elongate(sectionA, 2);
   509→   509→    const sequenceChain = this.sequence(augmented, 2, 'up');
   510→   510→    const aDoublePrime = sequenceChain[1] ?? augmented;
   511→   511→
   512→   512→    // Place each section at its bar range.
   513→   513→    this.placeMotifInPhrase(sectionA, 0, this.phraseEvents, true);
   514→   514→    this.placeMotifInPhrase(aPrime, 2, this.phraseEvents, true);
   515→   515→    this.placeMotifInPhrase(sectionB, 4, this.phraseEvents, true);
   516→   516→    this.placeMotifInPhrase(aDoublePrime, 6, this.phraseEvents, true);
   517→   517→
   518→   518→    // Cache the call motif and build the arp's response.
   519→   519→    this.lastCallMotif = sectionA;
   520→   520→    this.buildResponseEvents(energy, tension);
   521→   521→  }
   522→   522→
   523→   523→  /** Repeat a 2-3 note fragment to fill A' — classic motivic development. */
   524→   524→  private repeatFragment(motif: Motif): Motif {
   525→   525→    const fragLen = Math.min(3, motif.notes.length);
   526→   526→    const frag = this.fragment(motif, 0, fragLen);
   527→   527→    // Repeat the fragment 2-3 times to fill the section.
   528→   528→    const repeats = Math.max(2, Math.floor(8 / Math.max(1, frag.notes.length)));
   529→   529→    const notes: number[] = [];
   530→   530→    const durations: number[] = [];
   531→   531→    const velocities: number[] = [];
   532→   532→    const rests: boolean[] = [];
   533→   533→    for (let r = 0; r < repeats; r++) {
   534→   534→      // Optional transpose on each repeat for sequence feel.
   535→   535→      const transpose = r * 1;
   536→   536→      for (let i = 0; i < frag.notes.length; i++) {
   537→   537→        notes.push(frag.notes[i] + transpose);
   538→   538→        durations.push(frag.durations[i]);
   539→   539→        velocities.push(frag.velocities[i]);
   540→   540→        rests.push(frag.rests[i]);
   541→   541→      }
   542→   542→    }
   543→   543→    return { notes, durations, velocities, rests };
   544→   544→  }
   545→   545→
   546→   546→  /**
   547→   547→   * Place a motif's events into the phrase event table starting at `startBar`.
   548→   548→   * Snaps strong beats to chord tones (from PROGRESSIONS[scale]) so the lead
   549→   549→   * never clashes with the harmony engine's chord changes.
   550→   550→   */
   551→   551→  private placeMotifInPhrase(
   552→   552→    motif: Motif,
   553→   553→    startBar: number,
   554→   554→    events: (MotifEvent | null)[],
   555→   555→    snapChordTones: boolean,
   556→   556→  ): void {
   557→   557→    const sectionBars = 2;
   558→   558→    const startStep = startBar * this.stepsPerBar;
   559→   559→    const endStep = startStep + sectionBars * this.stepsPerBar;
   560→   560→    const prog = PROGRESSIONS[this.scale] || PROGRESSIONS.minor;
   561→   561→    let curStep = startStep;
   562→   562→
   563→   563→    for (let i = 0; i < motif.notes.length; i++) {
   564→   564→      if (curStep >= endStep) break;
   565→   565→      if (curStep >= this.phraseSteps) break;
   566→   566→
   567→   567→      const dur = Math.max(1, motif.durations[i]);
   568→   568→      const isRest = motif.rests[i];
   569→   569→
   570→   570→      if (!isRest) {
   571→   571→        let deg = motif.notes[i];
   572→   572→
   573→   573→        if (snapChordTones) {
   574→   574→          // Determine which bar (in absolute phrase terms) we're in.
   575→   575→          const barWithinPhrase = Math.floor(curStep / this.stepsPerBar);
   576→   576→          const stepWithinBar = curStep % this.stepsPerBar;
   577→   577→          const chordDeg = prog[barWithinPhrase % prog.length];
   578→   578→
   579→   579→          if (stepWithinBar === 0) {
   580→   580→            // Downbeat → snap to nearest chord tone.
   581→   581→            deg = this.nearestChordTone(deg, chordDeg);
   582→   582→          } else if (stepWithinBar === 8 && this.rng.chance(0.5)) {
   583→   583→            // Beat 3 → snap to a chord tone (3rd or 5th).
   584→   584→            deg = this.rng.pick([chordDeg + 2, chordDeg + 4]);
   585→   585→          }
   586→   586→          // Otherwise: keep the motif's scale degree (passing/neighbor tone).
   587→   587→        }
   588→   588→
   589→   589→        events[curStep] = {
   590→   590→          stepInPhrase: curStep,
   591→   591→          scaleDeg: deg,
   592→   592→          velocity: motif.velocities[i],
   593→   593→          // Cap duration at 4 (quarter note) to avoid synth-gate overflow.
   594→   594→          duration: Math.min(dur, 4),
   595→   595→          isRest: false,
   596→   596→        };
   597→   597→      }
   598→   598→
   599→   599→      curStep += dur;
   600→   600→    }
   601→   601→  }
   602→   602→
   603→   603→  /**
   604→   604→   * Build the arp's response events from the last call motif.
   605→   605→   * The response plays in bars 4-7 of the phrase (the "answer" portion),
   606→   606→   * typically during VARIATION sections.
   607→   607→   */
   608→   608→  private buildResponseEvents(energy: number, tension: number): void {
   609→   609→    this.responseEvents = new Array(this.phraseSteps).fill(null);
   610→   610→    if (!this.lastCallMotif) return;
   611→   611→
   612→   612→    const response = this.generateResponse(this.lastCallMotif);
   613→   613→    // Place the response starting at bar 4 (the B / A'' region).
   614→   614→    // No chord-tone snapping — the response is a counter-melody and can
   615→   615→    // use passing tones freely.
   616→   616→    this.placeMotifInPhrase(response, 4, this.responseEvents, false);
   617→   617→  }
   618→   618→
   619→   619→  // ─── Per-step note lookup ───────────────────────────────────────────────
   620→   620→
   621→   621→  /**
   622→   622→   * Get the next lead note for the given step/bar/energy.
   623→   623→   * Returns null for rests or steps with no scheduled event.
   624→   624→   *
   625→   625→   * The lead plays at root+12 (one octave above the bass root).
   626→   626→   *
   627→   627→   * Task V2c: if a HarmonyEngine is linked, the note is re-checked on strong
   628→   628→   * beats (step % 4 === 0) against the LIVE chord the pad is playing. If the
   629→   629→   * note's pitch class is NOT a chord tone, it's snapped to the nearest chord
   630→   630→   * tone (preserving the original octave as closely as possible). This
   631→   631→   * eliminates dissonance between the lead and the pad — the lead always
   632→   632→   * plays a chord tone on strong beats, regardless of which chord the pad's
   633→   633→   * progression has reached.
   634→   634→   *
   635→   635→   * On weak beats (step % 4 !== 0), the original note is preserved —
   636→   636→   * passing tones / neighbor tones on weak beats are musically valid (they
   637→   637→   * create tension that resolves on the next strong beat).
   638→   638→   */
   639→   639→  nextNote(
   640→   640→    step: number,
   641→   641→    bar: number,
   642→   642→    energy: number,
   643→   643→  ): { note: number; velocity: number; duration: number } | null {
   644→   644→    const phraseBar = ((bar % this.phraseBars) + this.phraseBars) % this.phraseBars;
   645→   645→    const phraseStep = phraseBar * this.stepsPerBar + (step % this.stepsPerBar);
   646→   646→
   647→   647→    const event = this.phraseEvents[phraseStep];
   648→   648→    if (!event || event.isRest) return null;
   649→   649→
   650→   650→    let midi = scaleNote(this.root + 12, this.scale, event.scaleDeg);
   651→   651→
   652→   652→    // ── Task V2c: melody-harmony synchronization ──
   653→   653→    // On strong beats (every quarter note = step % 4 === 0), snap the note
   654→   654→    // to the nearest chord tone of the LIVE chord (the one the pad is
   655→   655→    // playing right now). This makes the lead harmonize with the pad on
   656→   656→    // every downbeat — no clashing non-chord tones on strong beats.
   657→   657→    if (this.harmony) {
   658→   658→      const isStrongBeat = step % 4 === 0;
   659→   659→      if (isStrongBeat) {
   660→   660→        midi = this.snapToLiveChordTone(midi);
   661→   661→      }
   662→   662→    }
   663→   663→
   664→   664→    const vel = clamp(event.velocity * (0.6 + energy * 0.4), 0.1, 1.0);
   665→   665→    return { note: midi, velocity: vel, duration: event.duration };
   666→   666→  }
   667→   667→
   668→   668→  /**
   669→   669→   * Task V2c: snap a MIDI note to the nearest chord tone of the live chord.
   670→   670→   *
   671→   671→   * If the note is already a chord tone (any octave), it's returned unchanged.
   672→   672→   * Otherwise, the nearest chord-tone pitch class is found, and the note is
   673→   673→   * shifted to the octave that minimizes movement from the original note
   674→   674→   * (so the melodic contour is preserved as much as possible).
   675→   675→   *
   676→   676→   * If the harmony engine has no current chord (e.g. before the first chord
   677→   677→   * plays, or outside a lead section), the note is returned unchanged — the
   678→   678→   * static PROGRESSIONS[scale] snapping from placeMotifInPhrase still applies.
   679→   679→   */
   680→   680→  private snapToLiveChordTone(midi: number): number {
   681→   681→    if (!this.harmony) return midi;
   682→   682→    const chord = this.harmony.getCurrentChord();
   683→   683→    if (!chord || chord.notes.length === 0) return midi;
   684→   684→
   685→   685→    // Pitch classes of the live chord.
   686→   686→    const chordPCs = chord.notes.map(n => ((n % 12) + 12) % 12);
   687→   687→    const midiPC = ((midi % 12) + 12) % 12;
   688→   688→
   689→   689→    // Already a chord tone? Leave it alone — preserves melodic identity.
   690→   690→    if (chordPCs.includes(midiPC)) return midi;
   691→   691→
   692→   692→    // Find the nearest chord-tone pitch class (chromatic distance, wraparound).
   693→   693→    let nearestPC = chordPCs[0];
   694→   694→    let nearestDist = Infinity;
   695→   695→    for (const pc of chordPCs) {
   696→   696→      const d = Math.min(
   697→   697→        Math.abs(pc - midiPC),
   698→   698→        12 - Math.abs(pc - midiPC),
   699→   699→      );
   700→   700→      if (d < nearestDist) {
   701→   701→        nearestDist = d;
   702→   702→        nearestPC = pc;
   703→   703→      }
   704→   704→    }
   705→   705→
   706→   706→    // Place the snapped pitch class at the octave closest to the original
   707→   707→    // note. We pick the octave so the result is within a half-octave of the
   708→   708→    // original — this keeps the melodic contour smooth (no jumps).
   709→   709→    const octaveBase = midi - midiPC; // midi with PC zeroed out
   710→   710→    let snapped = octaveBase + nearestPC;
   711→   711→    // If we wrapped past 0/11, snap may be off by an octave — correct it.
   712→   712→    while (snapped - midi > 6) snapped -= 12;
   713→   713→    while (midi - snapped > 6) snapped += 12;
   714→   714→
   715→   715→    // Clamp to a sane lead range (don't snap above MIDI 96 or below 36).
   716→   716→    while (snapped > 96) snapped -= 12;
   717→   717→    while (snapped < 36) snapped += 12;
   718→   718→
   719→   719→    return snapped;
   720→   720→  }
   721→   721→
   722→   722→  /**
   723→   723→   * Get the next response note for the arp (counter-melody to the lead).
   724→   724→   * Returns null when no response event is scheduled.
   725→   725→   *
   726→   726→   * The arp plays at root+24 (two octaves above the bass root) — brighter
   727→   727→   * register so it sits above the lead as a true counter-melody.
   728→   728→   */
   729→   729→  nextResponseNote(
   730→   730→    step: number,
   731→   731→    bar: number,
   732→   732→    energy: number,
   733→   733→  ): { note: number; velocity: number; duration: number } | null {
   734→   734→    const phraseBar = ((bar % this.phraseBars) + this.phraseBars) % this.phraseBars;
   735→   735→    const phraseStep = phraseBar * this.stepsPerBar + (step % this.stepsPerBar);
   736→   736→
   737→   737→    const event = this.responseEvents[phraseStep];
   738→   738→    if (!event || event.isRest) return null;
   739→   739→
   740→   740→    const midi = scaleNote(this.root + 24, this.scale, event.scaleDeg);
   741→   741→    const vel = clamp(event.velocity * (0.55 + energy * 0.35), 0.1, 1.0);
   742→   742→    return { note: midi, velocity: vel, duration: event.duration };
   743→   743→  }
   744→   744→
   745→   745→  // ─── Incremental evolution ──────────────────────────────────────────────
   746→   746→
   747→   747→  /**
   748→   748→   * Per-bar evolution tick. Called from the engine's scheduler every bar.
   749→   749→   * Unlike LeadMotif (which mutated one note at a time), MelodyEngine
   750→   750→   * refreshes the B section (contrasting motif) at intervals — this gives
   751→   751→   * fresh variation without losing the A-section identity.
   752→   752→   *
   753→   753→   * Effective interval shrinks as `evolutionRate` grows (faster evolution
   754→   754→   * for more dynamic worlds like Goa / acid-psy).
   755→   755→   */
   756→   756→  tickEvolution(bar: number, evolutionRate = 0.4, intervalBars = 8): void {
   757→   757→    if (bar <= 0) return;
   758→   758→    if (bar <= this.lastEvolveBar) return;
   759→   759→    const effectiveInterval = Math.max(4, Math.round(intervalBars * (1.2 - evolutionRate)));
   760→   760→    if (bar % effectiveInterval !== 0) return;
   761→   761→    this.lastEvolveBar = bar;
   762→   762→    this.regenerateBSection();
   763→   763→  }
   764→   764→
   765→   765→  /** Refresh only the B section (bars 4-5) with a new contrasting motif. */
   766→   766→  private regenerateBSection(): void {
   767→   767→    // Clear B section events.
   768→   768→    for (let s = 4 * this.stepsPerBar; s < 6 * this.stepsPerBar; s++) {
   769→   769→      this.phraseEvents[s] = null;
   770→   770→    }
   771→   771→    const tension = clamp(this.lastTension + 0.15, 0, 1);
   772→   772→    const newB = this.generateMotif(this.lastEnergy, tension);
   773→   773→    this.placeMotifInPhrase(newB, 4, this.phraseEvents, true);
   774→   774→  }

   780→   780→    return {

   782→   782→      durations: [...this.currentMotif.durations],
   783→   783→      velocities: [...this.currentMotif.velocities],
   784→   784→      rests: [...this.currentMotif.rests],
   785→   785→    };
   786→   786→  }

   791→   791→    return {

   794→   794→      velocities: [...this.previousMotif.velocities],

   796→   796→    };
   797→   797→  }
