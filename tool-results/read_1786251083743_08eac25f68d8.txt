    58→const mtof = (m: number) => 440 * Math.pow(2, (m - 69) / 12);
    59→function scaleNote(root: number, scale: string, deg: number): number {
    60→  const sc = SCALES[scale] || SCALES.minor;
    61→  const n = sc.length, o = Math.floor(deg / n);
    62→  return root + 12 * o + sc[((deg % n) + n) % n];
    63→}

    74→// ─── RNG ────────────────────────────────────────────────────────
    75→
    76→class Rng {
    77→  s: number;
    78→  constructor(seed: number) { this.s = seed >>> 0; }
    79→  next(): number { this.s = (this.s * 1664525 + 1013904223) >>> 0; return this.s / 4294967296; }
    80→  int(min: number, max: number): number { return Math.floor(this.next() * (max - min + 1)) + min; }
    81→  pick<T>(a: T[]): T { return a[Math.floor(this.next() * a.length)]; }
    82→  chance(p: number): boolean { return this.next() < p; }
    83→  gauss(m: number, sd: number): number { return m + sd * (this.next() + this.next() + this.next() - 1.5); }
    84→}

    86→// ─── Motif (melodic identity with AABA structure) ───────────────
    87→
    88→class Motif {
    89→  private notes: number[] = [];
    90→  private rhythm: number[] = [];
    91→  private pos = 0;
    92→  private variationCount = 0;
    93→
    94→  constructor(root: number, scale: string, rng: Rng) {
    95→    // Generate a 4-note motif with contour (AAB A' structure)
    96→    // A: 2 notes, A: repeat with variation, B: contrasting, A': return
    97→    const contour = rng.pick([[1, 1, -2, 1], [2, -1, 1, -1], [-1, 2, -1, 1], [1, -1, 2, 0]]);
    98→    let prev = 0;
    99→    for (let i = 0; i < 4; i++) {
   100→      prev = Math.max(-3, Math.min(5, prev + contour[i]));
   101→      this.notes.push(prev);
   102→    }
   103→    // Rhythm: downbeat + offbeat + syncopated
   104→    this.rhythm = [0, 4, 8, 10];
   105→  }
   106→
   107→  next(): { degree: number; step: number } {
   108→    const n = this.notes[this.pos];
   109→    const r = this.rhythm[this.pos];
   110→    this.pos = (this.pos + 1) % 4;
   111→    return { degree: n, step: r };
   112→  }
   113→
   114→  /** Mutate one note slightly (preserve identity). */
   115→  mutate(rng: Rng) {
   116→    if (++this.variationCount % 4 === 0) {
   117→      const idx = rng.int(0, 3);
   118→      this.notes[idx] = Math.max(-3, Math.min(5, this.notes[idx] + rng.pick([-1, 1])));
   119→    }
   120→  }
   121→}

   409→    this.voiceSpecs = getVoiceSpecs(this.world.id);
   410→
   411→    // Load sound bank samples
   412→    this.soundBank.init(c).then(() => {
   413→      console.log('SoundBank loaded:', this.soundBank.listLoaded());
   414→    }).catch(e => console.warn('SoundBank load failed:', e));
   415→
   416→    // ── MASTER CHAIN (professional, from PSY3 style_master.py) ──
   417→    // sum → duck → HP → multiband → glue → saturation → truePeak → EQ → master → destination
   418→    this.sum = c.createGain();
   419→    this.duck = c.createGain();
   420→    
   421→    // HP on master (remove subsonic)
   422→    const masterHP = c.createBiquadFilter();

   424→    masterHP.frequency.value = 25;
   425→    masterHP.Q.value = 0.707;
   426→    
   427→    // DC blocker
   428→    const dcBlock = c.createWaveShaper();

   431→    
   432→    // EQ shelves (tonal shaping)
   433→    this.eqL = c.createBiquadFilter(); this.eqL.type = 'lowshelf';
   434→    this.eqL.frequency.value = 80; this.eqL.gain.value = 2;
   435→    this.eqH = c.createBiquadFilter(); this.eqH.type = 'highshelf';
   436→    this.eqH.frequency.value = 10000; this.eqH.gain.value = 1.5;
   437→    
   438→    // Professional processing nodes
   439→    this.multiband = new MultibandCompressor(c);
   440→    this.glue = new GlueCompressor(c);
   441→    this.saturation = new MasterSaturation(c, 1.15, 0.15);
   442→    this.truePeak = new TruePeakLimiter(c, 0.94);
   443→    
   444→    this.master = c.createGain(); this.master.gain.value = 0.88;
   445→
   446→    // Chain: sum → duck → HP → multiband → glue → saturation → truePeak → EQ → master → destination
   447→    this.sum.connect(this.duck);
   448→    this.duck.connect(masterHP);
   449→    masterHP.connect(this.multiband.inputNode);
   450→    this.multiband.connect(this.glue.inputNode);
   451→    this.glue.connect(this.saturation.inputNode);
   452→    this.saturation.connect(this.truePeak.inputNode);
   453→    this.truePeak.connect(this.eqL);
   454→    this.eqL.connect(this.eqH);
   455→    this.eqH.connect(this.master);
   456→    this.master.connect(c.destination);
   457→
   458→    this.analyser = c.createAnalyser(); this.analyser.fftSize = 2048;
   459→    this.master.connect(this.analyser);

   462→    // Real Moog ladder filter + band-limited saw oscillator.
   463→    // Voices check this.workletsReady and fall back to BiquadFilter until loaded.
   464→    this.workletLoadPromise = ensureWorkletsLoaded(c);
   465→    this.workletLoadPromise.then((ok) => { this.workletsReady = ok; });
   466→

   468→    // When ready, ALL synthesis happens in the audio thread.
   469→    // Zero per-hit node creation, sample-accurate timing.
   470→    this.engineNode = new Psy4EngineNode(c);
   471→    this.engineNode.init().then(async (ok) => {
   472→      this.useWorkletEngine = ok;
   473→      if (ok) {

   478→          engOut.connect(this.analyser);

   481→        this.sendWorldParamsToEngine();

   485→        // This is the key sound quality upgrade: the worklet plays the REAL
   486→        // kick.wav, hat_closed.wav, hat_open.wav, clap.wav samples instead of
   487→        // pure synth DSP. Samples are transferred as Float32Array (zero-copy).
   488→        this.sampleBank = new SampleBank(c);
   489→        const loaded = await this.sampleBank.loadAll();

   512→

   514→      }

   518→    // Each bus: input → lowShelf EQ → highShelf EQ → comp → saturation → out → sum
   519→    // This gives per-group tonal shaping + glue compression + harmonic cohesion.

   539→      comp.attack.value = 0.005; comp.release.value = 0.1; comp.knee.value = 4;

   541→      const satCurve = new Float32Array(1024);
   542→      for (let i = 0; i < 1024; i++) { const x = (i / 512) - 1; satCurve[i] = Math.tanh(x * cfg.satDrive); }
   543→      sat.curve = satCurve;
   544→      const out = c.createGain();
   545→      out.gain.value = Math.pow(10, cfg.makeupDb / 20);
   546→      input.connect(lowShelf); lowShelf.connect(highShelf);
   547→      highShelf.connect(comp); comp.connect(sat); sat.connect(out);
   548→      out.connect(this.sum);
   549→      this.buses.set(name, { input, comp, out });
   550→    }
   551→

   553→    // Each voice gets: input → HP filter → gain → pan → bus → sum
   554→    //                                → reverbSend → reverb
   555→    //                                → delaySend → delay
   556→    if (this.voiceSpecs) {
   557→      for (const [name, strip] of Object.entries(this.voiceSpecs.channels)) {
   558→        const input = c.createGain();
   559→        const hp = c.createBiquadFilter();
   560→        hp.type = 'highpass';
   561→        hp.frequency.value = strip.hpFreq;
   562→        const gain = c.createGain();
   563→        gain.gain.value = Math.pow(10, strip.gainDb / 20);

   567→        reverbSend.gain.value = strip.reverbSend;
   568→        const delaySend = c.createGain();
   569→        delaySend.gain.value = strip.delaySend;
   570→
   571→        // Chain: input → hp → gain → panner → bus (group-based routing)
   572→        input.connect(hp); hp.connect(gain); gain.connect(panner);
   573→        panner.connect(this.busForChannel(name));
   574→        // Sends (post-fader)
   575→        gain.connect(reverbSend); gain.connect(delaySend);
   576→
   577→        this.channelStrips.set(name, { input, hp, gain, reverbSend, delaySend, panner });
   578→      }
   579→    }
   580→
   581→    // stereo delay (ping-pong)
   582→    this.dSend = c.createGain();
   583→    const dL = c.createDelay(2), dR = c.createDelay(2);
   584→    dL.delayTime.value = 0.23; dR.delayTime.value = 0.31;
   585→    const dF = c.createBiquadFilter(); dF.type = 'lowpass'; dF.frequency.value = 3500;
   586→    const dFb = c.createGain(); dFb.gain.value = 0.35;
   587→    this.dOut = c.createGain(); this.dOut.gain.value = 0.3;
   588→    this.dSend.connect(dL); dL.connect(dF); dF.connect(dR); dR.connect(dFb); dFb.connect(dL);
   589→    const pl = c.createStereoPanner(); pl.pan.value = -0.5;
   590→    const pr = c.createStereoPanner(); pr.pan.value = 0.5;
   591→    dL.connect(pl); dR.connect(pr); pl.connect(this.dOut); pr.connect(this.dOut);
   592→    this.dOut.connect(this.sum);
   593→
   594→    // reverb
   595→    this.rSend = c.createGain(); this.rSend.gain.value = 0.25;
   596→    this.conv = c.createConvolver(); this.conv.buffer = this.makeImpulse(2.2, 2.5);
   597→    this.rSend.connect(this.conv); this.conv.connect(this.sum);
   598→
   599→    // Connect channel strip sends to FX returns
   600→    for (const [, strip] of this.channelStrips) {
   601→      strip.reverbSend.connect(this.rSend);
   602→      strip.delaySend.connect(this.dSend!);
   603→    }
   604→
   605→    // pre-generate buffers + waves
   606→    this.pink = this.makePink();
   607→    this.sawWave = this.makeWave('saw', 48);
   608→    this.sqWave = this.makeWave('square', 48);
   609→    this.triWave = this.makeWave('triangle', 48);
   610→  }

   635→  /**
   636→   * Create a Moog filter (real 4-stage tanh ladder via AudioWorklet if loaded,
   637→   * else fall back to BiquadFilter approximation). Returns a node with
   638→   * .inputNode / .outputNode / .cutoff / .resonance / .drive / .level.
   639→   */
   640→  private createVoiceFilter(opts: {
   641→    cutoff?: number; resonance?: number; drive?: number; level?: number;
   642→  }): { inputNode: AudioNode; outputNode: AudioNode; cutoff: AudioParam; resonance: AudioParam; drive: AudioParam; level: AudioParam; scheduleCutoff?: (t: number, start: number, end: number, dur: number) => void; dispose?: () => void } {
   643→    const c = this.ctx!;
   644→    if (this.workletsReady) {
   645→      const node = createMoogFilter(c, opts);
   646→      if (node) {
   647→        const n = node as MoogFilterNode;
   648→        return {
   649→          inputNode: n, outputNode: n,
   650→          cutoff: n.cutoff, resonance: n.resonance, drive: n.drive, level: n.level,
   651→          scheduleCutoff: (t, start, end, dur) => {
   652→            n.cutoff.cancelScheduledValues(t);
   653→            n.cutoff.setValueAtTime(start, t);
   654→            n.cutoff.exponentialRampToValueAtTime(Math.max(20, end), t + dur);
   655→          },
   656→          dispose: () => { try { n.disconnect(); } catch { /* noop */ } },
   657→        };
   658→      }
   659→    }
   660→    // Fallback: BiquadFilter + WaveShaper approximation
   661→    return this.createFallbackFilter(opts);
   662→  }
   663→
   664→  /** BiquadFilter + WaveShaper fallback (used until worklets load). */
   665→  private createFallbackFilter(opts: {
   666→    cutoff?: number; resonance?: number; drive?: number; level?: number;
   667→  }): { inputNode: AudioNode; outputNode: AudioNode; cutoff: AudioParam; resonance: AudioParam; drive: AudioParam; level: AudioParam; scheduleCutoff?: (t: number, start: number, end: number, dur: number) => void; dispose?: () => void } {
   668→    const c = this.ctx!;
   669→    const chain = new MoogFilterChain(c);
   670→    chain.setParams({
   671→      cutoff: opts.cutoff ?? 1000,
   672→      resonance: opts.resonance ?? 0.3,
   673→      drive: opts.drive ?? 1,
   674→      level: opts.level ?? 1,
   675→    });
   676→    return {
   677→      inputNode: chain.inputNode,
   678→      outputNode: chain.outputNode,
   679→      cutoff: (chain as unknown as { filter: BiquadFilterNode }).filter.frequency,
   680→      resonance: (chain as unknown as { filter: BiquadFilterNode }).filter.Q,
   681→      // drive/level on fallback aren't AudioParams (WaveShaper can't automate) — return dummy
   682→      drive: { value: opts.drive ?? 1, setValueAtTime: () => {}, setTargetAtTime: () => {}, linearRampToValueAtTime: () => {}, exponentialRampToValueAtTime: () => {}, cancelScheduledValues: () => {}, cancelAndHoldAtTime: () => {} } as unknown as AudioParam,
   683→      level: { value: opts.level ?? 1, setValueAtTime: () => {}, setTargetAtTime: () => {}, linearRampToValueAtTime: () => {}, exponentialRampToValueAtTime: () => {}, cancelScheduledValues: () => {}, cancelAndHoldAtTime: () => {} } as unknown as AudioParam,
   684→      scheduleCutoff: (t, start, end, dur) => chain.scheduleCutoff(t, start, end, dur),
   685→      dispose: () => { try { chain.disconnect(); } catch { /* noop */ } },
   686→    };
   687→  }

   689→  private makeWave(type: string, nH: number): PeriodicWave {
   690→    const c = this.ctx!;
   691→    const real = new Float32Array(nH + 1), imag = new Float32Array(nH + 1);
   692→    for (let k = 1; k <= nH; k++) {
   693→      if (type === 'saw') imag[k] = 2 / (Math.PI * k);
   694→      else if (type === 'square') imag[k] = (k % 2) ? 4 / (Math.PI * k) : 0;
   695→      else if (type === 'triangle') { const s = k % 2 ? 1 : -1; real[k] = s * 8 / (Math.PI * Math.PI * k * k); }
   696→    }
   697→    return c.createPeriodicWave(real, imag);
   698→  }

   700→  private makePink(): AudioBuffer {
   701→    const c = this.ctx!, n = c.sampleRate * 2, b = c.createBuffer(2, n, c.sampleRate);
   702→    for (let ch = 0; ch < 2; ch++) {
   703→      const d = b.getChannelData(ch);
   704→      let b0=0,b1=0,b2=0,b3=0,b4=0,b5=0,b6=0;
   705→      for (let i = 0; i < n; i++) {
   706→        const w = Math.random() * 2 - 1;
   707→        b0=0.99886*b0+w*0.0555179; b1=0.99332*b1+w*0.0750759; b2=0.969*b2+w*0.153852;
   708→        b3=0.8665*b3+w*0.3104856; b4=0.55*b4+w*0.5329522; b5=-0.7616*b5-w*0.0168980;
   709→        d[i] = (b0+b1+b2+b3+b4+b5+b6+w*0.5362)*0.11; b6=w*0.115926;
   710→      }
   711→    }
   712→    return b;
   713→  }

   715→  private makeImpulse(sec: number, dec: number): AudioBuffer {
   716→    const c = this.ctx!, n = Math.floor(c.sampleRate * sec), b = c.createBuffer(2, n, c.sampleRate);
   717→    for (let ch = 0; ch < 2; ch++) {
   718→      const d = b.getChannelData(ch);
   719→      for (let i = 0; i < n; i++) d[i] = (Math.random() * 2 - 1) * Math.pow(1 - i / n, dec);
   720→    }
   721→    return b;
   722→  }

   732→  /**
   733→   * Create a band-limited oscillator (worklet polyBLEP if loaded, else
   734→   * OscillatorNode+PeriodicWave fallback). Returns a uniform interface with
   735→   * start()/stop()/disconnect() so voices don't care which engine is active.
   736→   *
   737→   * For worklet nodes: start() is a no-op (worklet runs continuously), stop(t)
   738→   * schedules disconnect at time t. The envelope GainNode gates the audio.
   739→   */
   740→  private createVoiceOsc(freq: number, type: 'saw' | 'square' = 'saw'): {
   741→    node: AudioNode; frequency: AudioParam; detune: AudioParam;
   742→    start: (t: number) => void; stop: (t: number) => void;
   743→  } {
   744→    const c = this.ctx!;
   745→    if (this.workletsReady) {
   746→      if (type === 'saw') {
   747→        const n = createBLSaw(c, { frequency: freq });
   748→        if (n) {
   749→          const bl = n as BLSawNode;
   750→          // Worklet runs continuously; envelope gates it. Schedule disconnect after stop.
   751→          return {
   752→            node: bl, frequency: bl.frequency,
   753→            // BL saw has no detune param; emulate via frequency scaling at call site
   754→            detune: { value: 0, setValueAtTime: () => {}, setTargetAtTime: () => {}, linearRampToValueAtTime: () => {}, exponentialRampToValueAtTime: () => {}, cancelScheduledValues: () => {}, cancelAndHoldAtTime: () => {} } as unknown as AudioParam,
   755→            start: () => {},
   756→            stop: (tt) => {
   757→              const delay = Math.max(0, (tt - c.currentTime) * 1000) + 50;
   758→              setTimeout(() => { try { bl.disconnect(); } catch { /* noop */ } }, delay);
   759→            },
   760→          };
   761→        }
   762→      } else {
   763→        const n = createBLSquare(c, { frequency: freq });
   764→        if (n) {
   765→          const bl = n;
   766→          return {
   767→            node: bl, frequency: bl.frequency,
   768→            detune: { value: 0, setValueAtTime: () => {}, setTargetAtTime: () => {}, linearRampToValueAtTime: () => {}, exponentialRampToValueAtTime: () => {}, cancelScheduledValues: () => {}, cancelAndHoldAtTime: () => {} } as unknown as AudioParam,
   769→            start: () => {},
   770→            stop: (tt) => {
   771→              const delay = Math.max(0, (tt - c.currentTime) * 1000) + 50;
   772→              setTimeout(() => { try { bl.disconnect(); } catch { /* noop */ } }, delay);
   773→            },
   774→          };
   775→        }
   776→      }
   777→    }
   778→    // Fallback: native OscillatorNode + PeriodicWave (aliases at high freq but works)
   779→    const o = c.createOscillator();
   780→    const wave = type === 'square' ? this.sqWave : this.sawWave;
   781→    if (wave) o.setPeriodicWave(wave);
   782→    o.frequency.value = freq;
   783→    return { node: o, frequency: o.frequency, detune: o.detune, start: (tt) => o.start(tt), stop: (tt) => o.stop(tt) };
   784→  }

   786→  /** Disconnect a list of nodes after a delay (cleanup for worklet nodes). */
   787→  private scheduleCleanup(nodes: AudioNode[], t: number, delaySec = 0.1) {
   788→    const c = this.ctx!;
   789→    if (!c) return;
   790→    const delayMs = Math.max(0, (t - c.currentTime + delaySec) * 1000);
   791→    setTimeout(() => { for (const n of nodes) { try { n.disconnect(); } catch { /* noop */ } } }, delayMs);
   792→  }

   794→  kick(t: number, amp = 1) {
   795→    // ── WORKLET ENGINE: push event, no node creation ──
   796→    if (this.useWorkletEngine) {
   797→      this.scheduleEngineEvent(t, VOICE.KICK, 0, amp, this.world.kickDecay, 0);
   798→      return;
   799→    }
   800→    const c = this.ctx!;
   801→    const kickInput = this.getChannelInput('kick');
   802→    const fund = this.world.kickFundamental;
   803→    const spec = this.voiceSpecs?.kick;
   804→
   805→    // ── HYBRID KICK: PSY3 sample + synthetic mid + click ──
   806→    // Sample provides 93.6% sub body. Synth adds definition.
   807→
   808→    // 1. SAMPLE LAYER
   809→    if (spec?.useSample && this.soundBank.has(spec.sampleName || '')) {
   810→      const sampleBuf = this.soundBank.get(spec.sampleName!);
   811→      if (sampleBuf) {
   812→        const src = c.createBufferSource();
   813→        src.buffer = sampleBuf;
   814→        src.playbackRate.value = fund / 50; // pitch-shift to match world fundamental
   815→        const sG = c.createGain();
   816→        sG.gain.setValueAtTime(spec.subLevel * amp, t);
   817→        sG.gain.exponentialRampToValueAtTime(0.001, t + spec.decay);
   818→        // Saturation on sample for character
   819→        const sat = c.createWaveShaper();
   820→        const satCurve = new Float32Array(256);
   821→        const drive = 1 + spec.saturation * 1.5;
   822→        for (let i = 0; i < 256; i++) { const x = (i / 128) - 1; satCurve[i] = Math.tanh(x * drive); }
   823→        sat.curve = satCurve;
   824→        src.connect(sat); sat.connect(sG); sG.connect(kickInput);
   825→        src.start(t); src.stop(t + spec.decay + 0.05);
   826→      }
   827→    } else {
   828→      // Fallback: synthetic sub
   829→      const sub = c.createOscillator(); sub.type = 'sine';
   830→      sub.frequency.setValueAtTime(fund * 2.4, t);
   831→      sub.frequency.exponentialRampToValueAtTime(fund, t + 0.008);
   832→      sub.frequency.exponentialRampToValueAtTime(fund * 0.85, t + 0.09);
   833→      const subG = c.createGain();
   834→      subG.gain.setValueAtTime(0, t);
   835→      subG.gain.linearRampToValueAtTime(0.9 * amp, t + 0.001);
   836→      subG.gain.exponentialRampToValueAtTime(0.001, t + this.world.kickDecay);
   837→      sub.connect(subG); subG.connect(kickInput);
   838→      sub.start(t); sub.stop(t + this.world.kickDecay + 0.02);
   839→    }
   840→
   841→    // 2. SYNTHETIC MID PUNCH (always — adds definition on top of sample)
   842→    const mid = c.createOscillator(); mid.type = 'triangle';
   843→    mid.frequency.setValueAtTime(fund * 2, t);
   844→    mid.frequency.exponentialRampToValueAtTime(fund * 1.5, t + 0.02);
   845→    const midSat = c.createWaveShaper();
   846→    const midCurve = new Float32Array(256);
   847→    for (let i = 0; i < 256; i++) { const x = (i / 128) - 1; midCurve[i] = Math.tanh(x * 1.5); }
   848→    midSat.curve = midCurve;
   849→    const midG = c.createGain();
   850→    midG.gain.setValueAtTime((spec?.midLevel || 0.4) * amp, t);
   851→    midG.gain.exponentialRampToValueAtTime(0.001, t + 0.05);
   852→    mid.connect(midSat); midSat.connect(midG); midG.connect(kickInput);
   853→    mid.start(t); mid.stop(t + 0.06);
   854→
   855→    // 3. CLICK: noise, very short
   856→    const clickSrc = c.createBufferSource(); clickSrc.buffer = this.pink;
   857→    const clickHP = c.createBiquadFilter(); clickHP.type = 'highpass'; clickHP.frequency.value = 3000;
   858→    const clickG = c.createGain();
   859→    clickG.gain.setValueAtTime((spec?.clickLevel || 0.08) * amp, t);
   860→    clickG.gain.exponentialRampToValueAtTime(0.001, t + 0.003);
   861→    clickSrc.connect(clickHP); clickHP.connect(clickG); clickG.connect(kickInput);
   862→    clickSrc.start(t); clickSrc.stop(t + 0.01);
   863→
   864→    // sidechain
   865→    if (this.duck) {
   866→      const d = this.duck.gain;
   867→      d.cancelScheduledValues(t);
   868→      d.setValueAtTime(1 - this.world.duck * (0.5 + this.macros.aggression * 0.5), t);
   869→      d.setTargetAtTime(1, t + 0.02, 0.08 + this.macros.groove * 0.04);
   870→    }
   871→  }

   873→  bass(t: number, midi: number, dur: number, amp = 0.5, acid = false) {
   874→    // ── WORKLET ENGINE: push event, no node creation ──
   875→    if (this.useWorkletEngine) {
   876→      this.scheduleEngineEvent(t, VOICE.BASS, mtof(midi), amp, dur, acid ? 1 : 0);
   877→      return;
   878→    }
   879→    const c = this.ctx!, f = mtof(midi);
   880→    const spec = this.voiceSpecs?.bass;
   881→    const bassInput = this.getChannelInput('bass');
   882→
   883→    // ── BASS REBUILD: multi-layer architecture ──
   884→    //   SUB   : clean sine at f/2, mono, bypasses filter for clean low end
   885→    //   BODY  : BL saw → real Moog ladder filter (cutoff envelope) → drive
   886→    //   The Moog filter's tanh saturation adds harmonic character that
   887→    //   BiquadFilter cannot — this is the PSY3 sound.
   888→
   889→    // 1. SUB LAYER (clean fundamental, no filter)
   890→    const sub = c.createOscillator(); sub.type = 'sine'; sub.frequency.value = f / 2;
   891→    const subG = c.createGain();
   892→    subG.gain.setValueAtTime(0, t);
   893→    subG.gain.linearRampToValueAtTime((spec?.subLevel ?? 0.6) * amp, t + 0.003);
   894→    subG.gain.linearRampToValueAtTime(0, t + dur);
   895→    sub.connect(subG); subG.connect(bassInput);
   896→    sub.start(t); sub.stop(t + dur + 0.03);
   897→
   898→    // 2. BODY LAYER: BL saw → real Moog filter → amp envelope
   899→    const oscWrap = this.createVoiceOsc(f, acid ? 'square' : 'saw');
   900→
   901→    // Real Moog ladder filter (4-stage tanh, sample-accurate via AudioWorklet)
   902→    const cutoffStart = acid ? 2500 : (spec?.cutoffStart ?? 1200);
   903→    const cutoffEnd = acid ? this.world.bassCutoff : (spec?.cutoffEnd ?? 150);
   904→    const resQ = acid ? this.world.bassResonance : (spec?.resonance ?? 3);
   905→    const filter = this.createVoiceFilter({
   906→      cutoff: cutoffStart,
   907→      resonance: Math.min(1, resQ / 20),   // map Q→0..1 for Moog self-osc
   908→      drive: 1 + (spec?.saturation ?? 0.3) * 2 + this.macros.aggression * 0.5,
   909→      level: 1,
   910→    });
   911→    // Filter envelope: cutoff sweeps high→low (psytrance bass pluck character)
   912→    if (filter.scheduleCutoff) {
   913→      filter.scheduleCutoff(t, cutoffStart, cutoffEnd, Math.min(dur, 0.08));
   914→    }
   915→
   916→    // Amp envelope: snappy attack, sustain, quick release (PSY3 uses 0.42 level)
   917→    const g = c.createGain();
   918→    g.gain.setValueAtTime(0, t);
   919→    g.gain.linearRampToValueAtTime((spec?.ampLevel ?? 0.42) * amp, t + 0.003);
   920→    g.gain.linearRampToValueAtTime(0, t + dur);
   921→
   922→    // Route: osc → filter → gain → bassInput
   923→    oscWrap.node.connect(filter.inputNode);
   924→    filter.outputNode.connect(g);
   925→    g.connect(bassInput);
   926→
   927→    oscWrap.start(t);
   928→    oscWrap.stop(t + dur + 0.05);
   929→    // Clean up worklet nodes after note ends
   930→    this.scheduleCleanup([filter.inputNode as AudioNode, g], t + dur + 0.1, 0.05);
   931→
   932→    // Sidechain ducking (kick has priority over bass)
   933→    if (this.duck) {
   934→      const d = this.duck.gain;
   935→      d.cancelScheduledValues(t);
   936→      d.setValueAtTime(1 - this.world.duck * (0.5 + this.macros.aggression * 0.5), t);
   937→      d.setTargetAtTime(1, t + 0.02, 0.08 + this.macros.groove * 0.04);
   938→    }
   939→  }

   941→  lead(t: number, midi: number, dur: number, amp = 0.2, pan = 0) {
   942→    // ── WORKLET ENGINE: push event, no node creation ──
   943→    if (this.useWorkletEngine) {
   944→      this.scheduleEngineEvent(t, VOICE.LEAD, mtof(midi), amp, dur, pan);
   945→      return;
   946→    }
   947→    const c = this.ctx!, f = mtof(midi);
   948→    const spec = this.voiceSpecs?.lead;
   949→    const leadInput = this.getChannelInput('lead');
   950→
   951→    // ── LEAD REBUILD: BL supersaw → real Moog filter → amp envelope ──
   952→    // The Moog filter's resonance + tanh saturation gives the warm, vocal
   953→    // character that makes psytrance leads sing instead of buzz.
   954→    const baseCut = (spec?.cutoff ?? 1800) * (0.7 + this.macros.brightness * 0.6);
   955→
   956→    // Real Moog ladder filter with filter envelope (open → settle)
   957→    const filter = this.createVoiceFilter({
   958→      cutoff: baseCut * 2,
   959→      resonance: Math.min(1, ((spec?.resonance ?? 2) + this.macros.psychedelia * 3) / 20),
   960→      drive: 1 + (spec?.saturation ?? 0.2) * 1.5,
   961→      level: 1,
   962→    });
   963→    if (filter.scheduleCutoff) {
   964→      filter.scheduleCutoff(t, baseCut * 2, baseCut, dur);
   965→    }
   966→
   967→    // LFO modulation on filter cutoff (psychedelic movement)
   968→    if (this.macros.psychedelia > 0.3) {
   969→      const lfo = c.createOscillator(); lfo.type = 'sine';
   970→      lfo.frequency.value = 0.5 + this.macros.psychedelia * 3;
   971→      const lfoGain = c.createGain();
   972→      lfoGain.gain.value = baseCut * 0.3 * this.macros.psychedelia;
   973→      lfo.connect(lfoGain); lfoGain.connect(filter.cutoff);
   974→      lfo.start(t); lfo.stop(t + dur + 0.1);
   975→    }
   976→
   977→    // Amp envelope
   978→    const g = c.createGain();
   979→    g.gain.setValueAtTime(0, t);
   980→    g.gain.linearRampToValueAtTime((spec?.level ?? 0.16) * amp / 0.2, t + 0.006);
   981→    g.gain.linearRampToValueAtTime(0, t + dur);
   982→
   983→    // Supersaw: N detuned BL saws, stereo-spread
   984→    const numOscs = spec?.numOscs ?? 5;
   985→    const detune = (spec?.detune ?? 10) * (0.5 + this.macros.psychedelia);
   986→    const spread = spec?.stereoSpread ?? 0.4;
   987→    const oscType = (spec?.oscType === 'square') ? 'square' : 'saw';
   988→    const cleanupNodes: AudioNode[] = [];
   989→    for (let i = 0; i < numOscs; i++) {
   990→      const oscWrap = this.createVoiceOsc(f, oscType);
   991→      // Detune (BL saw has no detune param — adjust frequency directly)
   992→      const detuneCents = (i - (numOscs - 1) / 2) * detune;
   993→      const detuneMult = Math.pow(2, detuneCents / 1200);
   994→      oscWrap.frequency.setValueAtTime(f * detuneMult, t);
   995→      const pp = c.createStereoPanner();
   996→      pp.pan.value = (i - (numOscs - 1) / 2) * (spread * 2 / Math.max(1, numOscs - 1));
   997→      oscWrap.node.connect(pp); pp.connect(filter.inputNode);
   998→      oscWrap.start(t);
   999→      oscWrap.stop(t + dur + 0.05);
  1000→      cleanupNodes.push(pp);
  1001→    }
  1002→    filter.outputNode.connect(g); g.connect(leadInput);
  1003→    if (this.dSend) g.connect(this.dSend);
  1004→    if (this.rSend) g.connect(this.rSend);
  1005→    this.scheduleCleanup([filter.inputNode as AudioNode, ...cleanupNodes], t + dur + 0.1, 0.05);
  1006→  }

  1008→  acid(t: number, midi: number, dur: number, amp = 0.25) {
  1009→    // ── WORKLET ENGINE: push event, no node creation ──
  1010→    if (this.useWorkletEngine) {
  1011→      this.scheduleEngineEvent(t, VOICE.ACID, mtof(midi), amp, dur, 0);
  1012→      return;
  1013→    }
  1014→    const c = this.ctx!, f = mtof(midi);
  1015→    const acidInput = this.getChannelInput('lead'); // acid shares lead channel/music bus
  1016→
  1017→    // ── ACID REBUILD: BL square → high-resonance Moog filter → distortion ──
  1018→    // The acid sound is ALL about the resonant filter sweep. The real Moog
  1019→    // ladder's self-oscillation at high resonance gives the squelchy, vocal
  1020→    // "acid" character that BiquadFilter's linear resonance cannot.
  1021→
  1022→    const oscWrap = this.createVoiceOsc(f, 'square');
  1023→
  1024→    // High-resonance Moog filter (near self-oscillation) with envelope sweep
  1025→    const cutoffStart = 200 + this.macros.brightness * 3000;
  1026→    const cutoffEnd = 100;
  1027→    const filter = this.createVoiceFilter({
  1028→      cutoff: cutoffStart,
  1029→      resonance: Math.min(1, (12 + this.macros.psychedelia * 8) / 20), // high res for squelch
  1030→      drive: 2 + this.macros.aggression * 2,  // drive adds harmonic grit
  1031→      level: 1,
  1032→    });
  1033→    if (filter.scheduleCutoff) {
  1034→      filter.scheduleCutoff(t, cutoffStart, cutoffEnd, dur * 0.7);
  1035→    }
  1036→
  1037→    // Distortion after filter (classic acid chain: osc → filter → overdrive)
  1038→    const dist = c.createWaveShaper();
  1039→    const curve = new Float32Array(1024);
  1040→    const driveAmt = 2 + this.macros.aggression * 2;
  1041→    for (let i = 0; i < 1024; i++) { const x = (i / 512) - 1; curve[i] = Math.tanh(x * driveAmt); }
  1042→    dist.curve = curve;
  1043→
  1044→    // Amp envelope
  1045→    const g = c.createGain();
  1046→    g.gain.setValueAtTime(0, t);
  1047→    g.gain.linearRampToValueAtTime(amp, t + 0.003);
  1048→    g.gain.linearRampToValueAtTime(0, t + dur);
  1049→
  1050→    oscWrap.node.connect(filter.inputNode);
  1051→    filter.outputNode.connect(dist); dist.connect(g);
  1052→    g.connect(acidInput);
  1053→    if (this.dSend) g.connect(this.dSend);
  1054→    if (this.rSend) g.connect(this.rSend);
  1055→
  1056→    oscWrap.start(t);
  1057→    oscWrap.stop(t + dur + 0.05);
  1058→    this.scheduleCleanup([filter.inputNode as AudioNode, dist, g], t + dur + 0.1, 0.05);
  1059→  }

  1190→  pad(t: number, root: number, chord: number[], dur: number, amp = 0.08) {

  1200→    // Pads need width, movement, and warmth. The Moog filter's smooth cutoff

  1590→    const typ = arrSection.type;

  1592→    const density = Math.max(0.15, Math.min(1, arrSection.density * (0.5 + 0.7 * this.macros.energy)));

  1594→    const motif = new Motif(this.world.root, this.world.scale, rng);
  1595→    const energy = arrSection.energy;
  1596→
  1597→    // ── PSY3-STYLE MUSICAL GRAMMAR ──
  1598→    const grammarRng = new SeededRng(this.seed * 1000 + this.sectionIdx + 999);
  1599→    const leadMotif = arrSection.leadOn
  1600→      ? new LeadMotif(this.world.root, this.world.scale, grammarRng)
  1601→      : null;

  1603→      ? new AcidPattern(this.world.root, this.world.scale, grammarRng)
  1604→      : null;

  1606→    const bassPatternIdx = grammarRng.int(0, bassPatterns.length - 1);
  1607→    const tensionShape: TensionShape = typ === 'build' || typ === 'build2' ? 'rise' : typ === 'break' || typ === 'breakdown' ? 'fall' : 'arc';
  1608→
  1609→    // ── CALL/RESPONSE ENGINE ──
  1610→    this.callResponse = new CallResponseEngine(this.world.root, this.world.scale, grammarRng);
  1611→
  1612→    this.sec = {
  1613→      type: typ, bars, density, rng, motif, energy, chordIndex: 0,
  1614→      leadMotif, acidPattern, bassPatternIdx, tensionShape,
  1615→      bassOn: arrSection.bassOn, leadOn: arrSection.leadOn, acidOn: arrSection.acidOn,
  1616→      hatDensity: arrSection.hatDensity, percDensity: arrSection.percDensity,
  1617→      fxDensity: arrSection.fxDensity, variation: arrSection.variation, label: arrSection.label,
  1618→    };
  1619→    this.currentSection = arrSection.label;
  1620→    this.sectionIdx++;
  1621→  }
  1622→
  1623→  private s16(): number { return 60 / this.world.bpm / 4; }
  1624→
  1625→  private tick() {
  1626→    if (!this.playing || !this.ctx || !this.sec) return;
  1627→    // REDUCED LOOKAHEAD for lower latency:
  1628→    // Worklet mode: 0.1s (was 0.3s) — enough to batch ~4 events at 140bpm
  1629→    // Legacy mode: 0.15s (unchanged)
  1630→    // The worklet handles sample-accurate timing, so lookahead only affects
  1631→    // how far ahead we generate events, not when they play.
  1632→    const lookahead = this.useWorkletEngine ? 0.1 : 0.15;
  1633→    while (this.next < this.ctx.currentTime + lookahead) {
  1634→      this.step(this.si, this.next);
  1635→      this.si++;
  1636→      this.next += this.s16();
  1637→      if (this.si >= this.sec.bars * 16) {
  1638→        this.nextSection();
  1639→        this.si = 0;
  1640→        this.currentPhrase++;
  1641→        this.phrasesPlayed++;
  1642→        // Send 'newPhrase' to worklet — rotates phrase-locked samples
  1643→        // This gives sonic consistency (same kick for 8 bars) then variation
  1644→        if (this.useWorkletEngine && this.engineNode) {
  1645→          this.engineNode.notifyNewPhrase();
  1646→        }
  1647→      }
  1648→    }
  1649→    // Flush batched events to the engine worklet
  1650→    if (this.useWorkletEngine && this.engineNode) {
  1651→      this.engineNode.flushEvents();
  1652→    }
  1653→  }
  1654→
  1655→  // ─── The Musical Brain (v2) ───────────────────────────────────
  1656→
  1657→  private step(s: number, t: number) {
  1658→    if (!this.sec || !this.ctx) return;
  1659→    const S = this.sec;
  1660→    const sb = s % 16;           // step within bar (0-15)
  1661→    const bar = Math.floor(s / 16); // bar within section
  1662→    const phrase = Math.floor(s / 32); // 2-bar phrase
  1663→    const sw = this.world.swing * this.macros.groove;
  1664→    this.currentBar = bar;
  1665→    const e = this.macros.energy;
  1666→    const psy = this.macros.psychedelia;
  1667→    const dens = this.macros.density;
  1668→    const w = this.world;
  1669→
  1670→    // ─── SECTION AUTOMATION ──────────────────────────────────
  1671→    // DROP CONTRAST: last 2 bars of build — remove bass, narrow filter, create tension
  1672→    const isPreDrop = S.label.includes('BUILD') && bar >= S.bars - 2;
  1673→    const isDropStart = S.bassOn && S.leadOn && bar === 0;
  1674→    // ELEMENT REMOVAL: last bar of groove/build — remove hats for tension
  1675→    const isPreTransition = bar === S.bars - 1;
  1676→    // FILTER AUTOMATION: last 4 bars of build — riser + filter close
  1677→    const isBuildClimax = S.label.includes('BUILD') && bar >= S.bars - 4;
  1678→
  1679→    // Riser before drop (last 2 bars of build) — LONGER riser for more tension
  1680→    if (isPreDrop && sb === 0) {
  1681→      this.riser(t, this.s16() * 32);
  1682→    }
  1683→    // Additional riser at 4 bars before drop (earlier tension build)
  1684→    if (isBuildClimax && bar === S.bars - 4 && sb === 0) {
  1685→      this.sweep(t, this.s16() * 16);
  1686→    }
  1687→    // Impact at drop start — LOUDER for more payoff
  1688→    if (isDropStart && sb === 0) {
  1689→      this.impact(t);
  1690→      // Double impact for big drop entrance
  1691→      this.impact(t + this.s16() * 2);
  1692→    }
  1693→    // Filter sweep in breakdown
  1694→    if (!S.bassOn && bar === 0 && sb === 0) {
  1695→      this.sweep(t, this.s16() * 32);
  1696→    }
  1697→    // Sweep at section transitions (last bar) — vary sweep length by section
  1698→    if (isPreTransition && sb === 12 && S.bassOn) {
  1699→      const sweepLen = S.label.includes('BUILD') ? this.s16() * 8 : this.s16() * 4;
  1700→      this.sweep(t, sweepLen);
  1701→    }
  1702→    // Downlifter at drop start (descending sweep after impact = contrast)
  1703→    if (isDropStart && sb === 4) {
  1704→      this.downlifter(t, 0.1 + this.macros.energy * 0.05);
  1705→    }
  1706→    // ELEMENT REMOVAL: last 2 bars before break — remove hats (creates space)
  1707→    // This is what commercial tracks do — they strip elements before a section change
  1708→    const hatsMuted = (S.label.includes('BUILD') && bar >= S.bars - 2) ||
  1709→                      (S.bassOn && S.leadOn && bar >= S.bars - 2 && S.label !== 'FINAL DROP');
  1710→
  1711→    // ─── SECTION-AWARE REVERB/DELAY AUTOMATION ─────────────
  1712→    if (sb === 0 && bar === 0) {
  1713→      // Legacy path: adjust Web Audio reverb/delay sends
  1714→      if (this.rSend && this.ctx) {
  1715→        const reverbTarget = !S.bassOn ? 0.4 + this.macros.space * 0.3
  1716→                           : S.bassOn && S.leadOn ? 0.15 + this.macros.space * 0.2
  1717→                           : 0.2 + this.macros.space * 0.25;
  1718→        this.rSend.gain.setTargetAtTime(reverbTarget, t, 0.5);
  1719→      }
  1720→      if (this.dSend && this.ctx) {
  1721→        const delayTarget = !S.bassOn ? 0.3 + this.macros.space * 0.2
  1722→                          : S.label.includes('BUILD') ? 0.25 + this.macros.psychedelia * 0.15
  1723→                          : 0.15 + this.macros.psychedelia * 0.1;
  1724→        this.dSend.gain.setTargetAtTime(delayTarget, t, 0.5);
  1725→      }
  1726→
  1727→      // ── WORKLET ENGINE: section-aware FX automation ──
  1728→      // Build: more reverb+delay (tension/space)
  1729→      // Drop: less reverb (punch/dry), moderate delay
  1730→      // Break: max reverb (atmospheric), high delay (psychedelic)
  1731→      // Intro/Outro: medium reverb
  1732→      if (this.useWorkletEngine && this.engineNode) {
  1733→        const space = this.macros.space;
  1734→        const psy = this.macros.psychedelia;
  1735→        let revSends: number[], delSends: number[], revWet: number, delWet: number, delFb: number;
  1736→        if (!S.bassOn) {
  1737→          // Break: max space, atmospheric
  1738→          revSends = [0.12, 0.03, 0.40, 0.60, 0.45];
  1739→          delSends = [0.08, 0.0, 0.30, 0.20, 0.25];
  1740→          revWet = 0.45; delWet = 0.35; delFb = 0.45;
  1741→        } else if (S.label.includes('BUILD')) {
  1742→          // Build: rising tension, more delay
  1743→          revSends = [0.10, 0.02, 0.30, 0.45, 0.35];
  1744→          delSends = [0.06, 0.0, 0.25, 0.15, 0.20];
  1745→          revWet = 0.35; delWet = 0.30; delFb = 0.40;
  1746→        } else if (S.bassOn && S.leadOn || S.bassOn && S.leadOn) {
  1747→          // Drop: dry punch, less reverb on drums/bass, moderate on music
  1748→          revSends = [0.05, 0.01, 0.20, 0.35, 0.25];
  1749→          delSends = [0.04, 0.0, 0.15, 0.08, 0.12];
  1750→          revWet = 0.25; delWet = 0.20; delFb = 0.35;
  1751→        } else {
  1752→          // Intro/outro: medium space
  1753→          revSends = [0.08, 0.02, 0.25, 0.40, 0.30];
  1754→          delSends = [0.05, 0.0, 0.20, 0.10, 0.15];
  1755→          revWet = 0.30; delWet = 0.25; delFb = 0.38;
  1756→        }
  1757→        // Apply macro space/psychedelia modulation
  1758→        revWet *= (0.7 + space * 0.6);
  1759→        delWet *= (0.7 + psy * 0.6);
  1760→        this.engineNode.setFX({
  1761→          reverbSends: revSends, delaySends: delSends,
  1762→          reverbWet: revWet, delayWet: delWet, delayFeedback: delFb,
  1763→        });
  1764→      }
  1765→    }
  1766→
  1767→    // ─── PAD (chord progression, every 2 bars) ──────────────
  1768→    if (sb === 0 && bar % 2 === 0) {
  1769→      const progs = PROGRESSIONS[w.scale] || PROGRESSIONS.minor;
  1770→      const chord = progs[(bar / 2) % progs.length];
  1771→      const padAmp = 0.12 * (0.5 + e * 0.5) * (!S.bassOn ? 1.5 : 0.8);
  1772→      this.pad(t, w.root - 12, chord, this.s16() * 32, padAmp);
  1773→    }
  1774→
  1775→    // ─── TEXTURE (every 4 bars + continuous bed in drops) ───
  1776→    if (sb === 0 && bar % 4 === 0 && S.bassOn || S.leadOn) {
  1777→      this.texture(t, this.s16() * 64, w.textureLevel * (0.5 + psy * 0.5));
  1778→    }
  1779→    // Continuous texture bed in drops (every bar, lower volume)
  1780→    if (sb === 0 && bar % 2 === 1 && (S.bassOn && S.leadOn || S.bassOn && S.leadOn)) {
  1781→      this.texture(t, this.s16() * 32, w.textureLevel * 0.4 * psy);
  1782→    }
  1783→
  1784→    // ─── KICK (4 on floor — BALANCED: 0.5 not 0.9, kick was dominating 98% of mix energy)
  1785→    if (sb % 4 === 0) {
  1786→      const isDownbeat = sb === 0;
  1787→      const kickVel = isDownbeat ? 0.5 + e * 0.05 : 0.42 + e * 0.08;
  1788→      this.kick(t, kickVel);
  1789→    }
  1790→    // Ghost kick on syncopated step in drop
  1791→    if (S.bassOn && S.leadOn && sb === 14 && S.rng.chance(0.3 * dens)) {
  1792→      this.kick(t, 0.15);
  1793→    }
  1794→
  1795→    // ─── BASS (psytrance grammar — bar-to-bar variation) ──
  1796→    const isOff = sb % 2 === 1;
  1797→    const bt = isOff ? t + sw * this.s16() : t;
  1798→    // Use explicit bass patterns from BASS_PATTERNS (controlled, not random)
  1799→    const bassPatterns = BASS_PATTERNS[w.bass] || BASS_PATTERNS.off;
  1800→    // BAR-TO-BAR VARIATION: change pattern index every 4 bars for musical evolution
  1801→    // BEFORE: same pattern for entire section. AFTER: pattern rotates every 4 bars.
  1802→    const bassPatternRotIdx = (S.bassPatternIdx + Math.floor(bar / 4)) % bassPatterns.length;
  1803→    const bassPattern = bassPatterns[bassPatternRotIdx];
  1804→    const patternStep = Math.floor(sb / 2) % bassPattern.steps.length;
  1805→    let bassDegree = bassPattern.steps[patternStep];
  1806→    let bassAccent = bassPattern.accents[patternStep];
  1807→
  1808→    // BAR-TO-BAR NOTE VARIATION: every 2nd bar, add a passing tone or octave
  1809→    if (bar % 2 === 1 && sb === 6 && S.rng.chance(0.4)) {
  1810→      bassDegree = S.rng.pick([2, 4, 7]); // passing tone
  1811→      bassAccent = 0.6;
  1812→    }
  1813→    // Every 4th bar, octave up on the last beat for lift
  1814→    if (bar % 4 === 3 && sb === 14) {
  1815→      bassDegree = 7; // octave
  1816→      bassAccent = 0.8;
  1817→    }
  1818→
  1819→    // Determine if bass plays based on world type and pattern
  1820→    let bassOn = bassDegree >= 0 && bassAccent > 0;
  1821→    if (w.bass === 'roll') bassOn = bassOn && (isOff || sb % 4 === 0);
  1822→    else if (w.bass === 'off') bassOn = bassOn && sb % 4 === 2;
  1823→    else if (w.bass === 'acid') bassOn = bassOn && (isOff || sb === 0);
  1824→
  1825→    // Rest before drop (last 2 bars of build — creates tension/contrast)
  1826→    if (isPreDrop) bassOn = false;
  1827→    // No bass in breakdown
  1828→    if (!S.bassOn) bassOn = false;
  1829→
  1830→    if (bassOn) {
  1831→      const bassNote = grammarScaleNote(w.root, w.scale, bassDegree);
  1832→      // Velocity from pattern accent + energy + BAR POSITION VARIATION
  1833→      // Downbeats louder, ghost notes quieter — creates groove, not machine
  1834→      let bassVel = bassAccent * (0.35 + e * 0.15);
  1835→      let bassDur = this.s16() * 0.9;
  1836→      // Ghost bass: very quiet on step 0 of odd bars (lift)
  1837→      const isGhost = (bar % 2 === 1 && sb === 0 && S.rng.chance(0.3));
  1838→      if (isGhost) { bassVel = 0.2; bassDur = this.s16() * 0.4; }
  1839→      // SUSTAINED BASS: every 8th bar, play a longer sustained note for variation
  1840→      if (bar % 8 === 7 && sb === 0) {
  1841→        bassDur = this.s16() * 2.5; // sustained — breaks the pluck pattern
  1842→        bassVel *= 1.1;
  1843→      }
  1844→      this.bass(bt, bassNote, bassDur, bassVel, w.acid);
  1845→    }
  1846→
  1847→    // ─── ACID LINE (stored pattern with controlled mutation) ──
  1848→    if (w.acid && S.bassOn && S.leadOn && sb % 2 === 0 && S.rng.chance(0.5 * psy)) {
  1849→      // Use AcidPattern (stored pattern, not random pick)
  1850→      if (S.acidPattern) {
  1851→        const acidNote = S.acidPattern.next();
  1852→        if (acidNote !== null) {
  1853→          this.acid(t, acidNote, this.s16() * 1.5, 0.15 + psy * 0.1);
  1854→        }
  1855→      }
  1856→    }
  1857→
  1858→    // ─── HATS (with groove + velocity variation + GHOST NOTES + ELEMENT REMOVAL) ───
  1859→    // HATS MUTED before transitions — creates tension by removing elements
  1860→    if (!hatsMuted && w.hatPattern[sb] === 'x') {
  1861→      // VELOCITY CURVE: downbeats louder, offbeats lighter, bar position matters
  1862→      // BEFORE: sb%4===0 ? 0.12 : 0.08 (only 2 levels)
  1863→      // AFTER: 4 levels based on beat position + bar position + density
  1864→      const beatPos = sb % 4;
  1865→      let hatVel;
  1866→      if (beatPos === 0) hatVel = 0.25;           // downbeat — loudest
  1867→      else if (beatPos === 2) hatVel = 0.18;       // backbeat — medium
  1868→      else hatVel = 0.12;                           // offbeat — lightest
  1869→      // BAR VARIATION: every 4th bar, hats are louder (build tension)
  1870→      if (bar % 4 === 3) hatVel *= 1.2;
  1871→      // DENSITY: scale by section density
  1872→      hatVel *= (0.5 + dens * 0.5);
  1873→      const hatPan = 0.2 + Math.sin(s * 0.1) * 0.15;
  1874→      this.hat(t + (sb % 4 === 2 ? sw * this.s16() : 0), false, hatVel, hatPan);
  1875→    }
  1876→    // GHOST HATS: occasional quiet hat between main hats (adds groove, not machine)
  1877→    if (!hatsMuted && w.hatPattern[sb] === '.' && sb % 2 === 0 && S.rng.chance(0.15 * dens) && S.bassOn) {
  1878→      this.hat(t, false, 0.04 + dens * 0.02, 0.15 + Math.sin(s * 0.2) * 0.1);
  1879→    }
  1880→    // Open hat on step 4 — also muted during transition
  1881→    if (!hatsMuted && sb === 4 && S.bassOn) {
  1882→      const openVel = 0.06 + dens * 0.04;
  1883→      // BAR VARIATION: every 4th bar, open hat is longer (fill-like)
  1884→      const openIsLong = bar % 4 === 3;
  1885→      this.hat(t, openIsLong, openVel, -0.25);
  1886→    }
  1887→    // Shaker on offbeats in groove/drop — VELOCITY VARIATION
  1888→    if (S.bassOn && S.leadOn && sb % 2 === 1 && S.rng.chance(0.6 * dens)) {
  1889→      // Vary velocity based on position — creates groove, not static
  1890→      const shakerVel = (0.04 + dens * 0.03) * (sb % 4 === 3 ? 1.3 : 1.0); // accent on beat 4
  1891→      this.shaker(t, shakerVel, -0.15 + Math.sin(s * 0.07) * 0.1);
  1892→    }
  1893→
  1894→    // ─── CLAP / SNARE (on 2 & 4 with VARIATION) ─────────────────
  1895→    if (sb === 4 && S.bassOn || S.leadOn && S.bassOn) {
  1896→      // BAR VARIATION: every 4th bar, clap is harder (fill leading)
  1897→      const clapVel = 0.25 * (0.5 + e * 0.5) * (bar % 4 === 3 ? 1.2 : 1.0);
  1898→      this.clap(t, clapVel);
  1899→    }
  1900→    if (sb === 12 && S.bassOn && S.leadOn) {
  1901→      // Only in drops, extra clap for drive
  1902→      this.clap(t, 0.2 * (0.5 + e * 0.5));
  1903→    }
  1904→
  1905→    // ─── PERCUSSION (world-specific pattern with VARIATION) ─────
  1906→    if (w.percPattern[sb] === 'x' && S.bassOn && S.rng.chance(0.7 * dens)) {
  1907→      // VELOCITY VARIATION: accent on certain steps
  1908→      const percVel = (0.1 + dens * 0.05) * (sb % 8 === 0 ? 1.3 : 1.0);
  1909→      const percPan = 0.3 + Math.sin(s * 0.05) * 0.2;
  1910→      this.perc(t, percVel, percPan);
  1911→    }
  1912→
  1913→    // ─── DRUM FILL (last bar of phrase — REAL fills, not same pattern) ──
  1914→    // BEFORE: same 4 hits every time (perc, hat, perc, hat)
  1915→    // AFTER: different fill patterns based on bar position + section
  1916→    if (bar % 4 === 3 && sb >= 12 && S.bassOn) {
  1917→      const fillType = bar % 8 === 7 ? 2 : bar % 4 === 3 ? 1 : 0; // 3 fill types
  1918→      if (fillType === 0) {
  1919→        // Fill A: perc → hat → perc → open hat
  1920→        if (sb === 12) this.perc(t, 0.12, 0.4);
  1921→        if (sb === 13) this.hat(t, false, 0.1, -0.3);
  1922→        if (sb === 14) this.perc(t, 0.1, -0.3);
  1923→        if (sb === 15) this.hat(t, true, 0.08, 0.3);
  1924→      } else if (fillType === 1) {
  1925→        // Fill B: rapid hats → clap
  1926→        if (sb === 12) this.hat(t, false, 0.08, 0.2);
  1927→        if (sb === 13) this.hat(t, false, 0.09, -0.1);
  1928→        if (sb === 14) this.hat(t, false, 0.10, 0.15);
  1929→        if (sb === 15) this.clap(t, 0.15 * (0.5 + e * 0.5));
  1930→      } else {
  1931→        // Fill C: perc roll → impact (big fill before new phrase)
  1932→        if (sb === 12) this.perc(t, 0.10, 0.3);
  1933→        if (sb === 13) this.perc(t, 0.12, -0.2);
  1934→        if (sb === 14) this.perc(t, 0.14, 0.1);
  1935→        if (sb === 15) this.impact(t);
  1936→      }
  1937→    }
  1938→
  1939→    // ─── LEAD with CALL/RESPONSE (prevents MIDI soup) ────────
  1940→    // Primary lead and counter-lead alternate bars — never play simultaneously.
  1941→    // Creates musical conversation instead of everything-at-once.
  1942→    if (S.density > 0.3 && S.bassOn && this.callResponse) {
  1943→      // Determine which voice plays this bar (call/response)
  1944→      const phraseBar = bar % 8;
  1945→      const isPrimaryTurn = phraseBar < 2 || (phraseBar >= 4 && phraseBar < 6);
  1946→      const isCounterTurn = (phraseBar >= 2 && phraseBar < 4) || phraseBar >= 6;
  1947→
  1948→      if (isPrimaryTurn && S.leadMotif) {
  1949→        // Primary lead
  1950→        const leadResult = S.leadMotif.nextNote(sb, bar, e * (0.5 + psy * 0.5), S.rng);
  1951→        if (leadResult) {
  1952→          const leadDur = this.s16() * (1.5 + psy * 0.5);
  1953→          const leadPan = Math.sin(s * 0.03) * 0.2;
  1954→          this.lead(t, leadResult.note, leadDur, leadResult.velocity * 0.5, leadPan);
  1955→        }
  1956→        if (sb === 0 && bar % 4 === 0 && bar > 0) S.leadMotif.evolve();
  1957→      } else if (isCounterTurn) {
  1958→        // Counter lead (response) — different register, slightly different character
  1959→        const counterNote = this.callResponse.nextNote('counter-lead');
  1960→        if (counterNote > 0 && S.rng.chance(0.5 * e)) {
  1961→          const counterDur = this.s16() * (1.2 + psy * 0.4);
  1962→          const counterPan = Math.sin(s * 0.04 + 1.5) * 0.25; // different pan position
  1963→          // Counter lead plays at different octave (already +12 from CallResponseEngine)
  1964→          this.lead(t, counterNote, counterDur, 0.3 * e, counterPan);
  1965→        }
  1966→      }
  1967→    }
  1968→
  1969→    // ─── EAR CANDY (from PSY3: zap, blip, downlifter) ──────
  1970→    // Random FM zap — adds psychedelic sparkle
  1971→    if (S.rng.chance(0.03 * this.macros.surprise) && S.bassOn && S.leadOn) {
  1972→      this.zap(t, 0.08 + this.macros.psychedelia * 0.07);
  1973→    }
  1974→    // Random blip — ear candy accent
  1975→    if (S.rng.chance(0.04 * this.macros.surprise) && S.bassOn) {
  1976→      const blipFreq = 800 + S.rng.int(0, 6) * 200;
  1977→      this.blip(t, blipFreq, 0.06 + this.macros.brightness * 0.04);
  1978→    }
  1979→    // Downlifter at 8-bar boundaries in builds/drops
  1980→    if (bar % 8 === 7 && sb === 0 && (S.bassOn && S.leadOn || S.bassOn && S.leadOn)) {
  1981→      this.downlifter(t, 0.12 + this.macros.energy * 0.08);
  1982→    }
  1983→    // Random percussion hit with varying pan (spatial interest)
  1984→    if (S.rng.chance(0.02 * this.macros.surprise) && S.bassOn) {
  1985→      this.perc(t, 0.06, S.rng.gauss(0, 0.5));
  1986→    }
  1987→  }
