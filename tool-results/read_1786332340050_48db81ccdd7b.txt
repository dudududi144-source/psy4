    24→/**
    25→ * Map a reference spectral centroid (Hz) to a target synth cutoff (Hz)
    26→ * using log-linear interpolation. Anchor points:
    27→ *   500Hz  ->  800 Hz cutoff (dark / warm)
    28→ *   2000Hz -> 3000 Hz cutoff (balanced)
    29→ *   5000Hz -> 6000 Hz cutoff (bright / cutting)
    30→ * Falls back to clamp(centroid, 200, 12000) if out of typical range.
    31→ */
    32→function centroidToCutoff(centroid: number): number {
    33→  if (!isFinite(centroid) || centroid <= 0) return 1500;
    34→  const lc = Math.log(clamp(centroid, 80, 16000));
    35→  // Fit using endpoints (500,800) and (5000,6000):
    36→  //   ln(cut) = 0.8753 * ln(centroid) + 1.245
    37→  const lcut = 0.8753 * lc + 1.245;
    38→  return clamp(Math.exp(lcut), 200, 12000);
    39→}

   125→// ─── Pooled Synth Voice (from PSY6 — persistent oscillators) ────────────────
   126→
   127→class PooledSynthVoice {
   128→  osc1: OscillatorNode;
   129→  osc2: OscillatorNode;
   130→  g1: GainNode;
   131→  g2: GainNode;
   132→  filter: BiquadFilterNode;
   133→  vca: GainNode;
   134→  lfo: OscillatorNode;
   135→  lfoGain: GainNode;
   136→  bus: GainNode | null = null;
   137→
   138→  constructor(ctx: AudioContext) {
   139→    this.osc1 = ctx.createOscillator();
   140→    this.osc2 = ctx.createOscillator();
   141→    this.g1 = ctx.createGain();
   142→    this.g2 = ctx.createGain();
   143→    this.filter = ctx.createBiquadFilter();
   144→    this.filter.type = 'lowpass';
   145→    this.vca = ctx.createGain();
   146→    this.vca.gain.value = 0;
   147→    this.lfo = ctx.createOscillator();
   148→    this.lfoGain = ctx.createGain();
   149→    this.lfoGain.gain.value = 0;
   150→
   151→    this.osc1.connect(this.g1);
   152→    this.osc2.connect(this.g2);
   153→    this.g1.connect(this.filter);
   154→    this.g2.connect(this.filter);
   155→    this.filter.connect(this.vca);
   156→    this.lfo.connect(this.lfoGain);
   157→
   158→    // Start persistent oscillators (never stop them)
   159→    this.osc1.start();
   160→    this.osc2.start();
   161→    this.lfo.start();
   162→  }
   163→
   164→  connect(bus: GainNode) {
   165→    if (this.bus !== bus) {
   166→      this.vca.disconnect();
   167→      this.vca.connect(bus);
   168→      this.lfoGain.disconnect();
   169→      this.lfoGain.connect(this.filter.frequency);
   170→      this.bus = bus;
   171→    }
   172→  }
   173→
   174→  noteOn(p: SynthPreset, when: number, midi: number, vel: number, stepDur: number, bus: GainNode) {
   175→    this.connect(bus);
   176→    const f = mtof(clamp(midi, 12, 108));
   177→    const gate = p.gate || 0.6;
   178→    const dur = stepDur * gate * 2;
   179→    const rel = Math.max(p.rel, 0.02);
   180→    const end = when + dur;
   181→
   182→    this.osc1.type = p.wave1;
   183→    this.osc2.type = p.wave2;
   184→    this.osc1.frequency.setValueAtTime(f, when);
   185→    this.osc2.frequency.setValueAtTime(f * Math.pow(2, p.oct2 || 0), when);
   186→    this.osc2.detune.setValueAtTime(p.detune || 0, when);
   187→    this.g1.gain.setValueAtTime(0.6, when);
   188→    this.g2.gain.setValueAtTime(0.45, when);
   189→
   190→    const cut = clamp(p.cutoff, 60, 16000);
   191→    const res = clamp(p.res, 0.2, 24);
   192→    this.filter.type = p.fType;
   193→    this.filter.Q.setValueAtTime(res, when);
   194→    this.filter.frequency.cancelScheduledValues(when);
   195→    this.filter.frequency.setValueAtTime(Math.min(cut * 3, 16000), when);
   196→    this.filter.frequency.exponentialRampToValueAtTime(cut, when + Math.max((p.atk + p.dec * 0.7), 0.01));
   197→
   198→    if (p.lfoRate > 0 && p.lfoDest === 'cutoff') {
   199→      this.lfo.frequency.setValueAtTime(p.lfoRate, when);
   200→      this.lfoGain.gain.setValueAtTime(p.lfoDepth * 3000, when);
   201→    } else {
   202→      this.lfoGain.gain.setValueAtTime(0, when);
   203→    }
   204→
   205→    const vca = this.vca.gain;
   206→    const atk = Math.max(p.atk, 0.003);
   207→    vca.cancelScheduledValues(when);
   208→    vca.setValueAtTime(0, when);
   209→    vca.linearRampToValueAtTime(vel * 0.5, when + atk);
   210→    vca.setTargetAtTime(vel * 0.5 * p.sus, when + atk, Math.max(p.dec / 3, 0.01));
   211→    vca.setTargetAtTime(0.0001, end, Math.max(rel / 3, 0.008));
   212→  }
   213→
   214→  panic(ctx: AudioContext) {
   215→    try {
   216→      this.vca.gain.cancelScheduledValues(0);
   217→      this.vca.gain.setValueAtTime(0, ctx.currentTime);
   218→    } catch {}
   219→  }
   220→}

   222→// ─── Pooled Drum Voice (from PSY6 — multi-type) ────────────────────────────

   224→class PooledDrumVoice {
   225→  noise: AudioBufferSourceNode;
   226→  noiseGain: GainNode;
   227→  nFilter: BiquadFilterNode;
   228→  osc: OscillatorNode;
   229→  oscGain: GainNode;
   230→  out: GainNode;
   231→  bus: GainNode | null = null;
   232→
   233→  constructor(ctx: AudioContext, noiseBuffer: AudioBuffer) {
   234→    this.noise = ctx.createBufferSource();
   235→    this.noise.buffer = noiseBuffer;
   236→    this.noise.loop = true;
   237→    this.noiseGain = ctx.createGain();
   238→    this.noiseGain.gain.value = 0;
   239→    this.nFilter = ctx.createBiquadFilter();
   240→    this.nFilter.type = 'bandpass';
   241→    this.noise.connect(this.nFilter);
   242→    this.nFilter.connect(this.noiseGain);
   243→    this.osc = ctx.createOscillator();
   244→    this.osc.type = 'sine';
   245→    this.oscGain = ctx.createGain();
   246→    this.oscGain.gain.value = 0;
   247→    this.osc.connect(this.oscGain);
   248→    this.out = ctx.createGain();
   249→    this.noiseGain.connect(this.out);
   250→    this.oscGain.connect(this.out);
   251→    this.noise.start();
   252→    this.osc.start();
   253→  }
   254→
   255→  connect(bus: GainNode) {
   256→    if (this.bus !== bus) {
   257→      this.out.disconnect();
   258→      this.out.connect(bus);
   259→      this.bus = bus;
   260→    }
   261→  }
   262→
   263→  /**
   264→   * Trigger this voice. The optional `decayOverride` lets callers (e.g. reference
   265→   * pursuit) replace the preset decay with a blended value to chase the radio's
   266→   * kick decay. It is guarded against NaN/zero/out-of-range values.
   267→   */
   268→  hit(p: DrumPreset, when: number, vel: number, bus: GainNode, decayOverride?: number) {
   269→    this.connect(bus);
   270→    const tune = p.tune || 1;
   271→    const candidateDecay = (typeof decayOverride === 'number'
   272→      && isFinite(decayOverride) && decayOverride > 0.001 && decayOverride < 50)
   273→      ? decayOverride
   274→      : null;
   275→    const decay = candidateDecay ?? p.decay ?? 1;
   276→    const tone = p.tone || 1;
   277→    const punch = p.punch || 0;
   278→    const type = p.type;
   279→    const ng = this.noiseGain.gain;
   280→    const og = this.oscGain.gain;
   281→    ng.cancelScheduledValues(when);
   282→    og.cancelScheduledValues(when);
   283→    ng.setValueAtTime(0, when);
   284→    og.setValueAtTime(0, when);
   285→
   286→    if (type === 'kick') {
   287→      const dur = 0.12 + 0.5 * decay;
   288→      this.osc.frequency.setValueAtTime(180 * tune, when);
   289→      this.osc.frequency.exponentialRampToValueAtTime(Math.max(36 * tune, 24), when + 0.09);
   290→      og.setValueAtTime(vel * 1.1, when);
   291→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   292→      if (punch > 0) {
   293→        ng.setValueAtTime(vel * punch * 0.8, when);
   294→        ng.exponentialRampToValueAtTime(0.0001, when + 0.02);
   295→        this.nFilter.frequency.setValueAtTime(2500, when);
   296→      }
   297→    } else if (type === 'clap') {
   298→      const dur = 0.25 + 0.15 * decay;
   299→      this.nFilter.type = 'bandpass';
   300→      this.nFilter.frequency.setValueAtTime(1150 * tone, when);
   301→      this.nFilter.Q.value = 1.3;
   302→      ng.setValueAtTime(0, when);
   303→      [0, 0.014, 0.03].forEach(t2 => {
   304→        ng.setValueAtTime(0, when + t2);
   305→        ng.linearRampToValueAtTime(vel * 0.9, when + t2 + 0.002);
   306→        ng.exponentialRampToValueAtTime(0.02, when + t2 + 0.012);
   307→      });
   308→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   309→    } else if (type === 'hatC' || type === 'hatO') {
   310→      const open = type === 'hatO';
   311→      const dur = open ? 0.26 + 0.5 * decay : 0.03 + 0.05 * decay;
   312→      this.nFilter.type = 'highpass';
   313→      this.nFilter.frequency.setValueAtTime(7200 * Math.sqrt(tone), when);
   314→      ng.setValueAtTime(vel * (open ? 0.4 : 0.5), when);
   315→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   316→    } else if (type === 'tom') {
   317→      const dur = 0.22 + 0.35 * decay;
   318→      this.osc.type = 'sine';
   319→      this.osc.frequency.setValueAtTime(180 * tune, when);
   320→      this.osc.frequency.exponentialRampToValueAtTime(92 * tune, when + dur * 0.7);
   321→      og.setValueAtTime(vel * 0.9, when);
   322→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   323→    } else if (type === 'shaker') {
   324→      const dur = 0.04 + 0.07 * decay;
   325→      this.nFilter.type = 'highpass';
   326→      this.nFilter.frequency.setValueAtTime(6000 * tone, when);
   327→      ng.setValueAtTime(vel * 0.45, when);
   328→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   329→    } else if (type === 'glitch') {
   330→      const dur = 0.08 + 0.14 * decay;
   331→      this.nFilter.type = 'bandpass';
   332→      this.nFilter.frequency.setValueAtTime(1500 * tone + 800, when);
   333→      this.nFilter.Q.value = 4;
   334→      ng.setValueAtTime(vel * 0.7, when);
   335→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   336→    }
   337→  }
   338→
   339→  panic(ctx: AudioContext) {
   340→    try {
   341→      this.noiseGain.gain.cancelScheduledValues(0);
   342→      this.noiseGain.gain.setValueAtTime(0, ctx.currentTime);
   343→      this.oscGain.gain.cancelScheduledValues(0);
   344→      this.oscGain.gain.setValueAtTime(0, ctx.currentTime);
   345→    } catch {}
   346→  }
   347→}

   465→    }
   466→    this.saturator.curve = satCurve;
   467→
   468→    // Tone shelves (final EQ)
   469→    this.toneLow = c.createBiquadFilter();
   470→    this.toneLow.type = 'lowshelf';
   471→    this.toneLow.frequency.value = 110;
   472→    this.toneLow.gain.value = 1.5;
   473→    this.toneHigh = c.createBiquadFilter();
   474→    this.toneHigh.type = 'highshelf';
   475→    this.toneHigh.frequency.value = 8500;
   476→    this.toneHigh.gain.value = -2;
   477→
   478→    this.comp = c.createDynamicsCompressor();
   479→    this.comp.threshold.value = -8;
   480→    this.comp.knee.value = 12;
   481→    this.comp.ratio.value = 6;
   482→    this.comp.attack.value = 0.003;
   483→    this.comp.release.value = 0.2;
   484→    this.analyser = c.createAnalyser();
   485→    this.analyser.fftSize = 2048;
   486→
   487→    // Connect: master → saturator → toneLow → toneHigh → comp → analyser → destination
   488→    this.master.connect(this.saturator);
   489→    this.saturator.connect(this.toneLow);
   490→    this.toneLow.connect(this.toneHigh);
   491→    this.toneHigh.connect(this.comp);
   492→    this.comp.connect(this.analyser);
   493→    this.analyser.connect(c.destination);
   494→
   495→    // Delay (ping-pong with band-limited feedback)
   496→    this.delaySend = c.createGain();
   497→    this.delaySend.gain.value = 0.15;
   498→    this.delay = c.createDelay(0.5);
   499→    this.delay.delayTime.value = 0.375;
   500→    this.delayFb = c.createGain();
   501→    this.delayFb.gain.value = 0.35;
   502→    this.delayReturn = c.createGain();
   503→    this.delayReturn.gain.value = 0.5;
   504→    const dLP = c.createBiquadFilter();
   505→    dLP.type = 'lowpass';
   506→    dLP.frequency.value = 3300;
   507→    this.delaySend.connect(this.delay);
   508→    this.delay.connect(dLP);
   509→    dLP.connect(this.delayFb);
   510→    this.delayFb.connect(this.delay);
   511→    this.delay.connect(this.delayReturn);
   512→    this.delayReturn.connect(this.master);
   513→
   514→    // Reverb
   515→    this.reverbSend = c.createGain();
   516→    this.reverbSend.gain.value = 0.12;
   517→    this.reverb = c.createConvolver();
   518→    const irLen = Math.floor(c.sampleRate * 1.5);
   519→    const ir = c.createBuffer(2, irLen, c.sampleRate);
   520→    for (let ch = 0; ch < 2; ch++) {
   521→      const d = ir.getChannelData(ch);
   522→      for (let i = 0; i < irLen; i++) {
   523→        d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (c.sampleRate * 0.4));
   524→      }
   525→    }
   526→    this.reverb.buffer = ir;
   527→    this.reverbReturn = c.createGain();
   528→    this.reverbReturn.gain.value = 0.6;
   529→    this.reverbSend.connect(this.reverb);
   530→    this.reverb.connect(this.reverbReturn);
   531→    this.reverbReturn.connect(this.master);
   532→
   533→    // Track buses (8 tracks) with HPF, pan, and duck
   534→    this.duckGain = c.createGain();
   535→    this.duckGain.gain.value = 1.0;
   536→    // duckGain connects to master ONCE (not twice — was causing feedback)
   537→    this.duckGain.connect(this.master);
   538→
   539→    for (let i = 0; i < 8; i++) {
   540→      const bus = c.createGain();
   541→      // HPF on each track to clean mud (except kick/bass)
   542→      const hpf = c.createBiquadFilter();
   543→      hpf.type = 'highpass';
   544→      hpf.frequency.value = i < 2 ? 20 : (i < 4 ? 80 : 120);
   545→      // Stereo panner
   546→      const panner = c.createStereoPanner();
   547→      const panValues = [0, -0.15, 0.2, -0.2, 0, 0.1, -0.1, 0.15];
   548→      panner.pan.value = panValues[i];
   549→      // Track gain
   550→      const gain = c.createGain();
   551→      gain.gain.value = 0.8;
   552→
   553→      bus.connect(hpf);
   554→      hpf.connect(panner);
   555→      // Bass track (4) goes through duck gain for sidechain
   556→      if (i === 4) {
   557→        panner.connect(this.duckGain);
   558→      } else {
   559→        panner.connect(gain);
   560→        gain.connect(this.master);
   561→      }
   562→
   563→      // Send to delay/reverb
   564→      const dSend = c.createGain();
   565→      dSend.gain.value = i >= 4 ? 0.15 : 0.02;
   566→      bus.connect(dSend);
   567→      dSend.connect(this.delaySend);
   568→      const rSend = c.createGain();
   569→      rSend.gain.value = i >= 5 ? 0.2 : 0.03;
   570→      bus.connect(rSend);
   571→      rSend.connect(this.reverbSend);
   572→
   573→      this.chains.push(bus);
   574→      this.trackGains.push(gain);
   575→    }
   576→    // duckGain already connected to master above (line 458)
   577→
   578→    // Allocate voice pools (20 synth + 24 drum — from PSY6)
   579→    for (let i = 0; i < 20; i++) this.synthPool.push(new PooledSynthVoice(c));
   580→    for (let i = 0; i < 24; i++) this.drumPool.push(new PooledDrumVoice(c, this.noiseBuffer));
   581→
   582→    // Initialize tracks
   583→    this.initTracks();
   584→  }

   586→  private initTracks(): void {
   587→    const names = ['KICK', 'SNARE', 'HATS', 'PERC', 'BASS', 'LEAD', 'PAD', 'ARP'];
   588→    const presets = ['PS-KICK-TIGHT', 'TR-CLAP', 'PS-HAT', 'PS-PERC', 'PS-BASS-ROLL', 'PS-LEAD-SQUELCH', 'PS-PAD-PSYCH', 'PS-ARP-ACID'];
   589→    // Boosted track volumes for commercial loudness
   590→    const vols = [1.0, 0.6, 0.5, 0.4, 1.2, 0.7, 0.5, 0.5];
   591→    this.tracks = [];

   593→      this.tracks.push({
   594→        idx: i,
   595→        kind: i < 4 ? 'drum' : 'synth',
   596→        name: names[i],
   597→        presetId: presets[i],
   598→        mix: { vol: vols[i], pan: 0, mute: false, sendA: 0, sendB: 0 },
   599→      });
   600→    }

   606→  start(worldId?: string): void {
   607→    this.init();
   608→    if (this.ctx!.state === 'suspended') this.ctx!.resume();
   609→
   610→    // ── World-driven configuration (Track A) ──
   611→    this.currentWorld = WORLDS[worldId as WorldId] || WORLDS['dark-psy'];
   612→    this._bpm = this.currentWorld.defaultBpm;
   613→    this.musicalKey = {
   614→      root: Math.floor((this.currentWorld.rootRange[0] + this.currentWorld.rootRange[1]) / 2),
   615→      scale: this.currentWorld.defaultScale,
   616→    };
   617→    // Re-create musical generators with the world's key (LeadMotif, AcidPattern)
   618→    this.refreshMusicalGenerators();
   619→    this.arpIdx = 0;
   620→    this.bassPatternIdx = 0;
   621→
   622→    // Apply world FX mix to reverb and delay sends
   623→    const fxMix = this.currentWorld.fxMix;
   624→    this.reverbSend.gain.setTargetAtTime(0.04 + fxMix * 0.22, this.ctx!.currentTime, 0.05);
   625→    this.delaySend.gain.setTargetAtTime(0.05 + fxMix * 0.30, this.ctx!.currentTime, 0.05);
   626→
   627→    if (this.playing) return;
   628→    this.playing = true;
   629→    this.step = 0;
   630→    this.bar = 0;
   631→    this.sectionIdx = 0;
   632→    this.currentSection = this.arrangement[0].label;
   633→    this.onSectionChange?.(this.currentSection);
   634→    this.nextTime = this.ctx!.currentTime + 0.03;
   635→    this.scheduleNextTick();
   636→  }

   638→  stop(): void {

   641→    if (this.ctx) {
   642→      for (const v of this.synthPool) v.panic(this.ctx);
   643→      for (const v of this.drumPool) v.panic(this.ctx);
   644→    }

   656→  applyMusicalUnderstanding(understanding: {
   657→    key: { root: number; scale: string; confidence: number };
   658→    bpm: number;
   659→    bpmConfidence: number;
   660→    style?: string;
   661→    styleConfidence?: number;
   662→  }): void {
   663→    // ── KEY PURSUIT — refresh lead/acid generators when key changes ──
   664→    if (understanding.key.confidence > 0.3) {
   665→      const newRoot = 36 + understanding.key.root;
   666→      const newScale = understanding.key.scale;
   667→      const changed = newRoot !== this.musicalKey.root || newScale !== this.musicalKey.scale;
   668→      this.musicalKey = { root: newRoot, scale: newScale };
   669→      if (changed) this.refreshMusicalGenerators();
   670→    }
   671→
   672→    // ── CONTINUOUS BPM TRACKING (smooth ramp for large diffs) ──
   673→    // Always honor the new BPM when confidence is high. Small diffs (<=2 BPM)
   674→    // are applied immediately; larger diffs ramp across 4 bars to avoid tempo
   675→    // glitches and keep the scheduler's lookahead stable.
   676→    if (understanding.bpm > 0 && understanding.bpmConfidence > 0.5) {
   677→      const target = Math.round(understanding.bpm);
   678→      const diff = target - this._bpm;
   679→      if (Math.abs(diff) > 2) {
   680→        this.targetBpm = clamp(target, 60, 200);
   681→        this.bpmRampBarsLeft = 4;
   682→        this.bpmRampPerBar = diff / 4;
   683→      } else {
   684→        this._bpm = clamp(target, 60, 200);
   685→        this.targetBpm = 0;
   686→        this.bpmRampBarsLeft = 0;
   687→        this.bpmRampPerBar = 0;
   688→      }
   689→    }
   690→
   691→    // Auto-switch presets based on detected style
   692→    if (understanding.style && (understanding.styleConfidence ?? 0) > 0.4) {
   693→      this.applyStyle(understanding.style);
   694→    }
   695→  }

   697→  /**
   698→   * Re-create the LeadMotif and AcidPattern with the current musicalKey.
   699→   * Called whenever the reference listener reports a new key — this is what
   700→   * makes the engine actually pursue the radio's tonal center, not just store it.
   701→   */
   702→  private refreshMusicalGenerators(): void {
   703→    const seed = (this.musicalKey.root * 31 + this.musicalKey.scale.length * 7 + 11) >>> 0;
   704→    this.musicRng = new SeededRng(seed);
   705→    this.leadMotif = new LeadMotif(this.musicalKey.root, this.musicalKey.scale, this.musicRng);
   706→    this.acidPattern = new AcidPattern(this.musicalKey.root, this.musicalKey.scale, this.musicRng);
   707→  }

   730→  private targetEnergy = 0.5;
   731→
   732→  /**
   733→   * Receive fresh metrics from the reference radio and store them as pursuit
   734→   * targets. Also applies the smooth sub/high energy gain ramps immediately
   735→   * (these are the only adjustments that can ramp across long time constants
   736→   * without per-note cooperation — kick decay / centroid / transient density
   737→   * are applied per-note in triggerDrum / triggerSynth / scheduleStep).
   738→   */
   739→  liveTrack(refMetrics: {
   740→    lufs: number;
   741→    kickDecayMs: number;
   742→    spectralCentroid: number;
   743→    energy?: number;
   744→    subEnergy?: number;
   745→    highEnergy?: number;
   746→    transientDensity?: number;
   747→    bassDecayMs?: number;
   748→  }): void {
   749→    if (isFinite(refMetrics.lufs)) this.targetLufs = refMetrics.lufs;
   750→    if (refMetrics.energy !== undefined && isFinite(refMetrics.energy)) {
   751→      this.targetEnergy = clamp(refMetrics.energy, 0, 1);
   752→    }
   753→
   754→    // ── KICK DECAY target (seconds) ──
   755→    if (refMetrics.kickDecayMs > 50 && refMetrics.kickDecayMs < 800) {
   756→      this.refKickDecay = clamp(refMetrics.kickDecayMs / 1000, 0.05, 0.8);
   757→    }
   758→
   759→    // ── SPECTRAL CENTROID target (Hz) ──
   760→    if (isFinite(refMetrics.spectralCentroid) && refMetrics.spectralCentroid > 0) {
   761→      this.refSpectralCentroid = clamp(refMetrics.spectralCentroid, 100, 12000);
   762→    }
   763→
   764→    // ── TRANSIENT DENSITY target (transients/sec) ──
   765→    const td = refMetrics.transientDensity ?? 0;
   766→    if (isFinite(td) && td > 0) {
   767→      this.refTransientDensity = clamp(td, 0, 40);
   768→    }
   769→
   770→    // ── SUB / HIGH ENERGY targets (0..1) ──
   771→    if (refMetrics.subEnergy !== undefined && isFinite(refMetrics.subEnergy)) {
   772→      this.refSubEnergy = clamp(refMetrics.subEnergy, 0, 1);
   773→    }
   774→    if (refMetrics.highEnergy !== undefined && isFinite(refMetrics.highEnergy)) {
   775→      this.refHighEnergy = clamp(refMetrics.highEnergy, 0, 1);
   776→    }
   777→
   778→    // ── BASS DECAY target (seconds) ──
   779→    if (refMetrics.bassDecayMs !== undefined && refMetrics.bassDecayMs > 20 && refMetrics.bassDecayMs < 1500) {
   780→      this.refBassDecay = clamp(refMetrics.bassDecayMs / 1000, 0.05, 1.5);
   781→    }
   782→
   783→    // ── SUB / HIGH energy balancing — smooth ramp on track gains ──
   784→    // Boost bass track (4) when ref has more sub than we do; boost lead/pad/arp
   785→    // (5,6,7) when ref has more high energy. Time constants 0.8-1.0s (timbre).
   786→    if (this.ctx) {
   787→      const now = this.ctx.currentTime;
   788→      if (this.refSubEnergy > 0 && this.ownSubEnergy > 0) {
   789→        const subDiff = this.refSubEnergy - this.ownSubEnergy;
   790→        if (Math.abs(subDiff) > 0.05) {
   791→          const bassAdj = clamp(subDiff * 0.4, -0.3, 0.3);
   792→          const bassTarget = clamp(0.8 + bassAdj, 0.3, 2.0);
   793→          this.trackGains[4]?.gain.setTargetAtTime(bassTarget, now, 0.8);
   794→          if (subDiff > 0.05) {
   795→            const kickTarget = clamp(1.0 + subDiff * 0.2, 0.6, 1.8);
   796→            this.trackGains[0]?.gain.setTargetAtTime(kickTarget, now, 0.8);
   797→          }
   798→        }
   799→      }
   800→      if (this.refHighEnergy > 0 && this.ownHighEnergy > 0) {
   801→        const highDiff = this.refHighEnergy - this.ownHighEnergy;
   802→        if (Math.abs(highDiff) > 0.05) {
   803→          const adj = clamp(highDiff * 0.5, -0.3, 0.3);
   804→          for (const ti of [5, 6, 7]) {
   805→            const target = clamp(0.8 + adj, 0.3, 1.6);
   806→            this.trackGains[ti]?.gain.setTargetAtTime(target, now, 1.0);
   807→          }
   808→        }
   809→      }
   810→    }
   811→  }
   812→
   813→  /**
   814→   * Receive fresh metrics from our own engine output (via SelfAnalyzer).
   815→   * Stores own sub/high/centroid/transient for getPursuitStatus() and the
   816→   * sub/high balancing done in liveTrack().
   817→   */
   818→  selfTrack(selfMetrics: {
   819→    lufs: number;
   820→    energy?: number;
   821→    spectralCentroid?: number;
   822→    transientDensity?: number;
   823→    subEnergy?: number;
   824→    highEnergy?: number;
   825→  }): void {
   826→    this.ownLufs = selfMetrics.lufs;
   827→    if (selfMetrics.spectralCentroid !== undefined && isFinite(selfMetrics.spectralCentroid) && selfMetrics.spectralCentroid > 0) {
   828→      this.ownSpectralCentroid = selfMetrics.spectralCentroid;
   829→    }
   830→    if (selfMetrics.transientDensity !== undefined && isFinite(selfMetrics.transientDensity)) {
   831→      this.ownTransientDensity = selfMetrics.transientDensity;
   832→    }
   833→    if (selfMetrics.subEnergy !== undefined && isFinite(selfMetrics.subEnergy)) {
   834→      this.ownSubEnergy = selfMetrics.subEnergy;
   835→    }
   836→    if (selfMetrics.highEnergy !== undefined && isFinite(selfMetrics.highEnergy)) {
   837→      this.ownHighEnergy = selfMetrics.highEnergy;
   838→    }
   839→    // LUFS matching
   840→    if (this.targetLufs !== 0 && Math.abs(selfMetrics.lufs - this.targetLufs) > 1.0) {
   841→      const diff = this.targetLufs - selfMetrics.lufs;
   842→      const adj = diff > 0 ? 0.08 : -0.08;
   843→      const newMaster = clamp(this.master.gain.value + adj, 0.3, 2.0);
   844→      this.master.gain.setTargetAtTime(newMaster, this.ctx!.currentTime, 0.15);
   845→    }
   846→    // Energy matching — adjust track volumes
   847→    if (this.targetEnergy > 0 && selfMetrics.energy !== undefined) {
   848→      const energyDiff = this.targetEnergy - selfMetrics.energy;
   849→      if (Math.abs(energyDiff) > 0.05) {
   850→        // Boost/reduce all track gains slightly
   851→        const volAdj = energyDiff > 0 ? 0.02 : -0.02;
   852→        for (let i = 0; i < 8; i++) {
   853→          const g = this.trackGains[i];
   854→          if (g) {
   855→            const newVol = clamp(g.gain.value + volAdj, 0.1, 2.0);
   856→            g.gain.setTargetAtTime(newVol, this.ctx!.currentTime, 0.5);
   857→          }
   858→        }
   859→      }
   860→    }
   861→  }
   862→
   863→  setWorld(params: Record<string, number>): void {
   864→    if (params.masterLevel !== undefined) {
   865→      this.master.gain.setTargetAtTime(params.masterLevel, this.ctx!.currentTime, 0.1);
   866→    }
   867→    if (params.bassLevel !== undefined) {
   868→      this.trackGains[4].gain.setTargetAtTime(params.bassLevel * 0.8, this.ctx!.currentTime, 0.1);
   869→    }
   870→    if (params.leadLevel !== undefined) {
   871→      this.trackGains[5].gain.setTargetAtTime(params.leadLevel * 0.8, this.ctx!.currentTime, 0.1);
   872→    }
   873→    if (params.kickLevel !== undefined) {
   874→      this.trackGains[0].gain.setTargetAtTime(params.kickLevel * 0.8, this.ctx!.currentTime, 0.1);
   875→    }
   876→  }

   878→  getAnalyser(): AnalyserNode | null { return this.analyser; }

   880→  private scheduleNextTick(): void {
   881→    if (!this.playing) return;
   882→    this.timer = setTimeout(() => {
   883→      this.tick();
   884→      this.scheduleNextTick();
   885→    }, 15);
   886→  }
   887→
   888→  private tick(): void {
   889→    if (!this.playing || !this.ctx) return;
   890→    const lookahead = 0.06;
   891→
   892→    while (this.nextTime < this.ctx.currentTime + lookahead) {
   893→      // Recompute s16 each step so the BPM ramp changes tempo smoothly
   894→      // without invalidating the scheduler's lookahead window.
   895→      const s16 = 60 / this.bpm / 4;
   896→      this.scheduleStep(this.step, this.bar, this.nextTime);
   897→      this.step++;
   898→      this.nextTime += s16;
   899→      if (this.step >= 16) {
   900→        this.step = 0;
   901→        this.bar++;
   902→        // ── BPM ramp smoothing (one step per bar, over 4 bars total) ──
   903→        if (this.bpmRampBarsLeft > 0 && this.bpmRampPerBar !== 0) {
   904→          const stepped = this._bpm + this.bpmRampPerBar;
   905→          this._bpm = clamp(Math.round(stepped * 10) / 10, 60, 200);
   906→          this.bpmRampBarsLeft--;
   907→          if (this.bpmRampBarsLeft <= 0) {
   908→            // Snap to final target at end of ramp
   909→            this._bpm = this.targetBpm || this._bpm;
   910→            this.targetBpm = 0;
   911→            this.bpmRampPerBar = 0;
   912→          }
   913→        }
   914→        const section = this.arrangement[this.sectionIdx % this.arrangement.length];
   915→        if (this.bar >= section.bars) {
   916→          this.sectionIdx++;
   917→          this.bar = 0;
   918→          const next = this.arrangement[this.sectionIdx % this.arrangement.length];
   919→          this.currentSection = next.label;
   920→          this.onSectionChange?.(this.currentSection);
   921→          // Evolve lead motif at section boundaries for musical development
   922→          this.leadMotif?.evolve();
   923→          // Mutate arp shape at section boundaries, rate controlled by world.evolutionRate
   924→          if (this.musicRng && this.musicRng.chance(this.currentWorld.evolutionRate)) {
   925→            this.arpIdx = (this.arpIdx + 1) % 4;
   926→          }
   927→        }
   928→        // Change preset every 4 bars
   929→        if (this.bar % 4 === 0 && this.bar > 0) {
   930→          this.presetIdx = (this.presetIdx + 1) % 3;
   931→          this.rotatePresets();
   932→          // Rotate bass pattern every 4 bars for variation
   933→          const bassStyle = this.deriveBassStyle();
   934→          const bps = BASS_PATTERNS[bassStyle] || BASS_PATTERNS.off;
   935→          this.bassPatternIdx = (this.bassPatternIdx + 1) % bps.length;
   936→        }
   937→      }
   938→    }
   939→  }
   940→
   941→  private rotatePresets(): void {
   942→    // Rotate drum presets for variety
   943→    const kickPresets = ['PS-KICK-TIGHT', 'PS-KICK-DEEP'];
   944→    const bassPresets = ['PS-BASS-ROLL', 'PS-BASS-DEEP'];
   945→    const leadPresets = ['PS-LEAD-SQUELCH', 'PS-LEAD-FMTEX'];
   946→    this.tracks[0].presetId = kickPresets[this.presetIdx % kickPresets.length];
   947→    this.tracks[4].presetId = bassPresets[this.presetIdx % bassPresets.length];
   948→    this.tracks[5].presetId = leadPresets[this.presetIdx % leadPresets.length];
   949→  }
   950→
   951→  /** Derive bass style from world id (worlds.ts doesn't have a 'bass' field). */
   952→  private deriveBassStyle(): 'roll' | 'acid' | 'off' {
   953→    const id = this.currentWorld.id;
   954→    if (id.includes('dark') || id.includes('forest')) return 'roll';
   955→    if (id.includes('goa') || id.includes('acid')) return 'acid';
   956→    return 'off';
   957→  }

   959→  private scheduleStep(step: number, bar: number, time: number): void {
   960→    const w = this.currentWorld;
   961→    const section = this.arrangement[this.sectionIdx % this.arrangement.length];
   962→    const key = this.musicalKey;
   963→    const root = key.root;
   964→    const sc = key.scale;
   965→    const sd = 60 / this.bpm / 4;
   966→
   967→    // ── Energy from world's energyCurve, modulated by section density ──
   968→    const eIdx = clamp(
   969→      Math.floor((bar / Math.max(1, section.bars)) * w.energyCurve.length),
   970→      0,
   971→      w.energyCurve.length - 1
   972→    );
   973→    const baseEnergy = w.energyCurve[eIdx];
   974→    const energy = clamp(baseEnergy * (0.4 + 0.6 * section.density), 0, 1);
   975→
   976→    // ── Swing: delay offbeat steps by swing * halfStep ──
   977→    let stepTime = time;
   978→    if (step % 2 === 1 && w.swing > 0) {
   979→      stepTime += w.swing * sd * 0.5;
   980→    }
   981→
   982→    const isPreDrop = (section.label === 'BUILD' || section.label === 'BUILD 2') && bar >= section.bars - 2;
   983→    const isDropStart = section.label.includes('DROP') && bar === 0 && step === 0;
   984→
   985→    // ── RISER FX (last 2 bars of build) — uses raw time, not swung ──
   986→    if (isPreDrop && step === 0 && bar === section.bars - 2) {
   987→      this.triggerRiser(time, sd * 32);
   988→    }
   989→
   990→    // ── IMPACT FX (drop start) ──
   991→    if (isDropStart) {
   992→      this.triggerImpact(time);
   993→    }
   994→
   995→    // Reference pursuit — scale hat/perc probability by refTransientDensity.
   996→    const tScale = this.refTransientDensity > 0
   997→      ? clamp(0.5 + this.refTransientDensity / 24, 0.3, 1.8)
   998→      : 1.0;
   999→    const tVelBoost = tScale > 1 ? (tScale - 1) * 0.5 : 0;
  1000→
  1001→    // ── Pre-compute world timbre overrides (cutoff/resonance modulated by character) ──
  1002→    const leadTimbre = {
  1003→      cutoff: w.leadTimbre.cutoff * (0.7 + 0.6 * w.brightness),
  1004→      res: 2 + w.leadTimbre.resonance * 12,
  1005→      drive: w.leadTimbre.drive,
  1006→    };
  1007→    const bassTimbre = {
  1008→      cutoff: w.bassTimbre.cutoff * (0.7 + 0.6 * (1 - w.darkness)),
  1009→      res: 2 + w.bassTimbre.resonance * 12,
  1010→      drive: w.bassTimbre.drive,
  1011→    };
  1012→    const padTimbre = {
  1013→      cutoff: w.padTimbre.cutoff * (0.6 + 0.8 * w.brightness),
  1014→      res: 2 + w.padTimbre.resonance * 12,
  1015→      drive: w.padTimbre.drive,
  1016→    };
  1017→    const arpTimbre = {
  1018→      cutoff: w.textureTimbre.cutoff * (0.7 + 0.6 * w.psychedelia),
  1019→      res: 2 + w.textureTimbre.resonance * 12,
  1020→      drive: w.textureTimbre.drive,
  1021→    };
  1022→
  1023→    // ── KICK (track 0) — world-driven kickPattern (16-char gate string) ──
  1024→    if (w.kickPattern.length === 16 && w.kickPattern.charAt(step) === 'x') {
  1025→      const isDownbeat = step % 4 === 0;
  1026→      const aggressionBoost = 0.7 + 0.6 * w.aggression;
  1027→      const vel = isDownbeat
  1028→        ? 0.5 + section.density * 0.4 * aggressionBoost
  1029→        : 0.4 * aggressionBoost;
  1030→      this.triggerDrum(0, stepTime, vel);
  1031→    }
  1032→
  1033→    // ── CLAP (track 1) — backbeat on 2 and 4, gated by section density ──
  1034→    if ((step === 4 || step === 12) && section.density > 0.4) {
  1035→      this.triggerDrum(1, stepTime, 0.3 + energy * 0.1);
  1036→    }
  1037→
  1038→    // ── HATS (track 2) — probability from world.hatDensity per eligible offbeat ──
  1039→    const hatProb = clamp(w.hatDensity * (0.5 + 0.5 * energy) * tScale, 0, 1);
  1040→    if (step % 2 === 1 && this.musicRng?.chance(hatProb)) {
  1041→      const vel = 0.15 + (step % 4 === 3 ? 0.1 : 0) + energy * 0.1 + tVelBoost;
  1042→      this.triggerDrum(2, stepTime, vel);
  1043→    }
  1044→
  1045→    // ── PERC (track 3) — probability from world.percDensity ──
  1046→    const percProb = clamp(w.percDensity * energy * tScale, 0, 1);
  1047→    if (section.density > 0.5 && (step === 6 || step === 14) && this.musicRng?.chance(percProb)) {
  1048→      this.triggerDrum(3, stepTime, 0.2 + tVelBoost);
  1049→    }
  1050→
  1051→    // ── BASS (track 4) — world-driven bassPattern + BASS_PATTERNS by derived style ──
  1052→    if (section.bass && w.bassPattern.length === 16 && w.bassPattern.charAt(step) === 'x') {
  1053→      const bassStyle = this.deriveBassStyle();
  1054→      const bps = BASS_PATTERNS[bassStyle] || BASS_PATTERNS.off;
  1055→      const bp = bps[this.bassPatternIdx % bps.length];
  1056→      const bassStep = Math.floor((step - 1) / 2) % bp.steps.length;
  1057→      const bassDeg = bp.steps[bassStep];
  1058→      if (bassDeg >= 0) {
  1059→        const note = scaleNote(root, sc, bassDeg);
  1060→        const accent = bp.accents[bassStep] ?? 1;
  1061→        this.triggerSynth(4, stepTime, note, 0.5 * accent, sd, undefined, bassTimbre);
  1062→      }
  1063→    }
  1064→
  1065→    // ── LEAD (track 5) — LeadMotif with AABA structure, gated by section + energy ──
  1066→    if (section.lead && this.leadMotif && energy > 0.35) {
  1067→      const noteInfo = this.leadMotif.nextNote(step, bar, energy, this.musicRng!);
  1068→      if (noteInfo) {
  1069→        this.triggerSynth(5, stepTime, noteInfo.note, noteInfo.velocity, sd, sd * 0.5, leadTimbre);
  1070→      }
  1071→    }
  1072→
  1073→    // ── PAD (track 6) — chord progression from PROGRESSIONS[scale], on bar downbeat in drops ──
  1074→    if (section.lead && step === 0) {
  1075→      const prog = PROGRESSIONS[sc] || PROGRESSIONS.minor;
  1076→      const chordDeg = prog[bar % prog.length];
  1077→      const chordRoot = scaleNote(root + 12, sc, chordDeg);
  1078→      this.triggerSynth(6, stepTime, chordRoot, 0.25, sd * 4, undefined, padTimbre);
  1079→      // Also play fifth for full chord
  1080→      this.triggerSynth(6, stepTime + 0.01, scaleNote(root + 12, sc, chordDeg + 4), 0.15, sd * 4, undefined, padTimbre);
  1081→    }
  1082→
  1083→    // ── ARP (track 7) — generated from scale degrees, mutation from evolutionRate ──
  1084→    const arpProb = clamp(0.7 * energy, 0, 1);
  1085→    if (section.lead && step % 2 === 0 && this.musicRng?.chance(arpProb)) {
  1086→      const arpShapes = [
  1087→        [0, 2, 4, 7, 4, 2, 0, 7],
  1088→        [0, 4, 7, 4, 0, 7, 4, 0],
  1089→        [0, 7, 4, 2, 4, 7, 12, 7],
  1090→        [0, 2, 4, 7, 12, 7, 4, 2],
  1091→      ];
  1092→      const arp = arpShapes[this.arpIdx % arpShapes.length];
  1093→      const arpStep = Math.floor(step / 2) % arp.length;
  1094→      const deg = arp[arpStep];
  1095→      const note = scaleNote(root + 24, sc, deg);
  1096→      this.triggerSynth(7, stepTime, note, 0.25 * energy, sd, undefined, arpTimbre);
  1097→    }
  1098→
  1099→    // ── SHAKER (track 3 alt) — continuous offbeat in drops ──
  1100→    const shakerProb = clamp(0.4 * energy * tScale, 0, 1);
  1101→    if (section.bass && section.lead && step % 2 === 1 && this.musicRng?.chance(shakerProb)) {
  1102→      this.triggerDrum(3, stepTime, 0.15 + tVelBoost);
  1103→    }
  1104→  }

  1106→  private triggerDrum(trackIdx: number, time: number, vel: number, decayOverride?: number): void {
  1107→    const track = this.tracks[trackIdx];
  1108→    if (track.mix.mute) return;
  1109→    const preset = DRUM_PRESETS[track.presetId];
  1110→    if (!preset) return;
  1111→    const voice = this.drumPool[this.drumIdx];
  1112→    this.drumIdx = (this.drumIdx + 1) % this.drumPool.length;
  1113→
  1114→    // Reference pursuit — KICK DECAY: blend preset decay with refKickDecay.
  1115→    // The kick dur formula is `dur = 0.12 + 0.5 * decay`, so to hit a target
  1116→    // seconds T the equivalent decay param is `(T - 0.12) / 0.5`. We blend
  1117→    // 50/50 with the preset decay so the kick keeps its tonal character but
  1118→    // adopts the reference's tail length.
  1119→    let effectiveDecayOverride = decayOverride;
  1120→    if (trackIdx === 0 && this.refKickDecay > 0) {
  1121→      const targetDur = clamp(this.refKickDecay, 0.05, 0.8);
  1122→      const refDecayParam = clamp((targetDur - 0.12) / 0.5, 0.05, 4.0);
  1123→      const blended = preset.decay * 0.5 + refDecayParam * 0.5;
  1124→      effectiveDecayOverride = (isFinite(blended) && blended > 0) ? blended : undefined;
  1125→    }
  1126→
  1127→    voice.hit(preset, time, vel * track.mix.vol, this.chains[trackIdx], effectiveDecayOverride);
  1128→
  1129→    // Sidechain: when kick fires, duck the bass
  1130→    if (trackIdx === 0 && this.duckGain && this.ctx) {
  1131→      this.duckGain.gain.cancelScheduledValues(time);
  1132→      this.duckGain.gain.setValueAtTime(1 - 0.4, time); // 40% duck
  1133→      this.duckGain.gain.linearRampToValueAtTime(1.0, time + 0.25); // 250ms recovery
  1134→    }
  1135→  }
  1136→
  1137→  private triggerRiser(time: number, dur: number): void {
  1138→    if (!this.ctx) return;
  1139→    const c = this.ctx;
  1140→    // Noise through filter that opens up
  1141→    const noise = c.createBufferSource();
  1142→    noise.buffer = this.noiseBuffer;
  1143→    noise.loop = true;
  1144→    const filter = c.createBiquadFilter();
  1145→    filter.type = 'bandpass';
  1146→    filter.frequency.setValueAtTime(200, time);
  1147→    filter.frequency.exponentialRampToValueAtTime(8000, time + dur);
  1148→    filter.Q.value = 2;
  1149→    const gain = c.createGain();
  1150→    gain.gain.setValueAtTime(0.001, time);
  1151→    gain.gain.exponentialRampToValueAtTime(0.3, time + dur);
  1152→    gain.gain.exponentialRampToValueAtTime(0.001, time + dur + 0.1);
  1153→    noise.connect(filter);
  1154→    filter.connect(gain);
  1155→    gain.connect(this.master);
  1156→    noise.start(time);
  1157→    noise.stop(time + dur + 0.2);
  1158→  }
  1159→
  1160→  private triggerImpact(time: number): void {
  1161→    if (!this.ctx) return;
  1162→    const c = this.ctx;
  1163→    // Sub boom + noise crack
  1164→    const osc = c.createOscillator();
  1165→    osc.type = 'sine';
  1166→    osc.frequency.setValueAtTime(120, time);
  1167→    osc.frequency.exponentialRampToValueAtTime(35, time + 0.5);
  1168→    const og = c.createGain();
  1169→    og.gain.setValueAtTime(0.8, time);
  1170→    og.gain.exponentialRampToValueAtTime(0.001, time + 0.6);
  1171→    osc.connect(og);
  1172→    og.connect(this.master);
  1173→    osc.start(time);
  1174→    osc.stop(time + 0.7);
  1175→
  1176→    const noise = c.createBufferSource();
  1177→    noise.buffer = this.noiseBuffer;
  1178→    const hp = c.createBiquadFilter();
  1179→    hp.type = 'highpass';
  1180→    hp.frequency.value = 3000;
  1181→    const ng = c.createGain();
  1182→    ng.gain.setValueAtTime(0.4, time);
  1183→    ng.gain.exponentialRampToValueAtTime(0.001, time + 0.05);
  1184→    noise.connect(hp);
  1185→    hp.connect(ng);
  1186→    ng.connect(this.master);
  1187→    noise.start(time);
  1188→    noise.stop(time + 0.1);
  1189→  }

  1191→  private triggerSynth(
  1192→    trackIdx: number,
  1193→    time: number,
  1194→    midi: number,
  1195→    vel: number,
  1196→    stepDur: number,
  1197→    dur?: number,
  1198→    timbre?: { cutoff?: number; res?: number; drive?: number }
  1199→  ): void {
  1200→    const track = this.tracks[trackIdx];
  1201→    if (track.mix.mute) return;
  1202→    const basePreset = SYNTH_PRESETS[track.presetId];
  1203→    if (!basePreset) return;
  1204→    const voice = this.synthPool[this.synthIdx];
  1205→    this.synthIdx = (this.synthIdx + 1) % this.synthPool.length;
  1206→
  1207→    // ── Apply world timbre overrides on top of the factory preset ──
  1208→    let preset: SynthPreset = basePreset;
  1209→    if (timbre) {
  1210→      preset = {
  1211→        ...basePreset,
  1212→        cutoff: timbre.cutoff !== undefined ? clamp(timbre.cutoff, 60, 16000) : basePreset.cutoff,
  1213→        res: timbre.res !== undefined ? clamp(timbre.res, 0.2, 24) : basePreset.res,
  1214→      };
  1215→    }
  1216→
  1217→    let p: SynthPreset = dur ? { ...preset, gate: dur / (stepDur * 2) } : preset;
  1218→
  1219→    // Reference pursuit — SPECTRAL CENTROID matching for lead (5) and pad (6).
  1220→    // Applied on top of the world timbre so radio brightness nudges the world cutoff.
  1221→    if ((trackIdx === 5 || trackIdx === 6) && this.refSpectralCentroid > 0) {
  1222→      const targetCut = centroidToCutoff(this.refSpectralCentroid);
  1223→      const blended = preset.cutoff * 0.6 + targetCut * 0.4;
  1224→      if (isFinite(blended) && blended > 60) {
  1225→        p = { ...p, cutoff: clamp(blended, 200, 12000) };
  1226→      }
  1227→    }
  1228→
  1229→    // Reference pursuit — BASS DECAY matching for bass (4).
  1230→    if (trackIdx === 4 && this.refBassDecay > 0) {
  1231→      const desiredGate = clamp(this.refBassDecay / Math.max(stepDur * 2, 0.01), 0.05, 2.5);
  1232→      const blended = preset.gate * 0.7 + desiredGate * 0.3;
  1233→      if (isFinite(blended) && blended > 0) {
  1234→        p = { ...p, gate: clamp(blended, 0.05, 2.5) };
  1235→      }
  1236→    }
  1237→
  1238→    // Drive scales the voice velocity (per-voice drive isn't a SynthPreset field)
  1239→    const driveBoost = timbre?.drive ? clamp(timbre.drive / 1.5, 0.5, 1.8) : 1;
  1240→    voice.noteOn(p, time, midi, vel * track.mix.vol * driveBoost, stepDur, this.chains[trackIdx]);
  1241→
  1242→    // Add sub oscillator for bass track (sine one octave below)
  1243→    if (trackIdx === 4 && this.ctx) {
  1244→      const subFreq = mtof(midi - 12); // one octave below
  1245→      const subOsc = this.ctx.createOscillator();
  1246→      const subGain = this.ctx.createGain();
  1247→      subOsc.type = 'sine';
  1248→      subOsc.frequency.value = subFreq;
  1249→      // When chasing ref bass decay, lengthen the sub-osc tail to match.
  1250→      const baseDur = dur || stepDur * 0.3;
  1251→      const subDecay = (this.refBassDecay > 0)
  1252→        ? clamp(baseDur * 0.5 + this.refBassDecay * 0.5, 0.05, 1.5)
  1253→        : baseDur + 0.05;
  1254→      subGain.gain.setValueAtTime(0.5 * track.mix.vol * driveBoost, time);
  1255→      subGain.gain.exponentialRampToValueAtTime(0.001, time + subDecay);
  1256→      subOsc.connect(subGain);
  1257→      subGain.connect(this.chains[4]);
  1258→      subOsc.start(time);
  1259→      subOsc.stop(time + subDecay + 0.02);
  1260→    }
  1261→  }

  1263→  /**
  1264→   * Snapshot of reference pursuit state for UI display. Each entry pairs the
  1265→   * radio target with our current actual so the UI can render a delta.
  1266→   * Values are zero when no reference data has arrived yet.
  1267→   */
  1268→  getPursuitStatus(): {
  1269→    kickDecay: { target: number; actual: number };
  1270→    centroid: { target: number; actual: number };
  1271→    transientDensity: { target: number; actual: number };
  1272→    bpm: { target: number; actual: number };
  1273→    key: { root: number; scale: string };
  1274→  } {
  1275→    // Actual kick decay = current kick preset's dur, blended with ref if pursuing.
  1276→    let actualKickDur = 0;
  1277→    const kickPreset = DRUM_PRESETS[this.tracks[0]?.presetId ?? ''];
  1278→    if (kickPreset) {
  1279→      const presetDur = 0.12 + 0.5 * (kickPreset.decay || 1);
  1280→      if (this.refKickDecay > 0) {
  1281→        const refDecayParam = clamp((this.refKickDecay - 0.12) / 0.5, 0.05, 4.0);
  1282→        const blendedDecay = kickPreset.decay * 0.5 + refDecayParam * 0.5;
  1283→        actualKickDur = 0.12 + 0.5 * blendedDecay;
  1284→      } else {
  1285→        actualKickDur = presetDur;
  1286→      }
  1287→    }
  1288→    return {
  1289→      kickDecay: { target: this.refKickDecay, actual: actualKickDur },
  1290→      centroid: { target: this.refSpectralCentroid, actual: this.ownSpectralCentroid },
  1291→      transientDensity: { target: this.refTransientDensity, actual: this.ownTransientDensity },
  1292→      bpm: { target: this.targetBpm || this._bpm, actual: this._bpm },
  1293→      key: { root: this.musicalKey.root, scale: this.musicalKey.scale },
  1294→    };
  1295→  }
  1296→
  1297→  getMusicalKey(): { root: number; scale: string } { return this.musicalKey; }
  1298→  getOwnLufs(): number { return this.ownLufs; }
  1299→}
