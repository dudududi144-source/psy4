     1→/**
     2→ * MelodyEngine — professional melodic development engine.
     3→ *
     4→ * Replaces the simple EvolvingSequence-based LeadMotif with a real
     5→ * developmental melodic engine. Implements the classical techniques used
     6→ * by Beethoven, Bach, and modern film/game composers:
     7→ *
     8→ *   - Motif generation with SINGABLE contour (steps over leaps, chord-tone
     9→ *     start/end, resolution-after-leap, octave+3rd range).
    10→ *   - Development techniques: transpose, invert, retrograde, fragment,
    11→ *     elongate (augmentation), shorten (diminution), sequence.
    12→ *   - Phrase structure: A A' B A'' (8-bar developmental phrases, not AABA).
    13→ *   - Tension curves (low/medium/high/peak) drive register, density,
    14→ *     duration, leap probability.
    15→ *   - Call-response: lead plays a "call", arp plays a "response" (descending,
    16→ *     ending on a stable tone) during variation sections.
    17→ *
    18→ * Design notes:
    19→ *   - Motifs store scale degrees (integers, can be negative/>7) so they
    20→ *     transpose cleanly across any scale.
    21→ *   - Strong beats (downbeats, beat 3) snap to chord tones from
    22→ *     PROGRESSIONS[scale] — this keeps the lead compatible with Track H1's
    23→ *     harmony engine (no chord clashes on strong beats).
    24→ *   - Weak beats allow passing tones / neighbor tones.
    25→ *   - All randomness is driven by the supplied SeededRng (deterministic).
    26→ *
    27→ * Task ID: M1 (Melody track).
    28→ */
    29→
    30→import { SeededRng, scaleNote, PROGRESSIONS } from './musicalGrammar';
    31→
    32→// ─── Types ────────────────────────────────────────────────────────────────
    33→
    34→export interface Motif {
    35→  /** Scale degrees (integers; negative or >7 wrap octaves via scaleNote). */
    36→  notes: number[];
    37→  /** Note durations in 16th-note steps (1 = 16th, 2 = 8th, 4 = quarter...). */
    38→  durations: number[];
    39→  /** Velocities 0..1. */
    40→  velocities: number[];
    41→  /** Per-position rest flag (true = this slot is a rest). */
    42→  rests: boolean[];
    43→}
    44→
    45→export type ContourShape = 'arch' | 'descending' | 'ascending' | 'wave';
    46→
    47→interface MotifEvent {
    48→  /** Absolute step within the 8-bar phrase (0..127). */
    49→  stepInPhrase: number;
    50→  /** Scale degree at this event. */
    51→  scaleDeg: number;
    52→  /** Velocity 0..1 (already section/energy weighted). */
    53→  velocity: number;
    54→  /** Duration in 16th steps (capped at 4 for synth gate safety). */
    55→  duration: number;
    56→  /** True = this slot is explicitly a rest. */
    57→  isRest: boolean;
    58→}
    59→
    60→// ─── Constants ────────────────────────────────────────────────────────────
    61→
    62→/** Stable scale degrees: 1st (root), 3rd, 5th. Used for phrase endings. */
    63→const STABLE_DEGREES = [0, 2, 4];
    64→
    65→/** Chord-tone offsets from a chord root (root, 3rd, 5th). */
    66→const CHORD_TONE_OFFSETS = [0, 2, 4];
    67→
    68→const clamp = (v: number, a: number, b: number) => (v < a ? a : v > b ? b : v);
    69→
    70→// ─── MelodyEngine ─────────────────────────────────────────────────────────
    71→
    72→export class MelodyEngine {
    73→  /** The current "call" motif — drives phrase A and the response. */
    74→  private currentMotif: Motif;
    75→  /** Previously generated motif — kept for variation source. */
    76→  private previousMotif: Motif | null = null;
    77→  /** Monotonic counter — used to vary tension across phrases. */
    78→  private phraseCount = 0;
    79→
    80→  /** Phrase geometry: 8 bars of 16 steps each = 128 16th-note steps. */
    81→  private readonly phraseBars = 8;
    82→  private readonly stepsPerBar = 16;
    83→  private readonly phraseSteps = 128;
    84→
    85→  /** Pre-built event table for the lead's current phrase. */
    86→  private phraseEvents: (MotifEvent | null)[] = new Array(128).fill(null);
    87→  /** Pre-built event table for the arp's response (counter-melody). */
    88→  private responseEvents: (MotifEvent | null)[] = new Array(128).fill(null);
    89→
    90→  /** Last call motif — used as source for generateResponse(). */
    91→  private lastCallMotif: Motif | null = null;
    92→  /** Cached energy/tension for incremental rebuilds. */
    93→  private lastEnergy = 0.5;
    94→  private lastTension = 0.5;
    95→
    96→  /** Track last bar at which we did an incremental evolution. */
    97→  private lastEvolveBar = -1;
    98→
    99→  constructor(
   100→    private root: number,
   101→    private scale: string,
   102→    private rng: SeededRng,
   103→  ) {
   104→    this.currentMotif = this.generateMotif(this.lastEnergy, this.lastTension);
   105→    this.buildPhrase(this.lastEnergy, this.lastTension);
   106→  }
   107→
   108→  // ─── Key / configuration ────────────────────────────────────────────────
   109→
   110→  /**
   111→   * Update root + scale (called on key change). Rebuilds the phrase so the
   112→   * lead immediately follows the new tonal center.
   113→   */
   114→  setKey(root: number, scale: string): void {
   115→    this.root = root;
   116→    this.scale = scale;
   117→    this.previousMotif = null;
   118→    this.lastCallMotif = null;
   119→    this.phraseCount = 0;
   120→    this.currentMotif = this.generateMotif(this.lastEnergy, this.lastTension);
   121→    this.buildPhrase(this.lastEnergy, this.lastTension);
   122→  }
   123→
   124→  // ─── Motif generation ──────────────────────────────────────────────────
   125→
   126→  /**
   127→   * Generate a fresh motif with SINGABLE contour (4-8 notes).
   128→   *
   129→   * Rules:
   130→   *   - Start on a chord tone (1st/3rd/5th).
   131→   *   - Prefer steps (2nds) over leaps (3rds+).
   132→   *   - After a leap, resolve by step in the opposite direction.
   133→   *   - End on a stable tone (1st/3rd/5th).
   134→   *   - Range: within an octave + a 3rd.
   135→   *   - Duration: mix of 8ths/16ths with occasional longer notes.
   136→   *
   137→   * The contour shape is chosen based on `tension`:
   138→   *   - High tension → ascending / wave (build energy)
   139→   *   - Low tension → descending / wave (release)
   140→   *   - Mid → arch (rise then fall)
   141→   */
   142→  generateMotif(energy: number, tension: number): Motif {
   143→    const numNotes = this.rng.int(4, 8);
   144→    const contour = this.pickContour(tension);
   145→
   146→    // Octave shift: high tension lifts the whole motif up; low tension drops it.
   147→    const octShift = tension > 0.65 ? 7 : tension < 0.3 ? -7 : 0;
   148→    // Singable range (scale degrees). -3 (low 3rd) to 9 (octave + 3rd).
   149→    const minDeg = -3 + (octShift < 0 ? 0 : 0);
   150→    const maxDeg = 9 + (octShift > 0 ? 7 : 0);
   151→
   152→    const notes: number[] = [];
   153→    const durations: number[] = [];
   154→    const velocities: number[] = [];
   155→    const rests: number[] = [];
   156→
   157→    // First note: chord tone.
   158→    const startDeg = this.rng.pick(STABLE_DEGREES) + octShift;
   159→    notes.push(startDeg);
   160→    durations.push(this.pickDuration(tension, true));
   161→    velocities.push(this.pickVelocity(energy, tension, true));
   162→    rests.push(0);
   163→
   164→    let prevDelta = 0;
   165→
   166→    for (let i = 1; i < numNotes; i++) {
   167→      const progress = i / numNotes; // 0..1 through motif
   168→      let delta: number;
   169→
   170→      // After a leap (3rd or larger), 75% chance to resolve by step in opposite direction.
   171→      if (Math.abs(prevDelta) >= 2 && this.rng.chance(0.75)) {
   172→        delta = prevDelta > 0 ? -1 : 1;
   173→      } else {
   174→        // Otherwise: prefer steps. Leap probability rises with tension.
   175→        const leapProb = 0.15 + tension * 0.35;
   176→        if (this.rng.chance(leapProb)) {
   177→          // Leap: 3rd, 4th, 5th (up or down).
   178→          delta = this.rng.pick([2, 2, 3, 4, -2, -2, -3, -4]);
   179→        } else {
   180→          // Step: mostly 2nds, occasionally a repeat or skip.
   181→          delta = this.rng.pick([1, 1, 1, -1, -1, -1, 0, 2, -2]);
   182→        }
   183→        // Bias direction by contour shape.
   184→        delta = this.applyContourBias(delta, contour, progress);
   185→      }
   186→
   187→      let nextDeg = notes[i - 1] + delta;
   188→      // Clamp to singable range — bounce back if exceeded.
   189→      if (nextDeg < minDeg) nextDeg = notes[i - 1] + 1;
   190→      if (nextDeg > maxDeg) nextDeg = notes[i - 1] - 1;
   191→
   192→      notes.push(nextDeg);
   193→      durations.push(this.pickDuration(tension, false));
   194→      velocities.push(this.pickVelocity(energy, tension, false));
   195→      rests.push(0);
   196→      prevDelta = delta;
   197→    }
   198→
   199→    // End on a stable tone (1st, 3rd, 5th) — pick nearest stable degree.
   200→    const lastIdx = notes.length - 1;
   201→    notes[lastIdx] = this.nearestStableDegree(notes[lastIdx]);
   202→
   203→    // Insert occasional rests (low tension → more rests).
   204→    const restProb = 0.05 + (1 - tension) * 0.2;
   205→    for (let i = 1; i < notes.length; i++) {
   206→      if (this.rng.chance(restProb)) {
   207→        rests[i] = 1; // mark as rest
   208→      }
   209→    }
   210→
   211→    return {
   212→      notes,
   213→      durations,
   214→      velocities,
   215→      rests: rests.map((r) => r === 1),
   216→    };
   217→  }
   218→
   219→  /** Choose a contour shape based on tension. */
   220→  private pickContour(tension: number): ContourShape {
   221→    if (tension > 0.8) return this.rng.pick<ContourShape>(['ascending', 'wave']);
   222→    if (tension > 0.6) return this.rng.pick<ContourShape>(['ascending', 'arch']);
   223→    if (tension < 0.3) return this.rng.pick<ContourShape>(['descending', 'wave']);
   224→    return this.rng.pick<ContourShape>(['arch', 'wave', 'descending']);
   225→  }
   226→
   227→  /**
   228→   * Bias a step direction by the chosen contour.
   229→   * - ascending: prefer up-steps
   230→   * - descending: prefer down-steps
   231→   * - arch: up in first half, down in second half
   232→   * - wave: alternate direction every 2 notes
   233→   */
   234→  private applyContourBias(delta: number, contour: ContourShape, progress: number): number {
   235→    switch (contour) {
   236→      case 'ascending':
   237→        if (delta < 0 && this.rng.chance(0.6)) return Math.abs(delta);
   238→        return delta;
   239→      case 'descending':
   240→        if (delta > 0 && this.rng.chance(0.6)) return -Math.abs(delta);
   241→        return delta;
   242→      case 'arch':
   243→        if (progress < 0.5 && delta < 0 && this.rng.chance(0.6)) return Math.abs(delta);
   244→        if (progress >= 0.5 && delta > 0 && this.rng.chance(0.6)) return -Math.abs(delta);
   245→        return delta;
   246→      case 'wave': {
   247→        const phase = Math.floor(progress * 4) % 2;
   248→        if (phase === 0 && delta < 0 && this.rng.chance(0.5)) return Math.abs(delta);
   249→        if (phase === 1 && delta > 0 && this.rng.chance(0.5)) return -Math.abs(delta);
   250→        return delta;
   251→      }
   252→    }
   253→    return delta;
   254→  }
   255→
   256→  /** Pick a duration (in 16th steps) based on tension. */
   257→  private pickDuration(tension: number, isFirst: boolean): number {
   258→    // Low tension → slow notes (4-8). Peak → 16ths only.
   259→    if (tension < 0.3) return this.rng.pick([4, 6, 8]);
   260→    if (tension < 0.6) return this.rng.pick([2, 4, 4]);
   261→    if (tension < 0.8) return this.rng.pick([1, 2, 2]);
   262→    // Peak: mostly 16ths, occasional 8th.
   263→    return this.rng.pick([1, 1, 2]);
   264→  }
   265→
   266→  /** Pick a velocity based on energy + whether this is a strong beat. */
   267→  private pickVelocity(energy: number, _tension: number, isStrong: boolean): number {
   268→    const base = isStrong ? 0.7 : 0.5;
   269→    return clamp(base * (0.6 + energy * 0.4), 0.1, 1.0);
   270→  }
   271→
   272→  /** Find the nearest stable degree (1st/3rd/5th) to a given scale degree. */
   273→  private nearestStableDegree(deg: number): number {
   274→    let nearest = STABLE_DEGREES[0];
   275→    let nearestDist = Infinity;
   276→    for (const s of STABLE_DEGREES) {
   277→      // Search across octaves (steps of 7 scale degrees).
   278→      for (let oct = -7; oct <= 7; oct += 7) {
   279→        const candidate = s + oct;
   280→        const dist = Math.abs(candidate - deg);
   281→        if (dist < nearestDist) {
   282→          nearestDist = dist;
   283→          nearest = candidate;
   284→        }
   285→      }
   286→    }
   287→    return nearest;
   288→  }
   289→
   290→  /** Find the nearest chord tone (root/3rd/5th of `chordDeg`) to `deg`. */
   291→  private nearestChordTone(deg: number, chordDeg: number): number {
   292→    let nearest = chordDeg;
   293→    let nearestDist = Infinity;
   294→    for (const off of CHORD_TONE_OFFSETS) {
   295→      for (let oct = -7; oct <= 7; oct += 7) {
   296→        const candidate = chordDeg + off + oct;
   297→        const dist = Math.abs(candidate - deg);
   298→        if (dist < nearestDist) {
   299→          nearestDist = dist;
   300→          nearest = candidate;
   301→        }
   302→      }
   303→    }
   304→    return nearest;
   305→  }

   307→  // ─── Development techniques ─────────────────────────────────────────────
   308→
   309→  /** Transpose a motif by N scale degrees (clean — stays in scale). */
   310→  transpose(motif: Motif, scaleSteps: number): Motif {
   311→    return {
   312→      notes: motif.notes.map((n) => n + scaleSteps),
   313→      durations: [...motif.durations],
   314→      velocities: [...motif.velocities],
   315→      rests: [...motif.rests],
   316→    };
   317→  }
   318→
   319→  /**
   320→   * Melodic inversion — flip the contour upside-down.
   321→   * First note stays; subsequent deltas are negated.
   322→   */
   323→  invert(motif: Motif): Motif {
   324→    if (motif.notes.length === 0) {
   325→      return { notes: [], durations: [], velocities: [], rests: [] };
   326→    }
   327→    const first = motif.notes[0];
   328→    const invertedNotes = [first];
   329→    for (let i = 1; i < motif.notes.length; i++) {
   330→      const delta = motif.notes[i] - motif.notes[i - 1];
   331→      invertedNotes.push(invertedNotes[i - 1] - delta);
   332→    }
   333→    return {
   334→      notes: invertedNotes,
   335→      durations: [...motif.durations],
   336→      velocities: [...motif.velocities],
   337→      rests: [...motif.rests],
   338→    };
   339→  }
   340→
   341→  /** Retrograde — play the motif backwards. */
   342→  retrograde(motif: Motif): Motif {
   343→    return {
   344→      notes: [...motif.notes].reverse(),
   345→      durations: [...motif.durations].reverse(),
   346→      velocities: [...motif.velocities].reverse(),
   347→      rests: [...motif.rests].reverse(),
   348→    };
   349→  }
   350→
   351→  /** Fragment — take a sub-range of the motif (a 2-3 note cell). */
   352→  fragment(motif: Motif, startIdx: number, length: number): Motif {
   353→    const end = Math.min(startIdx + length, motif.notes.length);
   354→    const actualStart = Math.max(0, Math.min(startIdx, motif.notes.length - 1));
   355→    return {
   356→      notes: motif.notes.slice(actualStart, end),
   357→      durations: motif.durations.slice(actualStart, end),
   358→      velocities: motif.velocities.slice(actualStart, end),
   359→      rests: motif.rests.slice(actualStart, end),
   360→    };
   361→  }
   362→
   363→  /** Elongate (rhythmic augmentation) — multiply durations by `factor`. */
   364→  elongate(motif: Motif, factor: number): Motif {
   365→    return {
   366→      notes: [...motif.notes],
   367→      durations: motif.durations.map((d) => Math.max(1, Math.round(d * factor))),
   368→      velocities: [...motif.velocities],
   369→      rests: [...motif.rests],
   370→    };
   371→  }
   372→
   373→  /** Shorten (rhythmic diminution) — divide durations by `factor`. */
   374→  shorten(motif: Motif, factor: number): Motif {
   375→    return this.elongate(motif, 1 / factor);
   376→  }
   377→
   378→  /**
   379→   * Sequence — repeat the motif at successively higher/lower scale degrees.
   380→   * The classic melodic development technique (Beethoven, Bach, film composers).
   381→   * Default shift = 2 scale degrees per repeat (up/down a 3rd).
   382→   */
   383→  sequence(motif: Motif, steps: number, direction: 'up' | 'down'): Motif[] {
   384→    const out: Motif[] = [motif];
   385→    const dir = direction === 'up' ? 1 : -1;
   386→    const shiftPerStep = 2;
   387→    for (let i = 1; i < Math.max(1, steps); i++) {
   388→      out.push(this.transpose(motif, dir * shiftPerStep * i));
   389→    }
   390→    return out;
   391→  }
   392→
   393→  // ─── Call / Response ────────────────────────────────────────────────────
   394→
   395→  /**
   396→   * Generate a "response" motif that answers a "call" motif.
   397→   * The response:
   398→   *   - Inverts the call's contour (call ascending → response descending).
   399→   *   - Ends on the root (most stable tone).
   400→   *   - Shortens durations for a lighter, "answering" feel.
   401→   */
   402→  generateResponse(prevPhrase: Motif): Motif {
   403→    const inverted = this.invert(prevPhrase);
   404→    // Force the response to end on the root (degree 0) — definitive "answer".
   405→    if (inverted.notes.length > 0) {
   406→      inverted.notes[inverted.notes.length - 1] = 0;
   407→    }
   408→    // Lighter rhythm (twice as fast).
   409→    const response = this.shorten(inverted, 2);
   410→    // Slightly lower velocity for the response (counter-melody feel).
   411→    response.velocities = response.velocities.map((v) => clamp(v * 0.8, 0.1, 1.0));
   412→    return response;
   413→  }
   414→
   415→  // ─── Phrase structure: A A' B A'' ───────────────────────────────────────
   416→
   417→  /**
   418→   * Force a new phrase at section boundaries.
   419→   * Computes internal tension from energy + phraseCount, generates a fresh
   420→   * call motif, and rebuilds the 8-bar developmental phrase.
   421→   */
   422→  newPhrase(energy: number): void {
   423→    this.phraseCount++;
   424→    this.lastEnergy = energy;
   425→    const tension = this.computeTension(energy, this.phraseCount);
   426→    this.lastTension = tension;
   427→    this.previousMotif = this.currentMotif;
   428→    this.currentMotif = this.generateMotif(energy, tension);
   429→    this.buildPhrase(energy, tension);
   430→  }
   431→
   432→  /**
   433→   * Tension curves — maps energy (0..1) to melodic behavior.
   434→   *   - Low (0..0.3): slow, low register, consonant, rests.
   435→   *   - Medium (0.3..0.6): mid register, mostly steps.
   436→   *   - High (0.6..0.8): faster, ascending sequences, leaps.
   437→   *   - Peak (0.8..1.0): fastest, highest register, climbing sequences.
   438→   *
   439→   * We add a small periodic variation so consecutive phrases don't all hit
   440→   * the same tension peak even at the same energy level.
   441→   */
   442→  private computeTension(energy: number, phraseCount: number): number {
   443→    const phase = (phraseCount % 4) / 4;
   444→    const variation = 0.12 * Math.sin(2 * Math.PI * phase);
   445→    return clamp(energy + variation, 0, 1);
   446→  }
   447→
   448→  /**
   449→   * Build the 8-bar phrase events from the current motif.
   450→   *   - A (bars 0-1): state the motif.
   451→   *   - A' (bars 2-3): variation — transpose up a 3rd OR fragment & repeat.
   452→   *   - B (bars 4-5): contrasting motif (fresh, different contour).
   453→   *   - A'' (bars 6-7): return + development — augment + sequence up.
   454→   */
   455→  private buildPhrase(energy: number, tension: number): void {
   456→    // Clear event tables.
   457→    this.phraseEvents = new Array(this.phraseSteps).fill(null);
   458→    this.responseEvents = new Array(this.phraseSteps).fill(null);
   459→
   460→    // A — original motif.
   461→    const sectionA = this.currentMotif;
   462→
   463→    // A' — variation: transpose up a 3rd (2 scale degrees) OR fragment+repeat.
   464→    const aPrime = this.rng.chance(0.5)
   465→      ? this.transpose(sectionA, 2)
   466→      : this.repeatFragment(sectionA);
   467→
   468→    // B — contrasting motif (fresh, with higher tension for contrast).
   469→    const sectionB = this.generateMotif(energy, clamp(tension + 0.2, 0, 1));
   470→
   471→    // A'' — augmented + sequenced up (slower, then sequenced).
   472→    const augmented = this.elongate(sectionA, 2);
   473→    const sequenceChain = this.sequence(augmented, 2, 'up');
   474→    const aDoublePrime = sequenceChain[1] ?? augmented;
   475→
   476→    // Place each section at its bar range.
   477→    this.placeMotifInPhrase(sectionA, 0, this.phraseEvents, true);
   478→    this.placeMotifInPhrase(aPrime, 2, this.phraseEvents, true);
   479→    this.placeMotifInPhrase(sectionB, 4, this.phraseEvents, true);
   480→    this.placeMotifInPhrase(aDoublePrime, 6, this.phraseEvents, true);
   481→
   482→    // Cache the call motif and build the arp's response.
   483→    this.lastCallMotif = sectionA;
   484→    this.buildResponseEvents(energy, tension);
   485→  }
   486→
   487→  /** Repeat a 2-3 note fragment to fill A' — classic motivic development. */
   488→  private repeatFragment(motif: Motif): Motif {
   489→    const fragLen = Math.min(3, motif.notes.length);
   490→    const frag = this.fragment(motif, 0, fragLen);
   491→    // Repeat the fragment 2-3 times to fill the section.
   492→    const repeats = Math.max(2, Math.floor(8 / Math.max(1, frag.notes.length)));
   493→    const notes: number[] = [];
   494→    const durations: number[] = [];
   495→    const velocities: number[] = [];
   496→    const rests: boolean[] = [];
   497→    for (let r = 0; r < repeats; r++) {
   498→      // Optional transpose on each repeat for sequence feel.
   499→      const transpose = r * 1;
   500→      for (let i = 0; i < frag.notes.length; i++) {
   501→        notes.push(frag.notes[i] + transpose);
   502→        durations.push(frag.durations[i]);
   503→        velocities.push(frag.velocities[i]);
   504→        rests.push(frag.rests[i]);
   505→      }
   506→    }
   507→    return { notes, durations, velocities, rests };
   508→  }
   509→
   510→  /**
   511→   * Place a motif's events into the phrase event table starting at `startBar`.
   512→   * Snaps strong beats to chord tones (from PROGRESSIONS[scale]) so the lead
   513→   * never clashes with the harmony engine's chord changes.
   514→   */
   515→  private placeMotifInPhrase(
   516→    motif: Motif,
   517→    startBar: number,
   518→    events: (MotifEvent | null)[],
   519→    snapChordTones: boolean,
   520→  ): void {
   521→    const sectionBars = 2;
   522→    const startStep = startBar * this.stepsPerBar;
   523→    const endStep = startStep + sectionBars * this.stepsPerBar;
   524→    const prog = PROGRESSIONS[this.scale] || PROGRESSIONS.minor;
   525→    let curStep = startStep;
   526→
   527→    for (let i = 0; i < motif.notes.length; i++) {
   528→      if (curStep >= endStep) break;
   529→      if (curStep >= this.phraseSteps) break;
   530→
   531→      const dur = Math.max(1, motif.durations[i]);
   532→      const isRest = motif.rests[i];
   533→
   534→      if (!isRest) {
   535→        let deg = motif.notes[i];
   536→
   537→        if (snapChordTones) {
   538→          // Determine which bar (in absolute phrase terms) we're in.
   539→          const barWithinPhrase = Math.floor(curStep / this.stepsPerBar);
   540→          const stepWithinBar = curStep % this.stepsPerBar;
   541→          const chordDeg = prog[barWithinPhrase % prog.length];
   542→
   543→          if (stepWithinBar === 0) {
   544→            // Downbeat → snap to nearest chord tone.
   545→            deg = this.nearestChordTone(deg, chordDeg);
   546→          } else if (stepWithinBar === 8 && this.rng.chance(0.5)) {
   547→            // Beat 3 → snap to a chord tone (3rd or 5th).
   548→            deg = this.rng.pick([chordDeg + 2, chordDeg + 4]);
   549→          }
   550→          // Otherwise: keep the motif's scale degree (passing/neighbor tone).
   551→        }
   552→
   553→        events[curStep] = {
   554→          stepInPhrase: curStep,
   555→          scaleDeg: deg,
   556→          velocity: motif.velocities[i],
   557→          // Cap duration at 4 (quarter note) to avoid synth-gate overflow.
   558→          duration: Math.min(dur, 4),
   559→          isRest: false,
   560→        };
   561→      }
   562→
   563→      curStep += dur;
   564→    }
   565→  }
   566→
   567→  /**
   568→   * Build the arp's response events from the last call motif.
   569→   * The response plays in bars 4-7 of the phrase (the "answer" portion),
   570→   * typically during VARIATION sections.
   571→   */
   572→  private buildResponseEvents(energy: number, tension: number): void {
   573→    this.responseEvents = new Array(this.phraseSteps).fill(null);
   574→    if (!this.lastCallMotif) return;
   575→
   576→    const response = this.generateResponse(this.lastCallMotif);
   577→    // Place the response starting at bar 4 (the B / A'' region).
   578→    // No chord-tone snapping — the response is a counter-melody and can
   579→    // use passing tones freely.
   580→    this.placeMotifInPhrase(response, 4, this.responseEvents, false);
   581→  }
   582→
   583→  // ─── Per-step note lookup ───────────────────────────────────────────────
   584→
   585→  /**
   586→   * Get the next lead note for the given step/bar/energy.
   587→   * Returns null for rests or steps with no scheduled event.
   588→   *
   589→   * The lead plays at root+12 (one octave above the bass root).
   590→   */
   591→  nextNote(
   592→    step: number,
   593→    bar: number,
   594→    energy: number,
   595→  ): { note: number; velocity: number; duration: number } | null {
   596→    const phraseBar = ((bar % this.phraseBars) + this.phraseBars) % this.phraseBars;
   597→    const phraseStep = phraseBar * this.stepsPerBar + (step % this.stepsPerBar);
   598→
   599→    const event = this.phraseEvents[phraseStep];
   600→    if (!event || event.isRest) return null;
   601→
   602→    const midi = scaleNote(this.root + 12, this.scale, event.scaleDeg);
   603→    const vel = clamp(event.velocity * (0.6 + energy * 0.4), 0.1, 1.0);
   604→    return { note: midi, velocity: vel, duration: event.duration };
   605→  }
   606→
   607→  /**
   608→   * Get the next response note for the arp (counter-melody to the lead).
   609→   * Returns null when no response event is scheduled.
   610→   *
   611→   * The arp plays at root+24 (two octaves above the bass root) — brighter
   612→   * register so it sits above the lead as a true counter-melody.
   613→   */
   614→  nextResponseNote(
   615→    step: number,
   616→    bar: number,
   617→    energy: number,
   618→  ): { note: number; velocity: number; duration: number } | null {
   619→    const phraseBar = ((bar % this.phraseBars) + this.phraseBars) % this.phraseBars;
   620→    const phraseStep = phraseBar * this.stepsPerBar + (step % this.stepsPerBar);
   621→
   622→    const event = this.responseEvents[phraseStep];
   623→    if (!event || event.isRest) return null;
   624→
   625→    const midi = scaleNote(this.root + 24, this.scale, event.scaleDeg);
   626→    const vel = clamp(event.velocity * (0.55 + energy * 0.35), 0.1, 1.0);
   627→    return { note: midi, velocity: vel, duration: event.duration };
   628→  }
   629→
   630→  // ─── Incremental evolution ──────────────────────────────────────────────
   631→
   632→  /**
   633→   * Per-bar evolution tick. Called from the engine's scheduler every bar.
   634→   * Unlike LeadMotif (which mutated one note at a time), MelodyEngine
   635→   * refreshes the B section (contrasting motif) at intervals — this gives
   636→   * fresh variation without losing the A-section identity.
   637→   *
   638→   * Effective interval shrinks as `evolutionRate` grows (faster evolution
   639→   * for more dynamic worlds like Goa / acid-psy).
   640→   */
   641→  tickEvolution(bar: number, evolutionRate = 0.4, intervalBars = 8): void {
   642→    if (bar <= 0) return;
   643→    if (bar <= this.lastEvolveBar) return;
   644→    const effectiveInterval = Math.max(4, Math.round(intervalBars * (1.2 - evolutionRate)));
   645→    if (bar % effectiveInterval !== 0) return;
   646→    this.lastEvolveBar = bar;
   647→    this.regenerateBSection();
   648→  }
   649→
   650→  /** Refresh only the B section (bars 4-5) with a new contrasting motif. */
   651→  private regenerateBSection(): void {
   652→    // Clear B section events.
   653→    for (let s = 4 * this.stepsPerBar; s < 6 * this.stepsPerBar; s++) {
   654→      this.phraseEvents[s] = null;
   655→    }
   656→    const tension = clamp(this.lastTension + 0.15, 0, 1);
   657→    const newB = this.generateMotif(this.lastEnergy, tension);
   658→    this.placeMotifInPhrase(newB, 4, this.phraseEvents, true);
   659→  }
