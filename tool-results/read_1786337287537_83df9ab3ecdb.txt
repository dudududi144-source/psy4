     1→/**
     2→ * ReferenceListener — connects to a live psytrance radio stream and extracts
     3→ * acoustic features for comparison with our engine.
     4→ *
     5→ * PRIORITY 3+4: Reference listener + feature extraction.
     6→ *
     7→ * CRITICAL: NO ScriptProcessorNode. NO per-block allocation.
     8→ *
     9→ * Architecture:
    10→ *   fetch(streamUrl) → ReadableStream → AudioDecoder/AudioContext.decodeAudioData
    11→ *   → OfflineAudioContext for analysis → feature extraction
    12→ *
    13→ * The stream is decoded in chunks (not played through speakers), analyzed
    14→ * via OfflineAudioContext + AnalyserNode, and features are extracted.
    15→ * The audio is NEVER stored — only features.
    16→ *
    17→ * Window: 20 seconds, hop: 10 seconds (50% overlap).
    18→ */
    19→
    20→import type { RadioStream } from './radioStreams';
    21→
    22→export interface ReferenceMetrics {
    23→  // Tempo
    24→  bpm: number;
    25→  bpmConfidence: number;       // 0..1
    26→
    27→  // Loudness
    28→  rms: number;
    29→  peak: number;
    30→  lufs: number;
    31→  crestFactor: number;
    32→
    33→  // Spectral bands (normalized 0..1)
    34→  subEnergy: number;           // 20-60 Hz
    35→  lowEnergy: number;           // 60-250 Hz
    36→  midEnergy: number;           // 250-2000 Hz
    37→  highEnergy: number;          // 2000-8000 Hz
    38→  airEnergy: number;           // 8000-20000 Hz
    39→
    40→  // Spectral shape
    41→  spectralCentroid: number;    // Hz
    42→  spectralFlatness: number;    // 0..1
    43→  spectralRolloff: number;     // Hz (85%)
    44→
    45→  // Transients
    46→  transientDensity: number;    // transients per second
    47→  kickDensity: number;         // kick hits per second
    48→  hatDensity: number;          // hat hits per second
    49→  percussionDensity: number;   // all percussion per second

    51→  // Stereo

    61→
    62→  // Overall
    63→  energy: number;              // 0..1

    68→  detectedBassNote?: { note: number; freq: number; confidence: number };

    96→const WINDOW_SECONDS = 20;
    97→const HOP_SECONDS = 10;
    98→const MAX_WINDOWS = 30;        // keep last 30 windows (5 minutes of data)
    99→const SAMPLE_RATE = 44100;

   101→/**
   102→ * ReferenceListener — connects to a radio stream and extracts features.
   103→ *
   104→ * Usage:
   105→ *   const listener = new ReferenceListener();
   106→ *   await listener.connect(stream);
   107→ *   listener.start();          // begins feature extraction
   108→ *   const metrics = listener.getLatestMetrics();
   109→ *   const profile = listener.getProfile();
   110→ *   listener.disconnect();
   111→ */
   112→export class ReferenceListener {
   113→  private stream: RadioStream | null = null;
   114→  private audioCtx: AudioContext | null = null;
   115→  private analyser: AnalyserNode | null = null;
   116→  private sourceNode: MediaElementAudioSourceNode | null = null;
   117→  private audioEl: HTMLAudioElement | null = null;
   118→
   119→  private connected = false;
   120→  private extracting = false;
   121→  private metricsHistory: ReferenceMetrics[] = [];
   122→  private profile: ReferenceProfile | null = null;
   123→
   124→  private extractInterval: ReturnType<typeof setInterval> | null = null;
   125→  private captureBuffer: Float32Array | null = null;     // reused
   126→  private captureBufferIdx = 0;
   127→  private captureCtx: OfflineAudioContext | null = null;
   128→
   129→  // Analyser data buffers (reused — NO per-frame allocation)
   130→  private freqData: Uint8Array | null = null;
   131→  private timeData: Uint8Array | null = null;
   132→
   133→  // Callbacks
   134→  private onMetricsCallback: ((m: ReferenceMetrics) => void) | null = null;
   135→  private onProfileCallback: ((p: ReferenceProfile) => void) | null = null;
   136→  private onErrorCallback: ((e: Error) => void) | null = null;
   137→
   138→  /**
   139→   * Connect to a radio stream.
   140→   * Creates an HTMLAudioElement (NOT played through speakers) and routes it
   141→   * through a MediaElementAudioSourceNode → AnalyserNode for feature extraction.
   142→   */
   143→  async connect(stream: RadioStream): Promise<boolean> {
   144→    if (this.connected) {
   145→      await this.disconnect();
   146→    }
   147→
   148→    this.stream = stream;
   149→
   150→    try {
   151→      // Create audio element (muted — we only analyze, never play to speakers)
   152→      // MUST be appended to DOM for some browsers to play reliably
   153→      this.audioEl = new Audio();
   154→      this.audioEl.crossOrigin = 'anonymous';
   155→      this.audioEl.preload = 'auto';
   156→      this.audioEl.volume = 0; // silent — we only analyze
   157→      // Some browsers require the element in the DOM
   158→      this.audioEl.style.display = 'none';
   159→      document.body.appendChild(this.audioEl);
   160→
   161→      // Create AudioContext (needs user gesture — connect() is called from a click)
   162→      const Ctx = window.AudioContext || (window as any).webkitAudioContext;
   163→      this.audioCtx = new Ctx({ sampleRate: SAMPLE_RATE });
   164→      // Resume immediately (we're in a user-gesture context from the button click)
   165→      if (this.audioCtx.state === 'suspended') {
   166→        await this.audioCtx.resume();
   167→      }
   168→
   169→      // Set source BEFORE creating MediaElementAudioSourceNode
   170→      this.audioEl.src = stream.url;
   171→      this.audioEl.load();
   172→
   173→      // Create source from audio element
   174→      this.sourceNode = this.audioCtx.createMediaElementSource(this.audioEl);
   175→
   176→      // Create analyser (FFT-based — NO ScriptProcessor)
   177→      this.analyser = this.audioCtx.createAnalyser();
   178→      this.analyser.fftSize = 4096;
   179→      this.analyser.smoothingTimeConstant = 0.3;
   180→      this.analyser.minDecibels = -90;
   181→      this.analyser.maxDecibels = -10;
   182→
   183→      // Route: source → analyser → destination (destination needed for analyser to process)
   184→      // Audio element volume=0, so nothing is heard.
   185→      this.sourceNode.connect(this.analyser);
   186→      this.analyser.connect(this.audioCtx.destination);
   187→
   188→      // Pre-allocate reusable buffers (NO per-frame allocation)
   189→      this.freqData = new Uint8Array(this.analyser.frequencyBinCount);
   190→      this.timeData = new Uint8Array(this.analyser.fftSize);
   191→      this.captureBuffer = new Float32Array(WINDOW_SECONDS * SAMPLE_RATE);
   192→
   193→      // Start playback (muted) — must happen after context is running
   194→      // Wait a moment for the source to be ready
   195→      await new Promise(r => setTimeout(r, 100));
   196→      const playPromise = this.audioEl.play();
   197→      if (playPromise) {
   198→        await playPromise.catch((err) => {
   199→          console.warn('[ReferenceListener] play() rejected:', err);
   200→          // Try again with a slight delay
   201→          return new Promise<void>((resolve) => {
   202→            setTimeout(() => {
   203→              this.audioEl!.play().then(() => resolve()).catch(() => resolve());
   204→            }, 500);
   205→          });
   206→        });
   207→      }
   208→
   209→      // Verify context is running
   210→      if (this.audioCtx.state !== 'running') {
   211→        await this.audioCtx.resume();
   212→      }
   213→
   214→      this.connected = true;
   215→      return true;
   216→    } catch (err) {
   217→      this.onErrorCallback?.(err instanceof Error ? err : new Error(String(err)));
   218→      await this.disconnect();
   219→      return false;
   220→    }
   221→  }
   222→
   223→  /** Start periodic feature extraction (every HOP_SECONDS). */
   224→  start(): void {
   225→    if (!this.connected || this.extracting) return;
   226→    this.extracting = true;
   227→
   228→    // Extract every HOP_SECONDS (10s)
   229→    this.extractInterval = setInterval(() => {
   230→      this.extractWindow().catch(err => {
   231→        this.onErrorCallback?.(err instanceof Error ? err : new Error(String(err)));
   232→      });
   233→    }, HOP_SECONDS * 1000);
   234→
   235→    // Also do an initial extraction after 2 seconds (once buffer has data)
   236→    setTimeout(() => {
   237→      this.extractWindow().catch(() => {});
   238→    }, 2000);
   239→  }
   240→
   241→  /** Stop feature extraction. */
   242→  stop(): void {
   243→    this.extracting = false;
   244→    if (this.extractInterval) {
   245→      clearInterval(this.extractInterval);
   246→      this.extractInterval = null;
   247→    }
   248→  }
   249→
   250→  /** Disconnect from the stream. */
   251→  async disconnect(): Promise<void> {
   252→    this.stop();
   253→    this.connected = false;
   254→
   255→    if (this.audioEl) {
   256→      this.audioEl.pause();
   257→      this.audioEl.src = '';
   258→      // Remove from DOM if appended
   259→      if (this.audioEl.parentNode) {
   260→        this.audioEl.parentNode.removeChild(this.audioEl);
   261→      }
   262→      this.audioEl = null;
   263→    }
   264→    if (this.sourceNode) {
   265→      try { this.sourceNode.disconnect(); } catch { /* noop */ }
   266→      this.sourceNode = null;
   267→    }
   268→    if (this.analyser) {
   269→      try { this.analyser.disconnect(); } catch { /* noop */ }
   270→      this.analyser = null;
   271→    }
   272→    if (this.audioCtx) {
   273→      try { await this.audioCtx.close(); } catch { /* noop */ }
   274→      this.audioCtx = null;
   275→    }
   276→    this.stream = null;
   277→  }

   313→  /**
   314→   * Extract features from the current analyser state.
   315→   * Captures WINDOW_SECONDS of audio via AnalyserNode (NOT ScriptProcessor).
   316→   */
   317→  private async extractWindow(): Promise<void> {
   318→    if (!this.analyser || !this.audioCtx || !this.freqData || !this.timeData) return;
   319→
   320→    // We can't capture a full 20s buffer via AnalyserNode directly.
   321→    // Instead, we sample the analyser state multiple times over the window
   322→    // and aggregate. This is the standard approach for real-time analysis.
   323→    //
   324→    // For a proper 20s window, we take ~20 snapshots (1/sec) and average
   325→    // the spectral features. Temporal features (transients, BPM) use the
   326→    // time-domain data.
   327→
   328→    const snapshots: {
   329→      freq: Uint8Array;
   330→      time: Uint8Array;
   331→    }[] = [];
   332→
   333→    // Take 20 snapshots over 2 seconds (fast window for responsiveness)
   334→    // Each snapshot is ~1 frame; we aggregate to approximate the 20s window
   335→    const snapshotCount = 20;
   336→    const snapshotInterval = 100; // ms
   337→
   338→    for (let i = 0; i < snapshotCount; i++) {
   339→      this.analyser.getByteFrequencyData(this.freqData);
   340→      this.analyser.getByteTimeDomainData(this.timeData);
   341→      // Copy current snapshot (small allocation, but only 20 times per window)
   342→      snapshots.push({
   343→        freq: new Uint8Array(this.freqData),
   344→        time: new Uint8Array(this.timeData),
   345→      });
   346→      if (i < snapshotCount - 1) {
   347→        await new Promise(r => setTimeout(r, snapshotInterval));
   348→      }
   349→    }
   350→
   351→    // Extract features from the snapshots
   352→    const metrics = this.extractFeatures(snapshots);
   353→    this.metricsHistory.push(metrics);
   354→
   355→    // Trim history
   356→    if (this.metricsHistory.length > MAX_WINDOWS) {
   357→      this.metricsHistory.shift();
   358→    }
   359→
   360→    // Update rolling profile
   361→    this.updateProfile();
   362→
   363→    // Fire callbacks
   364→    this.onMetricsCallback?.(metrics);
   365→    if (this.profile) {
   366→      this.onProfileCallback?.(this.profile);
   367→    }
   368→  }

   370→  /**
   371→   * Extract acoustic features from analyser snapshots.
   372→   */
   373→  private extractFeatures(snapshots: { freq: Uint8Array; time: Uint8Array }[]): ReferenceMetrics {
   374→    const sr = this.audioCtx?.sampleRate || SAMPLE_RATE;
   375→    const fftSize = this.analyser?.fftSize || 4096;
   376→    const binCount = snapshots[0].freq.length;
   377→    const binHz = sr / 2 / binCount;
   378→
   379→    // ── Average frequency spectrum across snapshots ──
   380→    const avgFreq = new Float32Array(binCount);
   381→    for (const snap of snapshots) {
   382→      for (let i = 0; i < binCount; i++) {
   383→        avgFreq[i] += snap.freq[i];
   384→      }
   385→    }
   386→    for (let i = 0; i < binCount; i++) avgFreq[i] /= snapshots.length;
   387→
   388→    // ── Spectral bands (0..1, normalized from byte 0..255) ──
   389→    const bandEnergy = (loHz: number, hiHz: number): number => {
   390→      const loBin = Math.max(1, Math.floor(loHz / binHz));
   391→      const hiBin = Math.min(binCount - 1, Math.ceil(hiHz / binHz));
   392→      let sum = 0;
   393→      let count = 0;
   394→      for (let i = loBin; i <= hiBin; i++) {
   395→        sum += avgFreq[i] / 255;
   396→        count++;
   397→      }
   398→      return count > 0 ? sum / count : 0;
   399→    };
   400→
   401→    const subEnergy = bandEnergy(20, 60);
   402→    const lowEnergy = bandEnergy(60, 250);
   403→    const midEnergy = bandEnergy(250, 2000);
   404→    const highEnergy = bandEnergy(2000, 8000);
   405→    const airEnergy = bandEnergy(8000, 20000);
   406→
   407→    // ── Spectral centroid ──
   408→    let weightedSum = 0;
   409→    let totalMag = 0;
   410→    for (let i = 1; i < binCount; i++) {
   411→      const freq = i * binHz;
   412→      const mag = avgFreq[i] / 255;
   413→      weightedSum += freq * mag;
   414→      totalMag += mag;
   415→    }
   416→    const spectralCentroid = totalMag > 0 ? weightedSum / totalMag : 0;
   417→
   418→    // ── Spectral flatness (geometric mean / arithmetic mean) ──
   419→    let logSum = 0;
   420→    let arithSum = 0;
   421→    let validCount = 0;
   422→    for (let i = 1; i < binCount; i++) {
   423→      const mag = avgFreq[i] / 255;
   424→      if (mag > 0.001) {
   425→        logSum += Math.log(mag);
   426→        arithSum += mag;
   427→        validCount++;
   428→      }
   429→    }
   430→    const geoMean = validCount > 0 ? Math.exp(logSum / validCount) : 0;
   431→    const arithMean = validCount > 0 ? arithSum / validCount : 0;
   432→    const spectralFlatness = arithMean > 0 ? Math.min(1, geoMean / arithMean) : 0;
   433→
   434→    // ── Spectral rolloff (85%) ──
   435→    const totalEnergy = totalMag;
   436→    const threshold85 = totalEnergy * 0.85;
   437→    let cumulative = 0;
   438→    let rolloffBin = binCount - 1;
   439→    for (let i = 1; i < binCount; i++) {
   440→      cumulative += avgFreq[i] / 255;
   441→      if (cumulative >= threshold85) {
   442→        rolloffBin = i;
   443→        break;
   444→      }
   445→    }
   446→    const spectralRolloff = rolloffBin * binHz;
   447→
   448→    // ── Time-domain: RMS, peak, crest ──
   449→    let sumSq = 0;
   450→    let peak = 0;
   451→    const allTimeSamples: number[] = [];
   452→    for (const snap of snapshots) {
   453→      for (let i = 0; i < snap.time.length; i++) {
   454→        const s = (snap.time[i] - 128) / 128; // convert byte to -1..1
   455→        sumSq += s * s;
   456→        const abs = Math.abs(s);
   457→        if (abs > peak) peak = abs;
   458→        allTimeSamples.push(s);
   459→      }
   460→    }
   461→    const n = allTimeSamples.length;
   462→    const rms = Math.sqrt(sumSq / n);
   463→    const crestFactor = rms > 0 ? peak / rms : 0;
   464→    // LUFS approximation (K-weighted RMS, simplified)
   465→    const lufs = 20 * Math.log10(rms + 1e-12) - 0.691;
   466→
   467→    // ── Transient detection (from time-domain peaks) ──
   468→    const transientThreshold = 0.15;
   469→    const minGapMs = 80;
   470→    const minGapSamples = (minGapMs / 1000) * sr;
   471→    let transientCount = 0;
   472→    let lastTransientIdx = -minGapSamples;
   473→    for (let i = 1; i < n - 1; i++) {
   474→      const s = allTimeSamples[i];
   475→      const abs = Math.abs(s);
   476→      if (abs > transientThreshold &&
   477→          abs > Math.abs(allTimeSamples[i - 1]) &&
   478→          abs > Math.abs(allTimeSamples[i + 1]) &&
   479→          i - lastTransientIdx >= minGapSamples) {
   480→        transientCount++;
   481→        lastTransientIdx = i;
   482→      }
   483→    }
   484→    // Snapshots cover ~2 seconds (20 × 100ms)
   485→    const windowDurationSec = (snapshots.length * snapshotInterval) / 1000;
   486→    const transientDensity = transientCount / windowDurationSec;
   487→
   488→    // ── Kick/hat density (from frequency bands) ──
   489→    // Kick: transients correlated with sub+low energy
   490→    // Hat: transients correlated with high energy
   491→    let kickCount = 0;
   492→    let hatCount = 0;
   493→    let percCount = 0;
   494→    const subLowAvg = (subEnergy + lowEnergy) / 2;
   495→    const highAvg = highEnergy;
   496→    lastTransientIdx = -minGapSamples;
   497→    for (let i = 1; i < n - 1; i++) {
   498→      const s = allTimeSamples[i];
   499→      const abs = Math.abs(s);
   500→      if (abs > transientThreshold &&
   501→          abs > Math.abs(allTimeSamples[i - 1]) &&
   502→          abs > Math.abs(allTimeSamples[i + 1]) &&
   503→          i - lastTransientIdx >= minGapSamples) {
   504→        // Classify by which band is dominant at this moment
   505→        // (simplified — we use the overall band averages as proxy)
   506→        if (subLowAvg > 0.3) kickCount++;
   507→        if (highAvg > 0.3) hatCount++;
   508→        percCount++;
   509→        lastTransientIdx = i;
   510→      }
   511→    }
   512→    const kickDensity = kickCount / windowDurationSec;
   513→    const hatDensity = hatCount / windowDurationSec;
   514→    const percussionDensity = percCount / windowDurationSec;
   515→
   516→    // ── BPM estimation (autocorrelation of low-frequency energy) ──
   517→    // Sample the low-band energy over time and find the periodicity
   518→    const lowBandEnergies: number[] = [];
   519→    for (const snap of snapshots) {
   520→      let lowSum = 0;
   521→      const lowLoBin = Math.floor(40 / binHz);
   522→      const lowHiBin = Math.ceil(120 / binHz);
   523→      for (let i = lowLoBin; i <= lowHiBin; i++) {
   524→        lowSum += snap.freq[i] / 255;
   525→      }
   526→      lowBandEnergies.push(lowSum / (lowHiBin - lowLoBin + 1));
   527→    }
   528→    // Autocorrelation to find tempo
   529→    const bpm = this.estimateBPM(lowBandEnergies, snapshotInterval / 1000);
   530→    const bpmConfidence = bpm > 0 ? 0.7 : 0; // simplified confidence
   531→
   532→    // ── Kick/bass decay (from transient analysis) ──
   533→    // Simplified: measure the decay of the low-band energy after a transient
   534→    const kickDecayMs = this.estimateDecay(allTimeSamples, sr, 'low') * 1000;
   535→    const bassDecayMs = this.estimateDecay(allTimeSamples, sr, 'low') * 1000;
   536→
   537→    // ── Stereo width (from analyser — we only have mono-ish data, so estimate from spectral balance) ──
   538→    // True stereo width requires L/R channels. With AnalyserNode on a stereo source,
   539→    // we can approximate by looking at the correlation between frequency bands.
   540→    // Simplified: assume moderate width for streaming audio.
   541→    const stereoWidth = 0.4; // placeholder — would need channel splitter for true measurement
   542→
   543→    // ── Rhythmic regularity (consistency of transient intervals) ──
   544→    const rhythmicRegularity = this.computeRhythmicRegularity(allTimeSamples, sr);
   545→
   546→    // ── Repetition score (spectral consistency across snapshots) ──
   547→    let spectralVariance = 0;
   548→    if (snapshots.length > 1) {
   549→      const meanSpectrum = avgFreq;
   550→      for (const snap of snapshots) {
   551→        let diff = 0;
   552→        for (let i = 0; i < binCount; i++) {
   553→          diff += Math.abs(snap.freq[i] - meanSpectrum[i]);
   554→        }
   555→        spectralVariance += diff / binCount;
   556→      }
   557→      spectralVariance /= snapshots.length;
   558→    }
   559→    // Lower variance = higher repetition
   560→    const repetitionScore = Math.max(0, Math.min(1, 1 - spectralVariance / 50));
   561→
   562→    // ── Overall energy ──
   563→    const energy = Math.min(1, rms * 3);
   564→
   565→    // ── Confidence ──
   566→    const overallConfidence = Math.min(1, (rms > 0.01 ? 0.3 : 0) +
   567→                                           (snapshots.length >= 10 ? 0.3 : 0) +
   568→                                           (bpm > 0 ? 0.2 : 0) +
   569→                                           (transientCount > 5 ? 0.2 : 0));
   570→
   571→    return {
   572→      bpm,
   573→      bpmConfidence,
   574→      rms,
   575→      peak,
   576→      lufs,
   577→      crestFactor,
   578→      subEnergy,
   579→      lowEnergy,
   580→      midEnergy,
   581→      highEnergy,
   582→      airEnergy,
   583→      spectralCentroid,
   584→      spectralFlatness,
   585→      spectralRolloff,
   586→      transientDensity,
   587→      kickDensity,
   588→      hatDensity,
   589→      percussionDensity,
   590→      stereoWidth,
   591→      kickDecayMs,
   592→      bassDecayMs,
   593→      rhythmicRegularity,
   594→      repetitionScore,
   595→      energy,
   596→      overallConfidence,
   597→      timestamp: Date.now(),
   598→      sourceStream: this.stream?.id || 'unknown',
   599→    };
   600→  }

   602→  /**
   603→   * Estimate BPM via autocorrelation of an energy envelope.
   604→   */
   605→  private estimateBPM(energies: number[], hopSec: number): number {
   606→    if (energies.length < 4) return 0;
   607→
   608→    // Normalize
   609→    const mean = energies.reduce((a, b) => a + b, 0) / energies.length;
   610→    const normalized = energies.map(e => e - mean);
   611→
   612→    // Autocorrelation — find lag with highest correlation
   613→    // BPM range: 120-160 → lag range in samples
   614→    // At hopSec interval, 120 BPM = 0.5s = 0.5/hopSec lags
   615→    // 160 BPM = 0.375s = 0.375/hopSec lags
   616→    const minLag = Math.floor(0.3 / hopSec); // ~200 BPM
   617→    const maxLag = Math.floor(1.0 / hopSec); // ~60 BPM
   618→    const maxLagClamped = Math.min(maxLag, Math.floor(energies.length / 2));
   619→
   620→    let bestLag = 0;
   621→    let bestCorr = 0;
   622→
   623→    for (let lag = minLag; lag <= maxLagClamped; lag++) {
   624→      let corr = 0;
   625→      let count = 0;
   626→      for (let i = 0; i < energies.length - lag; i++) {
   627→        corr += normalized[i] * normalized[i + lag];
   628→        count++;
   629→      }
   630→      corr = count > 0 ? corr / count : 0;
   631→      if (corr > bestCorr) {
   632→        bestCorr = corr;
   633→        bestLag = lag;
   634→      }
   635→    }
   636→
   637→    if (bestLag === 0 || bestCorr < 0.01) return 0;
   638→
   639→    const periodSec = bestLag * hopSec;
   640→    const bpm = 60 / periodSec;
   641→
   642→    // Snap to common psytrance range
   643→    if (bpm < 60) return 0;
   644→    if (bpm > 200) return 0;
   645→    // Fold into 120-160 range
   646→    let foldedBpm = bpm;
   647→    while (foldedBpm < 120) foldedBpm *= 2;
   648→    while (foldedBpm > 160) foldedBpm /= 2;
   649→
   650→    return Math.round(foldedBpm);
   651→  }

   653→  /**
   654→   * Estimate decay time of transients in a given frequency band.
   655→   */
   656→  private estimateDecay(samples: number[], sr: number, band: 'low' | 'high'): number {
   657→    // Find the first significant transient
   658→    const threshold = 0.2;
   659→    let peakIdx = -1;
   660→    for (let i = 1; i < samples.length - 1; i++) {
   661→      if (Math.abs(samples[i]) > threshold &&
   662→          Math.abs(samples[i]) > Math.abs(samples[i - 1]) &&
   663→          Math.abs(samples[i]) > Math.abs(samples[i + 1])) {
   664→        peakIdx = i;
   665→        break;
   666→      }
   667→    }
   668→    if (peakIdx < 0) return 0.1; // default 100ms
   669→
   670→    // Measure decay from peak to 10% of peak
   671→    const peakVal = Math.abs(samples[peakIdx]);
   672→    const targetVal = peakVal * 0.1;
   673→    for (let i = peakIdx; i < samples.length; i++) {
   674→      if (Math.abs(samples[i]) <= targetVal) {
   675→        return (i - peakIdx) / sr;
   676→      }
   677→    }
   678→    return 0.2; // didn't decay within window
   679→  }

   681→  /**
   682→   * Compute rhythmic regularity (0..1).
   683→   * Measures how evenly spaced the transients are.
   684→   */
   685→  private computeRhythmicRegularity(samples: number[], sr: number): number {
   686→    const threshold = 0.15;
   687→    const minGapSamples = Math.floor(0.08 * sr);
   688→    const transients: number[] = [];
   689→    let lastIdx = -minGapSamples;
   690→    for (let i = 1; i < samples.length - 1; i++) {
   691→      const abs = Math.abs(samples[i]);
   692→      if (abs > threshold &&
   693→          abs > Math.abs(samples[i - 1]) &&
   694→          abs > Math.abs(samples[i + 1]) &&
   695→          i - lastIdx >= minGapSamples) {
   696→        transients.push(i);
   697→        lastIdx = i;
   698→      }
   699→    }
   700→    if (transients.length < 3) return 0;
   701→
   702→    // Compute intervals
   703→    const intervals: number[] = [];
   704→    for (let i = 1; i < transients.length; i++) {
   705→      intervals.push(transients[i] - transients[i - 1]);
   706→    }
   707→    const meanInterval = intervals.reduce((a, b) => a + b, 0) / intervals.length;
   708→    const variance = intervals.reduce((a, b) => a + (b - meanInterval) ** 2, 0) / intervals.length;
   709→    const stdDev = Math.sqrt(variance);
   710→    const cv = meanInterval > 0 ? stdDev / meanInterval : 1;
   711→    return Math.max(0, Math.min(1, 1 - cv));
   712→  }

   714→  /**
   715→   * Update the rolling reference profile with the latest metrics.
   716→   */
   717→  private updateProfile(): void {
   718→    if (this.metricsHistory.length === 0) return;
   719→    const windows = this.metricsHistory;
   720→
   721→    const stats = (vals: number[]) => {
   722→      if (vals.length === 0) return { mean: 0, p10: 0, p90: 0 };
   723→      const sorted = [...vals].sort((a, b) => a - b);
   724→      const mean = vals.reduce((a, b) => a + b, 0) / vals.length;
   725→      return {
   726→        mean,
   727→        p10: sorted[Math.floor(sorted.length * 0.1)] || sorted[0],
   728→        p90: sorted[Math.floor(sorted.length * 0.9)] || sorted[sorted.length - 1],
   729→      };
   730→    };
   731→
   732→    this.profile = {
   733→      bpm: { ...stats(windows.map(w => w.bpm)), count: windows.length },
   734→      rms: stats(windows.map(w => w.rms)),
   735→      lufs: stats(windows.map(w => w.lufs)),
   736→      subEnergy: stats(windows.map(w => w.subEnergy)),
   737→      lowEnergy: stats(windows.map(w => w.lowEnergy)),
   738→      midEnergy: stats(windows.map(w => w.midEnergy)),
   739→      highEnergy: stats(windows.map(w => w.highEnergy)),
   740→      airEnergy: stats(windows.map(w => w.airEnergy)),
   741→      spectralCentroid: stats(windows.map(w => w.spectralCentroid)),
   742→      transientDensity: stats(windows.map(w => w.transientDensity)),
   743→      kickDecayMs: stats(windows.map(w => w.kickDecayMs)),
   744→      bassDecayMs: stats(windows.map(w => w.bassDecayMs)),
   745→      stereoWidth: stats(windows.map(w => w.stereoWidth)),
   746→      energy: stats(windows.map(w => w.energy)),
   747→      windowCount: windows.length,
   748→      lastUpdated: Date.now(),
   749→      sourceStream: this.stream?.id || 'unknown',
   750→    };
   751→  }
