    17→const SCALES: Record<string, number[]> = {
    18→  minor: [0, 2, 3, 5, 7, 8, 10],
    19→  major: [0, 2, 4, 5, 7, 9, 11],
    20→  dorian: [0, 2, 3, 5, 7, 9, 10],
    21→  phrygian: [0, 1, 3, 5, 7, 8, 10],
    22→};
    23→
    24→const mtof = (m: number) => 440 * Math.pow(2, (m - 69) / 12);
    25→const clamp = (v: number, a: number, b: number) => v < a ? a : (v > b ? b : v);
    26→
    27→function scaleNote(root: number, scale: string, deg: number): number {
    28→  const sc = SCALES[scale] || SCALES.minor;
    29→  const n = sc.length;
    30→  const oct = Math.floor(deg / n);
    31→  const idx = ((deg % n) + n) % n;
    32→  return root + 12 * oct + sc[idx];
    33→}

   116→// ─── Pooled Synth Voice (from PSY6 — persistent oscillators) ────────────────
   117→
   118→class PooledSynthVoice {
   119→  osc1: OscillatorNode;
   120→  osc2: OscillatorNode;
   121→  g1: GainNode;
   122→  g2: GainNode;
   123→  filter: BiquadFilterNode;
   124→  vca: GainNode;
   125→  lfo: OscillatorNode;
   126→  lfoGain: GainNode;
   127→  bus: GainNode | null = null;
   128→
   129→  constructor(ctx: AudioContext) {
   130→    this.osc1 = ctx.createOscillator();
   131→    this.osc2 = ctx.createOscillator();
   132→    this.g1 = ctx.createGain();
   133→    this.g2 = ctx.createGain();
   134→    this.filter = ctx.createBiquadFilter();
   135→    this.filter.type = 'lowpass';
   136→    this.vca = ctx.createGain();
   137→    this.vca.gain.value = 0;
   138→    this.lfo = ctx.createOscillator();
   139→    this.lfoGain = ctx.createGain();
   140→    this.lfoGain.gain.value = 0;
   141→
   142→    this.osc1.connect(this.g1);
   143→    this.osc2.connect(this.g2);
   144→    this.g1.connect(this.filter);
   145→    this.g2.connect(this.filter);
   146→    this.filter.connect(this.vca);
   147→    this.lfo.connect(this.lfoGain);
   148→
   149→    // Start persistent oscillators (never stop them)
   150→    this.osc1.start();
   151→    this.osc2.start();
   152→    this.lfo.start();
   153→  }
   154→
   155→  connect(bus: GainNode) {
   156→    if (this.bus !== bus) {
   157→      this.vca.disconnect();
   158→      this.vca.connect(bus);
   159→      this.lfoGain.disconnect();
   160→      this.lfoGain.connect(this.filter.frequency);
   161→      this.bus = bus;
   162→    }
   163→  }
   164→
   165→  noteOn(p: SynthPreset, when: number, midi: number, vel: number, stepDur: number, bus: GainNode) {
   166→    this.connect(bus);
   167→    const f = mtof(clamp(midi, 12, 108));
   168→    const gate = p.gate || 0.6;
   169→    const dur = stepDur * gate * 2;
   170→    const rel = Math.max(p.rel, 0.02);
   171→    const end = when + dur;
   172→
   173→    this.osc1.type = p.wave1;
   174→    this.osc2.type = p.wave2;
   175→    this.osc1.frequency.setValueAtTime(f, when);
   176→    this.osc2.frequency.setValueAtTime(f * Math.pow(2, p.oct2 || 0), when);
   177→    this.osc2.detune.setValueAtTime(p.detune || 0, when);
   178→    this.g1.gain.setValueAtTime(0.6, when);
   179→    this.g2.gain.setValueAtTime(0.45, when);
   180→
   181→    const cut = clamp(p.cutoff, 60, 16000);
   182→    const res = clamp(p.res, 0.2, 24);
   183→    this.filter.type = p.fType;
   184→    this.filter.Q.setValueAtTime(res, when);
   185→    this.filter.frequency.cancelScheduledValues(when);
   186→    this.filter.frequency.setValueAtTime(Math.min(cut * 3, 16000), when);
   187→    this.filter.frequency.exponentialRampToValueAtTime(cut, when + Math.max((p.atk + p.dec * 0.7), 0.01));
   188→
   189→    if (p.lfoRate > 0 && p.lfoDest === 'cutoff') {
   190→      this.lfo.frequency.setValueAtTime(p.lfoRate, when);
   191→      this.lfoGain.gain.setValueAtTime(p.lfoDepth * 3000, when);
   192→    } else {
   193→      this.lfoGain.gain.setValueAtTime(0, when);
   194→    }
   195→
   196→    const vca = this.vca.gain;
   197→    const atk = Math.max(p.atk, 0.003);
   198→    vca.cancelScheduledValues(when);
   199→    vca.setValueAtTime(0, when);
   200→    vca.linearRampToValueAtTime(vel * 0.5, when + atk);
   201→    vca.setTargetAtTime(vel * 0.5 * p.sus, when + atk, Math.max(p.dec / 3, 0.01));
   202→    vca.setTargetAtTime(0.0001, end, Math.max(rel / 3, 0.008));
   203→  }
   204→
   205→  panic(ctx: AudioContext) {
   206→    try {
   207→      this.vca.gain.cancelScheduledValues(0);
   208→      this.vca.gain.setValueAtTime(0, ctx.currentTime);
   209→    } catch {}
   210→  }
   211→}

   213→// ─── Pooled Drum Voice (from PSY6 — multi-type) ────────────────────────────
   214→
   215→class PooledDrumVoice {
   216→  noise: AudioBufferSourceNode;
   217→  noiseGain: GainNode;
   218→  nFilter: BiquadFilterNode;
   219→  osc: OscillatorNode;
   220→  oscGain: GainNode;
   221→  out: GainNode;
   222→  bus: GainNode | null = null;
   223→
   224→  constructor(ctx: AudioContext, noiseBuffer: AudioBuffer) {
   225→    this.noise = ctx.createBufferSource();
   226→    this.noise.buffer = noiseBuffer;
   227→    this.noise.loop = true;
   228→    this.noiseGain = ctx.createGain();
   229→    this.noiseGain.gain.value = 0;
   230→    this.nFilter = ctx.createBiquadFilter();
   231→    this.nFilter.type = 'bandpass';
   232→    this.noise.connect(this.nFilter);
   233→    this.nFilter.connect(this.noiseGain);
   234→    this.osc = ctx.createOscillator();
   235→    this.osc.type = 'sine';
   236→    this.oscGain = ctx.createGain();
   237→    this.oscGain.gain.value = 0;
   238→    this.osc.connect(this.oscGain);
   239→    this.out = ctx.createGain();
   240→    this.noiseGain.connect(this.out);
   241→    this.oscGain.connect(this.out);
   242→    this.noise.start();
   243→    this.osc.start();
   244→  }
   245→
   246→  connect(bus: GainNode) {
   247→    if (this.bus !== bus) {
   248→      this.out.disconnect();
   249→      this.out.connect(bus);
   250→      this.bus = bus;
   251→    }
   252→  }
   253→
   254→  hit(p: DrumPreset, when: number, vel: number, bus: GainNode) {
   255→    this.connect(bus);
   256→    const tune = p.tune || 1;
   257→    const decay = p.decay || 1;
   258→    const tone = p.tone || 1;
   259→    const punch = p.punch || 0;
   260→    const type = p.type;
   261→    const ng = this.noiseGain.gain;
   262→    const og = this.oscGain.gain;
   263→    ng.cancelScheduledValues(when);
   264→    og.cancelScheduledValues(when);
   265→    ng.setValueAtTime(0, when);
   266→    og.setValueAtTime(0, when);
   267→
   268→    if (type === 'kick') {
   269→      const dur = 0.12 + 0.5 * decay;
   270→      this.osc.frequency.setValueAtTime(180 * tune, when);
   271→      this.osc.frequency.exponentialRampToValueAtTime(Math.max(36 * tune, 24), when + 0.09);
   272→      og.setValueAtTime(vel * 1.1, when);
   273→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   274→      if (punch > 0) {
   275→        ng.setValueAtTime(vel * punch * 0.8, when);
   276→        ng.exponentialRampToValueAtTime(0.0001, when + 0.02);
   277→        this.nFilter.frequency.setValueAtTime(2500, when);
   278→      }
   279→    } else if (type === 'clap') {
   280→      const dur = 0.25 + 0.15 * decay;
   281→      this.nFilter.type = 'bandpass';
   282→      this.nFilter.frequency.setValueAtTime(1150 * tone, when);
   283→      this.nFilter.Q.value = 1.3;
   284→      ng.setValueAtTime(0, when);
   285→      [0, 0.014, 0.03].forEach(t2 => {
   286→        ng.setValueAtTime(0, when + t2);
   287→        ng.linearRampToValueAtTime(vel * 0.9, when + t2 + 0.002);
   288→        ng.exponentialRampToValueAtTime(0.02, when + t2 + 0.012);
   289→      });
   290→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   291→    } else if (type === 'hatC' || type === 'hatO') {
   292→      const open = type === 'hatO';
   293→      const dur = open ? 0.26 + 0.5 * decay : 0.03 + 0.05 * decay;
   294→      this.nFilter.type = 'highpass';
   295→      this.nFilter.frequency.setValueAtTime(7200 * Math.sqrt(tone), when);
   296→      ng.setValueAtTime(vel * (open ? 0.4 : 0.5), when);
   297→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   298→    } else if (type === 'tom') {
   299→      const dur = 0.22 + 0.35 * decay;
   300→      this.osc.type = 'sine';
   301→      this.osc.frequency.setValueAtTime(180 * tune, when);
   302→      this.osc.frequency.exponentialRampToValueAtTime(92 * tune, when + dur * 0.7);
   303→      og.setValueAtTime(vel * 0.9, when);
   304→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   305→    } else if (type === 'shaker') {
   306→      const dur = 0.04 + 0.07 * decay;
   307→      this.nFilter.type = 'highpass';
   308→      this.nFilter.frequency.setValueAtTime(6000 * tone, when);
   309→      ng.setValueAtTime(vel * 0.45, when);
   310→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   311→    } else if (type === 'glitch') {
   312→      const dur = 0.08 + 0.14 * decay;
   313→      this.nFilter.type = 'bandpass';
   314→      this.nFilter.frequency.setValueAtTime(1500 * tone + 800, when);
   315→      this.nFilter.Q.value = 4;
   316→      ng.setValueAtTime(vel * 0.7, when);
   317→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   318→    }
   319→  }
   320→
   321→  panic(ctx: AudioContext) {
   322→    try {
   323→      this.noiseGain.gain.cancelScheduledValues(0);
   324→      this.noiseGain.gain.setValueAtTime(0, ctx.currentTime);
   325→      this.oscGain.gain.cancelScheduledValues(0);
   326→      this.oscGain.gain.setValueAtTime(0, ctx.currentTime);
   327→    } catch {}
   328→  }
   329→}

   470→    for (let ch = 0; ch < 2; ch++) {
   471→      const d = ir.getChannelData(ch);
   472→      for (let i = 0; i < irLen; i++) {
   473→        d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (c.sampleRate * 0.4));
   474→      }
   475→    }
   476→    this.reverb.buffer = ir;
   477→    this.reverbReturn = c.createGain();
   478→    this.reverbReturn.gain.value = 0.6;
   479→    this.reverbSend.connect(this.reverb);
   480→    this.reverb.connect(this.reverbReturn);
   481→    this.reverbReturn.connect(this.master);
   482→
   483→    // Track buses (8 tracks) with HPF, pan, and duck
   484→    this.duckGain = c.createGain();
   485→    this.duckGain.gain.value = 1.0;
   486→    // duckGain connects to master ONCE (not twice — was causing feedback)
   487→    this.duckGain.connect(this.master);
   488→
   489→    for (let i = 0; i < 8; i++) {
   490→      const bus = c.createGain();
   491→      // HPF on each track to clean mud (except kick/bass)
   492→      const hpf = c.createBiquadFilter();
   493→      hpf.type = 'highpass';
   494→      hpf.frequency.value = i < 2 ? 20 : (i < 4 ? 80 : 120);
   495→      // Stereo panner
   496→      const panner = c.createStereoPanner();
   497→      const panValues = [0, -0.15, 0.2, -0.2, 0, 0.1, -0.1, 0.15];
   498→      panner.pan.value = panValues[i];
   499→      // Track gain
   500→      const gain = c.createGain();
   501→      gain.gain.value = 0.8;
   502→
   503→      bus.connect(hpf);
   504→      hpf.connect(panner);
   505→      // Bass track (4) goes through duck gain for sidechain
   506→      if (i === 4) {
   507→        panner.connect(this.duckGain);
   508→      } else {
   509→        panner.connect(gain);
   510→        gain.connect(this.master);
   511→      }
   512→
   513→      // Send to delay/reverb
   514→      const dSend = c.createGain();
   515→      dSend.gain.value = i >= 4 ? 0.15 : 0.02;
   516→      bus.connect(dSend);
   517→      dSend.connect(this.delaySend);
   518→      const rSend = c.createGain();
   519→      rSend.gain.value = i >= 5 ? 0.2 : 0.03;
   520→      bus.connect(rSend);
   521→      rSend.connect(this.reverbSend);
   522→
   523→      this.chains.push(bus);
   524→      this.trackGains.push(gain);
   525→    }
   526→    // duckGain already connected to master above (line 458)
   527→
   528→    // Allocate voice pools (20 synth + 24 drum — from PSY6)
   529→    for (let i = 0; i < 20; i++) this.synthPool.push(new PooledSynthVoice(c));
   530→    for (let i = 0; i < 24; i++) this.drumPool.push(new PooledDrumVoice(c, this.noiseBuffer));
   531→
   532→    // Initialize tracks
   533→    this.initTracks();
   534→  }

   646→  selfTrack(selfMetrics: { lufs: number; energy?: number }): void {
   647→    this.ownLufs = selfMetrics.lufs;
   648→    // LUFS matching
   649→    if (this.targetLufs !== 0 && Math.abs(selfMetrics.lufs - this.targetLufs) > 1.0) {
   650→      const diff = this.targetLufs - selfMetrics.lufs;
   651→      const adj = diff > 0 ? 0.08 : -0.08;
   652→      const newMaster = clamp(this.master.gain.value + adj, 0.3, 2.0);
   653→      this.master.gain.setTargetAtTime(newMaster, this.ctx!.currentTime, 0.15);
   654→    }
   655→    // Energy matching — adjust track volumes
   656→    if (this.targetEnergy > 0 && selfMetrics.energy !== undefined) {
   657→      const energyDiff = this.targetEnergy - selfMetrics.energy;
   658→      if (Math.abs(energyDiff) > 0.05) {
   659→        // Boost/reduce all track gains slightly
   660→        const volAdj = energyDiff > 0 ? 0.02 : -0.02;
   661→        for (let i = 0; i < 8; i++) {
   662→          const g = this.trackGains[i];
   663→          if (g) {
   664→            const newVol = clamp(g.gain.value + volAdj, 0.1, 2.0);
   665→            g.gain.setTargetAtTime(newVol, this.ctx!.currentTime, 0.5);
   666→          }
   667→        }
   668→      }
   669→    }

   689→  private scheduleNextTick(): void {
   690→    if (!this.playing) return;
   691→    this.timer = setTimeout(() => {
   692→      this.tick();
   693→      this.scheduleNextTick();
   694→    }, 15);

   697→  private tick(): void {
   698→    if (!this.playing || !this.ctx) return;
   699→    const lookahead = 0.06;
   700→    const s16 = 60 / this.bpm / 4;
   701→
   702→    while (this.nextTime < this.ctx.currentTime + lookahead) {
   703→      this.scheduleStep(this.step, this.bar, this.nextTime);
   704→      this.step++;
   705→      this.nextTime += s16;
   706→      if (this.step >= 16) {
   707→        this.step = 0;
   708→        this.bar++;
   709→        const section = this.arrangement[this.sectionIdx % this.arrangement.length];
   710→        if (this.bar >= section.bars) {
   711→          this.sectionIdx++;
   712→          this.bar = 0;
   713→          const next = this.arrangement[this.sectionIdx % this.arrangement.length];
   714→          this.currentSection = next.label;
   715→          this.onSectionChange?.(this.currentSection);
   716→        }
   717→        // Change preset every 4 bars
   718→        if (this.bar % 4 === 0 && this.bar > 0) {
   719→          this.presetIdx = (this.presetIdx + 1) % 3;
   720→          this.rotatePresets();
   721→        }
   722→      }
   723→    }
   724→  }

   726→  private rotatePresets(): void {
   727→    // Rotate drum presets for variety
   728→    const kickPresets = ['PS-KICK-TIGHT', 'PS-KICK-DEEP'];
   729→    const bassPresets = ['PS-BASS-ROLL', 'PS-BASS-DEEP'];
   730→    const leadPresets = ['PS-LEAD-SQUELCH', 'PS-LEAD-FMTEX'];
   731→    this.tracks[0].presetId = kickPresets[this.presetIdx % kickPresets.length];
   732→    this.tracks[4].presetId = bassPresets[this.presetIdx % bassPresets.length];
   733→    this.tracks[5].presetId = leadPresets[this.presetIdx % leadPresets.length];
   734→  }

   736→  private scheduleStep(step: number, bar: number, time: number): void {
   737→    const section = this.arrangement[this.sectionIdx % this.arrangement.length];
   738→    const key = this.musicalKey;
   739→    const root = key.root;
   740→    const sc = key.scale;
   741→    const sd = 60 / this.bpm / 4;
   742→    const isPreDrop = (section.label === 'BUILD' || section.label === 'BUILD 2') && bar >= section.bars - 2;
   743→    const isDropStart = section.label.includes('DROP') && bar === 0 && step === 0;
   744→
   745→    // ── RISER FX (last 2 bars of build) ──
   746→    if (isPreDrop && step === 0 && bar === section.bars - 2) {
   747→      this.triggerRiser(time, sd * 32);
   748→    }
   749→
   750→    // ── IMPACT FX (drop start) ──
   751→    if (isDropStart) {
   752→      this.triggerImpact(time);
   753→    }
   754→
   755→    // ── KICK (track 0) — 4 on the floor ──
   756→    if (step % 4 === 0) {
   757→      this.triggerDrum(0, time, 0.5 + section.density * 0.4);
   758→    }
   759→
   760→    // ── CLAP (track 1) — on 2 and 4 ──
   761→    if ((step === 4 || step === 12) && section.density > 0.4) {
   762→      this.triggerDrum(1, time, 0.3);
   763→    }
   764→
   765→    // ── HATS (track 2) — offbeat with velocity variation ──
   766→    if (step % 2 === 1) {
   767→      const vel = 0.15 + (step % 4 === 3 ? 0.1 : 0) + section.density * 0.1;
   768→      this.triggerDrum(2, time, vel);
   769→    }
   770→
   771→    // ── PERC (track 3) — sparse, with variation ──
   772→    if (section.density > 0.5 && (step === 6 || step === 14) && Math.random() < 0.6) {
   773→      this.triggerDrum(3, time, 0.2);
   774→    }
   775→
   776→    // ── BASS (track 4) — offbeat, evolving pattern with passing tones ──
   777→    if (section.bass && step % 2 === 1) {
   778→      // Rich bass pattern: changes every 4 bars with passing tones and octaves
   779→      let bassPattern: number[];
   780→      const bp = bar % 8;
   781→      if (bp < 2) bassPattern = [0, 0, 0, 0, 0, 0, 4, 0]; // root with fifth
   782→      else if (bp < 4) bassPattern = [0, 0, 2, 0, 4, 0, 3, 0]; // walking
   783→      else if (bp < 6) bassPattern = [0, 0, 0, 7, 0, 0, 4, 3]; // octave + passing
   784→      else bassPattern = [0, 4, 0, 3, 0, 2, 0, 4]; // rolling
   785→
   786→      const bassDeg = bassPattern[Math.floor(step / 2) % bassPattern.length];
   787→      if (bassDeg >= 0) {
   788→        const note = scaleNote(root, sc, bassDeg);
   789→        this.triggerSynth(4, time, note, 0.5, sd);
   790→      }
   791→    }
   792→
   793→    // ── LEAD (track 5) — evolving trance motif with SPACES ──
   794→    if (section.lead) {
   795→      const phraseBar = bar % 8;
   796→      const leadSteps = [0, 6, 10];
   797→      if (leadSteps.includes(step)) {
   798→        let deg: number; let dur: number; let vel: number;
   799→        if (phraseBar < 2) {
   800→          const motif = [7, 5, 3];
   801→          deg = motif[leadSteps.indexOf(step)];
   802→          dur = 0.3; vel = 0.25;
   803→        } else if (phraseBar < 4) {
   804→          const motif = [5, 3, 0];
   805→          deg = motif[leadSteps.indexOf(step)];
   806→          dur = 0.2; vel = 0.3;
   807→        } else if (phraseBar < 6) {
   808→          const motif = [10, 12, 10];
   809→          deg = motif[leadSteps.indexOf(step)];
   810→          dur = 0.15; vel = 0.35;
   811→        } else if (phraseBar === 6) {
   812→          deg = step === 0 ? 0 : step === 6 ? 2 : 0;
   813→          dur = 0.4; vel = 0.4;
   814→        } else {
   815→          return; // rest bar
   816→        }
   817→        const note = scaleNote(root + 12, sc, deg);
   818→        this.triggerSynth(5, time, note, vel, sd, dur);
   819→      }
   820→    }
   821→
   822→    // ── PAD (track 6) — sustained chord every bar in drops ──
   823→    if (section.lead && step === 0) {
   824→      // Chord progression: root, IV, V, III (psytrance progression)
   825→      const progDegs = [0, 3, 4, 2];
   826→      const chordDeg = progDegs[bar % 4];
   827→      const chordRoot = scaleNote(root + 12, sc, chordDeg);
   828→      this.triggerSynth(6, time, chordRoot, 0.25, sd * 4);
   829→      // Also play fifth for full chord
   830→      this.triggerSynth(6, time + 0.01, scaleNote(root + 12, sc, chordDeg + 4), 0.15, sd * 4);
   831→    }
   832→
   833→    // ── ARP (track 7) — rolling arp, 70% density in drops ──
   834→    if (section.lead && step % 2 === 0 && Math.random() < 0.7) {
   835→      const arpDegs = [0, 2, 4, 7, 4, 2, 0, 7];
   836→      const deg = arpDegs[(step / 2) % arpDegs.length];
   837→      const note = scaleNote(root + 24, sc, deg);
   838→      this.triggerSynth(7, time, note, 0.25, sd);
   839→    }
   840→
   841→    // ── SHAKER (track 3 alt) — continuous offbeat in drops ──
   842→    if (section.bass && section.lead && step % 2 === 1 && Math.random() < 0.4) {
   843→      this.triggerDrum(3, time, 0.15);
   844→    }
   845→  }

   847→  private triggerDrum(trackIdx: number, time: number, vel: number): void {
   848→    const track = this.tracks[trackIdx];
   849→    if (track.mix.mute) return;
   850→    const preset = DRUM_PRESETS[track.presetId];
   851→    if (!preset) return;
   852→    const voice = this.drumPool[this.drumIdx];
   853→    this.drumIdx = (this.drumIdx + 1) % this.drumPool.length;
   854→    voice.hit(preset, time, vel * track.mix.vol, this.chains[trackIdx]);
   855→
   856→    // Sidechain: when kick fires, duck the bass
   857→    if (trackIdx === 0 && this.duckGain && this.ctx) {
   858→      this.duckGain.gain.cancelScheduledValues(time);
   859→      this.duckGain.gain.setValueAtTime(1 - 0.4, time); // 40% duck
   860→      this.duckGain.gain.linearRampToValueAtTime(1.0, time + 0.25); // 250ms recovery
   861→    }
   862→  }
   863→
   864→  private triggerRiser(time: number, dur: number): void {
   865→    if (!this.ctx) return;
   866→    const c = this.ctx;
   867→    // Noise through filter that opens up
   868→    const noise = c.createBufferSource();
   869→    noise.buffer = this.noiseBuffer;
   870→    noise.loop = true;
   871→    const filter = c.createBiquadFilter();
   872→    filter.type = 'bandpass';
   873→    filter.frequency.setValueAtTime(200, time);
   874→    filter.frequency.exponentialRampToValueAtTime(8000, time + dur);
   875→    filter.Q.value = 2;
   876→    const gain = c.createGain();
   877→    gain.gain.setValueAtTime(0.001, time);
   878→    gain.gain.exponentialRampToValueAtTime(0.3, time + dur);
   879→    gain.gain.exponentialRampToValueAtTime(0.001, time + dur + 0.1);
   880→    noise.connect(filter);
   881→    filter.connect(gain);
   882→    gain.connect(this.master);
   883→    noise.start(time);
   884→    noise.stop(time + dur + 0.2);
   885→  }
   886→
   887→  private triggerImpact(time: number): void {
   888→    if (!this.ctx) return;
   889→    const c = this.ctx;
   890→    // Sub boom + noise crack
   891→    const osc = c.createOscillator();
   892→    osc.type = 'sine';
   893→    osc.frequency.setValueAtTime(120, time);
   894→    osc.frequency.exponentialRampToValueAtTime(35, time + 0.5);
   895→    const og = c.createGain();
   896→    og.gain.setValueAtTime(0.8, time);
   897→    og.gain.exponentialRampToValueAtTime(0.001, time + 0.6);
   898→    osc.connect(og);
   899→    og.connect(this.master);
   900→    osc.start(time);
   901→    osc.stop(time + 0.7);
   902→
   903→    const noise = c.createBufferSource();
   904→    noise.buffer = this.noiseBuffer;
   905→    const hp = c.createBiquadFilter();
   906→    hp.type = 'highpass';
   907→    hp.frequency.value = 3000;
   908→    const ng = c.createGain();
   909→    ng.gain.setValueAtTime(0.4, time);
   910→    ng.gain.exponentialRampToValueAtTime(0.001, time + 0.05);
   911→    noise.connect(hp);
   912→    hp.connect(ng);
   913→    ng.connect(this.master);
   914→    noise.start(time);
   915→    noise.stop(time + 0.1);
   916→  }

   918→  private triggerSynth(trackIdx: number, time: number, midi: number, vel: number, stepDur: number, dur?: number): void {
   919→    const track = this.tracks[trackIdx];
   920→    if (track.mix.mute) return;
   921→    const preset = SYNTH_PRESETS[track.presetId];
   922→    if (!preset) return;
   923→    const voice = this.synthPool[this.synthIdx];
   924→    this.synthIdx = (this.synthIdx + 1) % this.synthPool.length;
   925→    const p = dur ? { ...preset, gate: dur / (stepDur * 2) } : preset;
   926→    voice.noteOn(p, time, midi, vel * track.mix.vol, stepDur, this.chains[trackIdx]);
   927→
   928→    // Add sub oscillator for bass track (sine one octave below)
