   202→/**
   203→ * Map a reference spectral centroid (Hz) to a target synth cutoff (Hz)
   204→ * using log-linear interpolation. Anchor points:
   205→ *   500Hz  ->  800 Hz cutoff (dark / warm)
   206→ *   2000Hz -> 3000 Hz cutoff (balanced)
   207→ *   5000Hz -> 6000 Hz cutoff (bright / cutting)
   208→ * Falls back to clamp(centroid, 200, 12000) if out of typical range.
   209→ */
   210→function centroidToCutoff(centroid: number): number {
   211→  if (!isFinite(centroid) || centroid <= 0) return 1500;
   212→  const lc = Math.log(clamp(centroid, 80, 16000));
   213→  // Fit using endpoints (500,800) and (5000,6000):
   214→  //   ln(cut) = 0.8753 * ln(centroid) + 1.245
   215→  const lcut = 0.8753 * lc + 1.245;
   216→  return clamp(Math.exp(lcut), 200, 12000);
   217→}

   324→// ─── Synth voice pool ───────────────────────────────────────────────────────
   325→//
   326→// The synth pool uses AdvancedSynthVoice (from ./advancedVoice) which supports
   327→// 4 modes: classic (drop-in 2-osc), fm (carrier+modulator), supersaw (5-7 detuned
   328→// saws with stereo spread), and wavetable (2 crossfading periodic waves with LFO
   329→// morph). All nodes are preallocated per voice — zero per-note allocation.
   330→//
   331→// Each voice holds up to 7 OscillatorNodes. With 20 voices that's 140 oscillators
   332→// max, well within modern browser limits.
   333→
   334→// ─── Pooled Drum Voice (from PSY6 — multi-type) ────────────────────────────
   335→
   336→class PooledDrumVoice {
   337→  noise: AudioBufferSourceNode;
   338→  noiseGain: GainNode;
   339→  nFilter: BiquadFilterNode;
   340→  osc: OscillatorNode;
   341→  oscGain: GainNode;
   342→  out: GainNode;
   343→  bus: GainNode | null = null;
   344→
   345→  constructor(ctx: AudioContext, noiseBuffer: AudioBuffer) {
   346→    this.noise = ctx.createBufferSource();
   347→    this.noise.buffer = noiseBuffer;
   348→    this.noise.loop = true;
   349→    this.noiseGain = ctx.createGain();
   350→    this.noiseGain.gain.value = 0;
   351→    this.nFilter = ctx.createBiquadFilter();
   352→    this.nFilter.type = 'bandpass';
   353→    this.noise.connect(this.nFilter);
   354→    this.nFilter.connect(this.noiseGain);
   355→    this.osc = ctx.createOscillator();
   356→    this.osc.type = 'sine';
   357→    this.oscGain = ctx.createGain();
   358→    this.oscGain.gain.value = 0;
   359→    this.osc.connect(this.oscGain);
   360→    this.out = ctx.createGain();
   361→    this.noiseGain.connect(this.out);
   362→    this.oscGain.connect(this.out);
   363→    this.noise.start();
   364→    this.osc.start();
   365→  }
   366→
   367→  connect(bus: GainNode) {
   368→    if (this.bus !== bus) {
   369→      this.out.disconnect();
   370→      this.out.connect(bus);
   371→      this.bus = bus;
   372→    }
   373→  }
   374→
   375→  /**
   376→   * Trigger this voice. The optional `decayOverride` lets callers (e.g. reference
   377→   * pursuit) replace the preset decay with a blended value to chase the radio's
   378→   * kick decay. It is guarded against NaN/zero/out-of-range values.
   379→   */
   380→  hit(p: DrumPreset, when: number, vel: number, bus: GainNode, decayOverride?: number) {
   381→    this.connect(bus);
   382→    const tune = p.tune || 1;
   383→    const candidateDecay = (typeof decayOverride === 'number'
   384→      && isFinite(decayOverride) && decayOverride > 0.001 && decayOverride < 50)
   385→      ? decayOverride
   386→      : null;
   387→    const decay = candidateDecay ?? p.decay ?? 1;
   388→    const tone = p.tone || 1;
   389→    const punch = p.punch || 0;
   390→    const type = p.type;
   391→    const ng = this.noiseGain.gain;
   392→    const og = this.oscGain.gain;
   393→    ng.cancelScheduledValues(when);
   394→    og.cancelScheduledValues(when);
   395→    ng.setValueAtTime(0, when);
   396→    og.setValueAtTime(0, when);
   397→
   398→    if (type === 'kick') {
   399→      const dur = 0.12 + 0.5 * decay;
   400→      this.osc.frequency.setValueAtTime(180 * tune, when);
   401→      this.osc.frequency.exponentialRampToValueAtTime(Math.max(36 * tune, 24), when + 0.09);
   402→      og.setValueAtTime(vel * 1.1, when);
   403→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   404→      if (punch > 0) {
   405→        ng.setValueAtTime(vel * punch * 0.8, when);
   406→        ng.exponentialRampToValueAtTime(0.0001, when + 0.02);
   407→        this.nFilter.frequency.setValueAtTime(2500, when);
   408→      }
   409→    } else if (type === 'clap') {
   410→      const dur = 0.25 + 0.15 * decay;
   411→      this.nFilter.type = 'bandpass';
   412→      this.nFilter.frequency.setValueAtTime(1150 * tone, when);
   413→      this.nFilter.Q.value = 1.3;
   414→      ng.setValueAtTime(0, when);
   415→      [0, 0.014, 0.03].forEach(t2 => {
   416→        ng.setValueAtTime(0, when + t2);
   417→        ng.linearRampToValueAtTime(vel * 0.9, when + t2 + 0.002);
   418→        ng.exponentialRampToValueAtTime(0.02, when + t2 + 0.012);
   419→      });
   420→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   421→    } else if (type === 'hatC' || type === 'hatO') {
   422→      const open = type === 'hatO';
   423→      const dur = open ? 0.26 + 0.5 * decay : 0.03 + 0.05 * decay;
   424→      this.nFilter.type = 'highpass';
   425→      this.nFilter.frequency.setValueAtTime(7200 * Math.sqrt(tone), when);
   426→      ng.setValueAtTime(vel * (open ? 0.4 : 0.5), when);
   427→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   428→    } else if (type === 'tom') {
   429→      const dur = 0.22 + 0.35 * decay;
   430→      this.osc.type = 'sine';
   431→      this.osc.frequency.setValueAtTime(180 * tune, when);
   432→      this.osc.frequency.exponentialRampToValueAtTime(92 * tune, when + dur * 0.7);
   433→      og.setValueAtTime(vel * 0.9, when);
   434→      og.exponentialRampToValueAtTime(0.0001, when + dur);
   435→    } else if (type === 'shaker') {
   436→      const dur = 0.04 + 0.07 * decay;
   437→      this.nFilter.type = 'highpass';
   438→      this.nFilter.frequency.setValueAtTime(6000 * tone, when);
   439→      ng.setValueAtTime(vel * 0.45, when);
   440→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   441→    } else if (type === 'glitch') {
   442→      const dur = 0.08 + 0.14 * decay;
   443→      this.nFilter.type = 'bandpass';
   444→      this.nFilter.frequency.setValueAtTime(1500 * tone + 800, when);
   445→      this.nFilter.Q.value = 4;
   446→      ng.setValueAtTime(vel * 0.7, when);
   447→      ng.exponentialRampToValueAtTime(0.0001, when + dur);
   448→    }
   449→  }
   450→
   451→  panic(ctx: AudioContext) {
   452→    try {
   453→      this.noiseGain.gain.cancelScheduledValues(0);
   454→      this.noiseGain.gain.setValueAtTime(0, ctx.currentTime);
   455→      this.oscGain.gain.cancelScheduledValues(0);
   456→      this.oscGain.gain.setValueAtTime(0, ctx.currentTime);
   457→    } catch {}
   458→  }
   459→}

   799→

   801→    this.noiseBuffer = c.createBuffer(1, c.sampleRate * 2, c.sampleRate);
   802→    const nd = this.noiseBuffer.getChannelData(0);
   803→    for (let i = 0; i < nd.length; i++) nd[i] = Math.random() * 2 - 1;

   806→    this.master = c.createGain();

   810→    this.saturator = c.createWaveShaper();
   811→    this.saturator.oversample = '4x';
   812→    const satCurve = new Float32Array(1024);
   813→    for (let i = 0; i < 1024; i++) {
   814→      const x = (i / 512) - 1;
   815→      satCurve[i] = Math.tanh(x * 1.3) * 0.7 + x * 0.3;
   816→    }
   817→    this.saturator.curve = satCurve;

   821→    this.toneLow.type = 'lowshelf';
   822→    this.toneLow.frequency.value = 110;
   823→    this.toneLow.gain.value = 1.5;
   824→    this.toneHigh = c.createBiquadFilter();
   825→    this.toneHigh.type = 'highshelf';
   826→    this.toneHigh.frequency.value = 8500;
   827→    this.toneHigh.gain.value = -2;
   828→
   829→    this.comp = c.createDynamicsCompressor();
   830→    // Comp is now a SAFETY LIMITER after the multiband — gentle ratio, fast
   831→    // attack, threshold just below clipping. The multiband does the heavy
   832→    // lifting; this catches anything that slips through.
   833→    this.comp.threshold.value = -3;
   834→    this.comp.knee.value = 6;
   835→    this.comp.ratio.value = 3;
   836→    this.comp.attack.value = 0.002;
   837→    this.comp.release.value = 0.15;
   838→    this.analyser = c.createAnalyser();
   839→    this.analyser.fftSize = 2048;
   840→
   841→    // Multiband compressor (Task E1) — 3-band crossover on the master bus.
   842→    // Splits LOW/MID/HIGH, compresses each separately, sums back. This is
   843→    // what gives the "loud, glued" commercial sound.
   844→    this.multiband = new MultibandCompressor(c, {
   845→      crossoverLow: 200, crossoverHigh: 2000,
   846→      lowThreshold: -18, lowRatio: 4,  lowAttack: 0.012, lowRelease: 0.2,  lowKnee: 6,  lowMakeup: 1.2,
   847→      midThreshold: -20, midRatio: 3,  midAttack: 0.008, midRelease: 0.15, midKnee: 10, midMakeup: 1.1,
   848→      highThreshold: -22, highRatio: 2, highAttack: 0.003, highRelease: 0.08, highKnee: 12, highMakeup: 1.0,
   849→    });
   850→
   851→    // Connect: master → saturator → toneLow → toneHigh → multiband → comp (safety) → analyser → destination
   852→    this.master.connect(this.saturator);
   853→    this.saturator.connect(this.toneLow);
   854→    this.toneLow.connect(this.toneHigh);
   855→    this.toneHigh.connect(this.multiband.input);
   856→    this.multiband.output.connect(this.comp);
   857→    this.comp.connect(this.analyser);
   858→    this.analyser.connect(c.destination);
   859→
   860→    // Delay (ping-pong with band-limited feedback)
   861→    this.delaySend = c.createGain();
   862→    this.delaySend.gain.value = 0.15;
   863→    this.delay = c.createDelay(0.5);
   864→    this.delay.delayTime.value = 0.375;
   865→    this.delayFb = c.createGain();
   866→    this.delayFb.gain.value = 0.35;
   867→    this.delayReturn = c.createGain();
   868→    this.delayReturn.gain.value = 0.5;
   869→    const dLP = c.createBiquadFilter();
   870→    dLP.type = 'lowpass';
   871→    dLP.frequency.value = 3300;
   872→    this.delaySend.connect(this.delay);
   873→    this.delay.connect(dLP);
   874→    dLP.connect(this.delayFb);
   875→    this.delayFb.connect(this.delay);
   876→    this.delay.connect(this.delayReturn);
   877→    this.delayReturn.connect(this.master);
   878→
   879→    // Reverb
   880→    this.reverbSend = c.createGain();
   881→    this.reverbSend.gain.value = 0.12;
   882→    this.reverb = c.createConvolver();
   883→    const irLen = Math.floor(c.sampleRate * 1.5);
   884→    const ir = c.createBuffer(2, irLen, c.sampleRate);
   885→    for (let ch = 0; ch < 2; ch++) {
   886→      const d = ir.getChannelData(ch);
   887→      for (let i = 0; i < irLen; i++) {
   888→        d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (c.sampleRate * 0.4));
   889→      }
   890→    }
   891→    this.reverb.buffer = ir;
   892→    this.reverbReturn = c.createGain();
   893→    this.reverbReturn.gain.value = 0.6;
   894→    this.reverbSend.connect(this.reverb);
   895→    this.reverb.connect(this.reverbReturn);
   896→    this.reverbReturn.connect(this.master);
   897→
   898→    // ── Send effects: Chorus / Phaser / Distortion / Bitcrush (Task E1) ──
   899→    // Each is a global SEND effect: per-track rack sends a portion to the bus
   900→    // gain, the bus feeds the effect input, and the effect output returns to
   901→    // the master sum (post-multiband-input, so sends are also compressed).
   902→
   903→    // Chorus — modulated short delays, stereo spread. For lead/pad/arp.
   904→    this.chorusSend = c.createGain();
   905→    this.chorusSend.gain.value = 1.0; // bus level; per-track sends scale this
   906→    this.chorusEffect = new ChorusSend(c, {
   907→      rate: 0.5, depth: 0.004, baseDelay: 0.012, wet: 0.6, dry: 0.7,
   908→    });
   909→    this.chorusReturn = c.createGain();
   910→    this.chorusReturn.gain.value = 0.5;
   911→    this.chorusSend.connect(this.chorusEffect.input);
   912→    this.chorusEffect.output.connect(this.chorusReturn);
   913→    this.chorusReturn.connect(this.master);
   914→
   915→    // Phaser — allpass cascade + LFO sweep. For lead/arp (psychedelic motion).
   916→    this.phaserSend = c.createGain();
   917→    this.phaserSend.gain.value = 1.0;
   918→    this.phaserEffect = new PhaserSend(c, {
   919→      rate: 0.3, depth: 0.6, baseFreq: 800, feedback: 0.4, stages: 6,
   920→      wet: 0.55, dry: 0.6,
   921→    });
   922→    this.phaserReturn = c.createGain();
   923→    this.phaserReturn.gain.value = 0.5;
   924→    this.phaserSend.connect(this.phaserEffect.input);
   925→    this.phaserEffect.output.connect(this.phaserReturn);
   926→    this.phaserReturn.connect(this.master);
   927→
   928→    // Distortion — hard-clip waveshaper. For acid/lead (grit).
   929→    this.distortionSend = c.createGain();
   930→    this.distortionSend.gain.value = 1.0;
   931→    this.distortionEffect = new DistortionSend(c, {
   932→      drive: 4, tone: 4000, wet: 0.5, dry: 0.5,
   933→    });
   934→    this.distortionReturn = c.createGain();
   935→    this.distortionReturn.gain.value = 0.4;
   936→    this.distortionSend.connect(this.distortionEffect.input);
   937→    this.distortionEffect.output.connect(this.distortionReturn);
   938→    this.distortionReturn.connect(this.master);
   939→
   940→    // Bitcrush — stair-step quantizer + sample-and-hold. For lo-fi texture.
   941→    this.bitcrushSend = c.createGain();
   942→    this.bitcrushSend.gain.value = 1.0;
   943→    this.bitcrushEffect = new BitcrushSend(c, {
   944→      bits: 6, holdMs: 4, tone: 2500, wet: 0.5, dry: 0.6,
   945→    });
   946→    this.bitcrushReturn = c.createGain();
   947→    this.bitcrushReturn.gain.value = 0.35;
   948→    this.bitcrushSend.connect(this.bitcrushEffect.input);
   949→    this.bitcrushEffect.output.connect(this.bitcrushReturn);
   950→    this.bitcrushReturn.connect(this.master);
   951→
   952→    // Track buses (8 tracks) — each gets a full TrackEffectsRack insert chain.
   953→    this.duckGain = c.createGain();
   954→    this.duckGain.gain.value = 1.0;
   955→    // duckGain connects to master ONCE (not twice — was causing feedback)
   956→    this.duckGain.connect(this.master);
   957→
   958→    // Build per-track rack configs from the current world. The configs encode
   959→    // per-track EQ/comp/sat/pan/Haas + per-track send levels + per-world
   960→    // modulations (dark-psy → more distortion, goa → more phaser, etc.).
   961→    const rackConfigs = buildTrackRackConfigs(this.currentWorld);
   962→    for (let i = 0; i < 8; i++) {
   963→      const rack = new TrackEffectsRack(c, rackConfigs[i]);
   964→      this.racks.push(rack);
   965→      // Voices connect to rack.input; liveTrack/setWorld adjust rack.output.
   966→      this.chains.push(rack.input);
   967→      this.trackGains.push(rack.output);
   968→
   969→      // Wire rack.output → (duckGain for bass | master for others).
   970→      // Bass goes through duckGain for sidechain ducking on kick hits.
   971→      if (i === 4) {
   972→        rack.output.connect(this.duckGain);
   973→      } else {
   974→        rack.output.connect(this.master);
   975→      }
   976→
   977→      // Wire all 6 send taps to the global send bus inputs.
   978→      // reverb/delay are the existing global buses (their gain is the master
   979→      // FX mix controlled by world.fxMix); chorus/phaser/distortion/bitcrush
   980→      // are the new Task E1 buses (their gain stays at 1.0 — per-track send
   981→      // gains do the scaling).
   982→      rack.connectSend('reverb', this.reverbSend);
   983→      rack.connectSend('delay', this.delaySend);
   984→      rack.connectSend('chorus', this.chorusSend);
   985→      rack.connectSend('phaser', this.phaserSend);
   986→      rack.connectSend('distortion', this.distortionSend);
   987→      rack.connectSend('bitcrush', this.bitcrushSend);
   988→    }
   989→    // duckGain already connected to master above
   990→
   991→    // Allocate voice pools (P1: 8 synth + 10 drum — down from 20 + 24).
   992→    // Each AdvancedSynthVoice now lazy-allocates only its 8 common nodes on
   993→    // construction; per-osc nodes are added on noteOn and torn down on
   994→    // release-tail-end. With 8 voices × 8 common = 64 idle nodes (was 580).
   995→    // 10 drum voices × ~8 nodes = 80 nodes; with voice stealing this covers
   996→    // the busiest psytrance patterns.
   997→    for (let i = 0; i < 8; i++) this.synthPool.push(new AdvancedSynthVoice(c, i));
   998→    for (let i = 0; i < 10; i++) this.drumPool.push(new PooledDrumVoice(c, this.noiseBuffer));
   999→
  1000→    // Initialize tracks
  1001→    this.initTracks();
  1002→  }

  1004→  private initTracks(): void {

  1006→    const presets = ['PS-KICK-TIGHT', 'TR-CLAP', 'PS-HAT', 'PS-PERC', 'PS-BASS-ROLL', 'PS-LEAD-SQUELCH', 'PS-PAD-PSYCH', 'PS-ARP-ACID'];
  1007→    // Boosted track volumes for commercial loudness
  1008→    const vols = [1.0, 0.6, 0.5, 0.4, 1.2, 0.7, 0.5, 0.5];
  1009→    this.tracks = [];
  1010→    for (let i = 0; i < 8; i++) {
  1011→      this.tracks.push({
  1012→        idx: i,
  1013→        kind: i < 4 ? 'drum' : 'synth',
  1014→        name: names[i],
  1015→        presetId: presets[i],
  1016→        mix: { vol: vols[i], pan: 0, mute: false, sendA: 0, sendB: 0 },
  1017→      });
  1018→    }
  1019→    // Initialize musical generators with the default key so they exist before

  1024→  start(worldId?: string): void {
  1025→    this.init();
  1026→    if (this.ctx!.state === 'suspended') this.ctx!.resume();
  1027→
  1028→    // ── World-driven configuration (Track A) ──
  1029→    this.currentWorld = WORLDS[worldId as WorldId] || WORLDS['dark-psy'];
  1030→    this._bpm = this.currentWorld.defaultBpm;
  1031→    this.musicalKey = {
  1032→      root: Math.floor((this.currentWorld.rootRange[0] + this.currentWorld.rootRange[1]) / 2),
  1033→      scale: this.currentWorld.defaultScale,
  1034→    };
  1035→    // Re-create musical generators with the world's key (MelodyEngine, AcidPattern)
  1036→    this.refreshMusicalGenerators();
  1037→    this.arpIdx = 0;
  1038→    this.bassPatternIdx = 0;
  1039→    // Reset phrase-locked rotation counters for a clean start
  1040→    this.phraseCounter = 0;
  1041→    this.phrasePresetVariant = 0;
  1042→    // Reset auto-switch anti-thrash guard
  1043→    this.lastAutoSwitchTime = 0;
  1044→    this.lastAutoSwitchWorldId = null;
  1045→    this.styleMatches = [];
  1046→
  1047→    // Apply the world's preferred kick/bass/lead presets immediately so the
  1048→    // engine starts with the right timbres (Task 15).
  1049→    this.applyWorldPresets();
  1050→
  1051→    // Apply per-world send-effect settings (Task E1) — pushes per-track send
  1052→    // levels and global effect parameters for the current world.
  1053→    this.applyWorldEffectSettings(this.currentWorld);
  1054→
  1055→    // Apply world FX mix to reverb and delay sends
  1056→    const fxMix = this.currentWorld.fxMix;
  1057→    this.reverbSend.gain.setTargetAtTime(0.04 + fxMix * 0.22, this.ctx!.currentTime, 0.05);
  1058→    this.delaySend.gain.setTargetAtTime(0.05 + fxMix * 0.30, this.ctx!.currentTime, 0.05);
  1059→
  1060→    if (this.playing) return;
  1061→    this.playing = true;
  1062→    this.step = 0;
  1063→    this.bar = 0;
  1064→    this.sectionIdx = 0;
  1065→    // ── Task F1: create the dynamic flow engine ──
  1066→    // Replaces the fixed `arrangement` array with a creative, radio-responsive
  1067→    // flow. The engine decides when to transition (based on radio energy, time
  1068→    // since last transition, musical logic, and the world's flow profile) and
  1069→    // produces continuous automation parameters (filterCutoff, reverbAmount,
  1070→    // delayAmount, tension, surprise) that replace the old static section-based
  1071→    // automation.
  1072→    //
  1073→    // The seed combines the world id hash, key root, and scale length so each
  1074→    // world+key combo produces a different but reproducible flow. A fresh seed
  1075→    // on every start() means each play-through takes a different path through
  1076→    // the archetype graph (more creative, less formulaic).
  1077→    this.totalBars = 0;
  1078→    this.lastRefEnergyForFlow = 0;
  1079→    this.activeSurprise = null;
  1080→    this.surpriseReverseHitScheduled = false;
  1081→    const flowSeed = ((Date.now() & 0xffff) ^ (this.currentWorld.id.length * 131) ^
  1082→                      (this.musicalKey.root * 17) ^ (this.musicalKey.scale.length * 7) ^ 0x5a5a) >>> 0;
  1083→    this.flowEngine = new FlowEngine(new SeededRng(flowSeed || 1));
  1084→    this.flowEngine.setWorld(this.currentWorld);
  1085→    this.currentFlow = this.flowEngine.getCurrent();
  1086→    this.currentSection = this.currentFlow.label;
  1087→    // Task V2b: reset section-automation state so the first section's static
  1088→    // levels get pushed on the first tick. Also clear any leftover lead
  1089→    // cutoff override from a previous session so the lead starts clean.
  1090→    this.lastAutomationSection = '';
  1091→    this.leadCutoffOverride = -1;
  1092→    this.onSectionChange?.(this.currentSection);
  1093→    this.nextTime = this.ctx!.currentTime + 0.03;
  1094→    this.scheduleNextTick();
  1095→  }

  1097→  stop(): void {
  1098→    this.playing = false;
  1099→    // Task V2a: stop the Worker-based scheduler. The SchedulerWorker keeps
  1100→    // the underlying Worker instance alive across stop/start cycles (cheap
  1101→    // restart) — it just stops posting ticks. We also clear the legacy
  1102→    // `timer` field for the (currently unused) setTimeout fallback path.
  1103→    this.scheduler.stop();
  1104→    if (this.timer) { clearTimeout(this.timer); this.timer = null; }
  1105→    if (this.ctx) {
  1106→      for (const v of this.synthPool) v.panic(this.ctx);
  1107→      for (const v of this.drumPool) v.panic(this.ctx);
  1108→    }
  1109→    // ── Task D1: reset PhaseSync own-beat state ──
  1110→    // The reference phase + syncEnabled flag are preserved (the radio is
  1111→    // still playing and the user's toggle choice persists across restarts).
  1112→    // We only clear the own-beat ring buffer + phase offsets + beat-drop
  1113→    // state — these are engine-instance-specific and would be stale after
  1114→    // a restart (the new engine instance has a different audio context).
  1115→    this.phaseSync.reset();
  1116→  }

  1118→  private get bpm(): number {
  1119→    return this._bpm || 145;
  1120→  }
  1121→  private _bpm = 145;
  1122→
  1123→  setBpm(bpm: number): void {
  1124→    this._bpm = bpm;
  1125→  }
  1126→
  1127→  applyMusicalUnderstanding(understanding: {
  1128→    key: { root: number; scale: string; confidence: number };
  1129→    bpm: number;
  1130→    bpmConfidence: number;
  1131→    style?: string;
  1132→    styleConfidence?: number;
  1133→  }): void {
  1134→    // ── KEY PURSUIT — refresh lead/acid generators when key changes ──
  1135→    // The radio's key detector returns a CHROMATIC root (0-11, C=0..B=11).
  1136→    // Our internal musicalKey.root is a MIDI note (typically 36-60). So we
  1137→    // lift the chromatic root into the bass octave (36 + root) when it's in
  1138→    // the chromatic range. If the listener ever returns a MIDI note directly
  1139→    // (> 11), we trust it as-is. NaN/undefined confidence → no-op.
  1140→    const k = understanding.key;
  1141→    const conf = typeof k?.confidence === 'number' && isFinite(k.confidence) ? k.confidence : 0;
  1142→    const rawRoot = typeof k?.root === 'number' && isFinite(k.root) ? k.root : -1;
  1143→    if (conf > 0.2 && rawRoot >= 0) {
  1144→      // Snap the root into the world's preferred rootRange so the engine
  1145→      // stays in a useful octave for the current world. We pick the octave
  1146→      // of 36 + (root mod 12) that falls inside rootRange.
  1147→      const chroma = ((rawRoot % 12) + 12) % 12;
  1148→      const worldRange = this.currentWorld.rootRange;
  1149→      let newRoot: number;
  1150→      if (rawRoot >= 12 && rawRoot >= worldRange[0] && rawRoot <= worldRange[1]) {
  1151→        // Listener returned a MIDI note already in range — trust it.
  1152→        newRoot = rawRoot;
  1153→      } else {
  1154→        // Lift chroma into an octave inside the world's rootRange.
  1155→        const lo = worldRange[0];
  1156→        const candidate = lo + (((chroma - (lo % 12)) + 12) % 12);
  1157→        newRoot = candidate;
  1158→      }
  1159→      const newScale = (typeof k.scale === 'string' && k.scale.length > 0) ? k.scale : this.musicalKey.scale;
  1160→      const changed = newRoot !== this.musicalKey.root || newScale !== this.musicalKey.scale;
  1161→      // ALWAYS update musicalKey first, then refresh generators — order matters
  1162→      // so the generators pick up the new key.
  1163→      this.musicalKey = { root: newRoot, scale: newScale };
  1164→      // Also store the scale on the ref feature snapshot so the classifier
  1165→      // can use it next time it runs.
  1166→      this.refKeyScale = newScale;
  1167→      if (changed) {
  1168→        this.refreshMusicalGenerators();
  1169→        if (typeof console !== 'undefined') {
  1170→          console.log('[PSY4] Key updated:', this.musicalKey);
  1171→        }
  1172→      }
  1173→    }
  1174→
  1175→    // ── CONTINUOUS BPM TRACKING (smooth ramp for large diffs) ──
  1176→    // Always honor the new BPM when confidence is high. Small diffs (<=2 BPM)
  1177→    // are applied immediately; larger diffs ramp across 4 bars to avoid tempo
  1178→    // glitches and keep the scheduler's lookahead stable.
  1179→    if (understanding.bpm > 0 && understanding.bpmConfidence > 0.5) {
  1180→      const target = Math.round(understanding.bpm);
  1181→      this.refBpm = clamp(target, 30, 220);
  1182→      const diff = target - this._bpm;
  1183→      if (Math.abs(diff) > 2) {
  1184→        this.targetBpm = clamp(target, 60, 200);
  1185→        this.bpmRampBarsLeft = 4;
  1186→        this.bpmRampPerBar = diff / 4;
  1187→      } else {
  1188→        this._bpm = clamp(target, 60, 200);
  1189→        this.targetBpm = 0;
  1190→        this.bpmRampBarsLeft = 0;
  1191→        this.bpmRampPerBar = 0;
  1192→      }
  1193→    }
  1194→
  1195→    // ── STYLE CLASSIFICATION (Task 14) ──
  1196→    // Two paths to determine the style:
  1197→    //   (a) Reference listener provided an explicit style tag with high
  1198→    //       confidence → use it directly (legacy path, kept for compat).
  1199→    //   (b) No style tag (or low confidence) → LEARN the style from the
  1200→    //       acoustic features we've accumulated in liveTrack(). This is the
  1201→    //       new spectral classifier path that addresses the user's complaint
  1202→    //       that "style must be learned, not defined by fat bass".
  1203→    // In both cases we run the classifier on the stored features so the UI
  1204→    // can render the full ranking. Auto-switch only happens when the top
  1205→    // match's confidence exceeds the threshold AND the anti-thrash cooldown
  1206→    // has elapsed.
  1207→    const features = this.buildRefFeatures();
  1208→    if (features) {
  1209→      const matches = classifyStyle(features);
  1210→      this.styleMatches = matches;
  1211→
  1212→      const explicitStyle = understanding.style;
  1213→      const explicitConf = understanding.styleConfidence ?? 0;
  1214→      const topMatch = matches[0];
  1215→
  1216→      if (explicitStyle && explicitConf > 0.4) {
  1217→        // Path (a): use the explicit tag — but still record matches for UI.
  1218→        // Only auto-switch if explicit tag strongly disagrees with current world.
  1219→        const explicitWorldId = styleToWorld(explicitStyle);
  1220→        if (explicitWorldId !== this.currentWorld.id && explicitConf > 0.6) {
  1221→          this.tryAutoSwitch(explicitWorldId, `explicit style tag '${explicitStyle}' (${(explicitConf * 100).toFixed(0)}%)`);
  1222→        }
  1223→      } else if (topMatch &&
  1224→                 topMatch.confidence >= Psy4EngineV2.AUTO_SWITCH_CONFIDENCE_THRESHOLD) {
  1225→        // Path (b): learn from features and auto-switch if confident.
  1226→        const targetWorldId = styleToWorld(topMatch.style);
  1227→        if (targetWorldId !== this.currentWorld.id) {
  1228→          const reason = topMatch.reasons[0] || `style '${topMatch.style}' (${(topMatch.confidence * 100).toFixed(0)}%)`;
  1229→          this.tryAutoSwitch(targetWorldId, reason);
  1230→        }
  1231→      }
  1232→    }
  1233→  }
  1234→
  1235→  /**
  1236→   * Apply a style classification result directly (Task 14).
  1237→   * If the top match's confidence exceeds the auto-switch threshold AND it
  1238→   * differs from the current world, switch worlds smoothly.
  1239→   * Public so the UI/tests can drive it explicitly if desired.
  1240→   */
  1241→  applyStyleClassification(matches: StyleMatch[]): void {
  1242→    this.styleMatches = matches;
  1243→    const top = matches[0];
  1244→    if (!top) return;
  1245→    if (top.confidence >= Psy4EngineV2.AUTO_SWITCH_CONFIDENCE_THRESHOLD) {
  1246→      const targetWorldId = styleToWorld(top.style);
  1247→      if (targetWorldId !== this.currentWorld.id) {
  1248→        const reason = top.reasons[0] || `style '${top.style}' (${(top.confidence * 100).toFixed(0)}%)`;
  1249→        this.tryAutoSwitch(targetWorldId, reason);
  1250→      }
  1251→    }
  1252→  }

  1256→   * Empty array if no reference features have arrived yet.

  1258→  getStyleClassification(): StyleMatch[] {

  1269→
  1270→  /**
  1271→   * Attempt an automatic world switch. Respects the 30-second anti-thrash
  1272→   * cooldown and skips if the target world is the same as the last switch.
  1273→   * This is the only place auto-switches should happen.
  1274→   */
  1275→  private tryAutoSwitch(worldId: string, reason?: string): void {
  1276→    if (!this.ctx) return;
  1277→    // Defensive: validate that worldId is one of the known WorldIds before
  1278→    // attempting to switch. The classifier's styleToWorld() should always
  1279→    // return a valid id, but we don't want a bad id to slip through and
  1280→    // silently no-op in switchWorld().
  1281→    if (!(worldId in WORLDS)) {
  1282→      if (typeof console !== 'undefined') {
  1283→        console.warn('[PSY4] tryAutoSwitch: unknown worldId', worldId);
  1284→      }
  1285→      return;
  1286→    }
  1287→    const now = this.ctx.currentTime * 1000; // ms since ctx start
  1288→    if (this.lastAutoSwitchTime > 0 &&
  1289→        (now - this.lastAutoSwitchTime) < Psy4EngineV2.AUTO_SWITCH_COOLDOWN_MS) {
  1290→      return; // too soon — thrash guard
  1291→    }
  1292→    if (this.lastAutoSwitchWorldId === worldId) {
  1293→      return; // we already switched to this world recently — no-op
  1294→    }
  1295→    this.switchWorld(worldId as WorldId);
  1296→    this.lastAutoSwitchTime = now;
  1297→    this.lastAutoSwitchWorldId = worldId;
  1298→    // Notify the UI that an auto-switch happened.
  1299→    try { this.onWorldChange?.(worldId, reason); } catch {}
  1300→  }

  1302→  /**
  1303→   * Switch to a different world smoothly (Task 14). Does NOT restart playback.
  1304→   *   - Updates currentWorld, musicalKey (root + scale), refreshes generators
  1305→   *   - Ramps BPM over 4 bars if the new world's BPM differs by more than 2
  1306→   *   - Applies the new world's FX mix to reverb/delay sends
  1307→   *   - Resets phrase-locked rotation counters (start the new world's first
  1308→   *     phrase cleanly)
  1309→   *   - All audio parameter changes use setTargetAtTime / 4-bar ramp — no jumps
  1310→   */
  1311→  switchWorld(worldId: WorldId): void {
  1312→    const newWorld = WORLDS[worldId];
  1313→    if (!newWorld) return;
  1314→    this.currentWorld = newWorld;
  1315→
  1316→    // Key — keep the root if it's within the world's range; otherwise snap to
  1317→    // the midpoint of the world's rootRange.
  1318→    const newRoot = (this.musicalKey.root >= newWorld.rootRange[0] &&
  1319→                     this.musicalKey.root <= newWorld.rootRange[1])
  1320→      ? this.musicalKey.root
  1321→      : Math.floor((newWorld.rootRange[0] + newWorld.rootRange[1]) / 2);
  1322→    // PRESERVE the listener-detected scale if the reference listener has set
  1323→    // one (refKeyScale) AND the new world allows that scale. This prevents
  1324→    // switchWorld from clobbering the key pursuit when the world auto-switches
  1325→    // (the original bug: radio detected "F major" but the engine reverted to
  1326→    // the world's defaultScale "phrygianDominant" immediately after switching).
  1327→    const listenerScale = this.refKeyScale;
  1328→    const scaleAllowed = (s?: string): s is string =>
  1329→      !!s && newWorld.scales.includes(s);
  1330→    const newScale = scaleAllowed(listenerScale)
  1331→      ? listenerScale
  1332→      : newWorld.defaultScale;
  1333→    const keyChanged = newRoot !== this.musicalKey.root || newScale !== this.musicalKey.scale;
  1334→    this.musicalKey = { root: newRoot, scale: newScale };
  1335→    if (keyChanged) this.refreshMusicalGenerators();
  1336→
  1337→    // BPM — ramp over 4 bars if diff > 2
  1338→    const newBpm = newWorld.defaultBpm;
  1339→    if (Math.abs(newBpm - this._bpm) > 2) {
  1340→      this.targetBpm = clamp(newBpm, 60, 200);
  1341→      this.bpmRampBarsLeft = 4;
  1342→      this.bpmRampPerBar = (newBpm - this._bpm) / 4;
  1343→    } else {
  1344→      this._bpm = clamp(newBpm, 60, 200);
  1345→      this.targetBpm = 0;
  1346→      this.bpmRampBarsLeft = 0;
  1347→      this.bpmRampPerBar = 0;
  1348→    }
  1349→
  1350→    // FX mix — smooth ramp
  1351→    if (this.ctx) {
  1352→      const now = this.ctx.currentTime;
  1353→      const fxMix = newWorld.fxMix;
  1354→      this.reverbSend.gain.setTargetAtTime(0.04 + fxMix * 0.22, now, 0.5);
  1355→      this.delaySend.gain.setTargetAtTime(0.05 + fxMix * 0.30, now, 0.5);
  1356→    }
  1357→
  1358→    // Reset phrase-locked counters — start the new world's first phrase cleanly
  1359→    this.phraseCounter = 0;
  1360→    this.phrasePresetVariant = 0;
  1361→    this.arpIdx = 0;
  1362→    this.bassPatternIdx = 0;
  1363→
  1364→    // Apply the new world's preferred kick/bass/lead presets immediately so
  1365→    // the next phrase starts with the right timbres. The phrase-locked
  1366→    // rotation will then alternate between the two variants from here on.
  1367→    this.applyWorldPresets();
  1368→
  1369→    // Apply per-world send-effect settings (Task E1) — pushes per-track send
  1370→    // levels and global effect parameters for the new world.
  1371→    this.applyWorldEffectSettings(newWorld);
  1372→
  1373→    // ── Task F1: update the flow engine's world profile ──
  1374→    // The new world's flow characteristics (baseline energy, section length
  1375→    // range, archetype weights, surprise rate) take effect on the NEXT
  1376→    // transition. We DON'T force a transition here — the music keeps flowing
  1377→    // organically and the new world's character shapes the next section
  1378→    // change. This avoids jarring mid-section character shifts while still
  1379→    // adapting the flow to the new world's identity.
  1380→    if (this.flowEngine) {
  1381→      this.flowEngine.setWorld(newWorld);
  1382→    }
  1383→  }

  1385→  /**
  1386→   * Apply the current world's preferred kick/bass/lead/pad/arp presets.
  1387→   * Called by switchWorld() and at start(). Phrase-locked rotation in tick()
  1388→   * will alternate between the two variants every 8 bars from here on.
  1389→   *
  1390→   * Task S1: leads/pads/arp now use ADVANCED_PRESETS:
  1391→   *   - Goa/Acid leads → FM presets (PS-FM-GOA / PS-FM-SQUELCH) — metallic/squelchy
  1392→   *   - Dark worlds    → PS-FM-SQUELCH (acid character)
  1393→   *   - Bright worlds  → PS-FM-BELL (cleaner, bell-like)
  1394→   *   - Pads           → PS-SUPERSAW-PAD (thick, rich, 7-osc supersaw)
  1395→   *   - Arp            → PS-WT-MORPH (wavetable, evolving texture)
  1396→   *   - Bass           → classic (PS-BASS-ROLL/DEEP) — bass doesn't need FM
  1397→   *   - Kick           → classic drum presets (unchanged)
  1398→   *
  1399→   * synthModeOverrides (set by reference pursuit via setSynthMode) take
  1400→   * precedence over the world defaults — applied at triggerSynth() time.
  1401→   */
  1402→  private applyWorldPresets(): void {
  1403→    const id = this.currentWorld.id;
  1404→    // Dark worlds → DEEP kick + ROLL bass; bright worlds → TIGHT kick + DEEP bass.
  1405→    // Mid worlds → mix. Lead swaps for goa/acid (FM metallic) vs others (FM bell).
  1406→    const dark = id === 'dark-psy' || id === 'forest' || id === 'deep-psy' || id === 'hypnotic';
  1407→    const acid = id === 'goa' || id === 'acid-psy';
  1408→    const bright = id === 'morning-psy' || id === 'cosmic' || id === 'organic-psy';
  1409→
  1410→    this.tracks[0].presetId = dark ? 'PS-KICK-DEEP' : 'PS-KICK-TIGHT';
  1411→    // Bass stays classic — bass doesn't need FM/supersaw/wavetable
  1412→    this.tracks[4].presetId = dark ? 'PS-BASS-ROLL' : (bright ? 'PS-BASS-DEEP' : 'PS-BASS-ROLL');
  1413→    // Lead: FM for goa/acid (metallic goa leads), squelchy FM for dark worlds,
  1414→    // bell FM for bright worlds. The synthModeOverrides can still flip a track
  1415→    // to supersaw/wavetable/classic at runtime via setSynthMode().
  1416→    this.tracks[5].presetId = acid
  1417→      ? 'PS-FM-GOA'
  1418→      : (dark ? 'PS-FM-SQUELCH' : 'PS-FM-BELL');
  1419→    // Pad: supersaw for thick rich pads (7-osc with stereo spread)
  1420→    this.tracks[6].presetId = 'PS-SUPERSAW-PAD';
  1421→    // Arp: wavetable for evolving textures that morph over time
  1422→    this.tracks[7].presetId = 'PS-WT-MORPH';
  1423→  }

  1425→  /**
  1426→   * Apply per-world send-effect modulations to the existing racks (Task E1).
  1427→   * Called by start() and switchWorld() after applyWorldPresets(). Rebuilds
  1428→   * the per-track rack configs for the new world and pushes the SEND levels
  1429→   * (reverb/delay/chorus/phaser/distortion/bitcrush) to the racks via smooth
  1430→   * ramps. EQ/comp/sat/pan are NOT changed here — they're set once at init()
  1431→   * and left alone so a world switch doesn't reset the per-track tonal balance
  1432→   * mid-phrase.
  1433→   *
  1434→   * Safe to call before init() (no-op when racks is empty).
  1435→   */
  1436→  private applyWorldEffectSettings(world: World): void {
  1437→    if (this.racks.length === 0) return;
  1438→    const cfgs = buildTrackRackConfigs(world);
  1439→    for (let i = 0; i < this.racks.length && i < cfgs.length; i++) {
  1440→      const rack = this.racks[i];
  1441→      const cfg = cfgs[i];
  1442→      // Only send levels are ramped here — tonal chain (EQ/comp/sat/pan) is
  1443→      // set once at init() and left alone for stability.
  1444→      rack.setParameter('sendReverb', cfg.sendReverb);
  1445→      rack.setParameter('sendDelay', cfg.sendDelay);
  1446→      rack.setParameter('sendChorus', cfg.sendChorus);
  1447→      rack.setParameter('sendPhaser', cfg.sendPhaser);
  1448→      rack.setParameter('sendDistortion', cfg.sendDistortion);
  1449→      rack.setParameter('sendBitcrush', cfg.sendBitcrush);
  1450→    }
  1451→
  1452→    // Also nudge the global send-effect parameters based on world character.
  1453→    if (this.chorusEffect && this.phaserEffect && this.distortionEffect && this.bitcrushEffect) {
  1454→      // Bright worlds → faster chorus rate; dark worlds → slower.
  1455→      const chorusRate = 0.35 + world.brightness * 0.5;
  1456→      this.chorusEffect.setParameter('rate', chorusRate);
  1457→      // Psychedelic worlds → faster phaser sweep + more feedback.
  1458→      const phaserRate = 0.15 + world.psychedelia * 0.6;
  1459→      this.phaserEffect.setParameter('rate', phaserRate);
  1460→      this.phaserEffect.setParameter('feedback', 0.25 + world.psychedelia * 0.4);
  1461→      // Aggressive/dark worlds → harder distortion.
  1462→      const distDrive = 2.5 + world.aggression * 5 + world.darkness * 2;
  1463→      this.distortionEffect.setParameter('drive', distDrive);
  1464→      // Dark worlds → coarser bitcrush (fewer bits, longer hold).
  1465→      const bcBits = Math.round(8 - world.darkness * 4);
  1466→      const bcHold = 2 + world.darkness * 6;
  1467→      this.bitcrushEffect.setParameter('bits', bcBits);
  1468→      this.bitcrushEffect.setParameter('holdMs', bcHold);
  1469→    }
  1470→  }

  1472→  /**
  1473→   * Adjust a single per-track effect parameter in real-time (Task E1).
  1474→   * Routes to the named rack's setParameter(). Used by the reference pursuit
  1475→   * (and the future automated mixer) to nudge timbre as the radio's character
  1476→   * changes. Unknown trackIdx / effectName → silent no-op.
  1477→   *
  1478→   * Recognized effectNames (see TrackEffectsRack.setParameter):
  1479→   *   eqLowGain, eqMidFreq, eqMidGain, eqMidQ, eqHighGain,
  1480→   *   compThreshold, compRatio, compAttack, compRelease, compKnee,
  1481→   *   satDrive, satMix,
  1482→   *   pan, haasDelayMs, haasMix,
  1483→   *   outputGain,
  1484→   *   sendReverb, sendDelay, sendChorus, sendPhaser, sendDistortion, sendBitcrush
  1485→   *
  1486→   * Task V2b addition: 'cutoff' is special-cased for the LEAD track (idx 5).
  1487→   * The lead's filter cutoff lives inside the AdvancedSynthVoice (not the
  1488→   * rack — the rack is post-voice). So we store the override here and apply
  1489→   * it in triggerSynth() when the lead's voice is about to fire. Pass -1
  1490→   * to clear the override (revert to world timbre + reference pursuit blend).
  1491→   * Any non-negative value (Hz) is clamped to [200, 16000] and used directly.
  1492→   */
  1493→  setTrackEffect(trackIdx: number, effectName: string, value: number): void {
  1494→    if (!Number.isFinite(value)) return;
  1495→    // Task V2b: lead filter cutoff override (for BUILD-section filter sweeps).
  1496→    // Stored on the engine; applied in triggerSynth() when the lead fires.
  1497→    if (effectName === 'cutoff') {
  1498→      if (trackIdx !== 5) return; // only LEAD has a sweepable filter override
  1499→      this.leadCutoffOverride = value < 0 ? -1 : clamp(value, 200, 16000);
  1500→      return;
  1501→    }
  1502→    if (trackIdx < 0 || trackIdx >= this.racks.length) return;
  1503→    this.racks[trackIdx].setParameter(effectName, value);
  1504→  }
  1505→
  1506→  /**
  1507→   * Adjust a per-track SEND level in real-time (Task E1). Subset of
  1508→   * setTrackEffect() specialized for sends — handy for the arrangement
  1509→   * engine to push more reverb in breaks, less in drops, etc.
  1510→   *
  1511→   *   sendName ∈ { 'reverb', 'delay', 'chorus', 'phaser', 'distortion', 'bitcrush' }
  1512→   *   level ∈ [0, 1]
  1513→   */
  1514→  setSendLevel(trackIdx: number, sendName: 'reverb' | 'delay' | 'chorus' | 'phaser' | 'distortion' | 'bitcrush', level: number): void {
  1515→    if (!Number.isFinite(level)) return;
  1516→    if (trackIdx < 0 || trackIdx >= this.racks.length) return;
  1517→    this.racks[trackIdx].setParameter(`send${sendName.charAt(0).toUpperCase()}${sendName.slice(1)}` as any, level);
  1518→  }
  1519→
  1520→  /**
  1521→   * Adjust a global send-effect parameter (Task E1). Lets the reference
  1522→   * pursuit or arrangement engine tweak the chorus rate / phaser feedback /
  1523→   * distortion drive / bitcrush bits in real time.
  1524→   *
  1525→   *   effectName ∈ { 'chorus', 'phaser', 'distortion', 'bitcrush' }
  1526→   *   param depends on the effect (see each class's setParameter).

  1529→    if (!Number.isFinite(value)) return;

  1531→      case 'chorus':     this.chorusEffect?.setParameter(param, value);     break;
  1532→      case 'phaser':     this.phaserEffect?.setParameter(param, value);     break;
  1533→      case 'distortion': this.distortionEffect?.setParameter(param, value); break;
  1534→      case 'bitcrush':   this.bitcrushEffect?.setParameter(param, value);   break;
  1535→    }

  1538→  /**
  1539→   * Adjust a master multiband compressor parameter in real-time (Task E1).

  1541→   *   lowThreshold, lowRatio, lowAttack, lowRelease, lowKnee, lowMakeup,

  1545→  setMasterParam(name: string, value: number): void {

  1549→  // ─── Task F1: continuous flow automation (replaces V2b section automation) ──
  1550→  //
  1551→  // Commercial psytrance rides the mix alongside the arrangement: reverb
  1552→  // washes out in breaks, sucks dry in drops, delay throws in transitions,
  1553→  // filter sweeps open in builds, chorus thickens in the variation.
  1554→  //
  1555→  // The V2b approach pushed STATIC send levels on section changes — a DROP
  1556→  // and a BREAK had different reverb tails, but within a section the mix was
  1557→  // flat. Task F1 replaces this with CONTINUOUS automation: the flow engine
  1558→  // produces smooth filterCutoff / reverbAmount / delayAmount / tension /
  1559→  // surprise values every bar (interpolated toward the archetype target with
  1560→  // a 1-4 bar time constant), and applyFlowAutomation() pushes them every
  1561→  // step via setTrackEffect / setSendLevel (both use setTargetAtTime
  1562→  // internally, so re-pushing is a smooth no-op once settled).
  1563→  //
  1564→  // This means:
  1565→  //   - The lead filter OPENS continuously during a BUILD (tension rising →
  1566→  //     cutoff rising) instead of jumping at the section boundary.
  1567→  //   - The reverb wash CONTINUOUSLY recedes during a DROP approach (energy
  1568→  //     rising → reverb falling) instead of snapping dry at the drop start.
  1569→  //   - The delay amount RISES at the end of a VARIATION phrase (tension
  1570→  //     releasing → delay rising for echo throws) instead of being a flat
  1571→  //     0.40 throughout.
  1572→  //
  1573→  // Layered on top:
  1574→  //   - Per-section chorus/phaser profile (kept from V2b — these are timbral
  1575→  //     colors that don't benefit from continuous automation).
  1576→  //   - Active surprise event effects (dropOut mute, filterSweep progression,
  1577→  //     echoThrow delay boost, silence) — applied per-step.
  1578→  //
  1579→  // Tracks: 0=KICK 1=SNARE 2=HATS 3=PERC 4=BASS 5=LEAD 6=PAD 7=ARP.
  1580→  applyFlowAutomation(flow: FlowState, bar: number, step: number, time: number): void {
  1581→    if (!this.ctx) return;
  1582→
  1583→    // ── (1) Per-section chorus/phaser profile (pushed on label change) ──
  1584→    // These are timbral colors that don't benefit from continuous automation
  1585→    // — they're either on or off for a given section type. We track the last
  1586→    // applied label to avoid spamming the audio thread (the rack uses
  1587→    // setTargetAtTime(0.05s) so re-pushing is a no-op once settled).
  1588→    if (flow.label !== this.lastAutomationSection) {
  1589→      this.lastAutomationSection = flow.label;
  1590→      this.applySectionChorusPhaser(flow.label);
  1591→    }
  1592→
  1593→    // ── (2) Continuous reverb send — driven by flow.reverbAmount ──
  1594→    // The flow engine smooths reverbAmount toward the archetype target:
  1595→    //   BREAK → 0.70 (wash), DROP → 0.25 (punchy), INTRO → 0.60, etc.
  1596→    // We push it to all melodic (5/6/7) and atmos (1/2/3) tracks. Kick (0)
  1597→    // and bass (4) stay at their world defaults — they need to stay punchy.
  1598→    const melReverb = clamp(flow.reverbAmount, 0, 0.8);
  1599→    const atmoReverb = clamp(flow.reverbAmount * 0.7, 0, 0.6);
  1600→    for (const ti of [5, 6, 7]) {
  1601→      this.setSendLevel(ti, 'reverb', melReverb);
  1602→    }
  1603→    for (const ti of [1, 2, 3]) {
  1604→      this.setSendLevel(ti, 'reverb', atmoReverb);
  1605→    }
  1606→
  1607→    // ── (3) Continuous delay send — driven by flow.delayAmount ──
  1608→    // The flow engine smooths delayAmount: VARIATION → 0.45 (echo throws),
  1609→    // BREAK → 0.50, DROP → 0.30, INTRO → 0.10. Same track routing as reverb.
  1610→    const melDelay = clamp(flow.delayAmount, 0, 0.6);
  1611→    const atmoDelay = clamp(flow.delayAmount * 0.5, 0, 0.4);
  1612→    for (const ti of [5, 6, 7]) {
  1613→      this.setSendLevel(ti, 'delay', melDelay);
  1614→    }
  1615→    for (const ti of [1, 2, 3]) {
  1616→      this.setSendLevel(ti, 'delay', atmoDelay);
  1617→    }
  1618→
  1619→    // ── (4) Continuous lead filter cutoff — driven by flow.filterCutoff ──
  1620→    // The flow engine smooths filterCutoff exponentially (ears hear log-Hz):
  1621→    //   BUILD → 3500 Hz (opening), DROP → 4000 Hz (bright), BREAK → 700 Hz
  1622→    //   (closing), INTRO → 1200 Hz. This naturally produces the signature
  1623→    //   "filter opening" build effect and "filter closing" break release
  1624→    //   WITHOUT hardcoded per-section sweeps — the smoothing does the work.
  1625→    //
  1626→    // During a filterSweep surprise, we OVERRIDE this with the surprise's
  1627→    // own sweep curve (computed below in the surprise handling section).
  1628→    let leadCutoffTarget = flow.filterCutoff;
  1629→
  1630→    // ── (5) Active surprise event effects ──
  1631→    // Per-step application of the active surprise (set in tick() when
  1632→    // maybeSurprise() returns an event).
  1633→    const surprise = this.activeSurprise;
  1634→    if (surprise) {
  1635→      const surpriseProgress = clamp(
  1636→        (this.totalBars - surprise.startBar) / Math.max(1, surprise.durationBars),
  1637→        0, 1,
  1638→      );
  1639→      switch (surprise.type) {
  1640→        case 'filterSweep': {
  1641→          // Open the filter wide, then close it back — a DJ-style EQ sweep.
  1642→          // Peak at mid-progress (intensity scales how wide the sweep goes).
  1643→          const peakHz = clamp(2000 + surprise.intensity * 6000, 2000, 12000);
  1644→          const baseHz = clamp(leadCutoffTarget, 400, 4000);
  1645→          // Triangle: baseHz → peakHz → baseHz
  1646→          const tri = 1 - Math.abs(surpriseProgress * 2 - 1);
  1647→          leadCutoffTarget = baseHz * Math.pow(peakHz / baseHz, tri);
  1648→          break;
  1649→        }
  1650→        case 'echoThrow': {
  1651→          // Boost the delay send + feedback for the echo throw duration.
  1652→          // The delay feedback is on this.delayFb (set at init to 0.35).
  1653→          const throwBoost = clamp(0.4 + surprise.intensity * 0.4, 0.4, 0.8);
  1654→          this.setSendLevel(5, 'delay', throwBoost);     // lead echo
  1655→          this.setSendLevel(7, 'delay', throwBoost * 0.7); // arp echo
  1656→          if (this.delayFb) {
  1657→            this.delayFb.gain.setTargetAtTime(
  1658→              clamp(0.45 + surprise.intensity * 0.3, 0.4, 0.8),
  1659→              time, 0.05,
  1660→            );
  1661→          }
  1662→          break;
  1663→        }
  1664→        case 'stutter': {
  1665→          // During a stutter, the lead retrigger is handled by the surprise
  1666→          // start handler (startSurprise) — here we just boost the delay send
  1667→          // so the stuttered notes echo.
  1668→          this.setSendLevel(5, 'delay', clamp(0.5 * surprise.intensity + 0.2, 0.2, 0.7));
  1669→          break;
  1670→        }
  1671→        case 'dropOut':
  1672→        case 'silence':
  1673→        case 'reverseHit':
  1674→          // No per-step FX change — the note gating in scheduleStep handles
  1675→          // dropOut/silence, and reverseHit is a one-shot fired in startSurprise.
  1676→          break;
  1677→      }
  1678→    } else {
  1679→      // No active surprise — relax the delay feedback back to its default.
  1680→      if (this.delayFb) {
  1681→        this.delayFb.gain.setTargetAtTime(0.35, time, 0.3);
  1682→      }
  1683→    }
  1684→

  1732→  //
  1733→  // startSurprise() is called once when a surprise event is popped from the
  1734→  // flow engine (in tick()). It fires any one-shot effects:
  1735→  //   - reverseHit: trigger a reversed impact (build tension)
  1736→  //   - dropOut:    ramp non-kick track gains to near-zero (DJ brake)
  1737→  //   - silence:    ramp the master to near-zero (dramatic pause)
  1738→  //   - stutter:    schedule a rapid lead retrigger via the existing triggerSynth
  1739→  //   - filterSweep / echoThrow: no one-shot — handled per-step in applyFlowAutomation
  1740→  //
  1741→  // endActiveSurprise() is called when the surprise's duration has elapsed.
  1742→  // It restores any muted tracks / boosted sends to their normal levels.
  1743→  // The continuous automation in applyFlowAutomation will re-push the
  1744→  // correct values on the next step, so we just need to clear the surprise-

  1747→  private startSurprise(event: SurpriseEvent, time: number): void {
  1748→    if (!this.ctx) return;
  1749→    const intensity = clamp(event.intensity, 0, 1);
  1750→
  1751→    switch (event.type) {
  1752→      case 'reverseHit': {
  1753→        // Fire a reversed impact — a sub-boom that swells IN instead of
  1754→        // decaying OUT. Builds tension before the next hit.

  1756→          this.triggerReverseImpact(time, intensity);
  1757→          this.surpriseReverseHitScheduled = true;
  1758→        }
  1759→        break;
  1760→      }
  1761→      case 'dropOut': {
  1762→        // DJ brake: ramp all non-kick track gains to near-zero over 50ms,
  1763→        // hold for the duration, then ramp back (the ramp-back happens in
  1764→        // endActiveSurprise). Kick (track 0) is NOT muted — it's the
  1765→        // heartbeat that keeps the groove alive during the brake.

  1767→        for (let i = 1; i < 8; i++) {
  1768→          const g = this.trackGains[i];
  1769→          if (g) {
  1770→            g.gain.setTargetAtTime(g.gain.value * muteDepth, time, 0.02);
  1771→          }
  1772→        }
  1773→        break;
  1774→      }
  1775→      case 'silence': {
  1776→        // Dramatic pause: ramp the master to near-zero. The next section's
  1777→        // first hit (or the end-of-surprise ramp-back) will be the payoff.
  1778→        const muteDepth = clamp(1 - intensity * 0.98, 0.02, 0.3);
  1779→        this.master.gain.setTargetAtTime(this.master.gain.value * muteDepth, time, 0.015);
  1780→        break;
  1781→      }
  1782→      case 'stutter': {
  1783→        // Rapid lead retrigger — fire 4-6 short lead notes at the current
  1784→        // chord root. Uses triggerSynth directly so the notes go through
  1785→        // the full voice + rack chain (with the flow's current lead timbre).
  1786→        if (this.currentChord && this.melody) {
  1787→          const root = this.currentChord.notes[0] + 12; // one octave up
  1788→          const stutters = 4 + Math.round(intensity * 2);
  1789→          const s16 = 60 / this.bpm / 4;
  1790→          for (let i = 0; i < stutters; i++) {
  1791→            this.triggerSynth(5, time + i * s16 * 0.5, root, 0.3 + intensity * 0.2, s16, s16 * 0.4);
  1792→          }
  1793→        }
  1794→        break;
  1795→      }
  1796→      case 'filterSweep':
  1797→      case 'echoThrow':
  1798→        // No one-shot — handled per-step in applyFlowAutomation
  1799→        break;

  1801→  }
